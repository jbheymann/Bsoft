92{

94 long converged(0), ilo(0), inhi(1), ihi(2);
95 long rand_max = get_rand_max(); // My own random maximum due to system differences
96 double irm(1.0L/rand_max);
97 double r, Rtry, Rsave, reltol(tolerance/10);
98
99 if ( maxcycles < 1 ) maxcycles = 1;
100
101 random_seed();
102
103 if ( verbose & VERB_DEBUG ) {
104 for ( i=0; i<10; i++ )
105 cout << "DEBUG Bsimplex::run: Random test: " <<
106 rand_max << " " << random() << " " << exp(random()*4.0/rand_max-2.0) << endl;
107 cout << "DEBUG Bsimplex::run: Initial parameters:" << endl;
108 for ( i=0; i<nparam; ++i )

110 }
111
112 // Initialize the R's
113 vector<double> R(npnt, 0);
114// for ( i=0; i<npnt; i++ ) R[i] = 0;
115 R[ihi] = 1e30;
116
117 for ( j=0; j<nparam; j++ ) {
118 if ( param[j] < lo[j] ) param[j] = lo[j];
119 if ( param[j] > hi[j] ) param[j] = hi[j];
120 }
121
122 // Initialize the simplex multiple point parameter array
123 vector<double> mp(npnt*nparam);
124 for ( j=0; j<nparam; j++ ) mp[j] = param[j]; // Set up the first point
125 for ( i=1; i<npnt; i++ ) { // Set up the other points
126 for ( j=0; j<nparam; j++ ) {
127 k = i*nparam+j;
128 r = random()*irm;
129 if ( lo.size() && hi.size() ) { // Stay within limits
130 mp[k] = lo[j] + (hi[j] - lo[j])*(0.25 + 0.5*r);
131 } else { // No limits are specified
132 if ( param[j] == 0 )
133 mp[k] = 2.0*r - 1.0;
134 else
135 mp[k] = param[j]*exp(4.0*r-2.0);
136 }
137 }
138 }
139
140 if ( verbose && report ) {
141 cout << "Cycle";
142 for ( j=0; j<nparam; j++ ) cout << "\tp" << j;
143 cout << "\tR" << endl;
144 }

146 while ( !converged ) {
147 for ( i=0; i<npnt; i++ ) { // Set up simplex points
148 for ( j=0; j<nparam; j++ ) param[j] = mp[i*nparam+j];
149 R[i] = (funk)(*this);
150 }
151 ilo = ihi = inhi = 0; // Rank the points
152 for ( i=0; i<npnt; i++ ) {
153 if ( R[i] <= R[ilo] ) ilo = inhi = i;
154 if ( R[i] > R[ihi] ) ihi = i;
155 }
156 for ( i=0; i<npnt; i++ )
157 if ( R[i] >= R[inhi] && i != ihi ) inhi = i;
158
159 // Exit here if the process converged
160 if ( cycle >= maxcycles || R[ihi] < tolerance ) {
161 converged = 1;
162 } else if ( (R[ihi]-R[ilo])/R[ilo] < reltol ) {
163 // Set the first point to the lowest
164 for ( j=0; j<nparam; j++ ) mp[j] = mp[ilo*nparam+j];
165 for ( i=1; i<npnt; i++ ) { // Randomize other points
166 for ( j=0; j<nparam; j++ ) {
167 k = i*nparam+j;
168 r = random()*irm;
169 if ( lo[j] < hi[j] ) { // Stay within limits
170 mp[k] = lo[j] + (hi[j] - lo[j])*(0.25 + 0.5*r);
171 } else { // If no limits are specified
172 if ( mp[k] == 0 )
173 mp[k] = 2.0*r - 1.0;
174 else
175 mp[k] = mp[k]*exp(4.0*r-2.0);
176 }
177 }
178 }
179 } else {

181 Rtry = amotry(mp, R, ihi, -1.0, funk);
182 // If it is better, try an additional extrapolation

194 mp[k] = param[j];
195 }
196 R[i] = (funk)(*this);
197 }
198 }
199 }
200 }

210 }
211

214 for ( i=0; i<npnt; i++ ) { // Show simplex points
215 cout << "DEBUG Bsimplex::run: Final: ";
216 for ( j=0; j<nparam; j++ ) cout << " " << mp[i*nparam+j];

218 }
219 }

247/*
248@brief Generates new points in parameter space for the simplex function.
249@param &mp multiple point matrix.
250@param &R R vector.
251@param ihi index of highest R.
252@param fac direction and magnitude of simplex modification.
253@fn (funk)(Bsimplex &) evaluation function returning an R value.
254@return double an R value.
255
256 Downhill simplex method of Nelder and Mead.
257Reference: Press W.H. et al (1992) Numerical Recipes in C.
258
259**/
260double Bsimplex::amotry(vector<double>& mp, vector<double>& R,
261 long ihi, double fac, double (funk)(Bsimplex&))
262{
263 long i, j, k, npnt(nparam + 1);
264 double psum, pnew, Rtry;
265 double fac1((1 - fac)/nparam);
266 double fac2(fac1 - fac);
267
268 for ( k=0; k<npnt; ++k ) {
269 for ( i=0; i<nparam; ++i ) {
270 avg[i] += mp[k*nparam+i];
271 for ( j=0; j<nparam; ++j )
272 cov[i][j] += mp[k*nparam+i]*mp[k*nparam+j];
273 }
274 }
275
276 for ( j=0; j<nparam; j++ ) {
277 psum = 0;
278 for ( i=0; i<npnt; i++ ) psum += mp[i*nparam+j];
279 param[j] = mp[ihi*nparam+j];
280 pnew = psum*fac1 - mp[ihi*nparam+j]*fac2;
281 if ( lo[j] < hi[j] ) {
282 if ( pnew < lo[j] ) pnew = param[j];
283 if ( pnew > hi[j] ) pnew = param[j];
284 }
285 param[j] = pnew;
286 }
287
288 Rtry = (funk)(*this);
289
290 if ( Rtry <= R[ihi] ) {
291 R[ihi] = Rtry;
292 for ( j=0; j<nparam; j++ ) mp[ihi*nparam+j] = param[j];
293 }
294
295 return Rtry;
296}
