23double molgroup_map_energy(Bmolgroup* molgroup, Bimage* map, double Kmap)
24{

27 Bmolgroup* mg;

38 for ( Emg=0, nmg=0, mg = molgroup; mg; mg = mg->next ) {

41 for ( atom = res->atom; atom; atom = atom->next ) {
42 x = (long) (atom->coord[0]/map->sampling(0)[0] + map->image->origin()[0] + 0.5);
43 y = (long) (atom->coord[1]/map->sampling(0)[1] + map->image->origin()[1] + 0.5);
44 z = (long) (atom->coord[2]/map->sampling(0)[2] + map->image->origin()[2] + 0.5);
45 if ( x>0 && x<map->sizeX()-1 && y>0 && y<map->sizeY()-1 && z>0 && z<map->sizeZ()-1 ) {
46 i = map->index(0, x, y, z, 0);
47 F = Vector3<double>((*map)[i+1] - (*map)[i-1], (*map)[i+map->sizeX()] - (*map)[i-map->sizeX()],
48 (*map)[i+slicesize] - (*map)[i-slicesize]);
49 Em -= (*map)[i];
50 atom->F += F * invunit;
51 }
52 nm++;
53 }
54 }
55 if ( nm ) mol->fom = (Em/nm + map->average())/map->standard_deviation() + 10;
56 else mol->fom = 100;
57 Emg += Em;
58 nmg += nm;
59 }
60 if ( nmg ) mg->fom = (Emg/nmg + map->average())/map->standard_deviation() + 10;
61 else mg->fom = 100;
62 E += Emg;
63 n += nmg;
64 }
65
66 if ( n < 1 ) {
67 if ( verbose & VERB_PROCESS )
68 cout << "Warning: Molecule group outside map boundaries!" << endl;
69 E = 1e100;
70 } else
71 E = Kmap*((E/n + map->average())/map->standard_deviation() + 10);
72
73 return E;
74}

250double molgroup_bond_fit_map_energy(Bmolgroup* molgroup, Bimage* map, double Kmap, int steps)
251{
252 map->change_type(Float) ;
253
254 Bbond* bond;
255 Batom *atom1, *atom2;
256 long i, n(0);
257 long slicesize = map->sizeX()*map->sizeY();
258 long x, y, z;
259 double frac, df, frac1, s1, s2, E(0);
260 double invunit = 0.5*Kmap/map->sampling(0)[0];
261 Vector3<double> F, F1, F2, loc;
262
263 for ( bond = molgroup->bond; bond; bond = bond->next ) {
264 atom1 = bond->atom1;
265 atom2 = bond->atom2;
266 df = 1.0L/steps;
267 s1 = s2 = 0;
268 F1 = F2 = 0;
269 for ( frac=0; frac<=1; frac+=df ) {
270 frac1 = 1 - frac;
271 loc = atom1->coord * frac1 + atom2->coord * frac;
272 x = (long) (loc[0]/map->sampling(0)[0] + map->image->origin()[0] + 0.5);
273 y = (long) (loc[1]/map->sampling(0)[1] + map->image->origin()[1] + 0.5);
274 z = (long) (loc[2]/map->sampling(0)[2] + map->image->origin()[2] + 0.5);
275 if ( x>0 && x<map->sizeX()-1 && y>0 && y<map->sizeY()-1 && z>0 && z<map->sizeZ()-1 ) {
276 i = map->index(0, x, y, z, 0);
277 F = Vector3<double>((*map)[i+1] - (*map)[i-1], (*map)[i+map->sizeX()] - (*map)[i-map->sizeX()],
278 (*map)[i+slicesize] - (*map)[i-slicesize]);
279 F1 += F * frac1;
280 F2 += F * frac;
281 if ( !isfinite((*map)[i]) ) cout << "Infinite value at " << x << " " << y << " " << z << endl;
282 E -= (*map)[i];
283 s1 += frac1;
284 s2 += frac;
285 }
286 n++;
287 }
288 atom1->F += F1 * (invunit/s1);
289 atom2->F += F2 * (invunit/s2);
290 }
291
292 if ( n < 1 ) {
293 if ( verbose & VERB_PROCESS )
294 cout << "Warning: Molecule outside map boundaries!" << endl;
295 E = 1e100;
296 } else
297 E = Kmap * ((E/n + map->average())/map->standard_deviation() + 10);
298
299 return E;
300}
