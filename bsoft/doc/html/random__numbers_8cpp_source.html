221double random_poisson(double avg)
222{
223 double r = 0;
224
225 if ( avg <= 0 ) return r;
226
227 double sq, lnavg, g, t, y(0);
228
229 if ( avg < 12.0 ) {
230 g = exp(-avg);
231 for ( r = -1, t = 1; t > g; r++ ) t *= random()*irm;
232 } else {
233 sq = sqrt(2.0*avg);
234 lnavg = log(avg);
235 g = avg*lnavg - lgamma(avg + 1.0L);
236 do {
237 for ( r = -1; r < 0; ) {
238 y = tan(M_PI*irm*random());
239 r = sq*y + avg;
240 }
241 r = floor(r);
242 t = 0.9*(1 + y*y)*exp(r*lnavg - lgamma(r + 1.0L) - g);
243 } while ( random()*irm > t );
244 }
245
246 return r;
247}
