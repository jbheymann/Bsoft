216long models_link_list_generate(Bmodel* model, double maxlength,
217 string type1, string type2, int flag)
218{
219 if ( !model ) return 0;
220 if ( !model->comp ) return 0;
221 if ( type1.length() < 1 ) return models_link_list_generate(model, maxlength);
222 if ( type2.length() < 1 ) type2 = type1;
223
224 int closest(flag&1); // Flag to find the closest second component to the first
225 long n(0);
226 double linkrad(0.1*maxlength);
227 double d, dmin(maxlength), d_avg(0), d_std(0);
228 Bmodel* mp;
229 Bcomponent* comp;
230 Bcomponent* comp2;
231 Bcomponent* compsel = NULL;
232 Bcomptype* ct1;
233 Bcomptype* ct2;
234 Blink* link = model->link;
235
236 if ( verbose ) {
237 cout << "Generating a model link list between component types:" << endl;
238 cout << "Maximum link length: " << maxlength << endl;
239 cout << "Component type 1: " << type1 << endl;
240 cout << "Component type 2: " << type2 << endl;
241 }
242
243 link = model->link;
244 if ( link ) linkrad = link->radius();
245
246 for ( mp = model; mp; mp = mp->next ) if ( mp->select() ) {
247 ct1 = mp->find_type(type1.c_str());
248 ct2 = mp->find_type(type2.c_str());
249 if ( !ct1 || !ct2 ) {
250 if ( !ct1 ) cerr << "Error: No component type " << type1 << " found!" << endl;
251 if ( !ct2 ) cerr << "Error: No component type " << type2 << " found!" << endl;
252 bexit(-1);
253 }
254
255 for ( comp = mp->comp; comp && comp->next; comp = comp->next ) {
256 if ( comp->select() && comp->type() == ct1 ) {
257 if ( closest ) {
258 dmin = maxlength;
259 compsel = NULL;
260 }
261 for ( comp2 = model->comp; comp2; comp2 = comp2->next ) {
262 if ( comp != comp2 && comp2->select() && comp2->type() == ct2 ) {
263 if ( mp->link->find(comp, comp2) == NULL ) {
264 d = comp->location().distance(comp2->location());
265 if ( closest ) {
266 if ( dmin > d ) {
267 dmin = d;
268 compsel = comp2;
269 }
270 } else if ( d <= maxlength ) {
271 link = link_add(&link, comp, comp2, d, linkrad);
272 if ( !mp->link ) mp->link = link;
273 n++;
274 d_avg += d;
275 d_std += d*d;
276 }
277 }
278 }
279 }
280 if ( closest && compsel ) {
281 d = comp->location().distance(compsel->location());
282 link = link_add(&link, comp, compsel, d, linkrad);
283 if ( !mp->link ) mp->link = link;
284 n++;
285 d_avg += d;
286 d_std += d*d;
287 }
288 }
289 }
290 }
291
292 if ( n ) {
293 d_avg /= n;
294 d_std = d_std/n - d_avg*d_avg;
295 if ( d_std > 0 ) d_std = sqrt(d_std);
296 else d_std = 0;
297 }
298
299 if ( verbose ) {
300 cout << "Number of links generated: " << n << endl;
301 cout << "Average and standard deviation: " << d_avg << tab << d_std << endl << endl;
302 }
303
304 return n;
305}
