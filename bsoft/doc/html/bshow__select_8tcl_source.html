131## @brief Updates selection object.
132#
133
134proc updateSelection { } {
135 global tool theimg
136# global px py
137# global mx my
138 global selectiontype
139 global sx1 sx2 sy1 sy2 sz1 sz2
140
141 set c [getImageCanvas $theimg]
142 $c delete selection
143
144 if { [winfo exists .wsel] && $tool == "select" && $sx1 != $sx2 && $sy1 != $sy2 } {
145 set wc [getControlWindow $theimg]
146 set slice_num [$wc.slice.scale get]
147 if { $sz1 <= $slice_num && $sz2 >= $slice_num } {
148 set scale [$wc.scale.scale get]
149 set h [lindex [$c cget -scrollregion] 3]
150 if { $selectiontype == "Square" || $selectiontype == "Rectangle" || $selectiontype == "Cube" } {
151 set cx1 [expr int($sx1 * $scale)]
152 set cy1 [expr int($h - $sy1 * $scale) ]
153 set cx2 [expr int($sx2 * $scale)]
154 set cy2 [expr int($h - $sy2 * $scale) ]
155 $c create rectangle $cx1 $cy1 $cx2 $cy2 -tags selection -outline red
156 }
157 if { $selectiontype == "Circle" || $selectiontype == "Ellipse" || $selectiontype == "Sphere" } {
158 set t 0.0
159 set z [expr ($sz1 + $sz2)*0.5 - $slice_num]
160 if { $z != 0 } {
161 set r [expr ($sz1 - $sz2)*0.5]
162 set t [expr $z/$r]
163 }
164 set t [expr sqrt(1.0 - $t*$t)]
165 set tn [expr 0.5*(1.0 - $t)]
166 set tp [expr 0.5*(1.0 + $t)]
167 set cx1 [expr $sx1 * $scale]
168 set cy1 [expr $h - $sy1 * $scale ]
169 set cx2 [expr $sx2 * $scale]
170 set cy2 [expr $h - $sy2 * $scale ]
171 set ccx1 [expr int($tp*$cx1 + $tn*$cx2)]
172 set ccx2 [expr int($tn*$cx1 + $tp*$cx2)]
173 set ccy1 [expr int($tp*$cy1 + $tn*$cy2)]
174 set ccy2 [expr int($tn*$cy1 + $tp*$cy2)]
175 $c create oval $ccx1 $ccy1 $ccx2 $ccy2 -tags selection -outline red
176 }
177 }
178 }
179}

184proc updateSelectionStats { } {
185 global theimg
186 global selectiontype calc_stats
187 global sx1 sx2 sy1 sy2 sz1 sz2
188 if { ! $calc_stats } { return }
189# set c [getImageCanvas $theimg]
190 set wc [getControlWindow $theimg]
191 set img_num [$wc.image.scale get]
192 set slice_num [$wc.slice.scale get]
193 set vlist "$sx1 $sy1 $sz1 $sx2 $sy2 $sz2"
194 if { $selectiontype == "Square" || $selectiontype == "Rectangle" || $selectiontype == "Cube" } {
195 set stats [Bimage stats $theimg $img_num 1 $vlist]
196 }
197 if { $selectiontype == "Circle" || $selectiontype == "Ellipse" || $selectiontype == "Sphere" } {
198 set stats [Bimage stats $theimg $img_num 2 $vlist]
199 }
200 if { $selectiontype == "Polygon" } {
201 set poly ""
202 append poly "$sx1 $sy1 $slice_num "
203 append poly "$sx1 $sy2 $slice_num "
204 append poly "$sx2 $sy2 $slice_num "
205 append poly "$sx2 $sy1 $slice_num "
206# puts $poly
207 set stats [Bimage stats $theimg $img_num 3 $poly]
208 }
209 .wsel.num.value config -text [lindex $stats 0]
210 .wsel.min.value config -text [lindex $stats 1]
211 .wsel.max.value config -text [lindex $stats 2]
212 .wsel.avg.value config -text [lindex $stats 3]
213 .wsel.std.value config -text [lindex $stats 4]
214 .wsel.sum.value config -text [lindex $stats 5]
215# puts "Expected size [expr (abs($sx1 - $sx2) + 1)*(abs($sy1 - $sy2) + 1)]"
216}
