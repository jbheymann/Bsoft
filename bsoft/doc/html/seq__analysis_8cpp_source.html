503{

506
507 Bstring temp;
508 Bresidue_type* rt = get_residue_properties(temp);
509 Bresidue_type* rt_curr = NULL;

511 string code1(nres, ' ');
512 for ( i = 0, rt_curr = rt; rt_curr; rt_curr = rt_curr->next, i++ )

514
515 kill_list((char *) rt, sizeof(Bresidue_type));
516
517 if ( verbose & VERB_DEBUG )

519
520 long nmol(0);
521 Bmolecule* mol = molgroup->mol;
522 for ( nmol = 0, mol = molgroup->mol; mol; mol = mol->next ) nmol++;
523
524 long npos(0);
525 for ( i=0; i<molgroup->maxlen; i++ ) npos += molgroup->seqflag[i];
526
527 long igap = code1.find_first_of('-');
528 vector<double> fsum(npos,0);
529 vector<double> info(npos,0);

531 string pattern(nres*npos,' ');
532 string seq(npos,' ');
533
534 if ( verbose & VERB_LABEL ) {
535 cout << "Information content analysis:" << endl;
536 cout << "Sequences: " << nmol << endl;
537 cout << "Alignment positions: " << npos << endl;
538 cout << "Moving average window: " << window << endl;
539 }
540
541 // Calculating the frequency of each residue type occurring at each
542 // position in the alignment
543 for ( mol = molgroup->mol; mol; mol = mol->next ) {
544 j = k = 0;
545 while ( j < mol->seq.length() ) {
546 if ( molgroup->seqflag[j] ) {
547 ii = code1.find_first_of(mol->seq[j]);
548 if ( ii > -1 ) freq[nres*k+ii] += 1;
549 k++;
550 }
551 j++;
552 }
553 }
554
555 // Calculating information content for a position in the alignment
556 multimap<double,char> ri;
557 double v;
558 k = 0;
559 mol = molgroup->mol;
560 for ( j=0; j<molgroup->maxlen; j++ ) {
561 if ( molgroup->seqflag[j] ) {
562 fgap = 0;
563 ri.clear();
564 for ( i=0; i<nres; i++ ) {
565 if ( i == igap )
566 fgap += freq[nres*k+i];
567 else
568 fsum[k] += freq[nres*k+i];
569 }
570 info[k] = 0;
571 if ( fsum[k] > 0 ) {
572 for ( i=0; i<nres; i++ ) if ( i != igap ) {
573 v = freq[nres*k+i]*log_2(freq[nres*k+i])/fsum[k];
574 ri.insert(pair<double,char>(v,code1[i]));
575 info[k] += v;
576 }
577 if ( fsum[k] > 20 )
578 info[k] += log(20.0/fsum[k])/log(2.0);
579 }
580 i = 0;
581 for ( auto it: ri ) {
582 pattern[nres*k+i] = it.second;
583 freq[nres*k+i] = it.first;
584 i++;
585 }
586 seq[k] = mol->seq[j];
587 k++;
588 if ( verbose & VERB_FULL )
589 cout << j << tab << fgap << endl;
590 }
591 }
592
593 // Calculate a moving average
594 vector<double> movavg = moving_average(info, window);
595
596 // Print out the list of results
597 if ( verbose & VERB_LABEL )
598 cout << " Pos\tBits\tMovAvg\tnSeq\tPattern" << endl;
599 for ( k=0; k<npos; k++ ) {
600 cout << k+1 << tab << info[k] << tab << movavg[k] << tab << fsum[k] << tab;
601 for ( i=0; i<nres; i++ ) if ( freq[nres*k+i] > 0.1 )
602 cout << pattern[nres*k+i];
603 cout << endl;
604 }
605
606 for ( k=0; k<npos; k++ ) {
607 if ( k > 0 ) cout << "-";
608 for ( i=n=0; i<nres; i++ ) if ( freq[nres*k+i] > 0.1 ) n++;
609 if ( n < 1 ) {
610 cout << "x";
611 } else {
612 if ( n > 1 ) cout << "[";
613 for ( i=0; i<nres; i++ ) if ( freq[nres*k+i] > 0.1 )
614 cout << pattern[nres*k+i];
615 if ( n > 1 ) cout << "]";
616 }
617 }
618 cout << endl;
619
620 vector<Complex<float>> per = seq_frequency_analysis(16, 4, 4, info);
621
622 vector<Complex<float>> per_avg = moving_average_complex(per, window);
623
624 Bstring title("Information");
625 if ( psfile.length() )
626 ps_seq_info(psfile, title, nres, movavg, fsum, per_avg,
627 freq, pattern);
628
629 return 0;
630}
