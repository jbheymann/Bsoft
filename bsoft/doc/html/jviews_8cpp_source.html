50int pdb_matrices_from_particles(Bparticle* part, Vector3<double> sampling, Bstring& filename) {
51
52 ofstream fpdb;
53 Matrix3 mat;

57 if ( fpdb.fail() ) return 0;

59 for ( ; part; part = part->next ) {
60
61 mat = (part->view.backward()).matrix();
62 shift = (part->loc - part->ori) * sampling;

64 for ( int i=0; i<3; i++ )

77int cmm_views(Bstring& filename, Bparticle* part, Vector3<double> sampling,
78 long part_sel, double fom_cutoff, double colorfom_min,
79 double colorfom_max, RGB<double> color, double diameter, double length, double thickness)
80{
81 ofstream fcmm;
82
83 fcmm.open(filename.c_str());
84 if ( fcmm.fail() ) return 0;
85
86 if ( verbose & VERB_PROCESS ) {
87 cout << endl << "Creating a Chimera marker file. Only the first datablock is used." << endl;
88 }
89
90 Bstring id = filename.base();
91 fcmm << "<marker_set name=\"" << id << "\">" << endl;
92
93 int n(0);
94 double fom;
95 View2<double> view;
96 Vector3<double> loc;
97
98// cout << part_sel << tab << fom_cutoff << endl;
99
100 for ( ; part; part = part->next ) {
101
102 fom = part->fom[0];
103 view = part->view2();
104 view = view.backward();
105 view.normalize();
106 loc = part->loc * sampling;
107
108// cout << part->id << tab << part->sel << tab << fom << endl;
109
110 if ( ((part_sel < 0) && (fom_cutoff < 0 )) || (part->sel == part_sel) || ((fom_cutoff > 0) && (fom >= fom_cutoff)) ) {
111
112 if ( colorfom_min >= 0 ) {
113 // truncate between min and max
114 if ( fom < colorfom_min ) { fom = colorfom_min; }
115 if ( fom > colorfom_max ) { fom = colorfom_max; }
116
117 // scale between 0 and 1
118 fom = fom - colorfom_min;
119 fom = fom / (colorfom_max - colorfom_min);
120
121 // calculate rgb from the fom value
122 color[0] = fom;
123 color[1] = 0;
124 color[2] = 1 - fom;
125 }
126
127 // put marker at the origin
128 fcmm << "<marker id=\"" << part->id << "99999\" x=\"" << loc[0]
129 << "\" y=\"" << loc[1]
130 << "\" z=\"" << loc[2]
131 << "\" r=\"" << color.r() << "\" g=\"" << color.g() << "\" b=\"" << color.b()
132 << "\" radius=\"" << thickness/2 << "\"/>" << endl;
133
134 // marker at the end of the current view vector
135 fcmm << "<marker id=\"" << part->id << "\" x=\"" << loc[0] + view[0] * length
136 << "\" y=\"" << loc[1] + view[1] * length
137 << "\" z=\"" << loc[2] + view[2] * length
138 << "\" r=\"" << color.r() << "\" g=\"" << color.g() << "\" b=\"" << color.b()
139 << "\" radius=\"" << diameter/2 << "\"/>" << endl;
140
141 //link this point to the origin
142 fcmm << "<link id1=\"" << part->id << "\" id2=\"" << part->id
143 << "99999\" r=\"" << color.r() << "\" g=\"" << color.g() << "\" b=\"" << color.b()
144 << "\" radius=\"" << thickness/2 << "\"/>" << endl;
145
146 n++;
147 }
148
149 }
150
151 fcmm << "</marker_set>" << endl;
152
153 fcmm.close();
154
155 cout << n << " markers written to " << filename << endl;
156
157 return n;
158}
