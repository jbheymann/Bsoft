176int Bimage::place(long nn, Bimage* p, Vector3<double> loc,
177 double radius, double scale, double shift, int operation)
178{
179 if ( radius <= 0 ) radius = (x > y)? x: y;
180
181 long i, si, xx, yy, zz;
182 long sz, sy, sx;
183 double sx2, sy2, sz2, r2(radius*radius), v;
184 Vector3<double> start = loc - p->image[nn].origin() + 0.5;
185 Vector3<long> lo(start);
186 Vector3<long> hi(p->size());
187 hi += lo;
188 lo = lo.max(0);
189 hi = hi.max(1);
190 hi = hi.min(size());
191
192 if ( verbose & VERB_FULL ) {
193 cout << "Placing a small image into a larger image:" << endl;
194 cout << "Small image size: " << p->size() << endl;
195 cout << "Location: " << loc << endl;
196 cout << "Low limits: " << lo << endl;
197 cout << "High limits: " << hi << endl << endl;
198 }
199
200 for ( zz=lo[2]; zz<hi[2]; zz++ ) {
201 sz = (long) zz - start[2];
202 sz2 = p->image[nn].origin()[2] - sz;
203 sz2 *= sz2;
204 for ( yy=lo[1]; yy<hi[1]; yy++ ) {
205 sy = (long) yy - start[1];
206 sy2 = p->image[nn].origin()[1] - sy;
207 sy2 *= sy2;
208 for ( xx=lo[0]; xx<hi[0]; xx++ ) {
209 sx = (long) xx - start[0];
210 sx2 = p->image[nn].origin()[0] - sx;
211 sx2 *= sx2;
212 if ( sx2 + sy2 + sz2 <= r2 ) {
213 i = index(xx, yy, zz, nn);
214 si = p->index(sx, sy, sz);
215 v = (*p)[si] * scale + shift;
216 switch ( operation ) {
217 case 1: if ( v < (*this)[i] ) set(i, v); break;
218 case 2: if ( v > (*this)[i] ) set(i, v); break;
219 default: add(i, v);
220 }
221 }
222 }
223 }
224 }
225
226 return 0;

229int Bimage::place(Bimage* p, long nn, Vector3<double> loc,
230 double radius, int operation)
231{
232 if ( radius <= 0 ) radius = (x > y)? x: y;
233 if ( operation==0 && !next ) next = new Bimage(Float, compoundtype, size(), n);
234
235 long i, j, xx, yy, zz, cc;
236 long pz, py, px;
237 double dx2, dy2, dz2, d2, w, r2(radius*radius), v;
238// Vector3<double> start = loc - p->image[nn].origin() + 0.5;
239 Vector3<double> start = loc - p->image[nn].origin();
240
241 if ( verbose & VERB_FULL )
242 cout << "Placing image " << nn << " at " << loc << endl;
243
244 for ( zz=0; zz<p->sizeZ(); ++zz ) {
245 pz = start[2] + zz;
246 dz2 = p->image[nn].origin()[2] - zz;
247 dz2 *= dz2;
248 if ( pz >= 0 && pz < z ) for ( yy=0; yy<p->sizeY(); ++yy ) {
249 py = start[1] + yy;
250 dy2 = p->image[nn].origin()[1] - yy;
251 dy2 *= dy2;
252 if ( py >= 0 && py < y ) for ( xx=0; xx<p->sizeX(); ++xx ) {
253 px = start[0] + xx;
254 dx2 = p->image[nn].origin()[0] - xx;
255 dx2 *= dx2;
256 d2 = dx2 + dy2 + dz2;
257 if ( px >= 0 && px < x && d2 <= r2 ) {
258 w = radius - sqrt(d2) + 1;
259 i = index(0, px, py, pz, 0);
260 j = p->index(0, xx, yy, zz, nn);
261// cout << xx << tab << yy << tab << zz << endl;
262 for ( cc=0; cc<c; ++cc, ++i, ++j ) {
263 v = (*p)[j];
264 switch ( operation ) {
265 case 1: if ( v < (*this)[i] ) set(i, v); break; // Maximum
266 case 2: if ( v > (*this)[i] ) set(i, v); break; // Minimum
267 default: add(i, v*w); next->add(i, w); // Sum
268 }
269 }
270 }
271 }
272 }
273 }
274
275 return 0;

331int Bimage::place_with_overlap(Bimage* p, long nn)
332{
333 long i, j, cc, tx, ty, tz, xx, yy, zz;
334 double wvol;
335 Vector3<double> d, w, imin, imax(p->size());
336
337 // Determine overlap in all dimensions
338 for ( i=0; i<p->n; i++ ) if ( i != nn ) {
339 d = p->image[nn].origin() - p->image[i].origin();
340 if ( d.abs() < p->size() ) {
341 if ( d[0] > p->x/2 ) imin[0] = p->x - d[0];
342 if ( d[0] < -p->x/2 ) imax[0] = -d[0];
343 if ( d[1] > p->y/2 ) imin[1] = p->y - d[1];
344 if ( d[1] < -p->y/2 ) imax[1] = -d[1];
345 if ( d[2] > p->z/2 ) imin[2] = p->z - d[2];
346 if ( d[2] < -p->z/2 ) imax[2] = -d[2];
347 }
348 }
349 imin = imin.max(0);
350 imax = imax.min(p->size());
351
352 Vector3<double> wmin(1,1,1);
353 Vector3<double> wmax(1,1,1);
354
355 for ( i=0; i<3; ++i ) {
356 if ( imin[i] ) wmin[i] = 1.0L/imin[i];
357 if ( imax[i] ) wmax[i] = 1.0L/(p->size()[i] - imax[i]);
358 }
359
360 if ( verbose & VERB_FULL ) {
361 cout << "Placing tile:" << endl;
362 cout << "Location: " << p->image[nn].origin() << endl;
363 cout << "Tile size: " << p->size() << endl;
364 cout << "Overlap minima: " << imin << endl;
365 cout << "Overlap maxima: " << imax << endl << endl;
366 }
367
368 zz = p->image[nn].origin()[2];
369 for ( tz=0; tz<p->z; tz++, zz++ ) {
370 w[2] = 1;
371 if ( tz < imin[2] ) w[2] = tz*wmin[2];
372 else if ( tz > imax[2] ) w[2] = (p->z - tz)*wmax[2];
373 yy = p->image[nn].origin()[1];
374 for ( ty=0; ty<p->y; ty++, yy++ ) {
375 w[1] = 1;
376 if ( ty < imin[1] ) w[1] = ty*wmin[1];
377 else if ( ty > imax[1] ) w[1] = (p->y - ty)*wmax[1];
378 xx = p->image[nn].origin()[0];
379 for ( tx=0; tx<p->x; tx++, xx++ ) {
380 w[0] = 1;
381 if ( tx < imin[0] ) w[0] = tx*wmin[0];
382 else if ( tx > imax[0] ) w[0] = (p->x - tx)*wmax[0];
383 i = p->index(0, tx, ty, tz, nn);
384 j = index(0, xx, yy, zz, 0);
385 wvol = w[0]*w[1]*w[2];
386// wvol = 1;
387 for ( cc=0; cc<c; cc++, i++, j++ ) add(j, (*p)[i] * wvol);
388 }
389 }
390 }
391
392 return 0;
393}

405int Bimage::place_central_part(Bimage* p, long nn)
406{
407 long i, j, cc, tx, ty, tz, xx, yy, zz, w;
408 Vector3<long> imin, imax(p->size());
409 Vector3<double> d;
410
411 // Determine overlap in all dimensions
412 for ( i=0; i<p->n; i++ ) if ( i != nn ) {
413 d = p->image[nn].origin() - p->image[i].origin();
414 for ( j=0; j<3; ++j ) {
415 if ( d[j] && fabs(d[j]) <= p->size()[j] ) {
416 if ( d[j] > 0 ) imin[j] = (long)(p->size()[j] - d[j])/2;
417 else imax[j] = (long)(p->size()[j] - d[j])/2;
418 }
419 }
420 }
421
422 // Special cases: The edges of the image
423 for ( i=0; i<3; ++i )
424 if ( p->image[nn].origin()[i] + p->size()[i] >= size()[i] )
425 imax[i] = size()[i] - p->image[nn].origin()[i];
426
427 if ( verbose & VERB_FULL ) {
428 cout << "Placing tile:" << endl;
429 cout << "Location: " << p->image[nn].origin() << endl;
430 cout << "Tile size: " << p->size() << endl;
431 cout << "Overlap minima: " << imin << endl;
432 cout << "Overlap maxima: " << imax << endl << endl;
433 }
434
435// if ( p->image[nn].origin()[0] < 1 ) cout << nn << tab << imin << endl;
436
437 zz = long(p->image[nn].origin()[2] + 0.5);
438 for ( tz=0; tz<p->z; tz++, zz++ ) {
439 w = 1;
440 if ( tz < imin[2] ) w = 0;
441 else if ( tz > imax[2] ) w = 0;
442 if ( w ) {
443 yy = long(p->image[nn].origin()[1] + 0.5);
444 for ( ty=0; ty<p->y; ty++, yy++ ) {
445 w = 1;
446 if ( ty < imin[1] ) w = 0;
447 else if ( ty > imax[1] ) w = 0;
448 if ( w ) {
449 xx = long(p->image[nn].origin()[0] + 0.5);
450 for ( tx=0; tx<p->x; tx++, xx++ ) {
451 w = 1;
452 if ( tx < imin[0] ) w = 0;
453 else if ( tx > imax[0] ) w = 0;
454 if ( w ) {
455 i = p->index(tx, ty, tz, nn);
456 j = index(xx, yy, zz);
457 for ( cc=0; cc<c; cc++, i++, j++ ) set(j, (*p)[i]);
458 }
459 }
460 }
461 }
462 }
463 }
464
465 return 0;

501 }

526double Bimage::linear_fit(Bimage* p, Bimage* pmask, double max_exclude)
527{
528 if ( !check_if_same_size(p) ) {
529 error_show("Bimage::linear_fit", __FILE__, __LINE__);
530 return -1.0;
531 }
532
533 if ( max_exclude >= 1 ) max_exclude /= 100; // Assume it is a percentage
534 if ( max_exclude > 0.5 ) max_exclude = 0.5; // At least half of the data must be fit
535
536 long i, j, nn, nm, nfit, best_excl(0);
537 long imgsize(x*y*z*c), num(imgsize);
538 int bin;
539 double sx, sx2, sy, sxy, sd, dy, denominator, a, b, R, bestR(1e30), best_a(0), best_b(0);
540 double excl_voxels, diff, maxdiff, diff_cutoff;
541
542 int hist[1024];
543 for ( i=0; i<1024; i++ ) hist[i] = 0;
544 vector<int> inc_mask(imgsize,1);
545
546 if ( pmask ) pmask->change_type(UCharacter);
547
548 if ( verbose & VERB_LABEL ) {
549 cout << "Linear least squares fit: image2 = intercept + slope x image1" << endl;
550 cout << "Maximum voxels to exclude: " << 100.0*max_exclude << " %" << endl;
551 }
552
553 if ( verbose & VERB_FULL )
554 cout << " %Set\t Voxels \t%Excluded\tIntercept\tSlope\t\tR" << endl;
555 else if ( verbose & VERB_PROCESS )
556 cout << "Intercept\t Slope\t R-factor\tExcluded\t%" << endl;
557
558 for ( nn=0; nn<n; nn++) {
559 best_excl = 0;
560 best_a = best_b = 0;
561 bestR = 1e30;
562 if ( pmask ) {
563 nm = nn;
564 if ( pmask->images() < n ) nm = 0;
565 for ( num=j=0, i=nm*imgsize; j<imgsize; j++,i++ ) {
566 if ( (*pmask)[i] ) {
567 inc_mask[i] = 1;
568 num++;
569 } else inc_mask[j] = -1;
570 }
571 }
572 for ( excl_voxels = 0; excl_voxels < max_exclude+0.01; excl_voxels += 0.01 ) {
573 nfit = 0;
574 a = b = sx = sx2 = sy = sxy = sd = dy = maxdiff = 0;
575 for ( j=0, i=nn*imgsize; j<imgsize; j++, i++ ) if ( inc_mask[j] > 0 ) {
576 sx += (*this)[i];
577 sy += (*p)[i];
578 sx2 += (*this)[i]*(*this)[i];
579 sxy += (*this)[i]*(*p)[i];
580 nfit++;
581 }
582 denominator = nfit*sx2 - sx*sx;
583 if ( denominator ) {
584 a = (sx2*sy - sx*sxy)/denominator;
585 b = (nfit*sxy - sx*sy) /denominator;
586 }
587 sy /= nfit;
588
589 for ( j=0, i=nn*imgsize; j<imgsize; j++, i++ ) if ( inc_mask[j] >= 0 ) {
590 diff = a + b*(*this)[i] - (*p)[i];
591 if ( inc_mask[j] ) {
592 dy += diff*diff;
593 sd += ((*p)[i]-sy)*((*p)[i]-sy);
594 }
595 diff = fabs(diff);
596 if ( maxdiff < diff ) maxdiff = diff;
597 }
598 R = 1e30;
599 if ( dy < 1e-30 ) R = 0;
600 else if ( sd ) R = sqrt(dy/sd);
601 if ( bestR > R ) {
602 bestR = R;
603 best_a = a;
604 best_b = b;
605 best_excl = num - nfit;
606 }
607
608 if ( verbose & VERB_FULL )
609 cout<< 100*excl_voxels << tab << nfit << tab
610 << 100.0-nfit*100.0/num << tab << a << tab << b << tab << R << endl;
611
612 if ( verbose & VERB_DEBUG )
613 cout << "DEBUG img_linear_fit: maxdiff=" << maxdiff << endl;
614 if ( maxdiff < 1e-30 ) maxdiff = 1;
615 for ( i=0; i<1024; i++ ) hist[i] = 0;
616 for ( j=0, i=nn*imgsize; j<imgsize; j++, i++ ) if ( inc_mask[j] >= 0 ) {
617 diff = fabs(a + b*(*this)[i] - (*p)[i]);
618 bin = (int) (1000*diff/maxdiff);
619 if ( bin > 999 ) bin = 999;
620 hist[bin]++;
621 }
622
623 i = 999;
624 j = hist[i];
625 while ( i > 1 && j < num*excl_voxels ) {
626 i--;
627 j += hist[i];
628 }
629 diff_cutoff = i*maxdiff/1000.0;
630
631 for ( j=0, i=nn*imgsize; j<imgsize; j++, i++ ) if ( inc_mask[j] >= 0 ) {
632 diff = fabs(a + b*(*this)[i] - (*p)[i]);
633 inc_mask[j] = 1;
634 if ( diff > diff_cutoff ) inc_mask[j] = 0;
635 }
636 }
637
638 for ( j=0, i=nn*imgsize; j<imgsize; j++, i++ )
639 set(i, best_a + best_b*(*this)[i] - (*p)[i]);
640
641 image[nn].FOM(bestR);
642
643 if ( verbose & VERB_PROCESS )
644 cout << best_a << tab << best_b << tab << bestR << tab << best_excl << tab << best_excl*100.0/num << " %" << endl;
645 }
646
647 if ( verbose >= VERB_PROCESS ) cout << endl;
648
649 return bestR;
650}

661int Bimage::histomatch(Bimage* p, long bins)
662{
663 if ( !check_if_same_size(p) ) {
664 error_show("Bimage::histomatch", __FILE__, __LINE__);
665 return -1;
666 }
667
668 long i;
669 long h1, h2;
670 double denom, fraction;
671 int* hist1 = new int[bins];
672 int* hist2 = new int[bins];
673 int* map = new int[bins];
674 double* dens = new double[bins];
675
676 for ( i=0; i<bins; i++ )
677 hist1[i] = hist2[i] = 0;
678
679 if ( verbose & VERB_LABEL )
680 cout << "Matching histograms in " << bins << " bins:" << endl << endl;
681
682 for ( i=0; i<datasize; i++ ) {
683 h1 = (long) (bins*((*this)[i] - min)/(max - min));
684 h2 = (long) (bins*((*p)[i] - p->minimum())/(p->maximum() - p->minimum()));
685 if ( h1 >= bins ) h1 = bins - 1;
686 if ( h2 >= bins ) h2 = bins - 1;
687 hist1[h1]++;
688 hist2[h2]++;
689 }
690
691 if ( verbose & VERB_DEBUG )
692 cout << "Histograms done" << endl;
693
694 for ( i=bins-1; i>0; i-- ) {
695 hist1[i-1] += hist1[i];
696 hist2[i-1] += hist2[i];
697 }
698
699 if ( verbose & VERB_DEBUG )
700 cout << "Integration done" << endl;
701
702 for ( h2=0; h2<bins; h2++ ) {
703 h1 = 0;
704 while ( ( h1 < bins ) && ( hist1[h1] > hist2[h2] ) ) h1++;
705 if ( h1 > 0 ) h1--;
706 if ( h2 < bins - 1 )
707 denom = hist1[h1+1] - hist1[h1] - hist2[h2+1] + hist2[h2];
708 else
709 denom = hist1[h1] - hist1[h1-1] - hist2[h2] + hist2[h2-1];
710 if ( denom != 0 )
711 fraction = (hist2[h2] - hist1[h1])/denom;
712 else
713 fraction = 0.5;
714 map[h2] = h1;
715 dens[h2] = (h2 + fraction)*(max - min)/bins + min;
716 }
717
718 if ( verbose & VERB_DEBUG ) {
719 cout << "Mapping calculated" << endl;
720 cout << "Bin\tHist1\tHist2\tMapping\tDensity" << endl;
721 for ( i=0; i<bins; i++ )
722 cout << i << tab << hist1[i] << tab << hist2[i] << tab
723 << map[i] << tab << dens[i] << endl;
724 }
725
726 for ( i=0; i<datasize; i++ ) {
727 h2 = (int) (bins*((*p)[i] - p->minimum())/(p->maximum() - p->minimum()));
728 if ( h2 >= bins ) h2 = bins - 1;
729 set(i, dens[map[h2]]);
730 }
731
732 statistics();
733
734 if ( verbose & VERB_DEBUG )
735 cout << "Histogram matching done" << endl;
736
737 delete[] hist1;
738 delete[] hist2;
739 delete[] map;
740 delete[] dens;
741
742 return 0;
743}
