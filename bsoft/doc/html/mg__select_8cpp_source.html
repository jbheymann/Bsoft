141Bmicrograph* field_find_micrograph_by_focus(Bfield* field, int focus_opt, int iselect)
142{
143 int i; // index
144 int nmgs; // number of micrographs in field
145 int count; // used to test if multiple micrographs have same focus level
146 double focus(0); // designated focus level, depends on focus_opt and index
147 Bmicrograph* mg = NULL; // pointer to micrograph at desired focal level
148
149 nmgs = field_count_micrographs(field);
150
151 float* focal_series = new float[nmgs];
152
153 // Put defocal values into an array, then put in order min to max
154 for ( i = 0, mg = field->mg; mg; mg = mg->next, i++ ) {
155 focal_series[i] = 0;
156 if ( mg->ctf ) focal_series[i] = mg->ctf->defocus_average();
157 else cerr << "Warning in field_find_micrograph_by_focus: CTF structure not allocated!" << endl;
158 }
159
160 qsort((void *) focal_series, nmgs, sizeof(float), (int (*)(const void *, const void *)) QsortSmallToLargeFloat);
161
162 // Select (index+1)th closest-to-focus or farthest-from-focus value
163 if ( focus_opt == 0 ) focus = focal_series[0+iselect];
164 if ( focus_opt == 1 ) focus = focal_series[nmgs-1-iselect];
165 for ( count = 0, i=0; i<nmgs; i++ ) if ( focus == focal_series[i] ) count++;
166
167 delete[] focal_series;
168
169 // Find micrograph with the selected focus level
170 for ( mg = field->mg; mg && mg->ctf && mg->ctf->defocus_average() != focus; mg = mg->next ) ;
171
172 if (verbose & VERB_PROCESS)
173 cout << "Focus of selected micrograph: " << focus << endl;
174
175 if ( (count > 1) || (count == 0) ) {
176 cerr << "Error: There are " << count << " micrographs with focus levels of " << focus << endl;
177 cerr << "Please check data or select with a different criterion" << endl;
178 }
179
180 return mg;
181}

201 double closest_angle(0); // Closest angle to requested angle
202 double mindiff = 1e37; // Minimum difference between requested and available angle
203 Bmicrograph* mg = NULL; // pointer to micrograph with specified rotang
204
205 // Find micrograph with the input rot. angle
206 for ( mindiff = 1e37, mg = field->mg; mg; mg = mg->next ) {
207 if ( fabs(rotang - mg->rot_angle) < mindiff ) {
208 mindiff = fabs(rotang - mg->rot_angle);
209 closest_angle = mg->rot_angle;
210 }
211 }
212
213 for ( count = 0, mg = field->mg; mg; mg = mg->next)
214 if ( fabs(closest_angle - mg->rot_angle) < 1e-10 ) count++;
215
216 for ( mg = field->mg; mg && closest_angle != mg->rot_angle; mg = mg->next ) ;
217
218 if ( (count > 1) || (count == 0) ) {
219 cerr << "Error: There are " << count << " micrographs with the rotational angle of "
220 << closest_angle*180/M_PI << endl;
