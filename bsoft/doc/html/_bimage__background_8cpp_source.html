27int Bimage::calculate_background(long nn, int flag)
28{
29 if ( !data_pointer() ) return -1;
30
31 if ( data_type() == Bit ) return 0;
32
33 long i, j, xx, yy, zz, cc, nm(0);
34 double x2(0), y2(0), z2(0), r;
35 double bkg(0);
36 Vector3<long> h(x/2, y/2, z/2);
37 Vector3<double> ori(image[nn].origin());
38
39 i = (long) (nn*x*y*z);
40 for ( zz=0; zz<z; zz++ ) {
41 if ( z > 1 ) {
42 z2 = (zz - ori[2])/h[2];
43 z2 *= z2;
44 }
45 for ( yy=0; yy<y; yy++ ) {
46 if ( y > 1 ) {
47 y2 = (yy - ori[1])/h[1];
48 y2 *= y2;
49 }
50 for ( xx=0; xx<x; xx++, i++ ) {
51 if ( x > 1 ) {
52 x2 = (xx - ori[0])/h[0];
53 x2 *= x2;
54 }
55 r = x2 + y2 + z2 - 1;
56 if ( flag == 2 ) r = -r;
57 else if ( flag == 3 ) r = -r - 0.5;
58 if ( r > 0 ) {
59 if ( compoundtype == TComplex ) {
60 bkg += complex(i).power();
61 nm++;
62 } else {
63 for ( cc=0, j=i*c; cc<c; cc++, j++ ) {
64 bkg += (*this)[j];
65 nm++;
66 }
67 }
68 }
69 }
70 }
71 }
72
73 if ( nm ) image[nn].background(bkg/nm);
74
75 if ( verbose & VERB_DEBUG )
76 cout << "DEBUG Bimage::calculate_background: img=" << nn << " nm=" << nm <<
77 " bkg=" << image[nn].background() << endl;
