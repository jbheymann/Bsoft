30Vector3<double> model_fit_plane(Bmodel* model)
31{
32 int i, j, n;
33 double offset, d, dstd(0), dmin(1e30);
34 vector<double> b(3);
35 Matrix a(3,3);
36 Vector3<double> cen, loc, normal;
37 Bcomponent* comp;
38
39 for ( i=0; i<3; i++ ) b[i] = 0;
40
41 for ( n=0, comp = model->comp; comp; comp = comp->next, n++ ) {
42 loc = comp->location();
43 cen += loc;
44 for ( i=0; i<3; i++ ) {
45 for ( j=0; j<=i; j++ )
46 a[j][i] += loc[i]*loc[j];
47 b[i] += loc[i];
48 }
49 }
50 cen /= n;
51 for ( i=1; i<3; i++ )
52 for ( j=0; j<i; j++ )
53 a[i][j] = a[j][i];
54
55 offset = 0;
56 if ( a[0][0] == 0 ) {
57 normal = Vector3<double>(1, 0, 0);
58 } else if ( a[1][1] == 0 ) {
59 normal = Vector3<double>(0, 1, 0);
60 } else if ( a[2][2] == 0 ) {
61 normal = Vector3<double>(0, 0, 1);
62 } else {
63 a.LU_decomposition(b);
64 normal = Vector3<double>(b[0], b[1], b[2]);
65 normal.normalize();
66 }
67
68 offset = cen.scalar(normal);
69
70 for ( n=0, comp = model->comp; comp; comp = comp->next, n++ ) {
71 d = comp->location().scalar(normal) - offset;
72 dstd += d*d;
73 if ( fabs(dmin) > fabs(d) ) dmin = d;
74 comp->FOM(d);
75 }
76
77 dstd /= n;
78 model->FOM(dstd);
79
80 if ( dmin < 0 ) normal = -normal;
81 for ( comp = model->comp; comp; comp = comp->next )
82 comp->view(View2<float>(normal));
83
84 return normal;
85}

102double model_distance_to_guide(Vector3<double> loc, Vector3<double> vec, Bmodel* guide, double sigma2)
103{
104 double dp, dl, dpa(0), w, wa(0);
105 Vector3<double> dvec;
106 Bcomponent* comp;
107
108 for ( comp = guide->comp; comp; comp = comp->next ) {
109 dvec = comp->location() - loc;
110 dp = dvec.scalar(vec); // distance to the general guide plane
111 dl = dvec.length2() - dp*dp; // lateral distance to the guide component
112 w = exp(-dl/sigma2);
113 dpa += dp*w;
114 wa += w;
115 }
116
117// cout << dpa << tab << wa << endl;
118
119 if ( wa ) dpa /= wa;
120
121 return dpa;
122}
