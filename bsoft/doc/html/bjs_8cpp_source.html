36JSvalue readJSON(string filename)
37{
38 string ext(extension(filename));
39
40 JSvalue root;
41
42 if ( ext == "json" ) {
43 JSparser parser(filename);
44 root = parser.parse();
45 } else if ( ext == "star" ) {
46 STARparser parser(filename);
47 root = parser.parse();
48 } else if ( ext == "mdoc" ) {
49 MDOCparser parser(filename);
50 root = parser.parse();
51 } else {
52 cerr << "Error: Extension " << ext << " not supported!" << endl;
53 }
