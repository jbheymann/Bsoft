40double Matrix::LU_decomposition()
41{
42 if ( rows() != columns() ) {
43 cerr << "Error: rows:" << rows() << " != columns:" << columns() << endl;
44 return 0;
45 }
46
47 long i, j, k, ibig, m(rows()), n(columns());
48 double big, det(1);
49 vector<long> ind(m,0);
50 vector<double> x(m,0);
51 Matrix a(m,m);
52
53 if ( verbose & VERB_DEBUG )
54 cout << "DEBUG Matrix::LU_decomposition: rows = " << m << endl;
55
56 // Find the largest element in each row for scaling and pivoting
57 // Additionally test for singularity
58 for ( i=0; i<n; i++ ) {
59 ind[i] = i; // Initialize the indices for tracking pivoting
60 big = 0;
61 for ( j=0; j<n; j++ )
62 if ( big < fabs(d[i][j]) ) big = fabs(d[i][j]);
63 if ( big < 1e-37 ) {
64 cerr << "Error: Singular matrix in Matrix::LU_decomposition" << endl;
65// if ( verbose & VERB_DEBUG ) {
66 cerr << *this << endl;
67// bexit(-1);
68// }
69 return -1e30;
70 } else
71 x[i] = 1.0/big;
72 }
73
74 // The LU decomposition
75 for ( j=0; j<n; j++ ) {
76 for ( i=0; i<j; i++ ) { // Calculating the upper triangle
77 for ( k=0; k<i; k++ ) // not the diagonal
78 d[i][j] -= d[i][k]*d[k][j];
79 }
80 big = 0; // Largest element for pivoting
81 ibig = -1;
82 for ( i=j; i<n; i++ ) { // Calculating the lower triangle
83 for ( k=0; k<j; k++ ) // and diagonal (i==j)
84 d[i][j] -= d[i][k]*d[k][j];
85 if ( x[i]*fabs(d[i][j]) >= big ) { // Get the biggest element
86 big = x[i]*fabs(d[i][j]); // for pivoting
87 ibig = i;
88 }
89 }
90 if ( j != ibig ) { // Interchange rows if necessary
91 for ( k=0; k<n; k++ ) swap(d[ibig][k], d[j][k]);
92 swap(x[ibig], x[j]); // Switch scales
93 swap(ind[ibig], ind[j]); // Switch indices
94 det = -det; // Change the sign of the determinant
95 }
96 if ( d[j][j] == 0 ) d[j][j] = 1e-37;
97 for ( i=j+1; i<n; i++ ) d[i][j] /= d[j][j]; // Divide by pivot element
98 }
99
100 // Calculate the determinant
101 for ( i=0; i<n; i++ ) det *= d[i][i];
102
103 // Invert matrix
104 for ( k=0; k<n; k++ ) {
105 for ( i=0; i<n; i++ ) x[i] = 0;
106 x[k] = 1;
107 for ( i=0; i<n; i++ ) // Forward substitution
108 for ( j=0; j<i; j++ ) x[i] -= d[i][j]*x[j];
109 for ( i=m-1; i>=0; i-- ) { // Backward substitution
110 for ( j=i+1; j<n; j++ ) x[i] -= d[i][j]*x[j];
111 x[i] /= d[i][i];
112 a[i][k] = x[i];
113 }
114 }
115
116 // Reorder the matrix back to the original row order
117 for ( i=0; i<n; i++ ) {
118 j = ind[i]; // Find the old row at this position
119 for ( k=0; k<n; k++ ) // Pack it back in order into matrix A
120 d[k][j] = a[k][i];
121 }
122
123 if ( verbose & VERB_DEBUG )
124 cout << "DEBUG Matrix::LU_decomposition: done" << endl << *this << endl;
125
126 return det;
127}

353{
354 double g, h;
355
356 g = d[i][j];
357 h = d[k][l];
358 d[i][j] = g - s*(h + g*tau);
359 d[k][l] = h + s*(g - h*tau);

679{

682
683 for ( auto loc: coor ) {
684 s += loc; // Sums
685 s2 += loc*loc; // Square sums
686 sx[0] += loc[0]*loc[1]; // Cross-term sums
687 sx[1] += loc[0]*loc[2];
688 sx[2] += loc[1]*loc[2];
689 n++;
690 }
691
692 if ( n < 1 ) return pax;
693
694 s /= n;
695 s2 /= n;
696 sx /= n;
697
698 a[0][0] = s2[0] - s[0]*s[0];
