142 fsc_node = xmlNewNode(NULL, BAD_CAST "fsc");
143 xmlAddSibling(node, fsc_node);
144 xmlNewProp(fsc_node, BAD_CAST "title", BAD_CAST "FSC Plot");
145 xmlNewProp(fsc_node, BAD_CAST "xaxis", BAD_CAST FSCX);
146 xmlNewProp(fsc_node, BAD_CAST "yaxis", BAD_CAST FSCY);
147 for ( i=cx*nrows, j=cy*nrows, k=0; k<nrows; i++, j++, k++ ) {
148 coor_node = xmlNewChild(fsc_node, NULL, BAD_CAST "coordinate", NULL);
149 node = xml_set_real(coor_node, "x", (*plot)[i], "%g");
150 node = xml_set_real(coor_node, "y", (*plot)[j], "%g");
151 }
152 xmlSaveFormatFile(filename1.c_str(), doc, 1);
153 xmlFreeDoc(doc);
154 }
