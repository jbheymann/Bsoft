36vector<long> histogram(vector<double> data, long bins, double& scale, double& offset)
37{
38 scale = 1;
39 offset = 0;
40
41 vector<long> histo(bins, 0);
42
43 if ( data.size() < 2 ) return histo;
44
45 long i;
46 double max(data[0]), min(data[0]);
47
48 for ( auto v: data ) {
49// cout << v << endl;
50 if ( min > v ) min = v;
51 if ( max < v ) max = v;
52 }
53
54 scale = (bins - 1)*1.0/(max - min);
55 offset = 0.5 - scale*min;
56
57 if ( verbose & VERB_DEBUG )
58 cout << "DEBUG histogram: range=" << min << "-"
59 << max << " bins=" << bins << " scale=" << scale
60 << " offset=" << offset << endl;
61
62 for ( auto v: data ) {
63 i = (long) (scale*v + offset);
64 if ( i >= 0 && i < bins ) histo[i]++;
65 }
66
67// for ( i=0; i<bins*c; i++ ) cout << i << tab << histo[i] << endl;
68
69 return histo;

184vector<double> histogram_thresholds(vector<long> h, long number)
185{
186 long i, j, bins(h.size());
187
188 // Set up matrices
189 Matrix P(bins,bins);
190 Matrix S(bins,bins);
191 Matrix H(bins,bins);
192
193 // Diagonals
194 for ( i=0; i<bins; ++i ) {
195 P[i][i] = h[i];
196 S[i][i] = h[i] * i;
197 }
198
199 // Second rows
200 for ( i=1, j=2; i<bins-1; ++i, ++j ) {
201 P[1][j] = P[1][i] + h[j];
202 S[1][j] = S[1][i] + j*h[j];
203 }
204
205 // Propagate sums
206 for ( i=2; i<bins; ++i ) {
207 for ( j=i+1; j<bins; ++j ) {
208 P[i][j] = P[1][j] - P[1][i-1];
209 S[i][j] = S[1][j] - S[1][i-1];
210 }
211 }
212
213 // H
214 for ( i=1; i<bins; ++i )
215 for ( j=i+1; j<bins; ++j )
216 if ( P[i][j] ) H[i][j] = S[i][j]*S[i][j]/P[i][j];
217
218 long depth(number-1);
219 double maxsum(0);
220 vector<double> t(depth,0);
221
222 find_maximum_sigma(H, 0, 0, depth, 0, maxsum, t);
223
224 if ( verbose )
225 cout << "Best FOM: " << maxsum << endl << endl;
226
227 return t;
228}

231double histogram_gaussian_R(Bsimplex& simp)
232{
233 long i, j, ng(simp.constant(0));
234 double v, df, R(0);
235 vector<double> amp(ng), pos(ng), invsig(ng);
236 for ( i=j=0; i<ng; ++i ) {
237 amp[i] = simp.parameter(j++);
238 pos[i] = simp.parameter(j++);
239 invsig[i] = 1/simp.parameter(j++);
240 }
241 vector<double>& f = simp.dependent_values();
242 vector<double>& x = simp.independent_values();
243
244 for ( i=0; i<simp.points(); i++ ) {
245 for ( j=0, df=0; j<ng; j++ ) {
246 v = (x[i] - pos[j])*invsig[j];
247 df += amp[j]*exp(-0.5*v*v);
248 }
249 df -= f[i];
250 R += df*df;
251 }
252
253 R = sqrt(R/i);
254
255 return R;
256}

270vector<double> plot_histogram_fit_gaussian(Bplot* plot, long ngauss)
271{
272 long i, j, k, l;
273 long bins(plot->rows()), hsum(0);
274 double hmax(0);
275 vector<double> mh(bins);
276 vector<double> xx(bins), fx(bins);
277 for ( i=0, j=bins; i<bins; ++i, ++j ) {
278 xx[i] = (*plot)[i];
279 fx[i] = (*plot)[j];
280 hsum += fx[i];
281 if ( hmax < fx[i] ) hmax = fx[i];
282 mh[i] = fx[i];
283 }
284
285 // Attempting to find peaks based on the gradient change
286 long w(bins/20);
287 if ( w < 5 ) w = 5;
288 mh = moving_gradient(mh, w);
289
290 vector<double> gmax(ngauss);
291 double ex;
292 for ( i=j=k=l=0, ex=0; i<bins; i++ ) {
293 if ( ex < mh[i] ) {
294 ex = mh[i];
295 j = i;
296 } else if ( mh[i] < 0 && mh[j] > 10 ) {
297 gmax[k++] = i;
298 ex = 0;
299 j = i;
300 }
301 }
302
303 // 3 parameters: amplitude, position, sigma
304 Bsimplex simp(1, 3*ngauss, 1, bins, xx, fx);
305 simp.constant(0, ngauss);
306
307 for ( i=j=0; j<ngauss; j++, i++ ) {
308 simp.parameter(i, hmax); // Amplitude
309 simp.limits(i, hsum/bins, 5*hmax);
310// cout << tab << simp.parameter(i);
311 i++;
312 simp.parameter(i, gmax[j]); // Position
313 if ( ngauss == 1 ) {
314 simp.limits(i, 0, bins-1);
315 } else if ( j == 0 ) {
316 simp.limits(i, 0, (gmax[0]+gmax[1])/2);
317 } else if ( j < ngauss-1 ) {
318 simp.limits(i, (gmax[j-1]+gmax[j])/2, (gmax[j]+gmax[j+1])/2);
319 } else {
320 simp.limits(i, (gmax[0]+gmax[1])/2, bins-1);
321 }
322
323 i++;
324 simp.parameter(i, bins*0.1/ngauss); // Sigma
325 simp.limits(i, 3, bins*0.5);
326// cout << tab << simp.parameter(i) << endl;
327 }
328
329 double R = simp.run(5000, 1e-6, histogram_gaussian_R);
330 R = bins*R/hsum;
331
332 vector<double> gauss;
333 for ( j=0; j<3*ngauss; j++ ) gauss.push_back(simp.parameter(j));
334 gauss.push_back(R);
335
336 if ( verbose & VERB_PROCESS ) {
337 cout << "Gaussian fit over " << bins << " bins:" << endl;
338 cout << "Amp\tOffset\tSigma\tSIP" << endl;
339 for ( j=0; j<ngauss; j++ )
340 cout << gauss[3*j] << tab << gauss[3*j+1] << tab << gauss[3*j+2] << tab << bins*gauss[3*j]/hsum << endl;
341 cout << "Sampling-independent RMSD: " << R << endl << endl;
342 }
343
344 return gauss;
345}
