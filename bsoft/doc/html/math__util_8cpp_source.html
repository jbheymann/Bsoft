228long smallest_prime(long number)
229{
230 if ( number < 2 ) {
231 cout << "Zero and one are neither composite nor prime numbers!" << endl;
232 return 1;
233 }
234
235 if ( number%2 == 0 ) return 2;
236 if ( number%3 == 0 ) return 3;
237
238 /* Primes generated by alternatingly adding 2 or 4 */
239 long i, inc(2);
240 double max = sqrt((double)number);
241 for( i=5; i<=max; i+=inc, inc=6-inc)
242 if ( number%i == 0 ) return i;
243
244 return number;

324double betacf(double a, double b, double x)
325{
326 int m,m2;
327 double aa, c, d, del, h, qab, qam, qap;
328 double fpmin(1.0e-30); //Number near the smallest representative
329 //floating-point number
330 double eps(3.0e-7); //Relative accuracy
331 int maxit(100); //Maximum allowed number of iterations
332
333 qab=a+b; // These q's will be used in factors that occur
334 qap=a+1.0; // in the coefficients (6.4.6).
335 qam=a-1.0;
336 c=1.0;
337 d=1.0-qab*x/qap;
338 if (fabs(d) < fpmin) d=fpmin;
339 d=1.0/d;
340 h=d;
341 for (m=1; m<=maxit;m++){
342 m2=2*m;
343 aa=m*(b-m)*x/((qam+m2)*(a+m2));
344 d=1.0+aa*d;
345 if (fabs(d) < fpmin) d=fpmin;
346 c=1.0+aa/c;
347 if (fabs(c) < fpmin) c=fpmin;
348 d=1.0/d;
349 h *= d*c;
350 aa = -(a+m)*(qab+m)*x/((a+m2)*(qap+m2));
351 d=1.0+aa*d; // Next step of the recurrence (the odd one).
352 if (fabs(d) < fpmin) d=fpmin;
353 c=1.0+aa/c;
354 if (fabs(c) < fpmin) c=fpmin;
355 d=1.0/d;
356 del=d*c;
357 h *= del;
358 if (fabs(del-1.0) < eps) break; // Are we done?
359 }
360 if (m > maxit) cerr << "Error: a or b too big, or maxit too small in betacf" << endl;
361 return h;
362}

376double betai (double a, double b, double x)
377{
378 double bt;
379
380 if(x < 0.0 || x > 1.0) cerr << "Error: Bad x in routine betai" << endl;
381 if(x == 0.0 || x == 1.0) bt=0.0;
382 else //Factors in front of the continued fraction.
383 bt=exp(lgamma(a+b)-lgamma(a)-lgamma(b)+a*log(x)+b*log(1.0-x));
384 if(x < (a+1.0)/(a+b+2.0)) //Use continued fraction directly.
385 return bt*betacf(a,b,x)/a;
386 else //Use continued fraction after making the sym
387 return 1.0-bt*betacf(b,a,1.0-x)/b; //metry transformation
388}

401double kmeans_threshold(long n, double* v)
402{
403 long i, nfg, nbg;
404 double t, vfg(v[0]), vbg(v[1]), dvfg(1), dvbg(1), dt(0.001), sumfg, sumbg;
405
406 for ( i=0; i<n && vfg == vbg; i++ ); // Ensure at least one value on each class
407
408 while ( fabs(dvfg) > dt || fabs(dvbg) > dt ) { // Exit on no more change
409 nfg = nbg = 0;
410 sumfg = sumbg = 0;
411 t = (vbg + vfg)/2;
412 for ( i=0; i<n; i++ ) { // Sum into two classes
413 if ( v[i] > t ) {
414 sumfg += v[i];
415 nfg++;
416 } else {
417 sumbg += v[i];
418 nbg++;
419 }
420 }
421 if ( nfg ) sumfg /= nfg;
422 if ( nbg ) sumbg /= nbg;
423 dvfg = sumfg - vfg;
424 dvbg = sumbg - vbg;
425 vfg = sumfg;
426 vbg = sumbg;
427 }
