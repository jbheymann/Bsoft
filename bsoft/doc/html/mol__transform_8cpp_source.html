450int molgroup_resolve_pbc(Bmolgroup* molgroup)
451{
452 Bmolecule* mol;
453 Bresidue* res;
454 Batom* atom;
455 double summass;
456 Vector3<double> csum, shift;
457 Vector3<double> box = molgroup->box;
458
459 if ( verbose & VERB_PROCESS ) {
460 cout << "Resolving periodic box coordinates:" << endl;
461 cout << "Box: " << molgroup->box << endl << endl;
462 }
463
464 for ( mol = molgroup->mol; mol; mol = mol->next ) {
465 csum = 0;
466 summass = 0;
467 for( res = mol->res; res; res = res->next ) {
468 for ( atom = res->atom; atom; atom = atom->next ) {
469 if ( summass == 0 ) {
470 csum = atom->coord * atom->mass;
471 } else {
472 shift = csum / summass;
473 shift += vector3_difference_PBC(atom->coord, shift, box);
474 csum += shift * atom->mass;
475 }
476 summass += atom->mass;
477 }
478 }
479 if ( summass ) csum /= summass;
480 if ( csum[0] < 0 ) csum[0] += box[0];
481 if ( csum[1] < 0 ) csum[1] += box[1];
482 if ( csum[2] < 0 ) csum[2] += box[2];
483 if ( csum[0] > molgroup->box[0] ) csum[0] -= box[0];
484 if ( csum[1] > molgroup->box[1] ) csum[1] -= box[1];
485 if ( csum[2] > molgroup->box[2] ) csum[2] -= box[2];
486 if ( verbose & VERB_FULL )
487 cout << "COM for molecule " << mol->id << ", mass " << summass << ": " << csum << endl;
488 for( res = mol->res; res; res = res->next ) {
489 for ( atom = res->atom; atom; atom = atom->next ) {
490 shift = atom->coord - csum;
491 if ( 2*shift[0] > box[0] ) shift[0] = -box[0];
492 else if ( 2*shift[0] < -box[0] ) shift[0] = box[0];
493 else shift[0] = 0;
494 if ( 2*shift[1] > box[1] ) shift[1] = -box[1];
495 else if ( 2*shift[1] < -box[1] ) shift[1] = box[1];
496 else shift[1] = 0;
497 if ( 2*shift[2] > box[2] ) shift[2] = -box[2];
498 else if ( 2*shift[2] < -box[2] ) shift[2] = box[2];
499 else shift[2] = 0;
500 atom->coord = atom->coord + shift;
501 }
502 }
503 }
504
505 return 0;
506}
