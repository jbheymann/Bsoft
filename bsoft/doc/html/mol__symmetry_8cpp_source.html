345static int QsortMassCOM(const void *x, const void *y)
346{
347 MassCOM* c1 = (MassCOM *) x;
348 MassCOM* c2 = (MassCOM *) y;
349
350 if ( fabs((c1->mass - c2->mass)/(c1->mass + c2->mass)) > 0.1 ) {
351 if ( c1->mass < c2->mass ) return 1;
352 else return -1;
353 } else {
354 if ( c1->com.length() < c2->com.length() ) return 1;
355 else return -1; // Sort from high to low
356 }

375int molgroup_find_standard_view(Bmolgroup* molgroup, Bsymmetry& sym, View2<double> ref_view)
376{
377 molgroup_shift_to_center_of_mass(molgroup);
378
379 int g, i, j, k, nmol, nunits(sym.order()), nmassif;
380 Bmolecule* mol;
381 Bresidue* res;
382 Batom* atom;
383
384 for ( nmol = 0, mol = molgroup->mol; mol; mol = mol->next ) nmol++;
385
386 if ( nmol < nunits ) {
387 cerr << "Error: Not enough molecules (" << nmol << ") to fit into " << nunits << " asymmetric units!" << endl << endl;
388 return -1;
389 }
390
391 // Get all chains' centers of mass and weight
392 double themass, themass2, maxmass(0);
393 MassCOM* c = new MassCOM[nmol];
394 for ( i=0; i<nmol; i++ ) c[i].com = c[i].mass = 0;
395
396 nmassif = 0;
397 for ( i=0, mol = molgroup->mol; mol; mol = mol->next, i++ ) {
398 for( res = mol->res; res; res = res->next ) {
399 for ( atom = res->atom; atom; atom = atom->next ) {
400 c[i].com += atom->coord * atom->mass;
401 c[i].mass += atom->mass;
402 }
403 }
404 if ( c[i].mass ) {
405 c[i].com /= c[i].mass;
406 nmassif++;
407 }
408 if ( maxmass < c[i].mass ) maxmass = c[i].mass;
409 }
410
411 // Sort the chains based on weight and distance from COM
412 qsort((void *) c, nmol, sizeof(MassCOM), (int (*)(const void *, const void *)) QsortMassCOM);
413
414 if ( verbose & VERB_FULL ) {
415 cout << endl << "Ordered list of centers-of-mass:" << endl;
416 cout << " x\t y\t z\tMass (Da)" << endl;
417 for ( i=0; i<nmol; i++ )
418 cout << c[i].com[0] << tab << c[i].com[1] << tab << c[i].com[2] << tab << c[i].mass << endl;
419 cout << endl << endl;
420 }
421
422 // Determine the number of groups of chains, each group likely symmetry-related
423 // A group is defined as a set of chains (number = symmetry order),
424 // that differ little in weight (stdev < 5%)
425 int ngroup(0);
426 Vector3<double> com, translate;
427
428 if ( verbose & VERB_FULL )
429 cout << endl << "Group\tMass\tStDev" << endl;
430 for ( i=0; i<nmol-nunits+1; i+=nunits ) { // Loop over all groups of size defined by symmetry
431 themass = themass2 = 0;
432 com = 0;
433 for ( j=i; j<i+nunits; j++ ) {
434 themass += c[j].mass;
435 themass2 += c[j].mass*c[j].mass;
436 com += c[j].com; // Center of group on symmetry axis
437 }
438 themass /= nunits;
439 themass2 = themass2/nunits - themass*themass;
440 if ( themass2 > 0 ) themass2 = sqrt(themass2);
441 else themass2 = 0;
442 if ( themass > 100 && themass2/themass < 0.05 ) {
443 ngroup++;
444 com *= -1.0/nunits; // Translation to symmetry axis
445 translate += com;
446 }
447 if ( verbose & VERB_FULL )
448 cout << ngroup << tab << themass << tab << themass2 << endl;
449 }
450 translate /= ngroup;
451
452 // Subtract the global center-of-mass from the chain COM's
453 com = 0;
454 for ( i=0; i<nmassif; i++ ) com += c[i].com;
455 com /= nmassif;
456 for ( i=0; i<nmassif; i++ ) c[i].com -= com;
457
458 // Generate pairs of vectors and calculate normals
459 int n(0), nvec = (ngroup*nunits*(nunits-1)*(nunits-2))/6;
460 double a;
461 Vector3<double> axis, v1, v2, d;
462 Vector3<double>* v = new Vector3<double>[nvec];
463
464 if ( verbose & VERB_FULL )
465 cout << endl << "Group\tv1\tv2\tv3\tx\ty\tz" << endl;
466 for ( g=0; g<ngroup; g++ ) {
467 // Calculate normal vectors
468 for ( i=2+nunits*g; i<nunits*(g+1); i++ ) {
469 for ( j=1+nunits*g; j<i; j++ ) {
470 v1 = c[i].com - c[j].com;
471 for ( k=nunits*g; k<j; k++ ) {
472 v2 = c[i].com - c[k].com;
473// v[n] = v1.cross(v2) * c[i].mass;
474 v[n] = v1.cross(v2);
475 v[n].normalize();
476 if ( verbose & VERB_FULL )
477 cout << g << tab << k << tab << j << tab << i << tab
478 << v[n][0] << tab << v[n][1] << tab << v[n][2] << endl;
479 n++;
480 }
481 }
482 }
483 }
484
485 if ( verbose & VERB_DEBUG )
486 cout << "DEBUG molgroup_find_standard_view: nvec=" << nvec << " n=" << n << endl;
487
488 // Find a common normal when comparing pairs of vectors and assign it as the major axis
489 int vbest = -1, m = 100;
490 int nx = 2*m, ny = 2*m, nz = 2;
491 int sv = nx * ny * nz;
492 int* nv = new int[sv];
493 for ( i=0; i<sv; i++ ) nv[i] = 0;
494
495 for ( i=1; i<n; i++ ) {
496 for ( j=0; j<i; j++ ) {
497 v1 = v[i];
498 v1.normalize();
499 a = fabs(v[i].angle(v[j]));
500 if ( a > M_PI_2 ) {
501 a = M_PI - a;
502 v1 = -v1;
503 }
504 v1 *= m;
505 if ( a < 0.05 ) {
506 k = (int) (nx*(v1[1]+m)) + (int) (v1[0]+m);
507 if ( k > nx*ny ) k = 0;
508 if ( v1[2] > 0 ) k += nx*ny;
509 nv[k]++;
510 if ( vbest < nv[k] ) {
511 vbest = nv[k];
512 axis = v1;
513 }
514 }
515 }
516 }
517 delete[] nv;
518
519 for ( i=0; i<n; i++ ) {
520 a = fabs(axis.angle(v[i]));
521 if ( a > M_PI_2 ) {
522 a = M_PI - a;
523 v[i] = -v[i];
524 }
525 if ( a < 0.05 ) axis += v[i];
526 }
527
528 axis.normalize();
529
530 // Find the 2-fold on the x-axis as the sum vector of two COM's
531 // that has a difference vector that is most parallel with the major axis
532 double ang(0), ang_min = 1000;
533 if ( sym.point() > 200 ) {
534 for ( i=1; i<nunits; i++ ) {
535 for ( j=0; j<i; j++ ) {
536 d = c[i].com - c[j].com;
537 ang = axis.angle(d);
538 if ( fabs(ang_min) > fabs(ang) ) {
539 ang_min = ang;
540 v2 = c[i].com + c[j].com;
541 v2.normalize();
542 }
543 }
544 }
545 }
546
547 delete[] c;
548 delete[] v;
549
550 if ( verbose & VERB_DEBUG ) {
551 cout << "Angle with major axis: " << ang_min*180/M_PI << endl;
552 cout << "Dihedral axis: " << v2 << endl;
553 cout << "Angle between axes: " << (axis.angle(v2))*180/M_PI << endl;
554 }
555
556 if ( verbose & VERB_LABEL ) {
557 cout << "Translate: " << translate << endl;
558 cout << "Major symmetry axis: " << axis << endl;
559 if ( sym.point() > 200 )
560 cout << "Minor symmetry axis: " << v2 << endl;
561 cout << endl;
562 }
563
564 // The first matrix rotates it to the major symmetry axis
565 // The second does the in-plane rotation
566 View2<double> view(-axis[0], -axis[1], axis[2], 0);
567 Matrix3 mat = view.matrix();
568 if ( sym.point() > 200 ) {
569 v1 = 0;
570 v1[0] = 1;
571 v2 = mat * v2;
572 ang = fmod(v1.angle(v2), M_PI*2.0/sym[0].order());
573 view = View2<double>(0, 0, 1, ang);
574 mat = view.matrix() * mat;
575 }
576
577 if ( verbose & VERB_DEBUG ) {
578 cout << "Dihedral axis: " << v2 << endl;
579 cout << "Angle: " << ang*180.0/M_PI << endl;
580 }
581
582 for ( mol = molgroup->mol; mol; mol = mol->next ) {
583 for( res = mol->res; res; res = res->next ) {
584 for ( atom = res->atom; atom; atom = atom->next ) {
585 atom->coord += translate;
586 atom->coord = mat * atom->coord;
587 }
588 }
589 }
590
591 return 0;
592}

616int molgroup_orient_to_standard_view(Bmolgroup* molgroup, Bsymmetry& sym,
617 View2<double> ref_view, Bresidue_matrix* simat)
618{
619 random_seed();
620
621 if ( sym.point() < 102 ) return 0;
622
623 long i, j, k, m, nmol, nres_cut, cut_max(30);
624 long nunit(sym.order()), ngroup, nid, n2=0, nn=0, no;
625 Vector3<double> a2[1000], an[1000];
626 Bmolecule* mol, *mol2, *symmol;
627 Bresidue* res;
628 Batom* atom;
629
630 for ( i=m=0; i<sym.operations(); i++ )
631 if ( m < sym[i].order() ) m = sym[i].order();
632
633 double a, refang = TWOPI/m;
634
635 for ( nmol = 0, mol = molgroup->mol; mol; mol = mol->next ) nmol++;
636 ngroup = nmol/nunit;
637
638 if ( ngroup < 1 ) {
639 cerr << "Error: Not enough molecules (" << nmol << ") to fit into " << nunit << " asymmetric units!" << endl << endl;
640 return -1;
641 }
642
643 molgroup_shift_to_center_of_mass(molgroup);
644
645 if ( verbose & VERB_PROCESS ) {
646 cout << "Orienting to the standard view for symmetry " << sym.label() << endl;
647 cout << "Number of molecules: " << nmol << endl;
648 cout << "Number of groups: " << ngroup << endl;
649 }
650
651 int* g = new int[nmol]; // Group indices
652 for ( i=0; i<nmol; i++ ) g[i] = 0;
653
654 Transform t;
655 int offset, notfound;
656
657 // Calculate a matrix of rotations between pairs of molecules
658 for ( k=0, i=1; k<nmol; k++, i++, i=(i>ngroup)?1:i ) g[k] = i;
659 for ( i=0, mol = molgroup->mol; mol->next; mol = mol->next, i++ ) {
660 nres_cut = mol->nres/2;
661 if ( nres_cut > cut_max ) for ( j=i+1, mol2 = mol->next; mol2; mol2 = mol2->next, j++ ) {
662 // if two sequences can be aligned, find the transformation to superimpose them
663 offset = seq_find_best_offset(mol, mol2, nid, simat);
664 if ( nid > cut_max || nid >= nres_cut ) {
665 g[j] = g[i];
666 t = mol_find_transformation(mol, mol2, offset);
667 symmol = molecule_copy(mol);
668 mol_coor_rotate(symmol, t);
669 t.fom = mol_calculate_rmsd(mol2, symmol);
670 molecule_kill(symmol);
671 angle_set_negPI_to_PI(t.angle);
672 a = t.angle/refang;
673 no = (long) (a + 0.5);
674 if ( fabs(t.angle - M_PI) < 0.1 ) {
675 for ( k=0, notfound=1; k<n2 && notfound; k++ ) {
676 if ( a2[k].angle(t.axis) < 0.1 ) {
677 a2[k] += t.axis;
678 notfound = 0;
679 }
680 }
681 if ( notfound ) {
682 a2[n2] = t.axis;
683 n2++;
684 }
685 }
686 if ( fabs(a - no) < 0.03 ) {
687 for ( k=0, notfound=1; k<nn && notfound; k++ ) {
688 if ( an[k].angle(t.axis) < 0.1 ) {
689 an[k] += t.axis;
690 notfound = 0;
691 }
692 }
693 if ( notfound ) {
694 an[nn] = t.axis;
695 nn++;
696 }
697 }
698 m++;
699 }
700 }
701 }
702
703 if ( verbose & VERB_PROCESS ) {
704 cout << "Num\tMol\tGroup\tLength" << endl;
705 for ( i=0, mol = molgroup->mol; mol; mol = mol->next, i++ )
706 cout << i+1 << tab << mol->id << tab << g[i] << tab << mol->nres << endl;

853 for ( k=0; k<nmol; k++ ) {

855 for ( mind=1e30, l=0; l<nmol; l++ ) {

860 n++;
861 }

868 cout << "Symmetry RMSD: " << R << " A (" << n << ")" << endl << endl;
