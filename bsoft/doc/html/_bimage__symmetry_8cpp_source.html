192Vector3<double> vector3_closest_to_intersection(Vector3<double> origin1,
193 Vector3<double> origin2, Vector3<double> axis1, Vector3<double> axis2)
194{
195 Vector3<double> a1 = axis1;
196 Vector3<double> a2 = axis2;
197 double a = a1.scalar(a1);
198 double b = a1.scalar(a2);
199 double c = a2.scalar(a2);
200 double d = a1.scalar(origin1 - origin2);
201 double e = a2.scalar(origin1 - origin2);
202
203 double s, t, denom = a*c - b*b;
204
205 if ( denom < 1e-30 ) { // Parallel lines
206 s = 0;
207 t = d/b;
208 } else {
209 s = (b*e - c*d)/denom;
210 t = (a*e - b*d)/denom;
211 }
212
213 if ( verbose & VERB_FULL )
214 cout << "Intersection distance: " << (origin1 + a1 * s).distance(origin2 + a2 * t) << endl;
215
216 Vector3<double> point = (origin1 + a1 * s + origin2 + a2 * t)/2;
217
218 return point;
219}

331 }
332

460 cout << "#\tax\tay\taz\tCC" << endl;

462 mat = Matrix3(bestaxis, sym[op0].axis());

465 amin = 0;
466 amax = M_PI;
467
468 // Count the number of points in the grid search

584int Bimage::change_symmetry(Bsymmetry& symold, Bsymmetry& symnu,
585 double radius, double z_slope)
586{
587 long i, xx, yy, zz, nn;
588 double dx, dy, dz, dxo, dyo, dzo, da, r, ro(1), ca, sa, s;
589 double rr(symold[0].order()*1.0L/symnu[0].order());
590 double af(1/rr - 1);
591
592 float* nudata = new float[datasize];
593
594 if ( verbose ) {
595 cout << "Changing the symmetry from " << symold.label() << " to " << symnu.label() << ":" << endl;
596 cout << "Radius: " << radius << endl;
597 cout << "Z-slope: " << z_slope << endl << endl;
598 }
599
600 for ( i=nn=0; nn<n; nn++ ) {
601 for ( zz=0; zz<z; zz++ ) {
602 dz = zz - image[nn].origin()[2];
603 s = -(radius + dz*z_slope)*af;
604 dzo = zz;
605 if ( dzo >= z - 1 ) dzo = z - 1.001;
606 for ( yy=0; yy<y; yy++ ) {
607 dy = yy - image[nn].origin()[1];
608 for ( xx=0; xx<x; xx++, i++ ) {
609 nudata[i] = 0;
610 dx = xx - image[nn].origin()[0];
611 da = af*atan2(dy, dx);
612 ca = cos(da);
613 sa = sin(da);
614 if ( radius > 0 ) {
615 r = sqrt(dx*dx + dy*dy);
616 if ( r > 0 ) ro = (1 + s/r);
617 else ro = 1;
618 dxo = ro*(dx*ca - dy*sa) + image[nn].origin()[0];
619 dyo = ro*(dx*sa + dy*ca) + image[nn].origin()[1];
620 } else {
621 dxo = rr*(dx*ca - dy*sa) + image[nn].origin()[0];
622 dyo = rr*(dx*sa + dy*ca) + image[nn].origin()[1];
623 }
624 if ( ro>0 && dxo>=0 && dxo<=x-1 && dyo>=0 && dyo<=y-1 )
625 nudata[i] = interpolate(dxo, dyo, dzo, nn, background(nn));
626 }
627 }
628 }
629 }
630
631 data_type(Float);
632
633 data_assign((unsigned char *) nudata);
634
635 statistics();
636
637 return 0;
638}

655Matrix3 Bimage::symmetry_equivalent(Bimage* pref, Bimage* pmask, Bsymmetry& sym)
656{
657 Matrix3 mat(1);
658 if ( sym.point() < 102 ) return mat;
659 if ( sym.point() > 600 ) return mat;
660 if ( !pref ) return mat;
661
662 if ( image->origin()[0] < 1 || image->origin()[1] < 1 || image->origin()[2] < 1 ) origin(size()/2);
663
664 if ( sym.point() < 200 )
665 return symmetry_equivalent_cyclic(pref, pmask, sym);
666
667 long nstep(1), step;
668 double dang(0), angle(0);
669
670 if ( sym.point() < 200 ) { // Cyclic
671 dang = M_PI/180.0; // 1 degree step size
672 nstep = 360/sym[0].order();
673 } else if ( sym.point() == 202 ) { // D2
674 nstep = 6;
675 } else if ( sym.point() < 300 ) { // Dihedral
676 dang = M_PI*1.0L/sym[0].order();
677 if ( sym[0].order()%2 == 0 ) nstep = 2;
678 else nstep = 4;
679 } else if ( sym.point() < 400 ) { // Tetrahedral
680 dang = M_PI_2;
681 nstep = 2;
682 } else if ( sym.point() < 500 ) { // Octahedral
683 dang = M_PI_2;
684 nstep = 1;
685 } else { // Icosahedral
686 dang = M_PI_2;
687 nstep = 4;
688 }
689
690 double scale(sampling(0)[0]/pref->sampling(0)[0]);
691 Vector3<double> axis(0,0,1), best_axis(0,0,1);
692 Vector3<double> vscale(scale, scale, scale);
693 Vector3<double> ori(image->origin()), translate((pref->size() - size())/2);
694 Bimage* pt;
695
696 double ccbest(-1);
697 double cc, best_angle(0);
698 Quaternion q(axis, M_PI_2), q2;
699
700 if ( verbose ) {
701 cout << "Finding the symmetry equivalent for " << sym.label() << endl;
702 cout << "Axis\t\t\tAngle\tCC" << setprecision(4) << endl;
703 }
704
705 for ( step = 0; step < nstep; ++step ) {
706 if ( sym.point() == 202 ) { // D2
707 switch ( step ) {
708 case 1: angle = M_PI_2; axis = Vector3<double>(0,0,1); break;
709 case 2: angle = M_PI_2; axis = Vector3<double>(0,1,0); break;
710 case 3: angle = M_PI_2; axis = Vector3<double>(1,0,0); break;
711 case 4: angle = M_PI/1.5L; axis = Vector3<double>(1,1,1); break;
712 case 5: angle = -M_PI/1.5L; axis = Vector3<double>(1,1,1); break;
713 default: angle = 0;
714 }
715 axis.normalize();
716 } else if ( sym.point() < 300 ) { // Dihedral
717 switch ( step ) {
718 case 1: angle = dang; break;
719 case 2: angle = M_PI_2; break;
720 case 3: angle = M_PI_2 - dang; break;
721 default: angle = 0;
722 }
723 axis.normalize();
724 } else if ( sym.point() == 532 ) { // Icosahedral
725 switch ( step ) {
726 case 1: angle = dang; break;
727 case 2: angle = atan(1/GOLDEN); axis = Vector3<double>(1,0,0); break;
728 case 3: q2 = Quaternion(axis, angle); q = q2*q;
729 axis = q.axis(); angle = q.angle(); break;
730 default: angle = 0;
731 }
732 axis.normalize();
733 } else { // Cyclic, tetra-, octahedral
734 angle = step*dang;
735 }
736 mat = Matrix3(axis, angle);
737 pt = transform(pref->size(), vscale, ori, translate, mat, FILL_BACKGROUND);
738 cc = pt->correlate(pref, 0, pref->sizeX()/2, pmask);
739 if ( ccbest < cc ) {
740 ccbest = cc;
741 best_angle = angle;
742 best_axis = axis;
743// if ( verbose & VERB_PROCESS )
744// cout << "Better CC: " << cc << " (" << angle*180.0/M_PI << ")" << endl;
745 }
746 delete pt;
747 if ( verbose )
748 cout << axis << tab << angle*180.0/M_PI << tab << cc << endl;
749 }
750
751 if ( verbose ) {
752 cout << "Best axis and angle: " << best_axis << tab << best_angle*180.0/M_PI << endl;
753 cout << "Best CC: " << ccbest << endl;
754 cout << "Fixing symmetry " << sym.label() << " orientation" << endl << endl;
755 }
756
757 mat = Matrix3(best_axis, best_angle);
758
759 return mat;
760}

777Matrix3 Bimage::symmetry_equivalent_cyclic(Bimage* pref, Bimage* pmask, Bsymmetry& sym)
778{
779 Matrix3 mat(1);
780 if ( sym.point() < 102 ) return mat;
781 if ( sym.point() > 200 ) return mat; // Not suitable for groups other than cyclic
782 if ( !pref ) return mat;

784 if ( image->origin()[0] < 1 || image->origin()[1] < 1 || image->origin()[2] < 1 ) origin(size()/2);

786 double hires(2*pref->sampling(0)[0]);

789 double accuracy(M_PI*0.1/180.0); // Angular accuracy ~ 0.1 degrees
790 if ( accuracy < 3.0/pref->sizeX() ) accuracy = 3.0/pref->sizeX(); // or 2/3 Nyquest
791
792 double dang(hires/pref->real_size()[0]); // Initial angular step size
793 double angle, angmin(0), angmax(TWOPI/sym[0].order()+0.01);
794 double scale(sampling(0)[0]/pref->sampling(0)[0]);

797 Vector3<double> ori(image->origin()), translate((pref->size() - size())/2);
798

801
802 if ( verbose )
803 cout << "Finding the symmetry equivalent for " << sym.label() << setprecision(4) << endl;
804
805 Bimage* pt = transform(pref->size(), vscale, ori, translate, mat, FILL_BACKGROUND);

807// long nannuli(pref->x/2);

809 Bimage* prcyl = pref->symmetrize_cylinder(1);
810
811 delete pt;
812
813 best_shift = ptcyl->find_shift(prcyl, NULL, 4*pref->image->sampling()[0], 0, pref->x/4, 0, 1, ccbest);
814
815 if ( verbose ) {
816 cout << "Shift\t\t\tAngle\tCC" << endl;
817 cout << best_shift << tab << 0 << tab << ccbest << endl;
818 }
819
820 ptcyl->reslice("-xyz");
821
822 shift = ptcyl->find_shift(prcyl, NULL, 4*pref->image->sampling()[0], 0, pref->x/4, 0, 1, cc);
823
824 if ( verbose )
825 cout << shift << tab << 180 << tab << cc << endl;
826
827 if ( ccbest < cc ) {
828 ccbest = cc;
829 best_shift = shift;
830 if ( verbose )
831 cout << "Map is upside down - rotating 180 degrees around x axis" << endl << endl;
832 axis = Vector3<double>(1,0,0);
833 rotate(axis, M_PI);
834 axis = Vector3<double>(0,0,1);
835// write_img("t.pif", this, 0);
836 }
837
838 delete ptcyl;
839 delete prcyl;
840
841 while ( dang >= accuracy ) {
842 ccbest = -1;
843 cout << "∆ang = " << dang*180.0/M_PI << tab << "res = " << hires << endl;
844 cout << "Shift\t\t\tAngle\tCC" << endl;
845 for ( angle = angmin; angle <= angmax; angle += dang ) {
846 mat = Matrix3(axis, angle);
847 pt = transform(pref->size(), vscale, ori, translate, mat, FILL_BACKGROUND);
848// cc = pt->correlate(pref, 0, pref->sizeX()/2, pmask);
849 if ( pmask ) pt->multiply(pmask);
850 shift = pt->find_shift(pref, NULL, hires, 0, 10, 0, 1, cc);
851 if ( ccbest < cc ) {
852 ccbest = cc;
853 best_angle = angle;
854 best_shift = shift;
855// if ( verbose & VERB_PROCESS )
856// cout << "Better CC: " << cc << " (" << angle*180.0/M_PI << ")" << endl;
857 }
858 delete pt;
859 if ( verbose )
860 cout << shift << tab << angle*180.0/M_PI << tab << cc << endl;
861 }
862 dang *= 0.75;
863 angmin = best_angle - 2*dang;
864 angmax = best_angle + 2.1*dang;
865 hires = dang*pref->real_size()[0];
866 }
867
868 best_shift *= scale;
869
870 if ( verbose ) {
871 cout << "Best shift and angle: " << best_shift << tab << best_angle*180.0/M_PI << endl;
872 cout << "Best CC: " << ccbest << endl;
873 cout << "Fixing symmetry " << sym.label() << " orientation" << endl << endl;
874 }
875
876 mat = Matrix3(axis, best_angle);
877 origin(size()/2 + best_shift);
878
879 return mat;
