634{
635 if ( resolution > real_size()[0] ) {
636 resolution = real_size()[0];
637 return;
638 }
639

641 Vector3<double> u(image->sampling());
642
643 if ( x > 1 ) invsum2 += 1/(u[0]*u[0]);
644 if ( y > 1 ) invsum2 += 1/(u[1]*u[1]);
645 if ( z > 1 ) invsum2 += 1/(u[2]*u[2]);
646 rm = 2/sqrt(invsum2);
647 if ( resolution < rm ) resolution = rm;
