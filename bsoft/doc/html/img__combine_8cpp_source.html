414Bimage* img_add_weighed(vector<string>& file_list, vector<double> weight,
415 double newavg, double newstd, int flags)
416{
417 if ( !file_list.size() ) return NULL;
418
419 int calcavg = flags & 1;
420 int calcfom = (flags & 2) >> 1;
421 int center = flags & 8;
422
423 long i, j, n(0), nf(0), nimg(0), c(0);
424 double v;
425 Bimage* p = NULL;
426
427 Bimage* psum = img_setup_combined(file_list, nimg, 0);
428 long nfiles(file_list.size());
429
430 double weightsum(0);
431 if ( weight.size() < nfiles )
432 for ( i=weight.size(); i<nfiles; i++ ) weight.push_back(1);
433
434 for ( i=0; i<nfiles; i++ ) weightsum += weight[i];;
435
436 double degrees_of_freedom = weightsum*(1 - 1.0L/nfiles);
437
438 if ( verbose & VERB_PROCESS ) {
439 cout << endl << "Adding " << nimg << " images together:" << endl;
440 cout << "New image size: " << psum->size() << endl;
441 cout << "Degrees of freedom: " << degrees_of_freedom << endl << endl;
442 } else if ( verbose & VERB_LABEL )
443 cout << endl << "Adding " << nimg << " images together" << endl << endl;
444
445 long imgsize = psum->sizeX()*psum->sizeY()*psum->sizeZ();
446 long datasize = imgsize*psum->images();
447
448 float* fom = NULL;
449 if ( calcfom ) {
450 psum->next = new Bimage(Float, TSimple, psum->size(), psum->images());
451 psum->next->sampling(psum->image->sampling());
452 psum->next->origin(psum->image->origin());
453 fom = (float *) psum->next->data_pointer();
454 }
455
456 vector<double> imgweight(psum->images());
457
458 for ( auto& filename: file_list ) {
459 p = read_img(filename, 1, -1);
460 if ( p != NULL ) {
461 if ( newstd > 0 ) p->rescale_to_avg_std(newavg, newstd);
462 if ( center ) p->center_wrap();
463 if ( verbose & VERB_LABEL )
464 cout << "Adding image " << nf << " with weight " << weight[nf] << endl << endl;
465 for ( n=j=0; n<p->images(); n++ ) {
466 imgweight[n] += weight[nf];
467 for ( i=n*imgsize; i<(n+1)*imgsize; i++ ) {
468 for ( c=0; c<psum->channels(); c++, j++ ) {
469 v = (*p)[j];
470 psum->add(j, weight[nf]*v);
471 if ( calcfom ) fom[i] += weight[nf]*v*v;
472 }
473 }
474 }
475 delete p;
476 }
477 nf++;
478 }
479
480 if ( calcfom ) {
481 for ( i=n=0; n<psum->images(); n++ )
482 for ( j=0; j<imgsize; j++, i++ )
483 fom[i] = (fom[i] - (*psum)[i]*(*psum)[i]/imgweight[n])/degrees_of_freedom;
484 if ( calcfom > 1 ) {
485 for ( i=0; i<datasize; i++ ) {
486 if ( fom[i] > 0 ) fom[i] = sqrt(fom[i]);
487 else fom[i] = 0;
488 }
489 }
490 }
491
492 if ( calcavg )
493 for ( i=n=0; n<psum->images(); n++ )
494 for ( j=0; j<imgsize; j++, i++ )
495 psum->set(i, (*psum)[i] / imgweight[n]);
496
497 psum->statistics();
498
499 return psum;
500}
