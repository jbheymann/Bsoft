57/*
58@brief Functions for particle picking.
59 Box interface syntax:
60 Bmg box [item] [action] [params]
61 Item: Must contain "Micrograph" or "Reconstruction"
62 Actions:
63 count [n] [|"bad"] counts the number of particles or bad areas
64 ids [fc|"bad"] [fi] returns the identifiers of particles or bad areas
65 location [id] returns the location of a particle or bad area
66 fom [id] [fi] returns the fom of a particle with the indicated index
67 fom_min returns minimum FOM value
68 fom_max returns maximum FOM value
69 select [id] [s] sets the selection number of a particle
70 select [x] [y] [z] returns the identifier at the given location
71 select_min returns minimum selection value
72 select_max returns maximum selection value
73 move [id] [dx] [dy] [dz] updates a particle location by the given vector
74 create [x] [y] [z] [n] [t] [s] creates a new particle at the given location
75 renumber [n] renumbers particles
76 delete [all|mg|rec|id] [n] deletes particles or bad areas
77 template updates picked template
78 pickcc [hr] [lr] [fmn] [fmx] [ed] [b] picks particles by cross-correlation with a template
79 pickvar [ak] [vk] [ed] [b] [ns] picks particles from a variance map
80 extract [f] [of] [bf] [nf] [ft] [fv] [mw] [sf] [p] extracts particles from a micrograph
81 Parameters:
82 id identifier
83 t box type: 1=particle, -1=bad area
84 fc FOM cutoff
85 fmn FOM minimum
86 fmx FOM maximum
87 fi FOM index (0-4)
88 s selection number
89 x x location
90 y y location
91 z z location
92 n image number
93 dx change in x
94 dy change in y
95 dz change in z
96 r particle radius
97 e gaussian edge for picking particles
98 hr high resolution limit
99 lr low resolution limit
100 ak averaging kernel edge size
101 av variance kernel edge size
102 ed exclusion distance for picking particles
103 b binning for picking particles
104 ns number of standard deviations for picking particles
105 f file name
106 of flag indicating odd box size
107 bf background correction flag
108 nf normalization flag
109 ft fill type
110 fv fill value
111 mw filament mask width
112 sf split flag
113 p particle path
114*/
115Tcl_Obj* do_box(Bproject* project, int objc, Tcl_Obj *CONST objv[])
116{
117 Tcl_Obj* returnObj = NULL;
118
119 Bstring item = Tcl_GetStringFromObj(objv[2], NULL);
120 if ( verbose & VERB_DEBUG )
121 cout << "DEBUG do_box: Item: " << item << " (" << item.length() << ")" << endl;
122
123 if ( item != "all" && !item.contains("Field") && !item.contains("Micrograph") &&
124 !item.contains("Reconstruction") ) {
125 cerr << "Error in do_box: Item " << item << " must be a micrograph or reconstruction!" << endl;
126 returnObj = Tcl_NewObj();
127 return returnObj;
128 }
129
130 Bstring id = item.post(':');
131 id = id.remove(' ');
132 item = item.pre(':');
133
134 Bfield* field = NULL;
135 Bmicrograph* mg = NULL;
136 Breconstruction* rec = NULL;
137 Bparticle* part = NULL;
138 Bbadarea* bad = NULL;
139 Bimage* p = NULL;
140 Bstring filename, filename2, base, delete_what;
141 double badrad(0);
142 Vector3<float> radius;
143
144 if ( item == "all" ) {
145 } else if ( item.contains("Field") ) {
146 for ( field = project->field; field && field->id != id; field = field->next ) ;
147 if ( field ) mg = field->mg;
148 if ( !mg )
149 cerr << "Error in do_box: Field " << id << " not found!" << endl;
150 } else if ( item.contains("Micrograph") ) {
151 for ( field = project->field; field; field = field->next ) {
152 for ( mg = field->mg; mg && mg->id != id; mg = mg->next ) ;
153 if ( mg ) break;
154 }
155 if ( !mg )
156 cerr << "Error in do_box: Micrograph " << id << " not found!" << endl;
157 } else if ( item.contains("Reconstruction") ) {
158 for ( rec = project->rec; rec && rec->id != id; rec = rec->next ) ;
159 if ( !rec )
160 cerr << "Error in do_box: Reconstruction " << id << " not found!" << endl;
161 } else {
162 cerr << "Error in do_box: Item " << item << " must be a field, micrograph or reconstruction!" << endl;
163 returnObj = Tcl_NewObj();
164 return returnObj;
165 }
166
167 if ( mg ) {
168 if ( mg->fmg.length() )
169 filename = mg->fmg;
170 else if ( mg->fframe.length() )
171 filename = mg->fframe;
172 part = mg->part;
173 bad = mg->bad;
174 radius = mg->box_size/2;
175 badrad = mg->bad_radius;
176 } else if ( rec ) {
177 filename = rec->frec;
178 part = rec->part;
179 bad = rec->bad;
180 radius = rec->box_size/2;
181 badrad = rec->bad_radius;
182 }
183
184 if ( filename.length() ) {
185 base = filename.base();
186 for ( p = imglist; p; p = p->next ) {
187 filename2 = p->file_name();
188 if ( filename2.base() == base ) break;
189 }
190 }
191
192 Bstring action = Tcl_GetStringFromObj(objv[3], NULL);
193 if ( verbose & VERB_DEBUG )
194 cout << "DEBUG do_box: Action: " << action << " (" << action.length() << ")" << endl;
195
196 if ( action == "count" ) {
197 if ( item == "all" )
198 returnObj = box_count(project, objc, objv);
199 else
200 returnObj = box_count(part, bad, objc, objv);
201 } else if ( action == "ids" ) {
202 returnObj = box_ids(part, bad, objc, objv);
203 } else if ( action == "location" ) {
204 returnObj = box_location(part, bad, objc, objv);
205 } else if ( action == "fom" ) {
206 returnObj = box_fom(part, bad, objc, objv);
207 } else if ( action == "fom_min" ) {
208 returnObj = box_fom_min(project, objc, objv);
209 } else if ( action == "fom_max" ) {
210 returnObj = box_fom_max(project, objc, objv);
211 } else if ( action == "select" ) {
212 if ( objc < 7 )
213 returnObj = box_select(part, objc, objv);
214 else
215 returnObj = box_select(part, bad, radius, badrad, objc, objv);
216 } else if ( action == "select_min" ) {
217 returnObj = box_select_min(project, objc, objv);
218 } else if ( action == "select_max" ) {
219 returnObj = box_select_max(project, objc, objv);
220 } else if ( action == "move" ) {
221 returnObj = box_move(part, bad, objc, objv);
222 } else if ( action == "create" ) {
223 if ( mg ) returnObj = box_create(mg, objc, objv);
224 else if ( rec ) returnObj = box_create(rec, objc, objv);
225 } else if ( action == "renumber" ) {
226 returnObj = box_renumber(part, objc, objv);
227 } else if ( action == "delete" ) {
228 if ( objc > 4 ) delete_what = Tcl_GetStringFromObj(objv[4], NULL);
229 if ( delete_what == "all" || delete_what == "mg" || delete_what == "rec" )
230 returnObj = box_delete(project, objc, objv);
231 else if ( mg )
232 returnObj = box_delete(mg, objc, objv);
233 else if ( rec )
234 returnObj = box_delete(rec, objc, objv);
235 } else if ( action == "center" ) {
236 if ( p ) returnObj = box_center(mg, rec, p, objc, objv);
237 } else if ( action == "template" ) {
238 if ( p ) returnObj = box_update_template(mg, rec, p, objc, objv);
239 } else if ( action == "pickcc" ) {
240 if ( p ) returnObj = box_pickcc(mg, rec, p, objc, objv);
241 } else if ( action == "pickvar" ) {
242 if ( p ) returnObj = box_pickvar(mg, rec, p, objc, objv);
243 } else if ( action == "extract" ) {
244 if ( p ) returnObj = box_extract(mg, rec, p, objc, objv);
245 } else {
246 cerr << "Error: Action " << action << " not recognized!" << endl;
247 bexit(-1);
248 }
249
250 return returnObj;
251}

438Tcl_Obj* box_select(Bparticle* part, Bbadarea* bad, Vector3<float> rad, double badrad, int objc, Tcl_Obj *CONST objv[])
439{
440 Tcl_Obj* returnObj = Tcl_NewObj();
441
442 int id(0), i(0);
443 double x(0), y(0), z(0);
444
445 if ( objc > 4 ) Tcl_GetDoubleFromObj(NULL, objv[4], &x);
446 if ( objc > 5 ) Tcl_GetDoubleFromObj(NULL, objv[5], &y);
447 if ( objc > 6 ) Tcl_GetDoubleFromObj(NULL, objv[6], &z);
448
449 double d, dmin;
450 Vector3<float> loc(x, y, z), b, vs, ve;
451
452 // Particle box
453 dmin = 2*rad.length();
454 for ( ; part; part = part->next ) {
455 d = loc.distance(part->loc);
456 if ( dmin > d ) {
457 dmin = d;
458 i = part->id;
459 b = part->loc;
460 }
461 }
462// if ( dmin <= rad.length() ) id = i;
463 if ( i ) {
464 vs = b - rad;
465 ve = b + rad;
466 if ( loc.within(vs, ve) ) id = i;
467 }
468
469 // Bad area
470 for ( ; bad; bad = bad->next ) {
471 d = loc.distance(bad->loc);
472 if ( dmin > d ) {
473 dmin = d;
474 i = bad->id;
475 }
476 }
477 if ( id == 0 ) if ( dmin <= badrad ) id = i;
478
479 Tcl_SetIntObj(returnObj, id);
480
481 return returnObj;
482}

847{
