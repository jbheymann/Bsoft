32int readMFF(Bimage* p, int readdata)
33{
34 ifstream* fimg = new ifstream;
35 fimg->open(p->file_name());
36 if ( fimg->fail() ) return -1;
37
38 if ( verbose & VERB_DEBUG )
39 cout << "DEBUG readMFF: File opened" << endl;
40
41 MFFhead* header = new MFFhead;
42
43 fimg->read((char *)header, MFFSIZE);
44 if ( fimg->fail() ) return -2;
45
46 // Determine byte order and swap bytes if from little-endian machine
47 unsigned char* b = (unsigned char *) header;
48 int i, sb = 0, vax = 0;
49 if ( abs(header->nz) > SWAPTRIG ) {
50 if ( verbose & VERB_PROCESS )
51 cerr << "Warning: Swapping header byte order for 4-byte types" << endl;
52 sb = 1;
53 for ( i=84; i<MFFSIZE; i+=4 ) swapbytes(b+i, 4);
54 }
55
56 // Convert VAX floating point types if necessary
57 if ( header->a > SWAPTRIG || header->b > SWAPTRIG || header->c > SWAPTRIG ) {
58 if ( verbose & VERB_PROCESS )
59 cerr << "Warning: Converting VAX floating point" << endl;
60 vax = 1;
61 for ( i=92; i<136; i+=4 ) // Unit cell parameters & step sizes
62 vax2ieee(b+i, sb);
63 for ( i=204; i<MFFSIZE; i+=4 ) // Origin & skew matrix
64 vax2ieee(b+i, sb);
65 }
66
67 // Map the parameters
68 p->size(header->nx, header->ny, header->nz);
69 p->images(1);
70 p->channels(1);
71 p->data_type(Float);
72 p->data_offset(MFFSIZE);
73
74 UnitCell uc(header->a, header->b, header->c, header->alpha, header->beta, header->gamma);
75 p->unit_cell(uc);
76
77 p->sampling(header->stepu, header->stepv, header->stepw);
78 p->label(header->title);
79
80 p->page_size(p->size()[0], p->size()[1], 1);
81
82 // Allocating the single sub-image and setting its origin
83 p->origin(-header->uvworg[0]/header->uvwmat[0][0],
84 -header->uvworg[1]/header->uvwmat[1][1],
85 -header->uvworg[2]/header->uvwmat[2][2]);
86
87 delete header;
88
89 if ( readdata )
90 p->read_data( fimg, -1, sb, vax, 8 );
91
92 fimg->close();
93 delete fimg;
94
95 return 0;
96}
