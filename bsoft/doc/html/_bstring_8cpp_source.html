876int string_sort(Bstring* slist, int descending, int value)
877{
878 int i, n(0);
879 Bstring* s;
880
881 for ( s = slist; s; s = s->next ) n++;
882
883 Bstring* ssort = new Bstring[n];
884
885 for ( i=0, s = slist; s; s = s->next, i++ ) ssort[i] = *s;
886
887 if ( value ) {
888 if ( descending ) qsort((void *) ssort, n, sizeof(Bstring), QsortLargeToSmallStringValue);
889 else qsort((void *) ssort, n, sizeof(Bstring), QsortSmallToLargeStringValue);
890 } else {
891 if ( descending ) qsort((void *) ssort, n, sizeof(Bstring), QsortLargeToSmallString);
892 else qsort((void *) ssort, n, sizeof(Bstring), QsortSmallToLargeString);
893 }
894
895 for ( i=0, s = slist; s; s = s->next, i++ ) *s = ssort[i];
896
897 delete[] ssort;
898
