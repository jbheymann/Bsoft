29 std::string GetFieldAsString(TIFF* tiff, ttag_t tag)
30 {
31 char* data = nullptr;
32 uint32_t count = 0;
33 if (TIFFGetField(tiff, tag, &count, &data) != 1) throw std::runtime_error("GetField: Field with tag " + std::to_string(tag) + " is not present.");
34 return std::string(data, count);
35 }
