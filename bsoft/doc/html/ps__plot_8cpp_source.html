496int ps_scale(ofstream* fps, double x1, double y1, double x2, double y2,
497 double min, double max, double increment,
498 double tick_length, double tick_angle,
499 int digits, int fontsize, int inverse)
500{
501 ios_base::fmtflags fl = fps->flags();
502
503 long i, j;
504 double value, showval;
505 double x_tick = tick_length*cos(tick_angle);
506 double y_tick = tick_length*sin(tick_angle);
507 double x_step = (x2-x1)*increment/(max - min);
508 double y_step = (y2-y1)*increment/(max - min);
509
510 if ( digits < 0 ) digits = 0;
511 if ( fontsize < 2 ) fontsize = 10;
512
513 long integer_size = (int) (log(fabs(max))/log(10.0) + 1);
514 if ( fabs(max) < fabs(min) ) integer_size = (int) (log(fabs(min))/log(10.0) + 1);
515 if ( inverse ) {
516 if ( min <= 0 ) integer_size = (int) (log(fabs(10/max))/log(10.0) + 1);
517 else integer_size = (int) (log(fabs(1/min))/log(10.0) + 1);
518 }
519 if ( integer_size < 1 ) integer_size = 1;
520 integer_size++; // Add one for the sign
521
522 long number_size = integer_size + 1 + digits;
523 double x_offset = x1 - number_size*fontsize/5;
524 double y_offset = y1 - fontsize/3;
525 if ( x2 - x1 > y2 - y1 )
526 y_offset -= fontsize;
527 else
528 x_offset -= number_size*fontsize/3;
529
530 if ( tick_angle > 0.75*M_PI && tick_angle < 1.25*M_PI )
531 x_offset = x1 + fontsize/5;
532
533 if ( verbose & VERB_DEBUG ) {
534 cout << "DEBUG ps_scale: min=" << min << " max=" << max << " increment=" << increment << endl;
535 cout << "DEBUG ps_scale: number_size=" << number_size << " digits=" << digits << endl;
536 }
537
538 // Ticks and labels
539 *fps << "/Helvetica findfont " << fontsize << " scalefont setfont " << endl;
540 for ( i=0, value=min; value<max+0.01*increment; value+=increment, i++ ) {
541 *fps << fixed << setprecision(2) <<
542 x1+i*x_step << " " << y1+i*y_step << " moveto " <<
543 x1+i*x_step+x_tick << " " << y1+i*y_step+y_tick << " lineto stroke" << endl;
544 showval = value;
545 if ( inverse && value ) { if ( value ) showval = 1/value; else showval = 999; }
546// cout << showval << endl;
547 *fps << x_offset + i*x_step << " " << y_offset + i*y_step << " moveto ("
548 << fixed << setprecision(digits) << right << setw(number_size) << showval << ") (";
549 for ( j=0; j<integer_size; j++ ) *fps << "x";
550 *fps << ".";
551 for ( j=0; j<digits; j++ ) *fps << "x";
552 *fps << ") cvs show" << endl;
553 }
554
555// fps->unsetf(fps->flags());
556 fps->flags(fl);
557
558 return 0;
559}
