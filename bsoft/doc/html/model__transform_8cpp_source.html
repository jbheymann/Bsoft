616 cout << "Mapping model " << model->identifier() << " to reference " << refmod->identifier() << endl;
617
618 long i, j;
619 Bcomponent* comp, *compr;
620 vector<double> bx(4), by(4), bz(4), v(4,0);
621 Matrix a(4,4);

626

628 Vector3<double> comr = refmod->center_of_coordinates();

634 loc = comp->location() - com;
635 locr = compr->location() - comr;
636 if ( verbose & VERB_FULL )
637 cout << loc[2] << tab << locr[2] << tab << loc[2] - locr[2] << endl;
638 v[0] = loc[0];
639 v[1] = loc[1];
640 v[2] = loc[2];
641 for ( i=0; i<4; i++ ) {
642 bx[i] += v[i]*locr[0];
643 by[i] += v[i]*locr[1];
644 bz[i] += v[i]*locr[2];
645 for ( j=0; j<=i; j++ ) a[i][j] += v[i]*v[j];
646 }
647 }
648
649 t = transform_matrix_solve(a, bx, by, bz, 0);
650 t.trans = comr - com;
651 t.origin = com;

664double model_random_displace_number(Bmodel* model, long number, double stdev, int flag)
665{
666 if ( stdev < 1e-30 ) return 0;
667
668 model_select_random(model, number);
669
670 long n(0);
671 double msd(0);
672 Vector3<double> vec;
673 Bmodel* mp;
674 Bcomponent* comp;
675
676 if ( verbose & VERB_PROCESS )
677 cout << "Randomizing " << number << " coordinates to standard deviation " << stdev << " A" << endl << endl;
678
679 for ( mp = model; mp; mp = mp->next ) if ( mp->select() ) {
680 for( comp = mp->comp; comp; comp = comp->next ) if ( comp->select() ) {
681 vec = vector3_random_gaussian(0, stdev);
682 comp->location() += vec;
683 if ( flag ) comp->set_within_boundary(model->minimum(), model->maximum());
684 msd += vec.length2();
685 n++;
686 }
687 }
688
689 if ( n ) msd /= n;
690
691 if ( verbose & VERB_PROCESS )
692 cout << "Mean square displacement: " << msd << " A2" << endl << endl;
693
694 model->select_all();
695
696 return msd;
697}
