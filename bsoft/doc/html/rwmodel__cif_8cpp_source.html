294long write_model_cif(string& filename, Bmodel* model, int splt)
295{
296 if ( verbose & VERB_DEBUG )
297 cout << "DEBUG write_model_cif: filename=" << filename << endl;
298
299 Bstar star;
300
301 star.line_length(120);
302
303 Bmodel* mp = NULL;
304 Bcomponent* comp = NULL;
305// Bcomponent* comp2 = NULL;
306// Blink* link = NULL;
307// Bgroup* group = NULL;
308
309 map<string,string> entities;
310
311 BstarBlock& block = star.add_block(model->identifier());
312
313 BstarLoop& loop1 = block.add_loop();
314 loop1.tags()[MOL_ID] = 0;
315 loop1.tags()[MOL_ENTITY] = 1;
316 for ( mp = model; mp; mp = mp->next ) {
317 vector<string>& vs = loop1.add_row(2);
318 vs[0] = mp->identifier();
319 vs[1] = mp->model_type();
320 entities[mp->model_type()] = mp->description();
321// cout << vs[0] << tab << vs[1] << endl;
322 }
323
324 BstarLoop& loop2 = block.add_loop();
325 loop2.tags()[ENTITY_ID] = 0;
326 loop2.tags()[ENTITY_TYPE] = 1;
327 loop2.tags()[ENTITY_DESCRIPTION] = 2;
328 for ( auto e1: entities ) {
329 vector<string>& vs = loop2.add_row(3);
330 vs[0] = e1.first;
331 vs[1] = "polymer";
332 vs[2] = '"' + e1.second + '"';
333 if ( verbose & VERB_FULL )
334 cout << vs[0] << tab << vs[2] << endl;
335 }
336
337 BstarLoop& loop = block.add_loop();
338 loop.tags()[ATOM_NUMBER] = 0;
339 loop.tags()[ATOM_SYMBOL] = 1;
340 loop.tags()[ATOM_TOPTYPE] = 2;
341 loop.tags()[ATOM_RESIDUE] = 3;
342 loop.tags()[ATOM_RESNUMBER] = 4;
343 loop.tags()[ATOM_RESINSERT] = 5;
344 loop.tags()[ATOM_CHAIN] = 6;
345 loop.tags()[ATOM_ENTITY] = 7;
346 loop.tags()[ATOM_X] = 8;
347 loop.tags()[ATOM_Y] = 9;
348 loop.tags()[ATOM_Z] = 10;
349 loop.tags()[ATOM_OCCUPANCY] = 11;
350 loop.tags()[ATOM_BFACTOR] = 12;
351 loop.tags()[ATOM_CHARGE] = 13;
352
353 for ( mp = model; mp; mp = mp->next ) {
354// cout << mp->identifier() << endl;
355 for ( comp = mp->comp; comp; comp = comp->next ) {
356 vector<string>& vd = comp->description();
357 vector<string>& vs = loop.add_row(14);
358 vs[0] = comp->identifier();
359 vs[1] = vd[0];
360 vs[2] = vd[1];
361 if ( vd.size() > 2 ) vs[3] = vd[2];
362 else vs[3] = "UNK";
363 if ( vd.size() > 4 ) vs[4] = vd[4];
364 else vs[4] = to_string(1);
365 if ( vd.size() > 5 ) vs[5] = vd[5];
366 else vs[5] = "?";
367 if ( vd.size() > 3 ) vs[6] = vd[3];
368 else vs[6] = "A";
369 vs[7] = mp->model_type();
370 vs[8] = to_string(comp->location()[0]);
371 vs[9] = to_string(comp->location()[1]);
372 vs[10] = to_string(comp->location()[2]);
373 vs[11] = to_string(comp->density());
374 vs[12] = to_string(comp->FOM());
375 vs[13] = to_string(comp->charge());
376 }
377 }
378
379 if ( verbose & VERB_DEBUG )
380 cout << "DEBUG write_model_cif: " << filename << endl;
381
382 star.write(filename, splt);
383
384 return 1;
