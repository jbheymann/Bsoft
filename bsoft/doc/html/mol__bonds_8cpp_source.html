151/*
152@brief Finds the top shortest bonds.
153@param *bondlist list of bonds.
154@param number number of shortest bonds to find.
155@return Bbond* short bond list.
156
157 This function finds the top number of shortests bonds with the aim
158 of assigning bonds based valence. The current bond length must be
159 given by the bond strength variable, k, in the bond structure.
160**/
161Bbond* md_bondlist_get_top(Bbond* bondlist, int number)
162{
163 if ( !bondlist ) return NULL;
164
165 int i, imax = 0;
166 double max = 0;
167 double* val = new double[number];
168 Bbond** bondarray = new Bbond*[number];
169 Bbond* newbond = NULL, *bond;
170
171 for ( i=0; i<number; i++ ) {
172 val[i] = 1e37;
173 bondarray[i] = NULL;
174 }
175
176 for ( bond = bondlist; bond; bond = bond->next ) {
177 for ( i=imax=0, max=0; i<number; i++ ) if ( max < val[i] ) {
178 max = val[i];
179 imax = i;
180 }
181 if ( val[imax] > bond->k ) {
182 val[imax] = bond->k;
183 bondarray[imax] = bond;
184 }
185 }
186
187 for ( i=0; i<number; i++ ) if ( bondarray[i] )
188 bond_add(&newbond, bondarray[i]->atom1, bondarray[i]->atom2, bondarray[i]->l, 1);
189
190 delete[] val;
191 delete[] bondarray;
192
193 return newbond;
194}

207Bbond* md_generate_bond_list_with_valence(Bmolgroup* molgroup, Bmd* md, int valence)
208{
209 if ( molgroup->bond ) {
210 if ( verbose ) cerr << "Warning: Bond list already defined!" << endl;
211 return molgroup->bond;

215
216 int i, ii, x, y, z, xx, yy, zz, ix, iy, iz;
217 Vector3<double> box = molgroup->box;
218 Vector3<double> sampling(cutoff, cutoff, cutoff);
219 Vector3<int> size((int) (box[0]/sampling[0] + 0.001),
220 (int) (box[1]/sampling[1] + 0.001), (int) (box[2]/sampling[2] + 0.001));
221 size = size.max(1);
222 for ( i=0; i<3; i++ ) sampling[i] = box[i]/size[i] + 0.001;
223 long boxsize = (long) size.volume();
224 double dist, bondlength;
225 Batom *atom, *atom2;
226 Latom *latom, *latom2;
227 Bbond* tbondlist = NULL;
228 Bbond* tbond = NULL;
229 Bbond* bond = NULL;

231 if ( verbose )
232 cout << "Sampling intervals for cutoff " << cutoff << ": " << sampling << endl;
233
234 Latom** alist = molgroup_atom_mesh_lists(molgroup, size, sampling);
235
236 for ( z=0; z<size[2]; z++ ) {
237 for ( y=0; y<size[1]; y++ ) {
238 for ( x=0; x<size[0]; x++ ) {
239 i = (z*size[1] + y)*size[0] + x;
240 tbond = tbondlist = NULL;
241 for ( latom = alist[i]; latom; latom = latom->next ) {
242 atom = latom->atom;
243 for ( zz=z-1; zz<=z+1; zz++ ) {
244 iz = zz;
245 if ( iz < 0 ) iz += size[2];
246 if ( iz >= size[2]) iz -= size[2];
247 for ( yy=y-1; yy<=y+1; yy++ ) {
248 iy = yy;
249 if ( iy < 0 ) iy += size[1];
250 if ( iy >= size[1] ) iy -= size[1];
251 for ( xx=x-1; xx<=x+1; xx++ ) {
252 ix = xx;
253 if ( ix < 0 ) ix += size[0];
254 if ( ix >= size[0] ) ix -= size[0];
255 ii = (iz*size[1] + iy)*size[0] + ix;
256 for ( latom2 = alist[ii]; latom2; latom2 = latom2->next ) {
257 atom2 = latom2->atom;
258 bondlength = md_find_bond_length(atom, atom2, md->bond);
259 if ( md->wrap )
260 dist = (vector3_difference_PBC(atom->coord, atom2->coord, box)).length();
261 else
262 dist = atom->coord.distance(atom2->coord);
263 if ( dist > 0.1*bondlength && dist < 1.3*bondlength ) {
264 if ( tbond ) tbond = bond_add(&tbond, atom, atom2, bondlength, dist);
265 else tbond = bond_add(&tbondlist, atom, atom2, bondlength, dist);
266 }
267 }
268 }
269 }
270 }
271 }
272 tbond = md_bondlist_get_top(tbondlist, valence);
273 if ( bond ) bond->next = tbond;
274 else bond = molgroup->bond = tbond;
275 if ( bond ) for ( ; bond->next; bond = bond->next ) ;
276 bond_kill(tbondlist);
277 }
278 }
279 }
280
281 for ( i=0; i<boxsize; i++ ) kill_list((char *) alist[i], sizeof(Latom));
282 delete[] alist;
283
284 if ( verbose & VERB_FULL )
285 md_show_bonds(molgroup);
286
287 return molgroup->bond;

817

819 for( res = mol->res; res; res = res->next )

823 radstd += radius*radius;
824 natom++;
