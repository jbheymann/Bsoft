13## @brief Configures scales for the image window
14#
15# These scales are used as stores for the slice and image numbers
16# as well as the scaling of the image itself
17#
18# @param w Control window to put scales in.

89## @brief Resets the scales when the image changes
90
91proc reconfigureScales { } {
92 global theimg
93 global window_scale montover
94 if ![Bimage exists $theimg] { return }
95 set c [getImageCanvas $theimg]
96 set wc [getControlWindow $theimg]
97 if { [info exists montover(active)] && $montover(active) } {
98 set width [expr $montover(ncol) * ($montover(nx) + 2*$montover(pad))]
99 set height [expr $montover(nrow) * ($montover(ny) + 2*$montover(pad))]
100 set nimages 1
101 set nslices 1
102 } else {
103 set width [Bimage get $theimg width]
104 set height [Bimage get $theimg height]
105 set nimages [Bimage get $theimg nimages]
106 set nslices [Bimage get $theimg nslices]
107 }
108 set width [expr $width * $window_scale]
109 set height [expr $height * $window_scale]
110 $wc.image.scale config -to [expr $nimages - 1] -tickinterval [expr $nimages - 1]
111 $wc.slice.scale config -to [expr $nslices - 1] -tickinterval [expr $nslices - 1]
112 reconfigureMinMax
113 set size "0 0 $width $height"
114 $c configure -scrollregion $size
115}

117## @brief Resets the min and max scales when the range step size changes
118
119proc reconfigureMinMax { } {
120 global theimg autoscale
121 set wc [getControlWindow $theimg]
122 set range_step [$wc.step.e get]
123 set min [Bimage get $theimg min]
124 set max [Bimage get $theimg max]
125 Bimage set $theimg show_min $min
126 Bimage set $theimg show_max $max
127 $wc.min.scale config -from $min -to $max -resolution $range_step
128 $wc.max.scale config -from $min -to $max -resolution $range_step
129 if $autoscale {
130 autoScale
131 } else {
132 $wc.min.scale set $min
133 $wc.max.scale set $max
134 }
135}

137## @brief Resets the range step size to the default value
138
139proc reconfigureRangeStep { } {
140 global theimg imgtype
141 if ![Bimage exists $theimg] { return }
142 set wc [getControlWindow $theimg]
143 set min [Bimage get $theimg min]
144 set max [Bimage get $theimg max]
145 set step 1000
146 if { $imgtype == "ps" || $imgtype == "ft" } {
147 set step 100000
148 }
149 set range_step [expr ($max - $min) / $step]
150 $wc.step.e delete 0 end
151 $wc.step.e insert 0 $range_step
152}

299## @brief Sets the minimum display value.

301# @param smin New display minimum.

303proc setMin { smin } {
304 global theimg mode
305 global img_load_phase autoscale
306 if ![Bimage exists $theimg] { return }
307 set wc [getControlWindow $theimg]
308 set step [$wc.step.e get]
309 if { [expr 2*abs([Bimage get $theimg show_min] - $smin)] < $step } { return }
310 if { [$wc.min.scale cget -from] || [$wc.min.scale cget -to] } {
311 set autoscale 0
312 set scale [$wc.scale.scale get]
313 set slice_num [$wc.slice.scale get]
314 set img_num [$wc.image.scale get]
315# puts "set show_min to $smin"
316 Bimage set $theimg show_min $smin
317 if { !$img_load_phase } {
318 Bimage show $theimg $img_num $slice_num $scale $mode
319 }
320 }

347## @brief Sets reasonable minimum and maximum display values.
348#
349
350proc autoScale {} {
351 global theimg autoscale std_scale imageonly
352 if ![Bimage exists $theimg] { return }
353 set wc [getControlWindow $theimg]
354
355# set autoscale [expr 1 - $autoscale]
356
357 if $imageonly {
358 set img_num [$wc.image.scale get]
359 set min [Bimage get $theimg min $img_num]
360 set max [Bimage get $theimg max $img_num]
361 set avg [Bimage get $theimg average $img_num]
362 set std [Bimage get $theimg standard_deviation $img_num]
363 } else {
364 set min [Bimage get $theimg min]
365 set max [Bimage get $theimg max]
366 set avg [Bimage get $theimg average]
367 set std [Bimage get $theimg standard_deviation]
368 }
369
370 set std_scale [$wc.as.scale get]
371
372 set smin [expr $avg - $std_scale*$std]
373 set smax [expr $avg + $std_scale*$std]
374 if { $smin >= $smax } {
375 set smin [expr $avg - 1]
376 set smax [expr $avg + 1]
377 }
378 if { $smin < $min } {
379 set smin $min
380 }
381 if { $smax > $max } {
382 set smax $max
383 }
384
385 $wc.min.scale set $smin
386 $wc.max.scale set $smax
387
388 update
389
390 set autoscale 1
391
392# puts "autoscale = $autoscale"
393}

408## @brief Pages through slices.
409#
410
411proc do_animate_slices {} {
412 global theimg aniproc
413 set wc [getControlWindow $theimg]
414 if { [Bimage get $theimg nimages] > [Bimage get $theimg nslices] } {
415 set scale_to_move images
416 } else {
417 set scale_to_move slices
418 }
419 if { [$wc.ani.animate get] == "" || [$wc.ani.animate get] == "0" } {
420 $wc.ani.animate delete 0 end
421 $wc.ani.animate insert 0 15
422 }
423 if { [$wc.ani.animate get] > 200 } {
424 $wc.ani.animate delete 0 end
425 $wc.ani.animate insert 0 200
426 }
427 if { $aniproc == 0} {
428 $wc.ani.animate_button configure -text "Stop"
429 set aniproc 1
430 $wc.ani.animate configure -state disabled
431 } else {
432 $wc.ani.animate_button configure -text "Animate"
433 set aniproc 0
434 $wc.ani.animate configure -state normal
435 }
436 if { $scale_to_move == "slices" } {
437 set nslices1 [expr [Bimage get $theimg nslices] - 1]
438 set slice [$wc.slice.scale get]
439 set dir 1
440 while { $aniproc } {
441 after [expr 1000 / [$wc.ani.animate get]]
442 if { $slice <= 0 } { set dir 1 }
443 if { $slice >= $nslices1 } { set dir -1 }
444 set slice [expr $slice + $dir]
445 $wc.slice.scale set $slice
446 update
447 }
448 } elseif { $scale_to_move == "images" } {
449 set nimages1 [expr [Bimage get $theimg nimages] - 1]
450 set currimage [$wc.image.scale get]
451 set dir 1
452 while { $aniproc } {
453 after [expr 1000 / [$wc.ani.animate get]]
454 if { $currimage <= 0 } { set dir 1 }
455 if { $currimage >= $nimages1 } { set dir -1 }
456 set currimage [expr $currimage + $dir]
457 $wc.image.scale set $currimage
458 update
459 }
460 }
