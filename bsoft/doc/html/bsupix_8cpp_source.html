228vector<Bsuperpixel> superpixel_from_json(JSvalue& root)
229{
230 vector<Bsuperpixel> seg;
231
232 vector<JSvalue> splist = root.array();
233
234 for ( auto it = splist.begin(); it != splist.end(); ++it ) {
235 JSvalue& jv = *it;
236 Bsuperpixel sp;
237 vector<double> coor = jv["coordinates"].array_real();
238 vector<double> avg = jv["channel_averages"].array_real();
239 vector<double> var = jv["channel_variances"].array_real();
240 vector<long> nb = jv["neighbors"].array_integer();
241 sp.channels(avg.size());
242 sp.index(jv["id"].integer());
243 sp.count(jv["count"].integer());
244 sp.coordinates(coor);
245 sp.channels(avg);
246 sp.variances(var);
247 sp.weight(jv["weight"].real());
248 sp.neighbors(nb);
249 seg.push_back(sp);
250 }
251
252 if ( verbose )
253 cout << "Superpixels: " << seg.size() << endl << endl;
254
255 return seg;
256}

269JSvalue json_from_superpixel(vector<Bsuperpixel>& seg)
270{
271 vector<JSvalue> splist;
272
273 for ( auto it = seg.begin(); it != seg.end(); ++it ) {
274 map<string, JSvalue> sp;
275 sp["id"] = it->index();
276 sp["count"] = it->count();
277 sp["coordinates"] = JSvalue(it->coordinates());
278 sp["channel_averages"] = JSvalue(it->channels());
279 sp["channel_variances"] = JSvalue(it->variances());
280 sp["weight"] = it->weight();
281 sp["neighbors"] = jsvalue_from_neighbors(*it);
282 splist.push_back(sp);
283 }
284
285 return splist;
