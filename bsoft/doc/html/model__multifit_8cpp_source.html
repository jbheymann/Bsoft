195Bmodel* model_from_peaks(Bimage* p, double threshold, int wrap)

219 for ( y=0; y<p->sizeY(); y++ ) {
220 for ( x=0; x<p->sizeX(); x++, i++ ) {
221 if ( (*pfom)[i] >= threshold ) {
222// id = string(++j, "%d");

228 comp->location(Vector3<double>(x, y, z) - origin);

231 if ( comp->location()[1] > h[1] ) comp->location()[1] -= p->sizeY();
232 if ( comp->location()[2] > h[2] ) comp->location()[2] -= p->sizeZ();
233 }
234 comp->scale(p->sampling(0));
235 comp->select(n+1);
236 comp->FOM((*pfom)[i]);
237 }
238 }
239 }
240 }
241 }
