134long project_add_files(Bproject* project, vector<string>& files, int ftype, long number)
135{
136 if ( files.size() < 1 ) return 0;
137
138 if ( verbose )
139 cout << "Adding files of type " << ftype << endl;
140
141 long i, j, nmg(0), mg_per_field(1);
142 Bfield* field = NULL;
143 Bmicrograph* mg = NULL;
144 Bparticle* part = NULL;
145 Bstring mgid;
146
147 if ( number > 1 && ( ftype == 1 || ftype == 2 ) )
148 mg_per_field = number;
149
150 if ( !project->field ) {
151 for ( auto fn: files ) {
152 if ( verbose )
153 cout << fn << endl;
154 mgid = Bstring(fn).base();
155 if ( nmg%mg_per_field == 0 ) {
156 mg = NULL;
157 field = field_add(&field, mgid);
158 if ( !project->field ) project->field = field; // First field
159 }
160 mg = micrograph_add(&mg, mgid);
161 if ( !field->mg ) field->mg = mg;
162 }
163 }
164
165 for ( i=0, field = project->field; field; field = field->next ) {
166 for ( mg = field->mg; mg; mg = mg->next, ++i ) {
167 if ( i >= files.size() ) break;
168 if ( verbose )
169 cout << i << tab << files[i] << endl;
170 Bimage* p = read_img(files[i], 0, -1);
171 if ( !p ) {
172 cerr << "File " << files[i] << " not found!" << endl;
173 } else {
174 if ( ftype == 1 ) {
175 mg->fframe = files[i];
176 mg->pixel_size = p->image->sampling();
177 mg->origin = p->image->origin();
178 } else if ( ftype == 2 ) {
179 mg->fmg = files[i];
180 mg->pixel_size = p->image->sampling();
181 mg->origin = p->image->origin();
182 } else if ( ftype == 3 ) {
183 mg->fpart = files[i];
184 mg->box_size = p->size();
185 for ( j=0; j<p->images(); ++j ) {
186 part = particle_add(&part, j+1);
187 if ( !mg->part ) mg->part = part;
188 part->pixel_size = p->image[j].sampling();
189 part->ori = p->image[j].origin();
190 part->view = p->image[j].view();
191 }
192 } else if ( ftype == 4 ) {
193 mg->fps = files[i];
194 }
195 delete p;
196 }
197 nmg++;
198 }
199 }
200
201 if ( verbose )
202 cout << nmg << " micrographs processed" << endl << endl;
203
204 return nmg;
205}

207Bproject* generate_project_from_files(Bstring& framepath, Bstring& mgpath, Bstring& partpath, Bstring& pspath, long mg_per_field, int flags)
208{
209 Bstring frameext = path_and_extension(framepath);
210 Bstring mgext = path_and_extension(mgpath);
211 Bstring partext = path_and_extension(partpath);
212 Bstring psext = path_and_extension(pspath);
213//cout << "-" << framepath << "-" << frameext << "-" << endl;
214 vector<string> frame_list = file_list(framepath.str(), frameext.str());
215//cout << "-" << mgpath << "-" << mgext << "-" << endl;
216 vector<string> mg_list = file_list(mgpath.str(), mgext.str());
217 vector<string> part_list = file_list(partpath.str(), partext.str());
218 vector<string> ps_list = file_list(pspath.str(), psext.str());
219
220 if ( verbose ) {
221 cout << "Creating file references:" << endl;
222 if ( frame_list.size() )
223 cout << "Movie references: " << framepath
224 << " (" << frameext << ", " << frame_list.size() << ")" << endl;
225 if ( mg_list.size() )
226 cout << "Micrograph references: " << mgpath
227 << " (" << mgext << ", " << mg_list.size() << ")" << endl;
228 if ( part_list.size() )
229 cout << "Particle references: " << partpath
230 << " (" << partext << ", " << part_list.size() << ")" << endl;
231 if ( ps_list.size() )
232 cout << "Power spectra references: " << pspath
233 << " (" << psext << ", " << ps_list.size() << ")" << endl;
234 }
235
236 // Ckecks
237 long err(0);
238 if ( frame_list.size() && mg_list.size() && frame_list.size() != mg_list.size() ) {
239 cerr << "The number of movie frame files and micrograph files must be the same!" << endl;
240 err--;
241 }
242 if ( mg_list.size() && ps_list.size() && ps_list.size() != mg_list.size() ) {
243 cerr << "The number ofpower spectrum files and micrograph files must be the same!" << endl;
244 err--;
245 }
246
247 if ( err ) {
248 cerr << "Too many errors! Abort" << endl;
249 bexit(-1);
250 }
251
252 Bproject* project = new Bproject;
253
254 if ( frame_list.size() ) project_add_files(project, frame_list, 1, mg_per_field);
255
256 if ( mg_list.size() ) project_add_files(project, mg_list, 2, mg_per_field);
257
258 if ( part_list.size() ) project_add_files(project, part_list, 3, mg_per_field);
259
260 if ( ps_list.size() ) project_add_files(project, ps_list, 4, mg_per_field);
261
262
263 return project;
264}
