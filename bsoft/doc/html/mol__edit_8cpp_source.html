254int molgroup_insert(Bmolgroup* molgroup, Bmolgroup* molinsert, double distance)
255{
256 if ( distance <= 0 ) distance = 2; // Default
257 if ( distance < 1 ) distance = 1; // Limits on cutoff distance in angstrom
258 if ( distance > 5 ) distance = 5;
259
260 long i;
261 int delete_res, nresdel(0), nmoldel(0);
262 long ii, x, y, z, xx, yy, zz, ix, iy, iz;
263 Vector3<double> sampling(distance, distance, distance);
264 Bmolecule* mol, *prevmol;
265 Bresidue* res, *prevres;
266 Batom* atom;
267 Vector3<double> box = molgroup->box;
268 Vector3<int> gridsize((int) (box[0]/sampling[0] + 0.001),
269 (int) (box[1]/sampling[1] + 0.001), (int) (box[2]/sampling[2] + 0.001));
270 gridsize = gridsize.max(1);
271 for ( i=0; i<3; i++ ) sampling[i] = box[i]/gridsize[i] + 0.001;
272 long gridvol = (long) gridsize.volume();
273
274 if ( verbose )
275 cout << "Inserting a molecule group and deleting overlapping residues" << endl;
276
277 if ( verbose & VERB_PROCESS )
278 cout << "Distance cutoff: " << distance << " A" << endl;
279
280 Latom* latom;
281 Latom** alist = molgroup_atom_mesh_lists(molinsert, gridsize, sampling);
282
283 // Find the atoms under the footprint to be deleted
284 for ( mol = prevmol = molgroup->mol; mol; ) {
285 for( res = prevres = mol->res; res; ) {
286 delete_res = 0;
287 for ( atom = res->atom; atom; atom = atom->next ) {
288 x = (long) ((atom->coord[0] - molgroup->min[0])/sampling[0]);
289 y = (long) ((atom->coord[1] - molgroup->min[1])/sampling[1]);
290 z = (long) ((atom->coord[2] - molgroup->min[2])/sampling[2]);
291 if ( x >=0 && x < gridsize[0] && y >= 0 && y < gridsize[1] && z >= 0 && z < gridsize[2] ) {
292 i = (z*gridsize[1] + y)*gridsize[0] + x;
293 for ( zz=z-1; zz<=z+1; zz++ ) {
294 iz = zz;
295 if ( iz < 0 ) iz += gridsize[2];
296 if ( iz >= gridsize[2]) iz -= gridsize[2];
297 for ( yy=y-1; yy<=y+1; yy++ ) {
298 iy = yy;
299 if ( iy < 0 ) iy += gridsize[1];
300 if ( iy >= gridsize[1] ) iy -= gridsize[1];
301 for ( xx=x-1; xx<=x+1; xx++ ) {
302 ix = xx;
303 if ( ix < 0 ) ix += gridsize[0];
304 if ( ix >= gridsize[0] ) ix -= gridsize[0];
305 ii = (iz*gridsize[1] + iy)*gridsize[0] + ix;
306 for ( latom = alist[ii]; latom; latom = latom->next ) {
307 if ( atom->coord.distance(latom->atom->coord) < distance )
308 delete_res = 1;
309 }
310 }
311 }
312 }
313 }
314 }
315 if ( delete_res ) {
316 if ( verbose & VERB_FULL )
317 cout << "Removing a residue" << endl;
318 if ( res == mol->res ) {
319 mol->res = prevres = res->next;
320 residue_kill(res);
321 res = mol->res;
322 } else {
323 prevres->next = res->next;
324 residue_kill(res);
325 res = prevres->next;
326 }
327 nresdel++;
328 } else {
329 prevres = res;
330 if ( res ) res = res->next;
331 }
332 }
333 if ( !mol->res ) {
334 if ( verbose & VERB_FULL )
335 cout << "Removing a molecule" << endl;
336 if ( molgroup->mol == mol ) {
337 molgroup->mol = prevmol = mol->next;
338 molecule_kill(mol);
339 mol = molgroup->mol;
340 } else {
341 prevmol->next = mol->next;
342 molecule_kill(mol);
343 mol = prevmol->next;
344 }
345 nmoldel++;
346 } else {
347 prevmol = mol;
348 if ( mol ) mol = mol->next;
349 }
350 }
351
352 for ( i=0; i<gridvol; i++ ) kill_list((char *) alist[i], sizeof(Latom));
353 delete[] alist;
354
355 if ( verbose & VERB_PROCESS )
356 cout << "Molecules and residues deleted: " << nmoldel << " " << nresdel << endl;
357
358 // Add the new molecules
359 if ( molgroup->mol ) {
360 for ( mol = molgroup->mol; mol->next; mol = mol->next ) ;
361 mol->next = molinsert->mol;
362 } else molgroup->mol = molinsert->mol;
363 molinsert->mol = NULL;
364
365 if ( verbose & VERB_PROCESS )
366 molgroup_density(molgroup) ;
367
368 molgroup_stats(molgroup);
369
370 return 0;
371}
