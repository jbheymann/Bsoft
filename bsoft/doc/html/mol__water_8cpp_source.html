26Bmolecule* mol_generate_one_water(Bmolecule** mollist, char* watername, Vector3<double> Ocoord)
27{
28 double rand_max = 1.0*get_rand_max();

30 double HOHangle = 109.47*M_PI/180.0;
31 char resname[12] = "HOH";
32 char atomname[12] = "O";
33 Vector3<double> H1vec, H2vec;
34 Matrix3 mat(1);
35
36 Bmolecule* mol = molecule_add(mollist, watername);
37 Bresidue* res = residue_add(&mol->res, resname);
38 res->num = 1;
39
40 strcpy(atomname, "O");
41 Batom* atom = atom_add(&res->atom, atomname);
42 atom->el[0] = 'O';
43 atom->num = 1;
44 atom->mass = 16;
45 atom->chrg = -0.834;
46 atom->coord = Ocoord;
47
48 strcpy(atomname, "HO1");
49 atom = atom_add(&atom, atomname);
50 atom->el[0] = 'H';
51 atom->num = 2;
52 atom->mass = 1;
53 atom->chrg = 0.417;
54 H1vec[0] = random()/rand_max - 0.5;
55 H1vec[1] = random()/rand_max - 0.5;
56 H1vec[2] = random()/rand_max - 0.5;
57 H1vec.normalize();
58 atom->coord = (H1vec * OHbond) + Ocoord;
59
60 strcpy(atomname, "HO2");
61 atom = atom_add(&atom, atomname);
62 atom->el[0] = 'H';
63 atom->num = 3;
64 atom->mass = 1;
65 atom->chrg = 0.417;
66 H2vec[0] = random()/rand_max - 0.5;
67 H2vec[1] = random()/rand_max - 0.5;
68 H2vec[2] = random()/rand_max - 0.5;
69 H2vec.normalize();
70
71 H2vec = H1vec.cross(H2vec);
72 mat = Matrix3(H2vec, HOHangle);
73 H2vec = mat * H1vec;
74 atom->coord = (H2vec * OHbond) + Ocoord;
