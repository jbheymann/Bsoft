269CTFparam* ctf_from_xml(xmlNodePtr node)
270{
271 if ( !xml_check_for_node(node, CTF_VOLTAGE) ) return NULL;
272
273 Bstring s;
274 vector<double> v;
275 CTFparam* ctf = new CTFparam;
276
277 ctf->identifier(xml_get_string(node, CTF_ID));
278 if ( xml_find_node(node, CTF_SELECT) )
279 ctf->select(xml_get_integer(node, CTF_SELECT));
280 if ( xml_find_node(node, CTF_FOM) )
281 ctf->fom(xml_get_real(node, CTF_FOM));
282 ctf->volt(xml_get_real(node, CTF_VOLTAGE));
283 ctf->alpha(xml_get_real(node, CTF_ALPHA));
284 ctf->dE(xml_get_real(node, CTF_DE));
285 ctf->Cc(xml_get_real(node, CTF_CC));
286
287 s = xml_copy_string(node, CTF_BASELINE);
288 ctf->parse_baseline_equation(s);
289 s = xml_copy_string(node, CTF_ENVELOPE);
290 ctf->parse_envelope_equation(s);
291
292 if ( xml_find_node(node, CTF_AMP_SHIFT) )
293 ctf->amp_shift(xml_get_real(node, CTF_AMP_SHIFT));
294 else
295 ctf->amp_shift(asin(xml_get_real(node, CTF_AMP)));
296 ctf->beam_tiltX(xml_get_real(node, CTF_TILT_X));
297 ctf->beam_tiltY(xml_get_real(node, CTF_TILT_Y));
298 ctf->defocus_average(xml_get_real(node, CTF_DEF_AVG));
299// ctf->defocus_deviation(xml_get_real(node, CTF_DEF_DEV));
300// ctf->astigmatism_angle(xml_get_real(node, CTF_AST_ANG) * M_PI/180.0);
301 ctf->astigmatism(xml_get_real(node, CTF_DEF_DEV), xml_get_real(node, CTF_AST_ANG) * M_PI/180.0);
302 ctf->Cs(xml_get_real(node, CTF_CS));
303
304 if ( xml_find_node(node, CTF_ABERRATION_ODD) ) {
305 v = parse_real_vector(xml_get_string(node, CTF_ABERRATION_ODD).substr(1));
306 ctf->aberration_odd_update(v);
307 }
308 if ( xml_find_node(node, CTF_ABERRATION_EVEN) ) {
309 v = parse_real_vector(xml_get_string(node, CTF_ABERRATION_EVEN).substr(1));
310 ctf->aberration_even_update(v);
311 }
312 ctf->zero(1);
313
314 long n, m;
315 xmlNodePtr ab_node;
316 for ( ab_node = node->xmlChildrenNode; ab_node; ab_node = ab_node->next ) {
317// cout << ab_node->name << endl;
318 if ( !xmlStrcmp(ab_node->name, BAD_CAST CTF_ABERRATION) ) {
319 n = xml_get_integer_attribute(ab_node, "n");
320 m = xml_get_integer_attribute(ab_node, "m");
321 ctf->aberration_coefficient(n, m, xml_get_real(ab_node));
322 }
323 }
324
325 return ctf;
326}

347{
348 int f, euler_flag, omega_flag;
349 Bstring tag;
350 xmlChar* part_id;
351 Bparticle* part = NULL;
352 Euler euler;
353

365 part->loc[0] = xml_get_real(node, PARTICLE_X);

367 part->loc[2] = xml_get_real(node, PARTICLE_Z);
368 part->pixel_size[0] = xml_get_real(node, PARTICLE_PIXEL_X);
369 part->pixel_size[1] = xml_get_real(node, PARTICLE_PIXEL_Y);
370 part->pixel_size[2] = xml_get_real(node, PARTICLE_PIXEL_Z);
371 part->ori[0] = xml_get_real(node, PARTICLE_ORIGIN_X);
372 part->ori[1] = xml_get_real(node, PARTICLE_ORIGIN_Y);
373 part->ori[2] = xml_get_real(node, PARTICLE_ORIGIN_Z);
374 euler_flag = xml_check_for_node(node, PARTICLE_THETA);
375 if ( euler_flag == 0 ) {
376 part->view[0] = xml_get_real(node, PARTICLE_VIEW_X);
377 part->view[1] = xml_get_real(node, PARTICLE_VIEW_Y);
378 part->view[2] = xml_get_real(node, PARTICLE_VIEW_Z);
379 part->view[3] = xml_get_real(node, PARTICLE_VIEW_ANGLE)*M_PI/180.0;
380 } else {
381 omega_flag = xml_check_for_node(node, PARTICLE_OMEGA);
382 if ( omega_flag )
383 euler[0] = -xml_get_real(node, PARTICLE_OMEGA)*M_PI/180.0;
384 else
385 euler[0] = xml_get_real(node, PARTICLE_PSI)*M_PI/180.0;
386 euler[1] = xml_get_real(node, PARTICLE_THETA)*M_PI/180.0;
387 euler[2] = xml_get_real(node, PARTICLE_PHI)*M_PI/180.0;
388 part->view = euler.view();
389 }
390 for ( f=0; f<NFOM; f++ ) if ( fom_tag[f] ) {
391 tag = get_fom_tag(fom_tag[f]);
392 part->fom[f] = xml_get_real(node, tag.c_str());
393 }
394 part->sel = xml_get_integer(node, PARTICLE_SELECT);
395
396 return part;
397}

493/*
494@brief Converts micrograph data from a XML document to a project structure.
495@param xmlDocPtr doc XML document pointer.
496@param *project project structure.
497@param &filename file name to resolve paths.
498@return int error code (<0 means failure).
499
500 The function sets up the project hierarchy from a XML file.
501 All angles given in the XML document are assumed to be in degrees,
502 and are converted here only to radians.
503
504**/
505int xml_to_project(xmlDocPtr doc, Bproject* project, Bstring& filename)
506{
507 int err(0);
508
509 Bfield* field = project->field;
510 Bmicrograph* mg = NULL;
511 Breconstruction* rec = NULL;
512
513 int i, f;
514 xmlChar *xstring, *id;
515 Bstring bid;
516 Bstring s;
517
518 xmlNodePtr project_node = xmlDocGetRootElement(doc);
519 xmlNodePtr child_node = NULL;
520 xmlNodePtr mg_node = NULL;
521 xmlNodePtr node = NULL;
522
523 // Make sure FOM tags are properly set up
524 if ( xml_check_for_node_in_tree(project_node, PARTICLE_FOM) ) project->fom_tag[0] = FOM;
525 if ( xml_check_for_node_in_tree(project_node, PARTICLE_FOM_CV) ) project->fom_tag[1] = FOM_CV;
526 if ( xml_check_for_node_in_tree(project_node, PARTICLE_HANDA_FOM) ) project->fom_tag[1] = FOM_HAND_A;
527 if ( xml_check_for_node_in_tree(project_node, PARTICLE_HANDB_FOM) ) project->fom_tag[2] = FOM_HAND_B;
528
529 for ( f=i=0; f<NFOM; f++ ) i += project->fom_tag[f];
530 if ( i < 2 ) project->fom_tag[0] = FOM;
531
532 for ( i=0, child_node = project_node->xmlChildrenNode; child_node; child_node = child_node->next ) {
533 if ( project->comment.empty() ) {
534 if ( !xmlStrcmp(child_node->name, BAD_CAST COMMENT) )
535 project->comment = (char *) xmlNodeGetContent(child_node);
536 }
537 if ( !project->reference ) {
538 if ( !xmlStrcmp(child_node->name, BAD_CAST MAP_REFERENCE) ) {
539 xstring = xmlNodeGetContent(child_node);
540 string_add(&project->reference, (char *) xstring);
541 xmlFree(xstring);
542 }
543 }
544 if ( !xmlStrcmp(child_node->name, BAD_CAST FIELD) ) {
545 id = xmlGetProp(child_node, BAD_CAST ID);
546 bid = (char *) id;
547 for ( field=project->field; field; field=field->next )
548 if ( field->id == bid ) break;
549 if ( !field ) field = field_add(&project->field, bid);
550 xmlFree(id);
551 for ( mg_node = child_node->xmlChildrenNode; mg_node; mg_node = mg_node->next ) {
552 if ( !xmlStrcmp(mg_node->name, BAD_CAST MICROGRAPH) ) {
553 id = xmlGetProp(mg_node, BAD_CAST ID);
554 bid = (char *) id;
555 mg = micrograph_add(&field->mg, bid);
556 xmlFree(id);
557 mg->block = i++;
558 mg->fmg = xml_get_string(mg_node, MICROGRAPH_FILE);
559 mg->fframe = xml_get_string(mg_node, MICROGRAPH_FRAMES_FILE);
560 mg->fpart = xml_get_string(mg_node, PARTICLE_FILE);
561 mg->ffil = xml_get_string(mg_node, FILAMENT_FILE);
562 mg->fft = xml_get_string(mg_node, MICROGRAPH_TRANSFORM_FILE);
563 mg->fps = xml_get_string(mg_node, MICROGRAPH_POWERSPEC_FILE);
564 mg->img_num = xml_get_integer(mg_node, MICROGRAPH_NUMBER);
565 mg->select = xml_get_integer(mg_node, MICROGRAPH_SELECT);
566 mg->fom = xml_get_real(mg_node, MICROGRAPH_FOM);
567 mg->magnification = xml_get_real(mg_node, MICROGRAPH_MAGNIFICATION);
568 mg->sampling = xml_get_real(mg_node, MICROGRAPH_SAMPLING);
569 mg->pixel_size[0] = xml_get_real(mg_node, MICROGRAPH_PIXEL_X);
570 mg->pixel_size[1] = xml_get_real(mg_node, MICROGRAPH_PIXEL_Y);
571 mg->frame_pixel_size[0] = xml_get_real(mg_node, MICROGRAPH_FRAME_PIXEL_X);
572 mg->frame_pixel_size[1] = xml_get_real(mg_node, MICROGRAPH_FRAME_PIXEL_Y);
573 mg->dose = xml_get_real(mg_node, MICROGRAPH_DOSE);
574 mg->exposure = xml_get_real(mg_node, MICROGRAPH_EXPOSURE);
575 mg->intensity = xml_get_real(mg_node, MICROGRAPH_INTENSITY);
576 mg->wri = xml_get_real(mg_node, MICROGRAPH_WATER_RING);
577 mg->tilt_axis = xml_get_real(mg_node, MICROGRAPH_TILT_AXIS);
578 mg->tilt_angle = xml_get_real(mg_node, MICROGRAPH_TILT_ANGLE);
579 mg->rot_angle = xml_get_real(mg_node, MICROGRAPH_ROT_ANGLE);
580 mg->origin[0] = xml_get_real(mg_node, MICROGRAPH_ORIGIN_X);
581 mg->origin[1] = xml_get_real(mg_node, MICROGRAPH_ORIGIN_Y);
582 mg->origin[2] = xml_get_real(mg_node, MICROGRAPH_ORIGIN_Z);
