658*/
659double img_recip_space_fom(Bimage* p, Bimage* pref, double hi_res, double lo_res,
660 int fom_type, vector<double>& weight, int flags, CTFparam* em_ctf)
661{
662 bool apply_ctf(flags & APPLY_CTF);
663 bool invert(flags & INVERT);
664
665 if ( pref->sampling(0)[0] < 0.01 ) pref->sampling(1,1,1);
666 if ( lo_res > 0 && hi_res > lo_res ) swap(hi_res, lo_res);
667 pref->check_resolution(hi_res);
668
669 double max_rad = pref->real_size()[0]/hi_res;
670 double max_rad_cv = 1.1*max_rad;
671 if ( max_rad_cv <= max_rad ) max_rad_cv = max_rad + 1;
672
673 double max_rad_sq = max_rad*max_rad;
674 double max_rad_cv_sq = max_rad_cv*max_rad_cv;
675 double min_rad_sq(0);
676 if ( lo_res > 0 ) {
677 min_rad_sq = pref->real_size()[0]/lo_res;
678 min_rad_sq *= min_rad_sq;
679 }
680
681 if ( verbose & VERB_DEBUG )
682 cout << "DEBUG img_recip_space_fom: max_rad=" << max_rad << " (" << max_rad_sq << ")" << endl;
683
684 double dphi, ctf_fac;
685 if ( !em_ctf ) apply_ctf = 0;
686 else if ( em_ctf->defocus_average() < 1 ) apply_ctf = 0;
687
688 long i, j, xx, yy;
689 long hx = (p->sizeX() - 1)/2, hy = (p->sizeY() - 1)/2;
690 double sx2, sy2, s2, d2, amp, denom, f, cv, wsum;
691 Vector3<double> m, d, iv;
692 Vector3<double> freq_scale(1.0/p->real_size()[0], 1.0/p->real_size()[1], 1.0);
693// cout << "size = " << p->size() << endl;
694// cout << "sampling = " << p->image->sampling() << endl;
695// cout << "real size = " << p->real_size() << endl;
696// bexit(-1);
697
698// double* w = new double[p->sizeX()];
699// for ( j=0; j<p->sizeX(); j++ ) w[j] = 1;
700// if ( weight.size() )
701// for ( size_t j=0; j<weight.size(); j++ ) if ( weight[j] ) w[j] = weight[j];
702
703// double* fom = new double[p->sizeX()];
704// double* dsum = new double[p->sizeX()];
705// double* msum = new double[p->sizeX()];
706// for ( i=0; i<p->sizeX(); i++ ) fom[i] = dsum[i] = msum[i] = 0;
707
708 vector<double> w(p->sizeX(), 1);
709 if ( weight.size() )
710 for ( j=0; j<weight.size(); ++j ) if ( weight[j] ) w[j] = weight[j];
711
712 vector<double> fom(p->sizeX(),0);
713 vector<double> dsum(p->sizeX(),0);
714 vector<double> msum(p->sizeX(),0);
715
716 Complex<float>* data = (Complex<float> *) p->data_pointer();
717 Complex<float>* refdata = (Complex<float> *) pref->data_pointer();
718 Complex<float> dtemp;
719
720 for ( yy=i=0; yy<p->sizeY(); yy++ ) {
721 iv[1] = yy;
722 if ( iv[1] > hy ) iv[1] -= p->sizeY();
723 sy2 = freq_scale[1]*iv[1];
724 sy2 *= sy2;
725 for ( xx=0; xx<p->sizeX(); xx++, i++ ) {
726 iv[0] = xx;
727 if ( iv[0] > hx ) iv[0] -= p->sizeX();
728 sx2 = freq_scale[0]*iv[0];
729 sx2 *= sx2;
730 d2 = iv.length2();
731 if ( d2 >= min_rad_sq && d2 <= max_rad_cv_sq ) {
732 s2 = sx2 + sy2;
733 j = (long) sqrt(d2);
734 dtemp = refdata[i];
735 if ( apply_ctf ) {
736 ctf_fac = em_ctf->calculate(s2, atan2(iv[1],iv[0]));
737 if ( invert ) ctf_fac = -ctf_fac;
738 dtemp *= ctf_fac;
739 }
740 if ( fom_type < 1 ) {
741 fom[j] += data[i].real()*dtemp.real() + data[i].imag()*dtemp.imag();
742 msum[j] += dtemp.power();
743 dsum[j] += data[i].power();
744 } else {
745 amp = data[i].amp() + dtemp.amp();
746 dphi = angle_set_negPI_to_PI(data[i].phi() - dtemp.phi());
747 fom[j] += amp*dphi*dphi;
748 msum[j] += amp;
749 }
750 }
751 }
752 }
753
754 if ( fom_type < 1 ) {
755 for ( f=wsum=0, j=0; j<max_rad; j++ ) {
756 wsum += w[j];
757 denom = dsum[j]*msum[j];
758 if ( denom > 0 ) f += w[j]*fom[j]/sqrt(denom);
759 }
760 f /= wsum;
761 for ( cv=wsum=0, j=(long)max_rad; j<max_rad_cv; j++ ) {
762 wsum += w[j];
763 denom = dsum[j]*msum[j];
764 if ( denom > 0 ) cv += w[j]*fom[j]/sqrt(denom);
765 }
766 cv /= wsum;
767 } else {
768 for ( f=wsum=0, j=0; j<max_rad; j++ ) {
769 wsum += w[j];
770 if ( msum[j] ) f += w[j]*cos(sqrt(fom[j]/msum[j]));
771 }
772 f /= wsum;
773 for ( cv=wsum=0, j=(long)max_rad; j<max_rad_cv; j++ ) {
774 wsum += w[j];
775 if ( msum[j] ) cv += w[j]*cos(sqrt(fom[j]/msum[j]));
776 }
777 cv /= wsum;
778 }
779
780// delete[] fom;
781// delete[] dsum;
782// delete[] msum;
783// delete[] w;
784
785 p->image->FOM(cv);
786
787 return f;
788}
