226long system_memory()
227{
228#if defined(CTL_HW) && (defined(HW_MEMSIZE) || defined(HW_PHYSMEM64))
229
230 int mib[2];
231 mib[0] = CTL_HW;
232#if defined(HW_MEMSIZE)
233 mib[1] = HW_MEMSIZE; // OSX
234#elif defined(HW_PHYSMEM64)
235 mib[1] = HW_PHYSMEM64; // NetBSD, OpenBSD
236#endif
237 u_int namelen = sizeof(mib) / sizeof(mib[0]);
238 long size(0);
239 size_t len = sizeof(size);
240
241 if (sysctl(mib, namelen, &size, &len, NULL, 0) == 0)
242 return size;
243
244#elif defined(_SC_PHYS_PAGES)
245
246#if defined(_SC_PAGESIZE) // FreeBSD, Linux, OpenBSD, and Solaris
247 return (long)sysconf( _SC_PHYS_PAGES ) *
248 (long)sysconf( _SC_PAGESIZE );
249
250#elif defined(_SC_PAGE_SIZE)
251 return (long)sysconf( _SC_PHYS_PAGES ) *
252 (long)sysconf( _SC_PAGE_SIZE );
253#endif
254
255#elif defined(_SC_AIX_REALMEM) // AIX
256 return (long)sysconf( _SC_AIX_REALMEM ) * (long)1024L;
257
258#endif
259
260 perror("system_memory: sysctl error");
261
262 return 0;
263}

562void vax2ieee(unsigned char* v, int sb)
563{
564 unsigned char t[4];
565 int i, shift, expnt;
566
567 if ( sb ) {
568 t[0] = v[2]; // Swap byte order
569 t[1] = v[3];
570 t[2] = v[0];
571 t[3] = v[1];
572 } else {
573 t[0] = v[1]; // Swap for VAX format
574 t[1] = v[0];
575 t[2] = v[3];
576 t[3] = v[2];
577 }
578 for ( i=0; i<4; i++ ) v[i] = t[i];
579
580 expnt = (v[0] << 1) | (v[1] >> 7); // Get exponent
581 if ( expnt == 0 )
582 if ( v[0] == 0 ) // Special case zero
583 memset(t, '\0', 4);
584 else { // Special case infinity or NaN
585 t[0] = 0xff;
586 t[1] = v[1] | 0x80;
587 }
588 else if ( expnt > 2 ) // Normalized value
589 t[0] = v[0] - 0x01; // Subtract 2 from the exponent
590 else { // Denormalized value
591 shift = 3 - expnt;
592 t[0] = v[0] & 0x80; // Keep sign, expnt = 0
593 t[1] = ( (v[1] & 0x7f) >> shift ) | ( 0x10 << expnt );
594 t[2] = ( v[1] << ( 8-shift ) ) | ( v[2] >> shift );
595 t[3] = ( v[2] << ( 8-shift ) ) | ( v[3] >> shift );
596 }
597 for ( i=0; i<4; i++ ) v[i] = t[i];
598}
