76 if ( verbose )

78
79 for ( ; field; field = field->next ) {

442 cerr << "Error: Not able to write " << filename << endl;

448 Bmicrograph* mg;

562int project_check(Bproject* project, int flags)
563{
564 long i, j, num_img_files(0), nwarn(0);
565// Vector3<double> d, ori;
566 double d;
567 Bfield* field;
568 Bmicrograph* mg;
569 Breconstruction* rec;
570 Bparticle* part;
571 Bbadarea* bad;
572 Bstring part_file;
573
574 if ( verbose & VERB_DEBUG )
575 cout << "DEBUG project_check: " << endl;
576
577 if ( project->field == NULL && project->rec ) project->select = 1;
578
579 for ( field=project->field; field; field=field->next ) {
580 i = 1;
581 for ( mg=field->mg; mg; mg=mg->next ) {
582 if ( verbose & VERB_DEBUG )
583 cout << "DEBUG project_check: mg=" << mg->id << endl;
584 if ( mg->next ) {
585 if ( mg->id == mg->next->id ) {
586 mg->id = mg->id + Bstring(i++, "_%03d");
587 if ( verbose )
588 cout << "Resetting micrograph ID: " << mg->id << endl;
589 if ( !mg->next->next ) {
590 mg->next->id = mg->next->id + Bstring(i, "_%03d");
591 if ( verbose )
592 cout << "Resetting micrograph ID: " << mg->next->id << endl;
593 }
594 }
595 }
596 if ( mg->fmg.length() ) num_img_files++;
597 if ( mg->fframe.length() ) num_img_files++;
598 if ( mg->fpart.length() ) num_img_files++;
599 if ( mg->ffil.length() ) num_img_files++;
600 if ( mg->fps.length() ) num_img_files++;
601 if ( num_img_files < 1 )
602 cerr << "Warning: No image files referenced for micrograph " << mg->id << endl;
603 if ( mg->sampling < 1e3 ) mg->sampling *= 1e4;
604 if ( mg->magnification < 1e3 ) mg->magnification *= 1e3;
605 mg->frame_pixel_size[2] = 1;
606 if ( mg->frame_pixel_size[0] < 0.01 || mg->frame_pixel_size[1] < 0.01 ) {
607 mg->frame_pixel_size = Vector3<double>(1,1,1);
608 if ( mg->magnification && mg->sampling ) {
609 mg->frame_pixel_size[0] = mg->sampling/mg->magnification;
610 if ( mg->frame_pixel_size[0] < 0.01 ) mg->frame_pixel_size[0] *= 1e4; // Assume um
611 mg->frame_pixel_size[1] = mg->frame_pixel_size[0];
612 }
613 if ( verbose && nwarn < 10 ) {
614 cerr << "Warning: Resetting frame pixel size for micrograph " <<
615 mg->id << " to " << mg->frame_pixel_size << endl;
616 nwarn++;
617 } else if ( nwarn == 10 ) {
618 cerr << "Warning: ..." << endl;
619 nwarn++;
620 }
621 }
622 mg->pixel_size[2] = 1;
623 if ( mg->pixel_size[0] < 0.01 || mg->pixel_size[1] < 0.01 ) {
624 if ( mg->frame_pixel_size.volume() ) {
625 mg->pixel_size = mg->frame_pixel_size;
626 } else {
627 mg->pixel_size = Vector3<double>(1,1,1);
628 if ( mg->magnification && mg->sampling ) {
629 mg->pixel_size[0] = mg->sampling/mg->magnification;
630 if ( mg->pixel_size[0] < 0.01 ) mg->pixel_size[0] *= 1e4; // Assume um
631 mg->pixel_size[1] = mg->pixel_size[0];
632 }
633 cerr << "Warning: Resetting pixel size for micrograph " <<
634 mg->id << " to " << mg->pixel_size << endl;
635 }
636 }
637 if ( flags & 32 && mg->intensity <= 0 ) micrograph_intensity(mg, flags);
638 if ( mg->scale[0] <= 0 ) mg->scale[0] = 1;
639 if ( mg->scale[1] <= 0 ) mg->scale[1] = 1;
640 if ( mg->scale[2] <= 0 ) mg->scale[2] = 1;
641 if ( mg->ctf ) {
642 if ( mg->ctf->volt() < 1e3 ) mg->ctf->volt(1e3*mg->ctf->volt());
643 if ( mg->ctf->volt() < 1e3 )
644 cerr << "Warning: The acceleration voltage must be specified!" << endl;
645 mg->ctf->zero(1);
646 }
647 if ( mg->part ) part_file = mg->part->fpart;
648 for ( part = mg->part; part; part = part->next ) {
649 if ( part->id < 1 )
650 cerr << "Error: Particle has an ID less than 1! (" << mg->id << ")" << endl;
651 part->mg = mg;
652 if ( part_file.length() && part_file != part->fpart ) part_file = 0;
653 if ( part->pixel_size[0] < 0.01 ) part->pixel_size = mg->pixel_size;
654 if ( part->pixel_size[1] < 0.01 ) part->pixel_size[1] = part->pixel_size[0];
655 part->pixel_size[2] = 1;
656 d = (part->ori - mg->box_size/2).length();
657 if ( d > mg->box_size[0]/4 )
658 part->ori = mg->box_size/2;
659 part->ori[2] = 0;
660 }
661 if ( part_file.length() ) {
662 mg->fpart = part_file;
663 for ( part = mg->part; part; part = part->next ) part->fpart = 0;
664 part_file = 0;
665 }
666 for ( j=0, bad = mg->bad; bad; bad = bad->next ) bad->id = --j;
667 }
668 }
669
670 if ( verbose & VERB_DEBUG )
671 cout << "DEBUG project_check: micrographs=" << project_count_micrographs(project) << endl;
672
673 i = 1;
674 for ( rec=project->rec; rec; rec=rec->next ) {
675 if ( rec->next ) {
676 if ( rec->id == rec->next->id ) {
677 rec->id = rec->id + Bstring(i++, "_%03d");
678 if ( verbose )
679 cout << "Resetting reconstruction ID: " << rec->id << endl;
680 if ( !rec->next->next ) {
681 rec->next->id = rec->next->id + Bstring(i, "_%03d");
682 if ( verbose )
683 cout << "Resetting reconstruction ID: " << rec->next->id << endl;
684 }
685 }
686 }
687 if ( rec->voxel_size[0] < 0.01 )
688 rec->voxel_size = Vector3<double>(1,1,1);
689 if ( rec->scale[0] <= 0 ) rec->scale[0] = 1;
690 if ( rec->scale[1] <= 0 ) rec->scale[1] = 1;
691 if ( rec->scale[2] <= 0 ) rec->scale[2] = 1;
692 for ( part = rec->part; part; part = part->next ) {
693 if ( part->id < 1 )
694 cerr << "Error: Particle has an ID less than 1! (" << mg->id << ")" << endl;
695 part->rec = rec;
696 if ( part->pixel_size[0] < 0.01 )
697 part->pixel_size = rec->voxel_size;
698 d = (part->ori - rec->box_size/2).length();
699 if ( d > rec->box_size[0]/4 )
700 part->ori = rec->box_size/2;
701 }
702 for ( j=0, bad = rec->bad; bad; bad = bad->next ) bad->id = --j;
703 }
704
705 if ( verbose & VERB_DEBUG )
706 cout << "DEBUG project_check: reconstructions=" << project_count_reconstructions(project) << endl;
707
708 field = project->field;
709 if ( ( flags & 1 ) && field ) {
710 if ( project_check_particles(project) < 0 ) bexit(-1);
711 }
712
713 project_update_first_zero(project);
714
715 return 0;
716}

871 for ( part = mg->part; part; part = part->next )
872 if ( max < part->sel ) max = part->sel;

875 for ( part = rec->part; part; part = part->next )
876 if ( max < part->sel ) max = part->sel;

886 mg->select = 0;
887 for ( part = mg->part; part; part = part->next ) if ( part->sel == n ) {
888 mg->select = 1;
889 c++;
890 }
891 }
892 }
893 for ( rec = project->rec; rec; rec = rec->next )
894 rec->select = 0;
895 } else {
896 for ( field = project->field; field; field = field->next )
897 for ( mg = field->mg; mg; mg = mg->next )
898 mg->select = 0;
899 for ( c=0, rec = project->rec; rec; rec = rec->next ) {
900 rec->select = 0;
901 for ( part = rec->part; part; part = part->next ) if ( part->sel == n ) {
902 rec->select = 1;
903 c++;
904 }
905 }
906 }
907 if ( c ) {
908 fn = filename.pre_rev('.') + Bstring(n, "_%04d.") + filename.post_rev('.');
909 if ( verbose )
910 cout << "Writing " << fn << " with " << c << " particles" << endl;
911 write_project(fn, project, 1, 1);
912 }
913 }
