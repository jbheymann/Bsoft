30Bimage* img_calc_wave_propagator(Vector3<long> size, Vector3<double> sam,
31 double thickness, double volt)
32{
33 Bimage* p = new Bimage(Float, TComplex, size[0], size[1], 1, 1);
34 p->sampling(sam);
35
36 long i, h, k, x, y;
37 double wavelength = electron_wavelength(volt);
38 double arg, sx2, sy2;
39 double afactor = M_PI*wavelength*thickness;
40// double nfactor = 1.0/(p->sizeX()*p->sizeY()); // Normalization for passing through FFT's
41 double xscale2 = 1.0/(p->sizeX()*p->sizeX()*sam[0]*sam[0]);
42 double yscale2 = 1.0/(p->sizeY()*p->sizeY()*sam[1]*sam[1]);
43
44 for ( i=y=0; y<p->sizeY(); ++y ) {
45 k = y;
46 if ( k > (p->sizeY()-1)/2 ) k -= p->sizeY();
47 sy2 = k*k*yscale2;
48 for ( x=0; x<p->sizeX(); ++x, ++i ) {
49 h = x;
50 if ( h > (p->sizeX()-1)/2 ) h -= p->sizeX();
51 sx2 = h*h*xscale2;
52 arg = afactor*(sx2 + sy2);
53// p->set(i, Complex<double>(cos(arg), sin(arg)) * nfactor);
54 p->set(i, Complex<double>(cos(arg), sin(arg)));
55 }
56 }
57
58// write_img("propagator.map", p);
59
60 return p;
61}

206 Vector3<double> sam, double thickness, double resolution, double Bfactor,

252

273int img_calc_phase_grating(Bimage* p, double volt, bool norm)
274{
275 p->simple_to_complex();
276
277 long i, j, n;
278 long imgsize = p->sizeX()*p->sizeY();
279 double arg;
280
281// double wavelength = electron_wavelength(volt);
282// double sigma = 0.0208886 * wavelength *
283// sqrt(1 + 5.8886579e-4/(wavelength*wavelength)); // According to Roar Kilaas
284 double sigma = 1e-10*TWOPI*ECHARGE/(PLANCK*LIGHTSPEED*beta(volt));
285
286// if ( verbose & VERB_FULL )
287 if ( verbose )
288 cout << "Sigma (interaction factor): " << sigma << endl;
289
290 if ( norm ) sigma /= p->images();
291
292 for ( i=n=0; n<p->images(); n++ ) {
293 for ( j=0; j<imgsize; i++, j++ ) {
294 arg = -sigma*(p->complex(i)).real();
295 p->set(i, Complex<double>(cos(arg), sin(arg)));
296 }
297 }
298
299 return 0;
