270int ps_marker_errors(ofstream* fps, Bmarker* set1, Bmarker* set2,
271 Transform t, Vector3<long> size, double err_scale)
272{
273 if ( !set1 || !set2 ) {
274 cerr << "Error: The two sets must have the same number of points!" << endl;
275 return -1;
276 }
277
278 long num(0);
279 double xdev(0), ydev(0);
280 Bmarker *m1, *m2;
281 Vector3<double> min = set1->loc;
282 Vector3<double> max = set1->loc;
283
284 //determine the deviation of every point in set2 from its predicted
285 //coordinates using the tranfsform parameters
286 for ( m1=set1, m2=set2; m1; m1=m1->next, m2=m2->next ) {
287 min = min.min(m1->loc);
288 max = max.max(m1->loc);
289 m2->err[0] = t.scale[0]*(m1->loc[0] *cos(t.angle) +
290 m1->loc[1]*sin(t.angle)) + t.trans[0] - m2->loc[0];
291 m2->err[1] = t.scale[1]*(-1* m1->loc[0] *sin(t.angle) +
292 m1->loc[1]*cos(t.angle)) + t.trans[1] - m2->loc[1];
293 xdev += m2->err[0]*m2->err[0];
294 ydev += m2->err[1]*m2->err[1];
295 num++;
296 }
297 xdev = sqrt(xdev/num);
298 ydev = sqrt(ydev/num);
299
300 long w = size[0];
301 long h = size[1];
302 double scale = 650.0/h;
303
304 if ( verbose & VERB_DEBUG )
305 cout << "DEBUG ps_marker_errors: xdev " << xdev << " ydev " << ydev << endl;
306
307 *fps << "/Helvetica-Bold findfont 10 scalefont setfont" << endl;
308 *fps << "40 60 moveto (Shift: " << t.trans[0] << " " << t.trans[1] << ") show" << endl;
309 *fps << "40 45 moveto (Scale: " << t.scale[0] << " " << t.scale[1] << ") show" << endl;
310 *fps << "40 30 moveto (Angle: " << t.angle*180.0/M_PI << ") show" << endl;
311 *fps << "40 15 moveto (Deviation: " << xdev << " " << ydev << " (scaled by " << err_scale << ")) show" << endl;
312 *fps << "/x_Origin 100 def \n/y_Origin 100 def \n/Height " << h << " def \n/Width " << w << " def" << endl;
313 *fps << "/Scale " << scale << " def" << endl;
314 *fps << "/err_scale " << err_scale << " def" << endl;
315 *fps << "/Frame { newpath 0 0 moveto 0 Height lineto Width Height lineto Width 0 lineto closepath } def" << endl;
316
317 *fps << "/Data [" << endl;
318 for ( m2=set2; m2; m2=m2->next )
319 *fps << m2->loc[0] << " " << m2->loc[1] << " " << m2->err[0] << " " << m2->err[1] << endl;
320 *fps << "] def\n" << endl;
321
322 *fps << "300 80 moveto (x axis) show" << endl;
323 *fps << "80 300 moveto gsave 90 rotate (y axis) show grestore" << endl;
