33int readEER(Bimage* p, int readdata, int img_select, int supres)
34{
35 if ( supres == 0 || supres == -1 ) supres = 1;
36
37 unsigned x, y, z, f;
38 ElectronCountedFramesDecompressor ecfd(p->file_name());
39
40 JSvalue mdata = json_from_xml(ecfd.GetAcquisitionMetadata().c_str());
41 if ( verbose & VERB_DEBUG_EER )
42 cout << mdata << endl;
43 double dose = mdata["totalDose"].real();
44 double exptime = mdata["exposureTime"].real();
45
46 (*p)["dose"] = dose;
47 (*p)["exposure"] = exptime;
48
49 ecfd.getSize(x,y,z);
50 f = ecfd.getNFrames();
51
52 if (supres > 0) { x *= supres; y *= supres; }
53 if (supres < 0) { x /= (-supres); y /= (-supres); }
54
55 if ( verbose & VERB_DEBUG ) {
56 cout << "DEBUG readEER: size: " << x << " " << y << " " << z << " " << f << endl;
57// cout << ecfd.GetAcquisitionMetadata() << endl;
58 }
59
60 p->size(x,y,1);
61 p->channels(1);
62 p->data_type(Float);
63
64 long nimg(f); // Read all frames
65 if ( img_select >= 0 ) nimg = 1; // Read one frame
66 else img_select = 0;
67 p->images(nimg);
68
69// p->information();
70 p->meta_data_update();
71
72 if ( !readdata ) return 0;
73
74 p->data_alloc();
75
76 long i, j, k(0), n, ne, nc(0), ncp(0);
77
78 if ( img_select >= 0 ) i = img_select;
79 for ( i=img_select; i<img_select+nimg; ++i ) {
80 vector<unsigned char> d(x*y, 0);
81 ecfd.decompressImage_AddTo(&d[0], supres, i);
82 if ( nimg == 1 ) k = 0; // Select one frames
83 for ( j=n=ne=0; j<x*y; ++j, ++k ) {
84 if ( d[j] ) {
85 ne += d[j];
86 p->add(k, d[j]);
87// cout << ne << "\t" << n << endl;
88 n = 0;
89 } else if ( n==127 ) {
90// cout << ne << "\t" << n << endl;
91 n = 1;
92 } else {
93 n++;
94 }
95 }
96 nc = ecfd.getNElectronsCounted();
97 dose = (nc-ncp)*1.0L/(x*y);
98 if ( nimg > 1 ) {
99 p->image[i].FOM(nc-ncp);
100 (*p)["image"][i]["electrons"] = nc-ncp;
101 (*p)["image"][i]["dose"] = dose;
102 (*p)["image"][i]["exposure"] = exptime/f;
103 }
104// cout << "Electrons counted: " << ne << "\t" << nc-ncp << "\t" << nc-ncp-ne << endl;
105 ncp = nc;
106 }
107
108 dose = nc*1.0L/(p->images()*x*y);
109 (*p)["electrons"] = nc;
110 (*p)["dose"] = dose;
111 (*p)["exposure"] = exptime;
112 if ( nimg == 1 ) {
113 p->image->FOM(nc);
114 (*p)["image"][0]["electrons"] = nc;
115 (*p)["image"][0]["dose"] = dose;
116 (*p)["image"][0]["exposure"] = exptime;
117 }
118
119 if ( verbose ) {
120 cout << "Electrons counted: " << nc << endl;
121 cout << "Electrons per pixel: " << dose << endl << endl;
122 }
123
124// for ( i=0; i<p->images(); ++i )
125// cout << i << tab << p->image[i].FOM() << endl;
126
127 return 0;

130vector<ElectronPos> read_eer_positions(string filename)
131{
132 unsigned x, y, z, f;
133 ElectronCountedFramesDecompressor ecfd(filename);
134
135 ecfd.getSize(x,y,z);
136 f = ecfd.getNFrames();
137
138 if ( verbose ) {
139 cout << "Size: " << x << " " << y << " " << z << " " << f << endl;
140// cout << ecfd.GetAcquisitionMetadata() << endl;
141 }
142
143 JSvalue mdata = json_from_xml(ecfd.GetAcquisitionMetadata().c_str());
144
145 long i, j, ne, nest, nep(0);
146 ElectronPos p0(0,0);
147 vector<ElectronPos> p;
148
149 for ( i=0; i<f; ++i ) {
150 nest = ecfd.nElectronFractionUpperLimit(i,i+1);
151// cout << "Estimated number of electrons: " << nest << endl;
152 vector<ElectronPos> p1(nest, p0);
153 ecfd.decompressCoordinateList((ElectronPos*)p1.data(), i);
154 ne = ecfd.getNElectronsCounted();
155 j = ne - nep;
156// cout << "Electrons counted: " << j << endl;
157// cout << p1[0].x << " " << p1[0].y << endl;
158// cout << pL1[j-1].x << " " << p1[j-1].y << endl;
159 nep = ne;
160 p.insert(p.end(), p1.begin(), p1.begin()+j);
161 }
162
163 if ( verbose ) {
164 cout << "Electrons counted: " << ne << endl;
165 cout << "Electrons per pixel: " << ne/(1.0*x*y*f) << endl;
166 }
167
168 return p;
169}
