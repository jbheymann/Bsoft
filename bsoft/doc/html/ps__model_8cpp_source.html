140int ps_model_fom_histogram(Bstring& filename, Bmodel* model)
141{
142 if ( !model ) return 0;

144 int nmod, i, j, nh(100), max(0);

151 if ( nmod < 2 )
152 cout << "Generating file " << filename << " with a FOM histogram" << endl << endl;

159 plot->title(title);
160
161 for ( i=0; i<=nh; i++ ) (*plot)[i] = i*1.0/nh;
162
163 for ( j=0, mp = model; mp; mp = mp->next ) if ( mp->select() ) {
164 j += plot->rows();
165 for ( comp = mp->comp; comp; comp = comp->next ) if ( comp->select() ) {
166 i = (long) (comp->FOM()*nh + 0.5);
167 if ( i < 0 ) i = 0;
168 if ( i > nh ) i = nh;
169 (*plot)[j+i] += 1;
170 }
171 }
172
173 for ( i=plot->rows(); i<plot->rows()*plot->columns(); i++ )
174 if ( max < (*plot)[i] ) max = (int) (*plot)[i];
175 max = 10*(max/10 + 1);
176
177 for ( i=0; i<nmod; i++ ) {
178 plot->page(i).title(title);
179 plot->page(i).columns(2);
180 plot->page(i).column(0).number(0);
181 plot->page(i).column(1).number(i+1);
182 plot->page(i).column(0).label("FOM");
183 plot->page(i).column(1).label("Count");
184 plot->page(i).column(0).axis(1);
185 plot->page(i).column(1).axis(3);
186 plot->page(i).column(1).type(1);
187 plot->page(i).column(1).element_size(0.5);
188 plot->page(i).axis(1).max(1);
189 plot->page(i).axis(3).max(max);
190 plot->page(i).axis(1).label("FOM");
191 plot->page(i).axis(3).label("Count");
192 }
193
194 ps_plot(filename, plot);
195
196 delete plot;
197
198 return 0;
