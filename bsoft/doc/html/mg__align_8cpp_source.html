98/*
99@brief Aligns the second image to the first.
100@param *p1 first image.
101@param *p2 second image (transformed)
102@param tile_size 3-valued vector for the size of sub-images.
103@param res_lo low resolution limit for cross-correlation.
104@param res_hi high resolution limit for cross-correlation.
105@param max_shift maximum shift allowed (default 1/4 of tile).
106@param filter_flag flag to filter micrograph extremes.
107@param refine_flag flag to turn on refinement of shift.
108@return Transform structure with shift, scale, rotation angle, and R factor.
109
110 The second image is transformed to fit on the first image.
111
112*/
113Transform img_align(Bimage* p1, Bimage* p2, Vector3<long> tile_size,
114 double res_lo, double res_hi, double max_shift, int filter_flag, int refine_flag)
115{
116 long i, iter, maxiter(5);
117 Vector3<double> shift;
118 Transform t, best_t;
119 best_t.fom = 1e37;
120
121 Vector3<long> start1, common_size, step_size;
122
123 // Find the largest area covering both images
124 common_size = p1->size().min(p2->size());
125 common_size = common_size.max(1);
126
127 // Ensure the tile fits into the images
128 tile_size = tile_size.max(1);
129 tile_size = tile_size.min(common_size);
130 if ( tile_size[0] >= common_size[0] && common_size[0] > 1 ) {
131 tile_size[0] = (int) (0.4*common_size[0]);
132// start1[0] = (common_size[0] - tile_size[0])/2;
133 }
134 if ( tile_size[1] >= common_size[1] && common_size[1] > 1 ) {
135 tile_size[1] = (int) (0.4*common_size[1]);
136// start1[1] = (common_size[1] - tile_size[1])/2;
137 }
138 if ( tile_size[2] >= common_size[2] && common_size[2] > 1 ) {
139 tile_size[2] = (int) (0.4*common_size[2]);
140// start1[2] = (common_size[2] - tile_size[2])/2;
141 }
142
143 if ( max_shift < 1 ) max_shift = tile_size[0]/4;
144
145 Vector3<double> half_tile;
146 half_tile[0] = (tile_size[0] - 1)/2;
147 half_tile[1] = (tile_size[1] - 1)/2;
148 half_tile[2] = (tile_size[2] - 1)/2;
149
150 Bimage* pex1 = NULL;
151 Bimage* pex2 = NULL;
152 Bmarker* set1 = NULL;
153 Bmarker* set2 = NULL;
154 Bmarker* m1 = NULL;
155 Bmarker* m2 = NULL;
156 Bstring fn1(p1->file_name()), fn2(p2->file_name());
157 Bstring filename(fn1.base() + "_" + fn2.base() + ".ps");
158
159 if ( verbose & VERB_LABEL ) {
160 cout << "Aligning images ";
161 if ( refine_flag ) cout << "with ";
162 else cout << "without ";
163 cout << "shift refinement." << endl << endl;
164 }
165
166 if ( common_size[0] > 2*tile_size[0] )
167 start1[0] = (common_size[0]
168 - (int) (tile_size[0]*(common_size[0]/tile_size[0])))/2;
169 if ( common_size[1] > 2*tile_size[1] )
170 start1[1] = (common_size[1]
171 - (int) (tile_size[1]*(common_size[1]/tile_size[1])))/2;
172 if ( common_size[2] > 2*tile_size[2] )
173 start1[2] = (common_size[2]
174 - (int) (tile_size[2]*(common_size[2]/tile_size[2])))/2;
175
176 if ( verbose & VERB_PROCESS ) {
177 cout << "Common size: " << common_size << endl;
178 cout << "Tile size: " << tile_size << endl;
179 cout << "Maximum shift: " << max_shift << endl;
180 cout << "Reference start: " << start1 << endl << endl;
181 }
182
183 if ( res_hi < 0.1 || res_hi > tile_size[0]*p1->sampling(0)[0] )
184 res_hi = 4.0*p1->sampling(0)[0];
185
186 if ( filter_flag ) {
187 p1->filter_extremes();
188 p2->filter_extremes();
189 }
190
191 pex1 = p1->extract_tiles(0, start1, common_size, tile_size, step_size, 0);
192
193// Vector3<long>* tile_start = new Vector3<long>[pex1->images()];
194 vector<Vector3<long>> tile_start(pex1->images());
195
196 for ( i=0; i<pex1->images(); i++ ) {
197 m1 = (Bmarker *) add_item((char **) &m1, sizeof(Bmarker));
198 m2 = (Bmarker *) add_item((char **) &m2, sizeof(Bmarker));
199 if ( !set1 ) set1 = m1;
200 if ( !set2 ) set2 = m2;
201 m1->sel = m2->sel = 1;
202 m1->fom = m2->fom = 1;
203 m1->loc = pex1->image[i].origin() + half_tile;
204 tile_start[i] = pex1->image[i].origin();
205 }
206
207 for ( iter=0; iter<maxiter && best_t.fom>0.1; iter++ ) {
208 pex2 = p2->extract_tiles(0, tile_start, tile_size);
209 // Determines the shift of the second image onto the first
210 pex1->find_shift(pex2, NULL, res_hi, res_lo, max_shift, 0, refine_flag);
211 for ( i=0, m2=set2; i<pex1->images(); i++, m2=m2->next ) {
212 shift = pex1->image[i].origin() - pex2->image[i].origin();
213 if ( shift[0] > half_tile[0] ) shift[0] -= tile_size[0];
214 if ( shift[1] > half_tile[1] ) shift[1] -= tile_size[1];
215 if ( shift[2] > half_tile[2] ) shift[2] -= tile_size[2];
216 // To get the location in the second image, the shift must be subtracted
217 m2->loc = tile_start[i] + half_tile - shift;
218 }
219 t = markers_find_rottrans(set1, set2, 0.01);
220 if ( verbose & VERB_PROCESS )
221 cout << "Fit " << iter+1 << ":\t" << p1->file_name() << tab << p2->file_name() << tab
222 << t.trans << tab
223 << t.scale << tab
224 << t.angle*180.0/M_PI << tab << t.fom << endl;
225 if ( best_t.fom > t.fom ) best_t = t;
226 else iter = maxiter;
227 for ( i=0, m1=set1, m2=set2; i<pex1->images(); i++, m1=m1->next, m2=m2->next ) {
228 m2->loc[0] = best_t.scale[0]*(m1->loc[0]*cos(best_t.angle) -
229 m1->loc[1]*sin(best_t.angle)) + t.trans[0] + 0.5;
230 m2->loc[1] = best_t.scale[1]*(m1->loc[0]*sin(best_t.angle) +
231 m1->loc[1]*cos(best_t.angle)) + t.trans[1] + 0.5;
232 m2->loc[2] = best_t.scale[2]*m1->loc[2] + t.trans[2];
233 tile_start[i] = m2->loc - half_tile;
234 }
235 delete pex2;
236 }
237
238 if ( verbose & VERB_RESULT )
239 cout << "Best fit:\t" <<
240 p1->file_name() << tab << p2->file_name() << tab <<
241 best_t.trans << tab <<
242 best_t.scale << tab <<
243 best_t.angle*180.0/M_PI << tab << best_t.fom << endl;
244
245 delete pex1;
246
247 ps_marker_errors(filename, set1, set2, t, p1->size(), 20);
248
249 kill_list((char *) set1, sizeof(Bmarker));
250 kill_list((char *) set2, sizeof(Bmarker));
251
252 return best_t;
253}

386/*
387@author Samuel Payne and Bernard Heymann
388@brief Finds local minimums in a grey scale image and uses those as features
389@param *p image (modified to floating point).
390@param extract_method The method to determine the center of the particle
391@param target_num Max number of desired features (-1 will return all)
392@param threshold Threshold value used to determine whether
393 a pixel is considered a feature.
394@return Bmarker* a set containing xyz coordinates of features
395
396 The input image is expected to have been cleaned in the following manner.
397 With each suggestion is listed the corresponding bsoft function call.
398 1. Any extremes such as white lettering, or black edges have been removed.
399 bfilter -e filname filename
400 2. Edge filter to smooth out the edges.
401 bedge -x<xsize>,<ysize>,<zsize> -o150,150,0 -g75 filename filename
402 3. Bandpass filtered to reduce the noise.
403 bfilter -b1000,2000 -df filename filename
404
405 The algorithm finds blocks of dark grey. The features are black dots
406 in the image, which could be particles, but don't have to be. As long
407 as the picking is consistent, it does not matter what is picked.
408 (it could be a pickle). The method used to find the center of the feature
409 depends on the input. (0=Center of Mass, 1=Center of Area, 2=Darkest pixel)
410
411**/
412Bmarker* img_extract_features(Bimage* p, int extract_method,
413 int target_num, double threshold)
414{
415 // The extreme considered as a feature depends on the relative threshold
416 int sign = 1;
417 if ( threshold < p->average() ) sign = -1;
418
419 long i, j, x, y, z, n;
420 long count(0);
421 Vector3<long> coor;
422 double temp;
423
424 if ( verbose & VERB_LABEL ) {
425 cout << "Using the ";
426 switch ( extract_method ) {
427 case 1: cout << "center of mass "; break;
428 case 2: cout << "center of volume "; break;
429 case 3: cout << "highest pixel density "; break;
430 default: cout << "center of mass "; break;
431 }
432 cout << "to determine the feature coordinates in " << p->file_name() << endl;
433 }
434
435 // Definition of features based on regions beyond a threshold
436 Bimage* pmask = p->regions(threshold, 0);
437 count = (long) pmask->maximum();
438
439 if ( verbose & VERB_LABEL )
440 cout << "Features extracted: " << count << endl << endl;
441
442 if ( count < 1 ) {
443 delete pmask;
444 return NULL;

446

448 Bmarker* m = NULL;
449 Bmarker** lut = new Bmarker*[count];

451 for ( i=0; i<count; i++ ) {

454 m->id = i+1;
455 m->sel = i+1;
456 lut[i] = m;
457 }
458
459 for ( i=n=0; n<p->images(); n++ ) {
460 for ( z=0; z<p->sizeZ(); z++ ) {
461 coor[2] = z;
462 for ( y=0; y<p->sizeY(); y++ ) {
463 coor[1] = y;
464 for ( x=0; x<p->sizeX(); x++, i++ ) {
465 coor[0] = x;
466 if ( (*pmask)[i] > 0 ) { //meaning that it belongs to a group
467 j = (long) ((*pmask)[i]-1);

475 m->loc += coor;
476 m->fom += 1;
477 }
478 if ( sign*(*p)[i] > sign*m->fom ) {
479 m->fom = (*p)[i];
480 if ( extract_method == 3 ) {
481 m->loc = coor;
482 }
483 }
484 m->img_num = n;
485 }
486 }
487 }
488 }
489 }
490
491 delete pmask;
492 delete[] lut;

495

497
498// if ( verbose & VERB_FULL )
499 cout << "Feature\tx\ty\tz\tex\tey\tez\tFOM" << endl;
500

502 if ( extract_method == 1 || extract_method == 2 )
503 m->loc /= m->fom;
504 m->err = m->loc - p->size() /2;
505// if ( verbose & VERB_FULL )
