268proc extractImage { } {
269 global theimg
270 global filename
271 global imgdir
272 global filetypes
273 global tool
274 global sx1 sx2 sy1 sy2 sz1 sz2
275 if ![Bimage exists $theimg] { return }
276 set wc [getControlWindow $theimg]
277 set x [Bimage get $theimg width]
278 set y [Bimage get $theimg height]
279 set z [$wc.slice.scale get]
280 set vlist "0 0 $z $x $y $z"
281 if { [winfo exists .wsel] && $tool == "select" } {
282 if { $sx1 != $sx2 || $sy1 != $sy2 } {
283 set vlist "$sx1 $sy1 $sz1 $sx2 $sy2 $sz2"
284 }
285 }
286# puts $vlist
287 set img_num [$wc.image.scale get]
288 set currdir $imgdir
289 if { [string length $currdir] < 2 } { set currdir [pwd] }
290 set extfile [tk_getSaveFile -filetypes $filetypes \
291 -initialfile [file tail $filename] -initialdir $currdir -defaultextension .mrc]
292 if { [string length $extfile] < 1 } { return }
293 Bimage extract $theimg $extfile $img_num $vlist
294}

488## @brief Modifies the image.
489#
490# @param w the window to destroy.
491
492proc modifyImage_and_DestroyWindow { w } {
493 global theimg
494 global datatype project_item
495 set do_trunc 0
496 set do_rescale 0
497 set ox [$w.panel.origin.entryx get]
498 set oy [$w.panel.origin.entryy get]
499 set oz [$w.panel.origin.entryz get]
500 set order [$w.panel.reslice.entry get]
501 set psi [$w.panel.rotate.entrypsi get]
502 set theta [$w.panel.rotate.entrytheta get]
503 set phi [$w.panel.rotate.entryphi get]
504 set min [$w.panel.truncate.entrymin get]
505 set max [$w.panel.truncate.entrymax get]
506 if { [expr abs($min - [Bimage get $theimg min])] > 0.001 \
507 || [expr abs($max - [Bimage get $theimg max])] > 0.001 } {
508 set do_trunc 1
509 }
510 set avg [$w.panel.rescale.entryavg get]
511 set std [$w.panel.rescale.entrystd get]
512 if { [expr abs($avg - [Bimage get $theimg average])] > 0.001 \
513 || [expr abs($std - [Bimage get $theimg standard_deviation])] > 0.001 } {
514 set do_rescale 1
515 }
516 destroy $w
517 Bimage set $theimg origin $ox $oy $oz
518 if [Bmg exists] { Bmg set $project_item origin $ox $oy $oz }
519 Bimage reslice $theimg $order
520 Bimage rotate $theimg $psi $theta $phi
521 if { $do_trunc } {
522 Bimage truncate $theimg $min $max
523 }
524 if { $do_rescale } {
525 Bimage rescale $theimg $avg $std
526 }
527 set olddatatype [Bimage get $theimg datatype]
528 if { $datatype != $olddatatype } {
529 Bimage set $theimg datatype $datatype
530 }
531 Update 1
532 reconfigureScales
533}

640## @brief This filters the image.
641#
642
643proc filterImage { w } {
644 global theimg filter_type kernel
645 set kernel [$w.kernel.e get]
646 destroy $w
647 if { [string compare $filter_type "Averaging"] == 0 } {
648 Bimage average $theimg $kernel $kernel $kernel
649 } elseif { [string compare $filter_type "Variance"] == 0 } {
650 Bimage variance $theimg $kernel $kernel $kernel
651 } elseif { [string compare $filter_type "Normalizing"] == 0 } {
652 Bimage normalize $theimg $kernel $kernel $kernel
653 } elseif { [string compare $filter_type "Gaussian"] == 0 } {
654 set sigma [expr $kernel/6.0]
655# puts "Gaussian filter: $sigma"
656 Bimage gaussian $theimg $kernel $sigma
657 }
658 Update 1
659 reconfigureScales
660}

1018proc calcPowerSpec {w} {
1019 global theimg
1020 global lg avgtileps avgframeps tiled
1021 global tilex tiley tilez
1022 global tilt_axis tilt_angle tilt_offset defocus
1023 global imgtype
1024 global filename project_item
1025 global window_width window_scale
1026
1027 set tilex [$w.tiles.x get]
1028 set tiley [$w.tiles.y get]
1029 set tilez [$w.tiles.z get]
1030 set tilt_axis [$w.tilt.axis.e get]
1031 set tilt_angle [$w.tilt.angle.e get]
1032 set tilt_offset [$w.tilt.offset.e get]
1033 set defocus [$w.tilt.defocus.e get]
1034 set cs [$w.tilt.cs.e get]
1035 set volt [$w.tilt.volt.e get]
1036 if { $tilt_angle != 0 } { set tiled 1 }
1037 destroy $w
1038# puts "Powerspectrum: $lg $tiled"
1039 set flags [expr 1 + 2*$avgtileps + 4 + 8*$lg + 16*$avgframeps]
1040 if $tiled {
1041 Bimage powerspec $theimg $flags $tilex $tiley $tilez $tilt_axis $tilt_angle $tilt_offset $defocus $cs $volt
1042 } else {
1043 Bimage powerspec $theimg $flags
1044 }
1045 set imgtype "ps"

1047 setMgParam
1048 Bmg set $project_item filename $filename ps
1049 Bmg set $project_item axis [expr $tilt_axis]
1050 Bmg set $project_item tilt [expr $tilt_angle]
1051 }
1052 reconfigureRangeStep
1053 reconfigureScales
1054 set scale 1.0
1055 set width [Bimage get $theimg width]
1056 if { $width > $window_width } { set scale [expr 0.1*((10*$window_width)/$width) ] }
1057 if { $scale < 0.1 } { set scale 0.1 }
1058 set window_scale $scale
1059 .scale.scale set $scale
1060 set ps [Bimage get $theimg pixel_size 0]
1061# puts "Power spectrum pixel size: $ps"
1062}
