773/*
774@brief Converts micrograph data from a STAR data base to a project structure.
775@param &star STAR data base.
776@param *project project structure.
777@param flag update tags.
778@return int error code (<0 means failure).
779
780 The function sets up the project hierarchy from a STAR database file.
781 Micrograph data blocks must contain the "micrograph.id" tag.
782 If the "micrograph.field_of_view_id" is present, the micrographs are
783 distributed into multi-micrograph fields-of-view. Otherwise, each
784 micrograph is assumed to be a unique field-of-view.
785 All angles given in the STAR data base are assumed to be in degrees,
786 and are converted here only to radians.
787
788**/
789int star_to_project(Bstar& star, Bproject* project, int flag)
790{
791 int err(0);
792 int nmg(0), nfield(0), nrec(0);
793
794 if ( verbose & VERB_DEBUG )
795 cout << "DEBUG star_to_project:" << endl;
796
797 if ( flag ) mg_star_update_tags(star);
798
799 nmg = star.number_of_blocks(MICROGRAPH_ID);
800 if ( nmg < 1 ) nmg = star.number_of_blocks(MICROGRAPH_FILE);
801 if ( nmg < 1 ) nmg = star.number_of_blocks(MICROGRAPH_FRAMES_FILE);
802 if ( nmg < 1 ) nmg = star.number_of_blocks(MICROGRAPH_TRANSFORM_FILE);
803 if ( nmg < 1 ) nmg = star.number_of_blocks(MICROGRAPH_POWERSPEC_FILE);
804
805 nrec = star.number_of_blocks(MAP_ID);
806 if ( nrec < 1 ) nrec = star.number_of_blocks(MAP_RECONSTRUCTION);
807 if ( nrec < 1 ) nrec = star.number_of_blocks(MAP_TRANSFORM_FILE);
808 if ( nrec < 1 ) nrec = star.number_of_blocks(MAP_POWERSPEC_FILE);
809
810 if ( nmg < 1 && nrec < 1 ) {
811 cerr << "Warning: No micrograph or reconstruction data blocks found" << endl;
812 return 0;
813 }
814
815 BstarBlock block;
816 Bfield* field = NULL;
817 Bmicrograph *mg, *mg2;
818 Breconstruction *rec, *rec2;
819
820 for ( rec2 = project->rec; rec2 && rec2->next; rec2 = rec2->next ) ;
821
822 if ( star.exists(MAP_REFERENCE) ) {
823 block = star.find(MAP_REFERENCE);
824 for ( auto il: block.loops() )
825 for ( auto ir: il.data() )
826 string_add(&project->reference, ir[0].c_str());
827 }
828
829 if ( ( verbose & VERB_DEBUG ) && project->reference )
830 cout << "DEBUG star_to_project: reference map = " << *(project->reference) << endl;
831
832 int i, j, f;
833 Bstring field_id, mgid;
834 Bstring base;
835
836 // Make sure FOM tags are properly set up
837 if ( star.number_of_blocks(PARTICLE_FOM) ) project->fom_tag[0] = FOM;
838 if ( star.number_of_blocks(PARTICLE_FOM_CV) ) project->fom_tag[1] = FOM_CV;
839 if ( star.number_of_blocks(PARTICLE_HANDA_FOM) ) project->fom_tag[1] = FOM_HAND_A;
840 if ( star.number_of_blocks(PARTICLE_HANDB_FOM) ) project->fom_tag[2] = FOM_HAND_B;
841
842 project_get_fom_tags(star, project->fom_tag);
843 for ( f=i=0; f<NFOM; f++ ) i += project->fom_tag[f];
844 if ( i < 2 ) project->fom_tag[0] = FOM;
845 if ( verbose & VERB_DEBUG )
846 for ( f=0; f<NFOM; f++ ) if ( project->fom_tag[f] )
847 cout << "DEBUG star_to_project: fom_tag=" << project->fom_tag[f] << endl;
848
849 // The orientation convention used is set based on whether the view angle
850 // array or the phi euler angle array is present in the file.
851 // If both are present, the one with the larger number of values wins,
852 // otherwise only the view vector and rotation angle
853 // representation will be used.
854 i = star.number_of_blocks(PARTICLE_PHI);
855 j = star.number_of_blocks(PARTICLE_VIEW_ANGLE);
856 project->euler_flag = 0;
857 project->omega_flag = 0;
858 if ( i > 0 && j > 0 )
859 cerr << "Error: The STAR file contains both views and Euler angles!" << endl;
860 else if ( i > j ) {
861 project->euler_flag = 1;
862 if ( star.number_of_blocks(PARTICLE_OMEGA) > 0 )
863 project->omega_flag = 1;
864 }
865
866 nmg = i = 0;
867 for ( auto ib: star.blocks() ) {
868 if ( ib.exists(MICROGRAPH_ID) || ib.exists(MICROGRAPH_FILE) ) {
869 mgid = 0;
870 field_id = 0;
871 /* This deals with old style access to micrograph data blocks */
872 if ( ib.exists(MICROGRAPH_ID) ) {
873 mgid = ib[MICROGRAPH_ID];
874 } else {
875 if ( ib.exists(MICROGRAPH_FILE) ) mgid = ib[MICROGRAPH_FILE];
876 else if ( ib.exists(MICROGRAPH_FRAMES_FILE) ) mgid = ib[MICROGRAPH_FRAMES_FILE];
877 else if ( ib.exists(PARTICLE_FILE) ) mgid = ib[PARTICLE_FILE];
878 else if ( ib.exists(FILAMENT_FILE) ) mgid = ib[FILAMENT_FILE];
879 else if ( ib.exists(MICROGRAPH_TRANSFORM_FILE) ) mgid = ib[MICROGRAPH_TRANSFORM_FILE];
880 else if ( ib.exists(MICROGRAPH_POWERSPEC_FILE) ) mgid = ib[MICROGRAPH_POWERSPEC_FILE];
881 if ( mgid.length() > 0 ) mgid = mgid.base();
882 }
883 if ( mgid.length() > 0 ) {
884 if ( ib.exists(MICROGRAPH_FIELD_ID) ) field_id = ib[MICROGRAPH_FIELD_ID];
885 if ( field_id.length() < 1 ) field_id = mgid;
886 if ( !project->field ) {
887 field_add(&field, field_id);
888 project->field = field; // First field
889 nfield++;
890 } else {
891 field = NULL;
892 if ( field_id.length() > 0 )
893 field = field_find_id(project->field, field_id);
894 if ( !field ) {
895 field = project->field;
896 field = field_add(&field, field_id);
897 nfield++;
898 }
899 }
900 mg = micrograph_from_starblock(ib, mgid, project->fom_tag);
901 mg->block = i++;
902 if ( verbose & VERB_FULL )
903 cout << "Reading field \"" << field->id << "\", micrograph \"" << mg->id <<"\"" << endl;
904 if ( field->mg ) {
905 for ( mg2 = field->mg; mg2->next; mg2 = mg2->next ) ;
906 mg2->next = mg;
907 } else field->mg = mg;
908 mgid = 0;
909 field_id = 0;
910 nmg++;
911 }
912 } else if ( ib.exists(MAP_ID) ||
913 ib.exists(MAP_RECONSTRUCTION) ||
914 ib.exists(MAP_TRANSFORM_FILE) ||
915 ib.exists(MAP_POWERSPEC_FILE) ) {
916 rec = reconstruction_from_starblock(ib, project->fom_tag);
917 if ( !project->rec ) project->rec = rec;
918 else rec2->next = rec;
919 rec2 = rec;
920 } else if ( ib.exists_loop(PARTICLE_ID) ) {
921 project->class_avg = particle_from_starblock(ib, project->fom_tag);
922 }
923 }
924
925 star.replace_tag(CTF_DEF_MIN, CTF_DEF_AVG);
926 star.replace_tag(CTF_DEF_MAX, CTF_DEF_DEV);
927
928 if ( verbose & VERB_PROCESS )
929 cout << nfield << " fields, " << nmg << " micrographs, " << nrec << " reconstructions converted" << endl;
930
931 return err;
932}

955 FOMType fom_tag[NFOM], int euler_flag, int omega_flag)
956{

958 long f, i, nt(0);
959 Bstring tag;
960 Euler euler;
961
962 if (verbose & VERB_DEBUG )
963 cout << "DEBUG particle_to_starblock: fom tags:" << endl;

965 BstarLoop& loop = block.add_loop();
966 loop.tags()[PARTICLE_ID] = 0;
967 loop.tags()[PARTICLE_GROUP] = 1;
968 loop.tags()[PARTICLE_DEFOCUS] = 2;
969 loop.tags()[PARTICLE_DEF_DEV] = 3;
970 loop.tags()[PARTICLE_AST_ANG] = 4;
971 loop.tags()[PARTICLE_MAGNIF] = 5;
972 loop.tags()[PARTICLE_X] = 6;
973 loop.tags()[PARTICLE_Y] = 7;
974 loop.tags()[PARTICLE_Z] = 8;
975 loop.tags()[PARTICLE_PIXEL_X] = 9;
976 loop.tags()[PARTICLE_PIXEL_Y] = 10;
977 loop.tags()[PARTICLE_PIXEL_Z] = 11;
978 loop.tags()[PARTICLE_ORIGIN_X] = 12;
979 loop.tags()[PARTICLE_ORIGIN_Y] = 13;
980 loop.tags()[PARTICLE_ORIGIN_Z] = 14;
981 if ( euler_flag < 1 ) {
982 loop.tags()[PARTICLE_VIEW_X] = 15;
983 loop.tags()[PARTICLE_VIEW_Y] = 16;
984 loop.tags()[PARTICLE_VIEW_Z] = 17;
985 loop.tags()[PARTICLE_VIEW_ANGLE] = 18;
986 nt = 19;
987 } else {
988 loop.tags()[PARTICLE_PHI] = 15;
989 loop.tags()[PARTICLE_THETA] = 16;
990 loop.tags()[PARTICLE_PSI] = 17;
991 nt = 18;
992 }
993 loop.tags()[PARTICLE_SELECT] = nt++;
994 for ( f=0; f<NFOM && fom_tag[f]; ++f ) {
995 tag = get_fom_tag(fom_tag[f]);
996 loop.tags()[tag.str()] = nt++;
997 if ( verbose & VERB_DEBUG )
998 cout << nt << tab << tag << endl;
999 }
1000 if ( part->fpart.length() )
1001 loop.tags()[PARTICLE_FILE] = nt++;
1002
1003 if (verbose & VERB_DEBUG )
1004 cout << "DEBUG particle_to_starblock: columns=" << nt << endl;
1005
1006 for ( p = part; p; p = p->next ) {
1007// cout << p->id << tab << "-" << p->fpart << "-" << endl;
1008 vector<string>& vs = loop.add_row(nt);
1009 vs[0] = to_string(p->id);
1010 vs[1] = to_string(p->group);
1011 vs[2] = to_string(p->def);
1012 vs[3] = to_string(p->dev);
1013 vs[4] = to_string(p->ast*180.0/M_PI);
1014 vs[5] = to_string(p->mag);
1015 vs[6] = to_string(p->loc[0]);
1016 vs[7] = to_string(p->loc[1]);
1017 vs[8] = to_string(p->loc[2]);
1018 vs[9] = to_string(p->pixel_size[0]);
1019 vs[10] = to_string(p->pixel_size[1]);
1020 vs[11] = to_string(p->pixel_size[2]);
1021 vs[12] = to_string(p->ori[0]);
1022 vs[13] = to_string(p->ori[1]);
1023 vs[14] = to_string(p->ori[2]);
1024 if ( euler_flag < 1 ) {
1025 vs[15] = to_string(p->view[0]);
1026 vs[16] = to_string(p->view[1]);
1027 vs[17] = to_string(p->view[2]);
1028 vs[18] = to_string(p->view[3]*180.0/M_PI);
1029 i = 19;
1030 } else {
1031 euler = Euler(p->view);
1032 vs[15] = to_string(euler.phi()*180.0/M_PI);
1033 vs[16] = to_string(euler.theta()*180.0/M_PI);
1034 vs[17] = to_string(euler.psi()*180.0/M_PI);
1035 i = 18;
1036 }
1037 vs[i++] = to_string(p->sel);
1038 for ( f=0; i<nt && f<NFOM && fom_tag[f]; ++f, ++i )
1039 vs[i] = to_string(p->fom[f]);
1040 if ( p->fpart.length() )
1041 vs[i] = p->fpart.str();
1042 }
1043
1044 if (verbose & VERB_DEBUG )
1045 cout << "DEBUG particle_to_starblock: columns converted=" << i << endl;
1046
1047 return 0;
1048}
