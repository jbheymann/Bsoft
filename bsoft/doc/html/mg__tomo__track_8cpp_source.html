253 if ( ccbest < cc[i] ) {
254 ccbest = cc[i];
255 abest = ar[i];

257 if ( verbose )
258 cout << ar[i]*180.0/M_PI << tab << cc[i] << endl;

260
261 delete[] ar;
262 delete[] cc;
263 delete pgold;

266 cout << "Best tilt axis angle: " << abest*180.0/M_PI << " degrees" << endl;

272 kill_list((char *) mgn->mark, sizeof(Bmarker));
273 mgp->mark = NULL;
274 mgn->mark = NULL;
275
276 for ( mg = project->field->mg; mg; mg = mg->next )
277 mg->tilt_axis = abest;
278
279 return ccbest;
