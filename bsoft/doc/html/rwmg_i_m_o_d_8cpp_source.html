20int read_imod_xf(Bstring& filename, Bproject* project)
21{
22 Bmicrograph* mg = NULL;
23 double a[4], dx, dy;
24
25 ifstream fprm(filename.c_str());
26 if ( fprm.fail() ) {
27 cerr << "Error: A text file specifying 2D transformations must be given!" << endl;
28 return -1;
29 }
30
31 if ( verbose & VERB_PROCESS )
32 cout << "Reading 2D transformations from " << filename << endl;
33
34 for ( mg = project->field->mg; mg; mg = mg->next ) {
35 fprm >> a[0] >> a[1] >> a[2] >> a[3] >> dx >> dy;
36 mg->tilt_axis = (atan2(-a[0], a[1]) + atan2(-a[3], -a[2]))/2.0;
37 mg->origin[0] -= a[0]*dx + a[2]*dy;
38 mg->origin[1] -= a[1]*dx + a[3]*dy;
39 if ( verbose )
40 cout << mg->origin << tab << mg->tilt_axis*180.0/M_PI << endl;
41 }
42
43 fprm.close();
44
45 return 0;
46}

74int read_imod_xyz(Bstring& filename, Breconstruction* rec)
75{
76 int mark_id(0), first(1);

78 string word;
79 Bmarker* mark = NULL;

81 ifstream fprm(filename.c_str());

83 cerr << "Error: A text file specifying the 3D marker model must be given!" << endl;
84 return -1;
85 }
86
87 fprm >> mark_id >> x >> y >> z >> dum >> dum;
88 while ( fprm >> word ) {
89 if ( word == "Pix:" ) fprm >> rec->voxel_size[0];
90 if ( word == "Dim:" ) break;
91 }
92 fprm >> rec->origin[0] >> rec->origin[1];
93 rec->origin /= 2;
94
95 if ( access(rec->frec.c_str(), R_OK) ) {
96 if ( verbose )
97 cout << "Getting reconstruction parameters from " << filename << endl;
98 } else {
99 if ( verbose )
100 cout << "Getting reconstruction parameters from " << rec->frec << endl;
101 Bimage* p = read_img(rec->frec, 0, -1);
102 rec->voxel_size = p->sampling(0);
103 rec->origin = p->size()/2;
104 delete p;
105 }
106 if ( verbose )
107 cout << setprecision(3) << "Voxel size = " << rec->voxel_size << " Origin = " << rec->origin << endl;
108
109 while ( !fprm.eof() ) {
110 if ( first ) first = 0;
111 else fprm >> mark_id >> x >> y >> z >> dum >> dum;
112 if ( fprm.fail() ) break;
113 mark = (Bmarker *) add_item((char **) &mark, sizeof(Bmarker));
114 if ( !rec->mark ) rec->mark = mark;
115 mark->id = mark_id;
116// mark->loc = Vector3<double>(x - rec->origin[0], y - rec->origin[1], z);
117 mark->loc = Vector3<double>(x, y, z);
118 mark->fom = 1;
119 mark->sel = 1;
120 if ( verbose )
121 cout << mark_id << tab << x << tab << y << tab << z << endl;
122 }
123
124 fprm.close();
125
126 return 0;

178 fprm.read((char *)tag, 4);

182 if ( fprm.fail() ) return -2;
183
184 swapbytes((unsigned char *) &cont->psize, 4);
185
186 if ( verbose & VERB_PROCESS )
187 cout << "number of points = " << cont->psize << endl;
188
189 psize = 3*cont->psize*sizeof(float);
190 point = new float[3*cont->psize];
191
192 fprm.read((char *)point, psize);
193 if ( fprm.fail() ) return -2;
194
195 swapbytes(psize, (unsigned char *) point, 4);
196
197 for ( j=0, mg = project->field->mg; mg; mg = mg->next, j+=3 ) {
198 mark = (Bmarker *) add_item((char **) &mg->mark, sizeof(Bmarker));
199 mark->id = i+1;
200 mark->loc = Vector3<double>(point[j], point[j+1], 0);
201 mark->fom = 1;
202 mark->sel = 1;
203 if ( verbose )
204 cout << mg->id << tab << mark->id << tab << mark->loc << endl;
205 }
206
207 delete[] point;
208 }
209
210 delete imod;
211 delete objt;
212 delete cont;
213
214 fprm.close();
215
216 return 0;
217}

309 }
310

312
313 for ( thisfile = file_list; thisfile; thisfile = thisfile->next ) {
314 ext = thisfile->extension();
315 if ( ext.contains("xf") ) {
316 err += read_imod_xf(*thisfile, project);
317 } else if ( ext.contains("tlt") ) {
318 err += read_imod_tlt(*thisfile, project);

322 rec->frec = base + ".rec";
