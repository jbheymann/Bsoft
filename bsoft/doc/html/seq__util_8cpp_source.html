418Bstring seq_find_protein_in_dna(Bmolgroup* molgroup, Bstring& seq, int seqlenmin, int seqlenmax,
419 int side1, int side2, double threshold, Bstring& gcfile)
420{
421 long n = seq.length();
422 if ( n < 1 ) return "";
423
424 Bmolecule* mol;
425 Bmolecule* bestmol = NULL;
426 if ( seqlenmin < n/2 ) seqlenmin = n/2;
427 if ( seqlenmax <= n ) seqlenmin = 2*n;
428
429 int linelength = 80;
430 int f, i, j, nres = 0, start, length, frame, term, seqlen = 0;
431 int bestscore = 0, bestframe = 0;
432 int bestindex = 0, bestseqlen = 0;
433 Bstring bestseq;
434 char format[100];
435 snprintf(format, 100, "%%.%ds%c", linelength, '\n');
436
437 Bstring gencode = get_genetic_code(gcfile);
438
439 if ( verbose ) {
440 cout << endl << "Finding an amino acid sequence in translated nucleotide sequences:" << endl;
441 cout << "Sequence length range: " << seqlenmin << " - " << seqlenmax << endl;
442 cout << "Score threshold (residues): " << threshold << endl;
443 }
444
445 for ( mol = molgroup->mol; mol; mol = mol->next ) { // Search all molecules
446 for ( f=0; f<6; f++ ) { // Search all translational frames
447 Bstring nucseq = mol->naseq;
448 frame = f;
449 if ( f > 2 ) {
450 frame = f - 3;
451 complement_sequence(nucseq);
452 }
453 Bstring aaseq = sequence_translate(nucseq, frame, gencode);
454 nres = aaseq.length();
455
456 vector<int> score(nres,0);
457
458 for ( i=0; i<=nres - n; i++ ) { // Loop over starting positions
459 if ( i == 0 || aaseq[i] == '*' ) {
460 seqlen = 0;
461 while ( ( i+seqlen+1 < nres ) && ( aaseq[i+seqlen+1] != '*' ) )
462 seqlen++;
463 }
464 if ( seqlen >= seqlenmin && seqlen <= seqlenmax ) {
465 term = 0;
466 for ( j=0; j<n && j<nres && !term; j++ ) {
467 if ( aaseq[i+j] == seq[j] ) score[i]++;
468 if ( aaseq[i+j] == '*' ) term = 1;
469 }
470 if ( !term && threshold > 0 && score[i] >= threshold ) {
471 if ( verbose )
472 cout << "Frame: " << f << " Position: " << 3*i + frame <<
473 " Length: " << seqlen << " Score: " << score[i] <<
474 " (" << score[i]*100.0/seq.length() << " %)" << endl;
475 for ( j=0; j<n; j+= linelength )
476 printf(format, &aaseq[i + j]);
477 }
478 if ( !term && bestscore < score[i] ) {
479 bestscore = score[i];
480 bestseqlen = seqlen;
481 bestmol = mol;
482 bestframe = f;
483 bestindex = 3*i + frame;
484 start = (i - side1)*3 + frame;
485 while ( start < 0 ) start += 3;
486 length = (n + side1 + side2)*3;
487 if ( length + start > nucseq.length() )
488 length = nucseq.length() - start;
489 bestseq = nucseq.substr(start, length);
490 }
491 }
492 }
493 }
494 }
495
496 cout << "Best sequence =" << bestseq << endl;
497
498 if ( bestmol && verbose ) {
499 cout << endl << "Best sequence score: " << bestscore <<
500 " (" << bestscore*100.0/seq.length() << "%)" << endl;
501 cout << "Molecule: " << bestmol->id << endl;
502 cout << "Length: " << bestseqlen << endl;
503 if ( bestframe < 3 ) cout << "Frame: " << bestframe << endl;
504 else cout << "Frame: " << bestframe - 3 << " (complement)" << endl;
505 cout << "Index: " << bestindex << endl;
506 cout << "Nucleotide sequence:" << endl;
507 for ( i=0; i<bestseq.length(); i+= linelength )
508 printf(format, bestseq.c_str() + i);
509 Bstring aaseq = sequence_translate(bestseq, 0, gencode);
510 cout << "Amino acid sequence:" << endl;
511 for ( i=0; i<aaseq.length(); i+= linelength )
512 printf(format, aaseq.c_str() + i);
513 cout << "Requested sequence:" << endl;
514 for ( i=0; i<seq.length(); i+= linelength )
515 printf(format, seq.c_str() + i);
516 }
517 cout << endl;
518
519 return bestseq;
520}

614Bstring sequence_translate(Bstring& nucseq, long frame, Bstring& gencode)
615{
616 long i, j;
617 long nnuc = nucseq.length();
618 long nres = (nnuc - frame)/3;
619 Bstring aaseq(nres, ' ');
620
621 for ( i=0, j=frame; i<nres && j<nnuc; i++, j+=3 )
622 aaseq[i] = gencode[index_from_codon(&nucseq[j])];
623
624 if ( i < nres ) aaseq[i] = 0;
625
626 return aaseq;
