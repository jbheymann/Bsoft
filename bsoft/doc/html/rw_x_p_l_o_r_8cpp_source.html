126/*
127@brief Reading a XPLOR structure factor file format.
128A structure factor text format used in X-ray crystallography.
129 A minimal implementation of the XPLOR reflection file format which
130 recognizes only the indices, amplitudes and phases on a line.
131@param *p the image structure.
132@param readdata flag to activate reading of image data.
133@return int error code (<0 means failure).
134**/
135int readXPLORsf(Bimage* p, int readdata)
136{
137 long i = 0, j, h, k, l, x, y, z, n = 0;
138 long hmin = 0, hmax = 0, kmin = 0, kmax = 0, lmin = 0, lmax = 0;
139 float amp, phi;
140
141 ifstream fhkl(p->file_name());
142
143 if ( fhkl.fail() ) return -1;
144
145 char aline[MAXLINELEN], tag[20];
146 fhkl.getline(aline, MAXLINELEN); // First line with number of reflections
147 sscanf( aline, "%s %ld", tag, &n);
148
149 fhkl.getline(aline, MAXLINELEN); // Second line with anomalous tag
150
151 while ( fhkl.getline(aline, MAXLINELEN) ) {
152 if ( strstr(aline, "INDE") || strstr(aline, "inde") ) {
153 i = sscanf( aline, "%s %ld %ld %ld %s %f %f", tag, &h, &k, &l,
154 tag, &amp, &phi ); // A reflection line
155 if ( i ) {
156 if ( hmin > h ) hmin = h;
157 if ( kmin > k ) kmin = k;
158 if ( lmin > l ) lmin = l;
159 if ( hmax < h ) hmax = h;
160 if ( kmax < k ) kmax = k;
161 if ( lmax < l ) lmax = l;
162 n++;
163 }
164 }
165 }
166
167 p->size(1,1,1);
168 if ( p->sizeX() < (long)2*hmax + 1 ) p->sizeX(2*(hmax + 1));
169 if ( p->sizeX() < 1 - (long)2*hmin ) p->sizeX(-2*(hmin - 1));
170 if ( p->sizeY() < (long)2*kmax + 1 ) p->sizeY(2*(kmax + 1));
171 if ( p->sizeY() < 1 - (long)2*kmin ) p->sizeY(-2*(kmin - 1));
172 if ( p->sizeZ() < (long)2*lmax + 1 ) p->sizeZ(2*(lmax + 1));
173 if ( p->sizeZ() < 1 - (long)2*lmin ) p->sizeZ(-2*(lmin - 1));
174
175 p->data_type(Float);
176 p->compound_type(TComplex);
177 p->channels(2);
178
179 fhkl.close();
180
181 if ( !readdata ) return 0;
182
183 Complex<float>* data = (Complex<float>*) p->data_alloc();
184
185 fhkl.open(p->file_name());
186
187 if ( fhkl.fail() ) return -1;
188
189 fhkl.getline(aline, MAXLINELEN); // Ignore the header line
190
191 while ( fhkl.getline(aline, MAXLINELEN) ) {
192 if ( strstr(aline, "INDE") || strstr(aline, "inde") ) {
193 i = sscanf( aline, "%s %ld %ld %ld %s %f %f", tag, &h, &k, &l,
194 tag, &amp, &phi ); // A reflection line
195 if ( i ) {
196 x = h;
197 if ( x < 0 ) x += p->sizeX();
198 y = k;
199 if ( y < 0 ) y += p->sizeY();
200 z = l;
201 if ( z < 0 ) z += p->sizeZ();
202 j = (z*p->sizeY() + y)*p->sizeX() + x;
203 phi *= M_PI/180;
204 data[j] = Complex<float>(amp*cos(phi), amp*sin(phi));
205 }
206 }
207 }
208
209 fhkl.close();
210
211 return 0;
212}
