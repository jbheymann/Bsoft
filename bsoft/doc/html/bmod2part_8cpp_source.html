204Bproject* project_part_from_model(Bmodel* model, Vector3<double> origin,
205 Vector3<double> sampling, Vector3<long> box_size)
206{
207 long i;
208
209 for ( i=0; i<3; i++ ) if ( sampling[i] <= 0 ) sampling[i] = 1;
210
211 Bstring b = base(model->mapfile());
212 Bproject* project = new Bproject;
213 Breconstruction* rec = reconstruction_add(&project->rec, b);
214 Bparticle* part = NULL;
215
216 rec->frec = model->mapfile();
217 rec->voxel_size = sampling;
218 rec->origin = origin;
219 rec->box_size = box_size;
220 rec->bad_radius = box_size[0]/4;
221
222 if ( verbose )
223 cout << "Generating particles from model components" << endl << endl;
224
225 Bmodel* mp;
226 Bcomponent* comp;
227 View2<float> v;
228
229 for ( i=0, mp = model; mp; mp = mp->next ) if ( mp->select() ) {
230 for ( comp = mp->comp; comp; comp = comp->next ) if ( comp->select() ) {
231 part = particle_add(&part, ++i);
232 if ( !rec->part ) rec->part = part;
233 part->ori = box_size/2;
234 part->loc = comp->location()/sampling + origin;
235 v = comp->view().backward();
236 part->view = View2<double>(v[0],v[1],v[2],v[3]);
237 part->sel = comp->select();
238 part->fom[0] = comp->FOM();
239 }
240 }
241
242 return project;
243}
