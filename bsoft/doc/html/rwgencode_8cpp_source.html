67Bstring read_gencode_star(Bstring& propfile)
68{
69 Bstar star;
70
71 if ( star.read(propfile.str()) < 0 )
72 error_show(propfile.c_str(), __FILE__, __LINE__);
73
74 if ( star.blocks().size() < 0 ) {
75 cerr << "No data blocks found in the STAR file!" << endl;
76 return NULL;
77 }
78
79 int i, j, index;
80 Bstring gc(' ', 64L);
81
82 for ( auto ib: star.blocks() ) {
83 for ( auto il: ib.loops() ) {
84 if ( ( i = il.find(RESPROP_CODE1) ) >= 0 &&
85 ( j = il.find(RESPROP_CODON) ) >= 0 ) {
86 if ( il.data().size() != 64 ) {
87 cerr << "Error: File " << propfile << " does contain only " << il.data().size() << " codons, 64 required!" << endl;
88 return "";
89 }
90 for ( auto ir: il.data() ) {
91 index = index_from_codon(ir[j].c_str());
92 gc[index] = ir[i][0];
93 }
94 }
95 }
96 }
97
98 if ( verbose & VERB_DEBUG )
99 cout << "DEBUG read_gencode_star: " << gc << endl;
100
101 return gc;
102}

111{
112 int i, j, index(0), e;
113 char acgt[8] = "ACGT";
114 char upcodon[4];
115
116 for ( i=0; i<3; i++ ) {
117 upcodon[i] = toupper(codon[i]);
118 e = (int) pow(4.0, 2.0 - i);
