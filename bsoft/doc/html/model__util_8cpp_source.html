261 Bmodel* mp;
262 Bcomptype* ct = model->type;
263 Bcomponent* comp = model->comp;
264 Blink* link = model->link;
265
266 if ( comp ) for ( ; comp->next; comp = comp->next ) ;
267 if ( link ) for ( ; link->next; link = link->next ) ;
268
269 for ( mp = model->next; mp; mp = mp->next ) {
270 if ( ct ) ct->next = mp->type;
271 else model->type = ct = mp->type;
272 mp->type = NULL;
273 if ( ct ) for ( ; ct->next; ct = ct->next ) ;
274 if ( comp ) comp->next = mp->comp;
275 else model->comp = comp = mp->comp;

278 if ( link ) link->next = mp->link;
279 else model->link = link = mp->link;
280 mp->link = NULL;
281 if ( link ) for ( ; link->next; link = link->next ) ;
282 }
283

285 model->next = NULL;
286
287 for ( nct=0, ct = model->type; ct; ct = ct->next ) nct++;
288
289 // Renumber all the components

291 comp->identifier() = to_string(++ncomp);

1337int model_insert(Bmodel* model, Bmodel* modinsert, double distance)
1338{
1339 if ( distance <= 0 ) distance = 2; // Default
1340 if ( distance < 1 ) distance = 1; // Limits on cutoff distance in angstrom
1341 if ( distance > 5 ) distance = 5;
1342
1343 long i, delete_comp, ncompdel(0), nmoddel(0);
1344 long ii, x, y, z, xx, yy, zz, ix, iy, iz;
1345 Vector3<double> sampling(distance, distance, distance);
1346 Bmodel *m, *pm;
1347 Bcomponent *c, *pc;
1348 Vector3<double> box = model->maximum() - model->minimum();
1349 Vector3<long> gridsize((long) (box[0]/sampling[0] + 0.001),
1350 (long) (box[1]/sampling[1] + 0.001), (long) (box[2]/sampling[2] + 0.001));
1351 gridsize = gridsize.max(1);
1352 for ( i=0; i<3; i++ ) sampling[i] = box[i]/gridsize[i] + 0.001;
1353// long gridvol = (long) gridsize.volume();
1354
1355 if ( verbose )
1356 cout << "Inserting a model and deleting overlapping components" << endl;
1357
1358 if ( verbose & VERB_PROCESS )
1359 cout << "Distance cutoff: " << distance << " A" << endl;
1360
1361 Vector3<double> gridori;
1362 vector<vector<Bcomponent*>> grid = models_component_grid(modinsert, gridsize, gridori, sampling);
1363
1364 // Find the atoms under the footprint to be deleted
1365 for ( m = pm = model; m; ) {
1366 for( c = pc = m->comp; c; ) {
1367 delete_comp = 0;
1368 x = (long) ((c->location()[0] - model->minimum()[0])/sampling[0]);
1369 y = (long) ((c->location()[1] - model->minimum()[1])/sampling[1]);
1370 z = (long) ((c->location()[2] - model->minimum()[2])/sampling[2]);
1371 if ( x >=0 && x < gridsize[0] && y >= 0 && y < gridsize[1] && z >= 0 && z < gridsize[2] ) {
1372 i = (z*gridsize[1] + y)*gridsize[0] + x;
1373 for ( zz=z-1; zz<=z+1; zz++ ) {
1374 iz = zz;
1375 if ( iz < 0 ) iz += gridsize[2];
1376 if ( iz >= gridsize[2]) iz -= gridsize[2];
1377 for ( yy=y-1; yy<=y+1; yy++ ) {
1378 iy = yy;
1379 if ( iy < 0 ) iy += gridsize[1];
1380 if ( iy >= gridsize[1] ) iy -= gridsize[1];
1381 for ( xx=x-1; xx<=x+1; xx++ ) {
1382 ix = xx;
1383 if ( ix < 0 ) ix += gridsize[0];
1384 if ( ix >= gridsize[0] ) ix -= gridsize[0];
1385 ii = (iz*gridsize[1] + iy)*gridsize[0] + ix;
1386 for ( auto& c2: grid[ii] ) {
1387 if ( c->location().distance(c2->location()) < distance )
1388 delete_comp = 1;
1389 }
1390 }
1391 }
1392 }
1393 }
1394 if ( delete_comp ) {
1395 if ( verbose & VERB_FULL )
1396 cout << "Removing a component" << endl;
1397 if ( c == m->comp ) {
1398 m->comp = pc = c->next;
1399 delete c;
1400 c = m->comp;
1401 } else {
1402 pc->next = c->next;
1403 delete c;
1404 c = pc->next;
1405 }
1406 ncompdel++;
1407 } else {
1408 pc = c;
1409 if ( c ) c = c->next;
1410 }
1411 }
1412 if ( !m->comp ) {
1413 if ( verbose & VERB_FULL )
1414 cout << "Removing a model" << endl;
1415 if ( model == m ) {
1416 model = pm = m->next;
1417 delete m;
1418 m = model;
1419 } else {
1420 pm->next = m->next;
1421 delete m;
1422 m = pm->next;
1423 }
1424 nmoddel++;
1425 } else {
1426 pm = m;
1427 if ( m ) m = m->next;
1428 }
1429 }
1430
1431 grid.clear();
1432
1433 if ( verbose & VERB_PROCESS )
1434 cout << "Models and components deleted: " << nmoddel << " " << ncompdel << endl;
1435
1436 // Add the new molecules
1437 if ( model ) {
1438 for ( m = model; m->next; m = m->next ) ;
1439 m->next = modinsert;
1440 } else model = modinsert;
1441 modinsert = NULL;
1442
1443 return 0;
1444}
