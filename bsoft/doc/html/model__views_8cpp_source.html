135/* The normal is defined as the average of all the rotation axes associated
136with the angles between the links */
137Vector3<double> component_normal(Bcomponent* comp)
138{
139 Vector3<double> normal;
140
141 if ( !comp->link[1] ) return normal;
142
143 long i, j;
144 Vector3<double> v1, v2;
145
146 for ( i=1; i<comp->link.size(); i++ ) {
147 for ( j=0; j<i; j++ ) {
148 v1 = comp->link[i]->location() - comp->location();
149 v2 = comp->link[j]->location() - comp->location();
150 v1 = v1.cross(v2);
151 if ( comp->location().angle(v1) > M_PI_2 ) v1 = -v1;
152 v1.normalize();
153 normal += v1;
154 }
155 }
156 if ( normal.length2() > 1e-10 ) normal.normalize();
157
158 return normal;
159}

161/* The view vector is the normal and the rotation angle is chosen such that
162the first link falls in the xz plane when the component is rotated to the
163standard view */
164View2<float> component_view(Bcomponent* comp)
165{
166 Vector3<double> w(comp->view().vector3());
167
168 if ( comp->link[1] ) w = component_normal(comp);
169
170 if ( w.length2() < 0.1 ) w = comp->location();
171
172 long i, im;
173 Vector3<double> v, u;
174 double a(0);
175
176 for ( i=1, im=0; i<comp->link.size() && comp->link[i]; i++ ) {
177 v = comp->location().normal(comp->link[im]->location(), comp->link[i]->location());
178 a = w.angle(v);
179 if ( a > M_PI_2 ) im = i;
180 }
181
182 w.normalize();
183 u = comp->link[im]->location() - comp->location();
184 u.normalize();
185 v = w.cross(u);
186 v.normalize();
187 u = v.cross(w);
188 a = angle_set_negPI_to_PI(atan2(w[1], w[0]) + atan2(v[2], -u[2]));
189
190 View2<float> view(w[0], w[1], w[2], a);
191
192 return view;
193}

414int component_hand(Bstring s)
415{
416 int h(0), len = s.length();
417 Bstring ss(s);
418 Bstring sh(s);
419
420 if ( len < 3 ) return (0);
421 else if ( len == 3 ) {
422 ss = ss.canonical(1);
423 sh = sh.swap(0,2);
424 sh = sh.canonical(1);
425 } else if ( len == 6 ) {
426 ss = ss.canonical(2);
427 sh = sh.swap(0,2);
428 sh = sh.swap(3,5);
429 sh = sh.canonical(2);
430 }
431
432 h = ss.compare(sh);
433
434 if ( h > 0 ) h = 1;
435 if ( h < 0 ) h = -1;
436
437 return h;
438}
