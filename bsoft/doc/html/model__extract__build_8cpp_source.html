19/*
20 The first axis is along the link.
21 The second axis is the average of the component normals, usually directed
22 away from the molecule group center.
23 The third axes is generated as the cross product of the first two. i.e.,
24 perpendicular to both.
25 A new second axis is generated as the cross product between the other two,
26 ensuring that all the axes are orthogonal.
27*/
28Matrix3 get_link_matrix(Blink* link)
29{
30 Matrix3 mat;
31 Vector3<double> u, v, w;
32
33 u = link->comp[1]->location() - link->comp[0]->location();
34 w = link->comp[0]->view().vector3();
35 w += link->comp[1]->view().vector3();
36 v = w.cross(u);
37 w = u.cross(v);
38 u.normalize();
39 v.normalize();
40 w.normalize();
41// for ( int i=0; i<3; i++ ) {
42// mat[i] = u[i];
43// mat[i+3] = v[i];
44// mat[i+6] = w[i];
45// }
46 mat[0] = u;
47 mat[1] = v;
48 mat[2] = w;
49 mat = mat.transpose();
50// cout << mat << endl;
51
52 return mat;
53}

55Vector3<double> component_find_shift(Bcomponent* comp, Bimage* p, Bimage* ptemp,
56 Bimage* pmask, Bimage* pfsmask, View2<float> view,
57 double hires, double lores, double shift_limit,
58 int shift_flag, fft_plan planf, fft_plan planb)
59{
60 long n, imax(0);
61 Vector3<long> size(ptemp->size());
62 Vector3<double> origin(ptemp->image->origin());
63 Vector3<double> scale(ptemp->sampling(0));
64 Vector3<double> loc = comp->location()/scale + p->image->origin();
65
66 Matrix3 mat = view.matrix();
67 Bimage* pcomp = p->extract(0, loc, size, origin, mat);
68 Bimage* pone = NULL;
69 Bimage* pmaskrot = NULL;
70
71 double cc, ccmax(-1);
72 Vector3<double> shift, best_shift;
73
74 if ( pmask ) {
75 pcomp->rescale_to_avg_std(0, 1, pmask);
76 pcomp->multiply(pmask);
77 }
78
79 if ( pfsmask ) pmaskrot = pfsmask->extract_wrap(0, size, mat);
80
81 for ( n=0; n<ptemp->images(); n++ ) {
82 pone = ptemp->extract(n);
83 pone->origin(origin);
84 shift = pcomp->find_shift(pone, pmaskrot, hires, lores, shift_limit, 0, 1, planf, planb, cc);
85 if ( ccmax < cc ) {
86 ccmax = cc;
87 imax = n;
88 if ( shift_flag == 1 ) shift[0] = shift[1] = 0; // Shift only along the normal
89 else if ( shift_flag == 2 ) shift[2] = 0; // Shift only lateral
90 best_shift = mat * shift; // Shift in voxels
91 }
92 delete pone;
93 }
94
95 delete pcomp;
96 delete pmaskrot;
97
98 comp->select(imax + 1);
99 comp->FOM(ccmax);
100
101 return best_shift;
102}

153int component_refine_random(Bcomponent* comp, Bimage* p, Bimage* ptemp, Bimage* pmask,
154 Bimage* pfsmask, int max_iter, double hires, double lores, double max_shift,
155 double max_view_angle, double max_rot_angle, int shift_flag, fft_plan planf, fft_plan planb)
156{
157 if ( verbose & VERB_FULL )
158 cout << "Refining component " << comp->identifier() << " by random view variations" << endl;
159
160 int imax(0), iter, itm;
161 double shift_limit(max_shift/ptemp->sampling(0)[0]);
162 double cc, ccmax(-1), a, s, irm(1.0/get_rand_max());
163 View2<float> view(comp->view());
164 View2<float> comp_view(comp->view());
165 Vector3<double> scale(ptemp->sampling(0));
166 Vector3<double> r, comp_shift;
167 Vector3<double> shift;
168
169 for ( iter=itm=0; iter<=max_iter; iter++ ) {
170 if ( iter > 0 ) {
171 r = vector3_random_gaussian(0, max_view_angle);
172 a = max_rot_angle*(random()*irm*2 - 1);
173 view[0] += r[0];
174 view[1] += r[1];
175 view[2] += r[2];
176 view[3] += a;
177 view.normalize();
178 }
179 shift = component_find_shift(comp, p, ptemp, pmask, pfsmask, view, hires, lores,
180 shift_limit, shift_flag, planf, planb);
181 cc = comp->FOM();
182 if ( ccmax < cc ) {
183 ccmax = cc;
184 imax = comp->select() - 1;
185 comp_shift = shift; // Shift in voxels
186 comp_view = view;
187 itm = iter;
188 }
189 }
190
191 comp_shift *= scale; // Shift in angstroms
192 s = comp_shift.length();
193 if ( s > max_shift ) {
194 comp_shift *= max_shift/s;
195 s = comp_shift.length();
196 }
197 comp->shift(comp_shift);
198 comp->force(comp_shift);
199 comp->view(comp_view);
200 comp->select(imax + 1);
201 comp->FOM(ccmax);
202
203 return itm;
204}

206int component_refine(Bcomponent* comp, Bimage* p, Bimage* ptemp, Bimage* pmask,
207 Bimage* pfsmask, double viewmax, double rotmax,
208 double viewstep, double rotstep,
209 double hires, double lores, double accuracy, double max_shift,
210 int shift_flag, fft_plan planf, fft_plan planb)
211{
212 if ( verbose & VERB_FULL )
213 cout << "Refining component " << comp->identifier() << " by contracting grid search" << endl;
214
215 if ( viewmax < 0 ) viewmax = 0;
216 if ( rotmax < 0 ) rotmax = 0;
217 if ( viewstep < 0 ) viewstep = 0;
218 if ( rotstep < 0 ) rotstep = 0;
219
220 int imax(0), itm;
221 double shift_limit(max_shift/ptemp->sampling(0)[0]);
222 double cc, ccp(-2), ccmax(-1), s;
223 Vector3<double> scale(ptemp->sampling(0));
224 Vector3<double> comp_shift;
225 Vector3<double> shift;
226
227 View2<double> comp_view(comp->view());
228 vector<View2<double>> views;
229
230 double tol(1e-6);
231 if ( viewstep && accuracy > viewstep ) accuracy = viewstep;
232 if ( rotstep && accuracy > rotstep ) accuracy = rotstep;
233
234 int notdone(1);
235 for ( itm=0; notdone; itm++ ) {
236 views = views_within_limits(comp_view, viewstep, viewstep, rotstep, viewmax, rotmax);
237 for ( auto v: views ) {
238 shift = component_find_shift2(comp, p, ptemp, pmask, pfsmask, v, hires, lores,
239 shift_limit, shift_flag, planf, planb);
240 cc = comp->FOM();
241 if ( ccmax < cc ) {
242 ccmax = cc;
243 imax = comp->select() - 1;
244 comp_shift = shift; // Shift in voxels
245 comp_view = v;
246 }
247 }
248 if ( fabs(ccp - ccmax) < tol ) {
249 viewstep /= 2; // Contract around best point
250 rotstep /= 2;
251 }
252 ccp = ccmax;
253 if ( viewstep < tol && rotstep < tol ) notdone = 0;
254 if ( viewstep <= accuracy && rotstep <= accuracy ) notdone = 0;
255 }
256
257 comp_shift *= scale; // Shift in angstroms
258 s = comp_shift.length();
259 if ( s > max_shift ) {
260 comp_shift *= max_shift/s;
261 s = comp_shift.length();
262 }
263 comp->shift(comp_shift);
264 comp->force(comp_shift);
265 comp->view(comp_view);
266 comp->select(imax + 1);
267 comp->FOM(ccmax);
268
269 return itm;
270}
