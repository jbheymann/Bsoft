713double project_tilt_from_particle_defocus(Bproject* project)
714{
715 long i, j, nmg(0), nmgsel(0), npart(0), nptot(0), npsel(0);
716 double axis, axis_avg(0), axis_std(0), a1(0), da;
717 double tilt, tilt_avg(0), tilt_std(0);
718 double ddef, ddef_rmsd(0);
719 Vector3<double> loc, normal;
720 vector<double> b(3);
721 Matrix a(3,3);
722
723 Bfield* field;
724 Bmicrograph* mg;
725 Bparticle* part;
726
727 nptot = project_count_mg_particles(project);
728
729 for ( field = project->field; field; field = field->next ) {
730 for ( mg = field->mg; mg ; mg = mg->next, nmg++ ) if ( mg->select ) {
731 for ( i=0; i<3; i++ ) {
732 for ( j=0; j<3; j++ )
733 a[j][i] = 0;
734 b[i] = 0;
735 }
736 npart = 0;
737 ddef_rmsd = 0;
738 for ( part = mg->part; part; part = part->next ) if ( part->sel ) {
739 loc = part->loc - mg->origin;
740 ddef = part->def - mg->ctf->defocus_average();
741 ddef_rmsd += ddef*ddef;
742 loc[2] = ddef/mg->pixel_size[0];
743 for ( i=0; i<3; i++ ) {
744 for ( j=0; j<3; j++ )
745 a[j][i] += loc[i]*loc[j];
746 b[i] += loc[i];
747 }
748 npart++;
749 }
750 npsel += npart;
751 normal = fit_plane(a, b);
752 if ( normal[2] < 0 ) normal = -normal;
753 axis = atan2(-normal[0], normal[1]);
754 tilt = acos(normal[2]);
755 if ( nmg > 0 ) {
756 da = fabs(angle_set_negPI_to_PI(axis - a1));
757 if ( da > M_PI_2 ) {
758 axis = angle_set_negPI_to_PI(axis + M_PI);
759 tilt = -tilt;
760 }
761 } else {
762 a1 = axis;
763 }
764 if ( npart )
765 ddef_rmsd = sqrt(ddef_rmsd/npart);
766 if ( npart > 5 ) {
767 axis_avg += axis;
768 axis_std += axis*axis;
769 tilt_avg += tilt;
770 tilt_std += tilt*tilt;
771 nmgsel++;
772 }
773 if ( verbose )
774 cout << mg->id << tab << npart << tab << ddef_rmsd << tab << axis*180.0/M_PI << tab << tilt*180.0/M_PI << endl;
775 }
776 }
777
778 if ( nmgsel ) {
779 axis_avg /= nmgsel;
780 axis_std = axis_std/nmgsel - axis_avg*axis_avg;
781 if ( axis_std > 0 ) axis_std = sqrt(axis_std);
782 tilt_avg /= nmgsel;
783 tilt_std = tilt_std/nmgsel - tilt_avg*tilt_avg;
784 if ( tilt_std > 0 ) tilt_std = sqrt(tilt_std);
785 }
786
787 if ( verbose ) {
788 cout << "Number of micrographs: " << nmgsel << " (" << nmgsel*100.0/nmg << "%)" << endl;
789 cout << "Number of particles: " << npsel << " (" << npsel*100.0/nptot << "%)" << endl;
790 cout << "Axis average & stdev: " << axis_avg*180.0/M_PI << tab << axis_std*180.0/M_PI << endl;
791 cout << "Tilt average & stdev: " << tilt_avg*180.0/M_PI << tab << tilt_std*180.0/M_PI << endl;
792 }
793
794 return axis_avg;
795}
