47Bmodel* read_model_bild(vector<string> file_list)
48{
49 int j, k, nc, first_comment;
50 Bmodel* model = NULL;
51 Bmodel* mp = NULL;
52 Bcomponent* comp = NULL;
53 Bcomponent* comp1 = NULL;
54 Bcomponent* comp2 = NULL;
55 Blink* link = NULL;
56 string id, line;
57 vector<string> tokens;
58 ifstream fmod;
59 char aline[1024];
60 RGBA<float> rgba(1,1,1,1);
61 Vector3<double> loc, vec;
62
63 for ( auto filename: file_list ) {
64 if ( verbose & VERB_LABEL )
65 cout << "Reading file: " << filename << endl;
66 first_comment = 1;
67 fmod.open(filename.c_str());
68 if ( fmod.fail() ) return NULL;
69// id = string(i, "%d");
70// mp = model_add(&mp, id);
71// if ( !model ) model = mp;
72 if ( model ) mp = mp->add(base(filename));
73 else model = mp = new Bmodel(base(filename));
74 link = NULL;
75 comp = NULL;
76 nc = 0;
77 while ( fmod.getline(aline, 1024) ) {
78 line = aline;
79 if ( verbose & VERB_DEBUG )
80 cout << "DEBUG read_model_bild: " << line << endl;
81 tokens = split(line);
82 if ( tokens.size() ) {
83 if ( verbose & VERB_DEBUG )
84 cout << "DEBUG read_model_bild: " << tokens[0];
85 if ( tokens[0] == ".comment" ) {
86 if ( first_comment ) {
87 for ( j=1; j<tokens.size(); ++j )
88 mp->comment() += tokens[j] + " ";
89 first_comment = 0;
90 }
91 } else if ( tokens[0] == ".color" ) {
92 if ( verbose & VERB_DEBUG )
93 cout << " " << tokens[1];
94 if ( isdigit(tokens[1][0]) )
95 for ( j=0; j<tokens.size() && j<3; ++j )
96 rgba[j] = to_real(tokens[j+1]);
97 else
98 ;
99 } else if ( tokens[0] == ".transparency" ) {
100 rgba[3] = to_real(tokens[1]);
101 } else if ( tokens[0] == ".dot" || tokens[0] == ".dotat" ) {
102 for ( j=0; j<tokens.size() && j<3; ++j )
103 loc[j] = to_real(tokens[j+1]);
104 comp = component_add_if_not_exist(loc, &mp->comp, nc);
105 comp->radius(1);
106 comp->color(rgba);
107 } else if ( tokens[0] == ".sphere" ) {
108 for ( j=0; j<tokens.size() && j<3; ++j )
109 loc[j] = to_real(tokens[j+1]);
110 comp = component_add_if_not_exist(loc, &mp->comp, nc);
111 if ( tokens.size() > 4 ) comp->radius(to_real(tokens[4]));
112 comp->color(rgba);
113 } else if ( tokens[0] == ".m" || tokens[0] == ".move" ) {
114 for ( j=0; j<tokens.size() && j<3; ++j )
115 loc[j] = to_real(tokens[j+1]);
116 comp = component_add_if_not_exist(loc, &mp->comp, nc);
117 comp->color(rgba);
118 } else if ( tokens[0] == ".mr" || tokens[0] == ".moverel" ) {
119 for ( j=0; j<tokens.size() && j<3; ++j )
120 vec[j] = to_real(tokens[j+1]);
121 loc += vec;
122 comp = component_add_if_not_exist(loc, &mp->comp, nc);
123 comp->color(rgba);
124 } else if ( tokens[0] == ".d" || tokens[0] == ".draw" ) {
125 for ( j=0; j<tokens.size() && j<3; ++j )
126 loc[j] = to_real(tokens[j+1]);
127 comp = component_add_if_not_exist(loc, &mp->comp, nc);
128 comp->color(rgba);
129 if ( comp && comp2 ) {
130 link = link_add(&link, comp, comp2, 0, 1);
131 if ( !mp->link ) mp->link = link;
132 link->color(rgba);
133 }
134 } else if ( tokens[0] == ".dr" || tokens[0] == ".drawrel" ) {
135 for ( j=0; j<tokens.size() && j<3; ++j )
136 vec[j] = to_real(tokens[j+1]);
137 loc += vec;
138 comp = component_add_if_not_exist(loc, &mp->comp, nc);
139 comp->color(rgba);
140 if ( comp && comp2 ) {
141 link = link_add(&link, comp, comp2, 0, 1);
142 if ( !mp->link ) mp->link = link;
143 link->color(rgba);
144 }
145 } else if ( tokens[0] == ".v" || tokens[0] == ".vector" || tokens[0] == ".cylinder" ) {
146 for ( j=0; j<tokens.size() && j<3; ++j )
147 loc[j] = to_real(tokens[j+1]);
148 comp = component_add_if_not_exist(loc, &mp->comp, nc);
149 comp->color(rgba);
150 for ( j=0; j<tokens.size() && j<3; ++j )
151 loc[j] = to_real(tokens[j+4]);
152 comp1 = component_add_if_not_exist(loc, &mp->comp, nc);
153 comp1->color(rgba);
154 if ( comp && comp1 ) {
155 link = link_add(&link, comp, comp1, 0, 1);
156 if ( !mp->link ) mp->link = link;
157 link->color(rgba);
158 if ( tokens.size() > 7 ) link->radius(to_real(tokens[7]));
159 if ( link->radius() < 0.001 ) link->radius(1);
160 }
161 } else if ( tokens[0] == ".polygon" ) {
162 comp1 = NULL;
163 for ( j=0, k=1; k<tokens.size(); ++j, ++k ) {
164 loc[j] = to_real(tokens[k]);
165 if ( j > 2 ) {
166 j = 0;
167 comp = component_add_if_not_exist(loc, &mp->comp, nc);
168 comp->color(rgba);
169 if ( comp && comp1 ) {
170 link = link_add(&link, comp, comp1, 0, 1);
171 if ( !mp->link ) mp->link = link;
172 link->color(rgba);
173 }
174 comp1 = comp;
175 }
176 }
177 }
178 if ( verbose & VERB_DEBUG )
179 cout << endl;
180 }
181 comp2 = comp; // Previous component
182 }
183 fmod.close();
184 }
185
186 return model;
187}

377 bild_polygon_start(fbld, v1);
378 bild_polygon_add(fbld, v2);
379 bild_polygon_end(fbld, v3);
380 }

382 }

384 fbld << endl;
385

768BILD format
769
770BILD is a simple text format that describes lines, polygons, and geometric primitives such as spheres, boxes, cylinders, and cones with commands (see example.bild and XYZ-axes.bild). Character strings can also be included. The commands are used to generate a VRML model in Chimera. The objects can be specified in absolute coordinates and/or transformed and scaled coordinates. See also: shape, define, geometric objects
771
772This file type is indicated by the filename suffix .bild or .bld, or by using bild:filename in the command line.
773
774Curved objects (spheres, cones, and cylinders) read from BILD format are described as perfectly smooth in exported files, but display within Chimera uses planar facets. The number of facets and the apparent smoothness can be increased by raising the subdivision level with the Effects tool or the set command.
775
776BILD Commands
777
778Square brackets denote optional parameters.
779
780.arrow x1 y1 z1 x2 y2 z2 [r1 [r2 [rho]]]
781Draw an arrow from (x1, y1, z1) to (x2, y2, z2). An arrow consists of a cylinder stretching from the initial point to an intermediary junction, and a cone stretching from the junction to the final point. The radius of the cylinder is r1 (default 0.1), the radius of the base of the cone is r2 (default 4*r1), and the fraction of the total distance taken up by the cylinder is rho (default 0.75).
782.box x1 y1 z1 x2 y2 z2

784.color name
785or
786.color r g b
787Set the color of subsequently defined items. The name can be a built-in name, a name defined previously with colordef, or an integer that refers to the old BILD color wheel (0-65, inclusive). Alternatively, a color can be described by its red (r), green (g), and blue (b) components, each in the range 0-1, inclusive. Any transparency in the color is ignored, but transparency can be set separately.

789Define the starting location of the next character string. Lines in the BILD file that do not start with a period (.) are character strings to be displayed. See also .font.
