21int CTFparam::update(CTFparam& ctf)
22{
23 int i;
24
25 if ( ctf.identifier().length() ) id = ctf.identifier(); // Optics group
26 sel = ctf.sel; // Optics group number
27 if ( ctf.volt() ) volt(ctf.volt()); // Acceleration voltage (volts)
28 if ( ctf.focal_length() ) fl = ctf.focal_length(); // Focal length (angstrom)
29 if ( ctf.objective_aperture() ) oa = ctf.objective_aperture(); // Objective aperture (angstrom)
30 if ( ctf.Cc() ) cc = ctf.Cc(); // Chromatic aberation (angstrom)
31 if ( ctf.alpha() ) a = ctf.alpha(); // Illumination half-angle (radians)
32 if ( ctf.dE() ) de = ctf.dE(); // Energy spread (eV)
33 if ( ctf.baseline_type() ) { // Baseline equation
34 bt = ctf.baseline_type();
35 for ( i=0; i<NCTFPARAM; i++ ) base[i] = ctf.baseline(i);
36 }
37 if ( ctf.envelope_type() ) { // Envelope equation
38 et = ctf.envelope_type();
39 for ( i=0; i<NCTFPARAM; i++ ) env[i] = ctf.envelope(i);
40 }
41 if ( ctf.abw.size() ) abw = ctf.abw;
42 if ( ctf.f ) f = ctf.f; // Figure-of-merit
43
44 update_terms();
45
46 return 0;
47}

386int CTFparam::parse_baseline_equation(Bstring base_eq)
387{
388 if ( verbose & VERB_DEBUG )
389 cout << "DEBUG parse_baseline_equation: Base eq = " << base_eq << endl;
390
391 base_eq = base_eq.remove('\"');
392
393 int n(0);
394 base[1] = base[2] = base[3] = base[4] = 0;
395
396 if ( base_eq.contains("s3") ) { // Polynomial
397 bt = 1;
398 sscanf(base_eq.c_str(), "%lf + %lf*$s + %lf*$s2 + %lf*$s3 + %lf*$s4%n",
399 &base[0], &base[1], &base[2], &base[3], &base[4], &n);
400 } else if ( base_eq.contains("sqrt") ) { // EMAN style

404 } else if ( base_eq.contains("exp") ) { // Double gaussian
405 bt = 2;

408 } else if ( base_eq.length() ) {
409 bt = 0;
410 sscanf(base_eq.c_str(), "%lf%n", &base[0], &n);
411 } else {
412 bt = 0;
413 base[0] = 1;
414 }
415
416 // Water ring
417 Bstring wr = base_eq.substr(n, 100);
418// char* ptr = &base_eq[n];
419// if ( n < base_eq.length() && strstr(ptr, "exp") ) {
420 if ( wr.contains("exp") ) {

432 base[0] << "," << base[1] << "," << base[2] << "," <<
433 base[3] << "," << base[4] << endl;

446int CTFparam::parse_envelope_equation(Bstring env_eq)
447{
448 env_eq = env_eq.remove('"');
449
450 if ( env_eq.contains("s2") ) {
451 et = 2*env_eq.count('s') - 1;
452 if ( env_eq.index('+') > 0 )
453 if ( env_eq.index('+') < env_eq.index('*') ) et++;
454 } else if ( env_eq.contains("pow") ) {
455 et = 6;
456 } else {
457 et = 5;
458 }
459
460 if ( verbose & VERB_DEBUG )
461 cout << "DEBUG: parse_envelope_equation: " << et << tab << env_eq << endl;
462
463 switch ( et ) {
464 case 1: // Single gaussian
465 sscanf(env_eq.c_str(), "%lf*exp(%lf*$s2)",
466 &env[0], &env[1]);
467 break;
468 case 2: // Single gaussian with constant
469 sscanf(env_eq.c_str(), "%lf + %lf*exp(%lf*$s2)",
470 &env[0], &env[1], &env[2]);
471 break;
472 case 3: // Double gaussian
473 sscanf(env_eq.c_str(), "%lf*exp(%lf*$s2) + %lf*exp(%lf*$s2)",
474 &env[0], &env[1], &env[2], &env[3]);
475 break;
476 case 4: // Double gaussian with constant
477 sscanf(env_eq.c_str(), "%lf + %lf*exp(%lf*$s2) + %lf*exp(%lf*$s2)",
478 &env[0], &env[1], &env[2], &env[3], &env[4]);
479 break;
480 case 5: // Hyperbolic decay with constant
481 sscanf(env_eq.c_str(), "%lf + %lf/$s",
482 &env[0], &env[1]);
483 break;
484 case 6: // Exponential decay with constant
485 sscanf(env_eq.c_str(), "%lf + %lf*pow($s, %lf)",
486 &env[0], &env[1], &env[2]);
487 break;
488 default:
489 et = 4;
490 env[0] = 1; // Default envelope
491 env[1] = 0;
492 env[2] = -10;
493 env[3] = 0;
494 env[4] = -10;
495 }
496
497 return 0;
498}

764int ctf_update_from_json(CTFparam &cp, JSvalue &js)
765{
766// cout << js << endl;
767 if ( js.exists("Optics_group") ) cp.identifier(js["Optics_group"].value());
768 if ( js.exists("Volt") ) cp.volt(js["Volt"].real());
769 if ( js.exists("Focal_length") ) cp.focal_length(js["Focal_length"].real());
770 if ( js.exists("Focus_step") ) cp.focus_step(js["Focus_step"].real());
771 if ( js.exists("Objective_aperture") ) cp.objective_aperture(js["Objective_aperture"].real());
772 if ( js.exists("Slit_width") ) cp.slit_width(js["Slit_width"].real());
773 if ( js.exists("Cc") ) cp.Cc(js["Cc"].real());
774 if ( js.exists("Beam_convergence") ) cp.alpha(js["Beam_convergence"].real());
775 if ( js.exists("Energy_spread") ) cp.dE(js["Energy_spread"].real());

777 if ( js.exists("Aberration_even") ) {
778 vector<double> v = parse_real_vector(js["Aberration_even"].value().substr(1));
779 cp.aberration_even_update(v);
780 }
781 if ( js.exists("Aberration_odd") ) {
782 vector<double> v = parse_real_vector(js["Aberration_odd"].value().substr(1));
783 cp.aberration_odd(v);
784 }
785
786 vector<string> tags = cp.aberration_tags();
787 for ( auto t: tags )
788 if ( js.exists(t) )
789 cp.aberration_coefficient(t,js[t].real());
790
791 if ( js.exists("Aberration") ) {
792 JSvalue& jsab = js["Aberration"];
793 for ( auto t: tags )
794 if ( jsab.exists(t) )
795 cp.aberration_coefficient(t,jsab[t].real());
796 }
797
798 // These modify the aberration terms
799 if ( js.exists("Amplitude_phase") ) cp.amp_shift(js["Amplitude_phase"].real());
800 if ( js.exists("Defocus_average") ) cp.defocus_average(js["Defocus_average"].real());
801 if ( js.exists("Defocus_deviation") && js.exists("Astigmatism_angle") )
802 cp.astigmatism(js["Defocus_deviation"].real(), js["Astigmatism_angle"].real()*M_PI/180.0);
803 if ( js.exists("Cs") ) cp.Cs(js["Cs"].real());
804
805 if ( js.exists("Baseline_type") ) cp.baseline_type(js["Baseline_type"].integer());

951vector<double> defocus_range_profile(CTFparam& ctf, double freq_step, double def_min, double def_max, double def_step)
952{
953 double scut = ctf.frequency_cutoff();
954 long n(scut/freq_step+1);
955
956 if ( def_min < 1 ) {
957 vector<double> c(n, 0.5);
958 return c;
959 }
960
961 long i, m(0);
962 double def, s2, v;
963 vector<double> c(n, 0);
964
965 double def_ori = ctf.defocus_average();
966
967 for ( def = def_min; def <= def_max; def += def_step, ++m ) {
968 ctf.defocus_average(def);
969 for ( i=0; i<n; ++i ) {
970 s2 = i*freq_step;
971 s2 *= s2;
972 v = ctf.calculate(s2, 0);
973 c[i] += v*v;
974 }
975 }
976
977 ctf.defocus_average(def_ori); // Preserve the original defocus
978
979 for ( i=0; i<n; ++i )
980 c[i] /= m;
981
982 return c;
983}

996double defocus_factor(CTFparam& ctf, double def_min, double def_max, double def_step)
997{
998 if ( def_min < 1 ) return 0.5;
999
1000 double def_fac(0);
1001 vector<double> c = defocus_range_profile(ctf, 0.001, def_min, def_max, def_step);
1002
1003 for ( auto it = c.begin(); it != c.end(); ++it ) def_fac += *it;
1004
1005 def_fac /= c.size();
1006
1007 return def_fac;
