166## @brief Returns the image coordinates given canvas coordinates.
167#
168# @param x
169# @param y Canvas coordinates.
170
171proc imageCoordinatesFromCanvas { x y } {
172 global theimg
173 global px py
174 set c [getImageCanvas $theimg]
175 set wc [getControlWindow $theimg]
176 set scale [$wc.scale.scale get]
177 set z [$wc.slice.scale get]
178 set px [expr floor( [$c canvasx $x] / $scale ) ]
179 set py [expr floor( [$c canvasy $y] / $scale ) ]
180 set pyi [yFlip $py]
181 return "$px $pyi $z"
182}

1107## @brief Displays updated coordinates whenever the mouse is moved.
1108#
1109
1110proc showCoords { } {
1111 global filename
1112 global theimg imgtype
1113 global tool
1114 global px py
1115 if { [string length $filename] < 1 } { return }
1116 if ![Bimage exists $theimg] { return }
1117# puts "px = $px py = $py"
1118 set wc [getControlWindow $theimg]
1119 set img_num [$wc.image.scale get]
1120 set slice_num [$wc.slice.scale get]
1121 set value [Bimage get $theimg pixel $px $py $slice_num $img_num]
1122 if { $value eq " " } { return }
1123 $wc.pixel.x config -text "$px"
1124 $wc.pixel.y config -text "$py"
1125 $wc.pixel.value config -text "$value"
1126 if { $tool != "measure" } {
1127 set ps [getPixelSizeEntry]
1128# puts "pixel size = $ps"
1129 if { $imgtype == "ft" || $imgtype == "ps" } {
1130 set width [Bimage get $theimg width]
1131 set height [Bimage get $theimg height]
1132 set depth [Bimage get $theimg nslices]
1133 set dx [expr ($px*1.0/$width - 0.5) / [lindex $ps 0]]
1134 set dy [expr ($py*1.0/$height - 0.5) / [lindex $ps 1]]
1135 if { $slice_num > 1 } {
1136 set dz [expr ($slice_num*1.0/$depth - 0.5) / [lindex $ps 2]]
1137 } else {
1138 set dz 0
1139 }
1140# puts "$dx $dy $dz"
1141 set d [expr sqrt($dx*$dx + $dy*$dy + $dz*$dz)]
1142 if { $d < 0.0001 } { set d 0.0001 }
1143 set d [expr 1/$d]
1144 set u "1/A"
1145 } else {
1146 set origin [Bimage get $theimg origin]
1147 set dx [expr [lindex $ps 0]*($px - [lindex $origin 0])]
1148 set dy [expr [lindex $ps 1]*($py - [lindex $origin 1])]
1149 set dz [expr [lindex $ps 2]*($slice_num - [lindex $origin 2])]
1150 set d [expr sqrt($dx*$dx + $dy*$dy + $dz*$dz)]
1151 set u "A"
1152 }
1153 $wc.distance.value config -text "$d"
1154 $wc.distance.unit config -text "$u"
1155 }
1156}
