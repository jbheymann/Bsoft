16// Given a pointer to packed TGA color data, and the number of bits used
17// per color return an unpacked 32-bit color.
18RGB<unsigned char> TGA_to_RGB(const unsigned char * pColorData, unsigned int bitsPerColor )
19{
20 RGB<unsigned char> color;
21 switch( bitsPerColor )
22 {
23 case 15:
24 // MSB LSB
25 // xRRRRRGG GGGBBBBB
26 color[0] = (pColorData[1] >> 2) & 0x1F;
27 color[1] = ((pColorData[1] << 3) & 0x1C) | ((pColorData[0] >> 5) & 0x07);
28 color[2] = (pColorData[0] & 0x1F);
29
30 // Convert 5-bit channels to 8-bit channels by left shifting by three
31 // and repeating the first three bits to cover the range [0,255] with
32 // even spacing. For example:
33 // channel input bits: 4 3 2 1 0
34 // channel output bits: 4 3 2 1 0 4 3 2
35 color[0] = (color[0] << 3) | (color[0] >> 2);
36 color[1] = (color[1] << 3) | (color[1] >> 2);
37 color[2] = (color[2] << 3) | (color[2] >> 2);
38 break;
39
40 case 24:
41 // MSB LSB
42 // RRRRRRRR GGGGGGGG BBBBBBBB
43 color[0] = pColorData[2];
44 color[1] = pColorData[1];
45 color[2] = pColorData[0];
46 break;
47 }
48
49 return color;
50}

54RGBA<unsigned char> TGA_to_RGBA(const unsigned char * pColorData, unsigned int bitsPerColor )

64 color[1] = ((pColorData[1] << 3) & 0x1C) | ((pColorData[0] >> 5) & 0x07);
65 color[2] = (pColorData[0] & 0x1F);
66

68 // and repeating the first three bits to cover the range [0,255] with
69 // even spacing. For example:
70 // channel input bits: 4 3 2 1 0
71 // channel output bits: 4 3 2 1 0 4 3 2
72 color[0] = (color[0] << 3) | (color[0] >> 2);
73 color[1] = (color[1] << 3) | (color[1] >> 2);
74 color[2] = (color[2] << 3) | (color[2] >> 2);
75 break;
76
77 case 16:
78 // MSB LSB
79 // ARRRRRGG GGGBBBBB
80 color[3] = 255 * ((pColorData[1] & 0x80) >> 7);

90 color[0] = (color[0] << 3) | (color[0] >> 2);
