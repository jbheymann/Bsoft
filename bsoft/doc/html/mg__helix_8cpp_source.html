102double nodes_center(Bfilnode* fnode, Bimage* p, long img_num, int width)
103{
104 long nspline(0);
105
106 Vector3<double>* spline = vector3_spline_from_nodes(fnode, nspline);
107
108 Vector3<long> size(width, nspline, 1);
109
110 Bimage* pfil = p->extract_filament(img_num, width, 2, nspline, spline);
111
112 delete[] spline;
113
114 long i, x, y, n(0);
115 long ns, ne;
116 Complex<float>* profile = new Complex<float>[pfil->sizeX()];
117 double len(0), max, pwr, off, cc(0);
118 Vector3<double> loc(fnode->loc), vec;
119 Bfilnode* fn;
120 Matrix3 mat(0,1,0,-1,0,0,0,0,1);
121
122 fft_plan planf = fft_setup_plan(pfil->sizeX(), 1, 1, FFTW_FORWARD, 1);

125 if ( verbose )
126 cout << "Node\tOffset\tCC" << endl;
127 for ( fn = fnode; fn; fn = fn->next ) {
128 len += fn->loc.distance(loc);
129 ns = (long) (len - width);
130 ne = (long) (len + width);
131 if ( ns < 0 ) ns = 0;
132 if ( ne > nspline ) ne = nspline;
133 for ( i=0; i<pfil->sizeX(); i++ ) profile[i] = 0;
134 for ( y=ns; y<ne; y++ ) {
135 i = y*pfil->sizeX();
136 for ( x=0; x<pfil->sizeX(); x++, i++ ) {
137 profile[x][0] += (*pfil)[i];
138 }
139 }
140 fftw(planf, profile);
141 for ( pwr=0, i=0; i<pfil->sizeX(); i++ ) {
142 pwr += profile[i].power();
143 profile[i] *= profile[i];
144 }
145 fftw(planb, profile);
146 for ( max=-1000, off=0, i=0; i<pfil->sizeX(); i++ ) if ( max < profile[i].real() ) {
147 max = profile[i].real();
148 off = i;
149 }
150// max /= sqrt(2*pwr);
151 max /= pwr;
152 cc += max;
153 n++;
154 if ( off > pfil->sizeX()/2 ) off -= (long)pfil->sizeX();
155 off /= 2;
156 if ( verbose )
157// cout << fn->id << tab << m - (long)pfil->sizeX()/2 << endl;
158 cout << fn->id << tab << off << tab << max << endl;
159 if ( fn->next ) vec = fn->next->loc;
160 else vec = fn->loc;
161 vec -= loc;
162 vec = mat * vec;
163 vec.normalize();
164 fn->loc += vec * off;
165 loc = fn->loc;
166 }
167
168 delete pfil;
169 delete[] profile;
170 fft_destroy_plan(planf);
171 fft_destroy_plan(planb);
172
173 if ( n ) cc /= n;
174
175 return cc;
176}
