207int img_tile_with_overlap(Bimage* p, Bstring outputfile, Vector3<int> size,
208 Vector3<int> overlap, int digits, int first_number, DataType nudatatype)
209{
210 Bstring filename;
211 char format[16];
212 Bimage* pone = NULL;
213 Vector3<double> start;
214
215 if ( digits < 2 ) digits = 2;
216
217 snprintf(format, 16, "%%0%dd.", digits);
218
219 if ( size[0] < 1 ) size[0] = 512;
220 if ( size[1] < 1 ) size[1] = 512;
221 if ( size[2] < 1 ) size[2] = 512;
222 if ( size[0] > p->sizeX() ) size[0] = p->sizeX();
223 if ( size[1] > p->sizeY() ) size[1] = p->sizeY();
224 if ( size[2] > p->sizeZ() ) size[2] = p->sizeZ();
225 if ( overlap[0] < 0 ) overlap[0] = 0;
226 if ( overlap[1] < 0 ) overlap[1] = 0;
227 if ( overlap[2] < 0 ) overlap[2] = 0;
228 if ( overlap[0] > p->sizeX() ) overlap[0] = p->sizeX()/2;
229 if ( overlap[1] > p->sizeY() ) overlap[1] = p->sizeY()/2;
230 if ( overlap[2] > p->sizeZ() ) overlap[2] = p->sizeZ()/2;
231
232 int n, i = first_number;
233 Vector3<long> step = size - overlap;
234 Vector3<long> xsize;
235
236 filename = outputfile.pre_rev('.') + ".tiles";
237 n = (p->sizeX()/step[0] + 1)*(p->sizeY()/step[1] + 1)*(p->sizeZ()/step[2] + 1);
238
239 if ( verbose ) {
240 cout << "Splitting image " << p->file_name() << " into tiles:" << endl;
241 cout << "Tile size: " << size << endl;
242 cout << "Overlap: " << overlap << endl;
243 cout << "Number of tiles: " << n << endl << endl;
244 }
245
246 ofstream fd(filename.c_str());
247 if ( fd.fail() ) return 0;
248
249 fd << p->sizeX() << " " << p->sizeY() << " " << p->sizeZ() << endl;
250 fd << overlap[0] << " " << overlap[1] << " " << overlap[2] << endl;
251
252 for ( n=0, start[2]=0; start[2]<p->sizeZ(); start[2]+=step[2] ) {
253 xsize[2] = p->sizeZ() - (long)start[2];
254 if ( xsize[2] > size[2] ) xsize[2] = size[2];
255 for ( start[1]=0; start[1]<p->sizeY(); start[1]+=step[1] ) {
256 xsize[1] = p->sizeY() - (long)start[1];
257 if ( xsize[1] > size[1] ) xsize[1] = size[1];
258 for ( start[0]=0; start[0]<p->sizeX(); start[0]+=step[0], i++ ) {
259 xsize[0] = p->sizeX() - (long)start[0];
260 if ( xsize[0] > size[0] ) xsize[0] = size[0];
261 pone = p->extract(0, start, xsize);
262 pone->label(p->label());
263 pone->change_type(nudatatype);
264 filename = outputfile.pre_rev('.') + Bstring(i, format) + outputfile.post_rev('.');
265 if ( verbose & VERB_LABEL )
266 cout << "File " << i << ": " << filename << endl;
267 write_img(filename, pone, 0);
268 delete pone;
269 fd << start[0] << " " << start[1] << " " << start[2] << endl;
270 n++;
271 }
272 }
273 }
274
275 fd.close();
276
277 if ( verbose & VERB_LABEL )
278 cout << "Number of tiles: " << n << endl << endl;
279
280 return n;
