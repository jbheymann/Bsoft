335int img_fft_times(int ndim, int minsize, int maxsize, int opt)
336{
337 long i, size;
338 vector<long> prime;
339 Bimage* p = NULL;
340 double tvs, tvp, tvf;
341 double dtp, dte;
342 fft_plan plan;
343
344 if ( ndim < 1 ) ndim = 1;
345 if ( ndim > 3 ) ndim = 3;
346 if ( minsize < 1 ) minsize = 1;
347 if ( maxsize < minsize ) maxsize = minsize;
348
349 long n = maxsize - minsize + 1;
350 int_float* t = new int_float[n];
351
352 cout << endl << "Timing the execution of fast Fourier transforms:" << endl;
353 cout << "FFTW planner option: " << opt << endl;
354 cout << "Number of dimensions: " << ndim << endl;
355 cout << "Size range: " << minsize << " - " << maxsize << endl << endl;
356 cout << "Size\tPtime\tEtime\tPrime factors" << endl;
357 for ( size = minsize; size <= maxsize; size++ ) {
358 switch ( ndim ) {
359 case 1:
360 p = new Bimage(Float, TComplex, size, 1, 1, 1);
361 break;
362 case 2:
363 p = new Bimage(Float, TComplex, size, size, 1, 1);
364 break;
365 case 3:
366 p = new Bimage(Float, TComplex, size, size, size, 1);
367 break;
368 }
369 prime = prime_factors(size);
370 tvs = getwalltime();
371 plan = p->fft_setup(FFTW_FORWARD, opt);
372 tvp = getwalltime();
373 if ( p->fft(plan, 0) )
374 return error_show("img_fft_times", __FILE__, __LINE__);
375 tvf = getwalltime();
376 fft_destroy_plan(plan);
377 dtp = tvp - tvs;
378 dte = tvf - tvp;
379 cout << size << tab << fixed << setprecision(5) << dtp << tab << dte;
380 for ( i=0; i<prime.size(); i++ ) cout << tab << prime[i];
381 cout << endl;
382 delete p;
383 t[size-minsize].i = size;
384 t[size-minsize].f = dte;
385 }
386 cout << endl;
387
388 qsort((void *) t, n, sizeof(int_float), QsortSmallToLargeIntFloat);
389
390 cout << "Ordered times:" << endl;
391 cout << "Size\tEtime" << endl;
392 for ( i=0, size=minsize; i<n; i++, size++ ) cout << t[i].i << tab << t[i].f << endl;
393 cout << endl;
394
395 delete[] t;
396
397 return 0;
