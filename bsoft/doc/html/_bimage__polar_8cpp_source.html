79Bimage* Bimage::radial(long minrad, long maxrad, double rad_step,
80 double ellipticity, double angle, Bimage* pmask, int wrap)
81{
82 long i, j, nn, xx, yy, zz, ir, nrad(0);
83 double dx, dy, dz, x2, y2, z2, a, r, f, f1;
84
85 if ( rad_step < 0.1 ) rad_step = 1;
86
87 Vector3<long> h(size()/2);
88 Vector3<double> origin;
89 Vector3<double> scale(1/rad_step, 1/rad_step, 1/rad_step);
90 if ( z < 2 ) scale[2] = 1;
91
92 if ( maxrad < minrad ) swap(minrad, maxrad);
93 if ( maxrad < 1 ) maxrad = size().max()/(2*rad_step);
94 else if ( maxrad > size().max()/(2*rad_step) ) maxrad = size().max()/(2*rad_step);
95
96 Bimage* prad = new Bimage(Float, TSimple, maxrad, 1, 1, n);
97 prad->sampling(rad_step*image->sampling()[0], 1, 1);
98
99 vector<float> num(maxrad);
100
101 if ( pmask )
102 if ( pmask->data_type() != UCharacter || pmask->minimum() != 0 || pmask->maximum() != 1 )
103 pmask->to_mask();
104
105 if ( verbose & VERB_PROCESS ) {
106 cout << "Calculating a radial profile:" << endl;
107 cout << "Minimum and maximum pixels: " << minrad << " " << maxrad << endl;
108 cout << "Radial step size: " << rad_step << endl;
109 cout << "Minimum and maximum radius: " << minrad*image->sampling()[0] << " " << maxrad*image->sampling()[0] << " A" << endl;
110 cout << "Radial step size: " << rad_step*image->sampling()[0] << " A" << endl;
111 if ( pmask )
112 cout << "Mask file: " << pmask->file_name() << endl;
113 }
114
115 if ( verbose & VERB_PROCESS )
116 cout << "Image\toriX\toriY\toriZ" << endl;
117 for ( i=nn=0; nn<n; nn++ ) {
118 for ( j=0; j<maxrad; j++ ) num[j] = 0;
119 if ( x > 1 ) origin[0] = image[nn].origin()[0];
120 if ( y > 1 ) origin[1] = image[nn].origin()[1];
121 if ( z > 1 ) origin[2] = image[nn].origin()[2];
122// if ( (vector3_origin_to_shift(origin, size())).length() < x/10 ) wrap = 1;
123// else wrap = 0;
124 if ( verbose & VERB_PROCESS )
125 cout << nn+1 << tab << origin << endl;
126 dz = -origin[2];
127 if ( wrap && dz < -h[2] ) dz += z;
128 for ( zz=0; zz<z; zz++, dz++ ) {
129 if ( wrap && dz > h[2] ) dz -= z;
130 z2 = dz * scale[2];
131 z2 *= z2;
132 dy = -origin[1];
133 if ( wrap && dy < -h[1] ) dy += y;
134 for ( yy=0; yy<y; yy++, dy++ ) {
135 if ( wrap && dy > h[1] ) dy -= y;
136 y2 = dy * scale[1];
137 y2 *= y2;
138 dx = -origin[0];
139 if ( wrap && dx < -h[0] ) dx += x;
140 for ( xx=0; xx<x; xx++, dx++, i++ ) if ( ( !pmask ) || ( pmask && (*pmask)[i] ) ) {
141 if ( wrap && dx > h[0] ) dx -= x;
142 x2 = dx * scale[0];
143 x2 *= x2;
144 r = sqrt(x2 + y2 + z2);
145 if ( fabs(ellipticity - 1) > 0.001 ) {
146 a = atan2(dy, dx) - angle;
147 r *= sqrt((ellipticity*ellipticity+1)/
148 (2*(ellipticity*ellipticity*cos(a)*cos(a)+
149 sin(a)*sin(a))));
150 }
151 ir = (long) r;
152 if ( ( ir >= minrad ) && ( ir < maxrad ) ) {
153 f = r - ir;
154 f1 = 1 - f;
155 j = ir + nn*maxrad; // Add for multiple images
156 num[ir] += f1;
157 prad->add(j, f1*(*this)[i]);
158 ir++;
159 if ( ir < maxrad ) {
160 j++;
161 num[ir] += f;
162 prad->add(j, f*(*this)[i]);
163 }
164 nrad++;
165 }
166 }
167 }
168 }
169 for ( j=0, ir = nn*maxrad; j<maxrad; j++, ir++ ) {
170 if ( num[j] ) {
171 prad->set(ir, (*prad)[ir]/num[j]);
172 }
173 }
174 prad->image[nn].origin(0.0,0.0,0.0);
175 }
176
177 if ( verbose & VERB_PROCESS )
178 cout << "Pixels in radial averages: " << nrad << endl;
179
180 if ( verbose & VERB_DEBUG )
181 cout << "DEBUG Bimage:radial: prad size = " << prad->sizeX() << endl;
182
183 prad->statistics();
184
185 return prad;
186}

261/*
262@brief Least squares function for the radial_fit function.
263@param &simp a simplex structure.
264@return double the least squares R value.
265
266 A radial profile is fitted to a reference profile using an iterative
267 simplex down-hill method. The equations solved are:
268 xref = x*m
269 yref(xref) = y(x)*a + b
270 where m is the dimension scaling (or magnification).
271 a is the amplitude scaling.
272 b is the amplitude shift.
273 The least squares function is:
274 sum(weight*(yref(xref) - y(x)*a - b)^2)
275 R = sqrt(---------------------------------------)
276 sum(weight)
277**/
278double radial_R(Bsimplex& simp)
279{
280 vector<double>& y = simp.dependent_values();
281 vector<double>& yref = simp.independent_values();
282
283 int i, iref, n(0);
284 double fraction, xref, df, R(0), sum(0), weight;
285
286 for ( i=0; i<simp.points(); i++ ) {
287 xref = simp.parameter(0)*i;
288 iref = (int) xref;
289 if ( iref >= 0 && iref < simp.points() - 1 ) {
290 fraction = xref - iref;
291 df = (1.0 - fraction)*yref[iref] + fraction*yref[iref + 1]
292 - simp.parameter(1)*y[i] - simp.parameter(2);
293// weight = i*(fabs(yref[iref]) + fabs(y[i]));
294 weight = i;
295// weight = fabs(y[i]);
296 sum += weight;
297 R += df*df*weight;
298 n++;
299 }
300 }
301 if ( sum && n > 0.2*simp.points() ) R = sqrt(R/sum);
302 else R = 1e30; // Out of range
303
304 if ( verbose & VERB_DEBUG )
305 cout << "DEBUG radial_R: " << simp.parameter(0) << " " <<
306 simp.parameter(1) << " " << simp.parameter(2) << " R=" << R << endl;
307
308 return R;
309}

325double* Bimage::radial_fit(Bimage* pref)
326{
327 long i, npoint(x);
328 if ( x < pref->sizeX() ) npoint = pref->sizeX();
329
330 // 3 parameters: Dimension scaling, amplitude scaling, amplitude shift
331 vector<double> ax(npoint), fx(npoint);
332 for ( i=0; i<npoint; i++ ) ax[i] = (*pref)[i];
333 for ( i=0; i<npoint; i++ ) fx[i] = (*this)[i];
334
335 Bsimplex simp(1, 3, 0, npoint, ax, fx);
336
337 // Dimension scaling estimated from the input units
338 simp.parameter(0, image->sampling()[0]/pref->sampling(0)[0]);
339
340 // Amplitude scaling estimated from the standard deviations
341 simp.parameter(1, 1);
342 if ( std != 0 && pref->standard_deviation() != 0 )
343 simp.parameter(1, pref->standard_deviation()/std);
344 else if ( max - min > 0 && pref->maximum() - pref->minimum() > 0 )
345 simp.parameter(1, (pref->maximum() - pref->minimum())/(max - min));
346
347 // Amplitude shift estimated from the averages
348 simp.parameter(2, pref->average() - avg*simp.parameter(1));
349// if ( simp.parameter(2] < 1e-6 && max - min > 0 && pref->maximum() - pref->minimum() > 0 )
350// simp.parameter(2, (pref->maximum() - pref->minimum() + max - min)/2.0;
351
352 // Parameter limits
353 simp.limits(0, 0.8*simp.parameter(0), 1.2*simp.parameter(0));
354 simp.limits(1, 0.1*simp.parameter(1), 10*simp.parameter(1));
355 simp.limits(2, simp.parameter(2) - 10*pref->standard_deviation(), simp.parameter(2) + 10*pref->standard_deviation());
356
357 if ( verbose & VERB_PROCESS ) {
358 cout << "Radial fit initial parameters:" << endl;
359 cout << "Dimension scale: " << simp.parameter(0) << endl;
360 cout << "Amplitude scale and shift: " << simp.parameter(1) << " " << simp.parameter(2) << endl;
361 }
362
363 // Do the downhill simplex iterative fit
364 double R = simp.run(10000, 1e-12*pref->standard_deviation(), radial_R) / pref->standard_deviation();
365
366 int idat;
367 double fraction, xdat, f;
368
369 if ( verbose & VERB_PROCESS ) {
370 cout << "Radial fit final parameters:" << endl;
371 cout << "Dimension scale: " << simp.parameter(0) << endl;
372 cout << "Relative dimension scale: " << simp.parameter(0)*pref->sampling(0)[0]/image->sampling()[0] << endl;
373 cout << "Amplitude scale and shift: " << simp.parameter(1) << " " << simp.parameter(2) << endl;
374 cout << "R: " << R << endl << endl;
375 cout << "Radial averages:" << endl;
376 cout << "Radius(A) " << file_name() << " " << pref->file_name() << endl;
377 }
378
379 if ( verbose & VERB_PROCESS ) {
380 for ( i=0; i<pref->sizeX(); i++ ) {
381 xdat = i/simp.parameter(0);
382 idat = (int) xdat;
383 if ( idat >= 0 && idat < x - 1 ) {
384 fraction = xdat - idat;
385 f = simp.parameter(1)*((1.0 - fraction)*(*this)[idat]
386 + fraction*(*this)[idat+1]) + simp.parameter(2);
387 } else {
388 f = 0;
389 }
390 cout << pref->sampling(0)[0]*i << " " << f << " " << (*pref)[i] << endl;
391 }
392 }
393
394 double* result = new double[simp.parameters()];
395 for ( i=0; i<simp.parameters(); i++ ) result[i] = simp.parameter(i);
396
397 return result;

410{

414 p->data_type(Float);
415 p->size(nusize);
416 if ( p->sizeX() < 2 ) origin[0] = 0;
417 if ( p->sizeY() < 2 ) origin[1] = 0;
418 if ( p->sizeZ() < 2 ) origin[2] = 0;
419 p->data_alloc();

421 long i, nn, xx, yy, zz, ir, nexp(0);
422 double r, f, v;
423 Vector3<double> d;
424
425 if ( verbose ) {
426 cout << "Calculating a full image from a radial profile:" << endl;
427 cout << "Size: " << p->size() << endl;
428 cout << "Origin: " << origin << endl;
429 }
430
431 for ( nn=i=0; nn<n; nn++ ) {
432 for ( zz=0; zz<p->sizeZ(); zz++ ) {
433 d[2] = zz - origin[2];
434 for ( yy=0; yy<p->sizeY(); yy++ ) {
435 d[1] = yy - origin[1];
436 for ( xx=0; xx<p->sizeX(); xx++, i++ ) {
437 d[0] = xx - origin[0];
438 r = d.length();
439 ir = (long) r;
440 v = (*p)[i];
441 if ( ir < x ) {
442 f = 1 - r + ir;
443 v += f*(*this)[ir];
444 ir++;
445 f = 1 - f;
446 if ( ir < x ) v += f*(*this)[ir];
447 else v += f*background(nn);
448 nexp++;
449 } else {
450 v = background(nn);
451 }
452 p->set(i, v);
453 }
454 }
455 }
456 p->image[nn].origin(origin);
457 }
458
459 if ( verbose )
460 cout << "Pixels generated: " << nexp << endl << endl;
461
462 p->statistics();
463
464 return p;
465}

526 }
527
528 // Interpolate cartesian points to get spherical voxel values
529 if ( verbose & VERB_FULL )
530 cout << "Image\tOrigin" << endl;
531 for ( nn=0; nn<n; nn++ ) {
532 if ( verbose & VERB_FULL )
533 cout << nn+1 << tab << image[nn].origin() << endl;
534 for ( theta=0; theta<ntheta; theta++ ) {
535 cos_theta = dannulus*cos(dtheta*theta);
536 sin_theta = dannulus*sin(dtheta*theta);
537 for ( phi=0; phi<nphi; phi++ ) {
538 cos_phi = cos(dphi*phi);
539 sin_phi = -sin(dphi*phi);
540 for ( ann=0; ann<nannuli; ann++ ) {
541 j = psph->index(ann, phi, theta, nn);
542 scale = ann*sin_theta;
543 coor = Vector3<double>(scale*cos_phi, scale*sin_phi, ann*cos_theta);
544 coor += image[nn].origin();
545 psph->set(j, interpolate(coor, nn, 0));
546 }
547 }
548 }
549 psph->image[nn].origin(0.0, nphi/2, ntheta/2);
550 }
551
552 if ( verbose & VERB_PROCESS ) cout << endl;
553
554 if ( verbose & VERB_DEBUG )
555 cout << "DEBUG Bimage::cartesian_to_spherical: Calculating statistics" << endl;
556
557 statistics();
558
559 return psph;
560}

696Bimage* Bimage::polar_transform(long nangles, long ann_min, long ann_max,
697 long dann, long zmin, long zmax, long zinc)
698{
699 change_type(Float);
700
701 if ( dann < 1 ) dann = 1;
702 if ( ann_min >= x/2 ) ann_min = x/2;
703 if ( ann_max < 1 || ann_max >= x/2 ) ann_max = x/2;
704 if ( ann_max >= y/2 ) ann_max = y/2;
705 if ( ann_min > ann_max ) swap(ann_min, ann_max);
706 if ( ann_min == ann_max ) ann_max = ann_min + dann;
707
708 if ( zinc < 1 ) zinc = 1;
709 if ( zmin > zmax ) swap(zmin, zmax);
710 if ( zmin == zmax ) zmax = zmin + zinc;
711 if ( zmax >= z ) zmax = z - 1;
712
713 long i, ann, ang, r, zz, iz;
714 long nannuli((ann_max - ann_min)/dann + 1);
715 long nz((zmax - zmin)/zinc + 1);
716 double a, cosa, sina;
717 double dang(M_PI*2.0/nangles);
718 Vector3<double> coor;
719
720 Bimage* p = new Bimage(Float, TSimple, nangles, nannuli, nz, 1);
721
722 if ( verbose & VERB_PROCESS ) {
723 cout << "Calculating a polar image from " << file_name() << endl;
724 cout << "Image size: " << nangles << " x " << nannuli << " x " << nz << endl;
725 cout << "Angular step size: " << dang*180/M_PI << " degrees" << endl;
726 cout << "Annular range: " << ann_min << " - " << ann_max << " pixels" << endl;
727 cout << "Annular step size: " << dann << " pixels" << endl;
728 cout << "Z range: " << zmin << " - " << zmax << " pixels" << endl;
729 cout << "Z step size: " << zinc << " pixels" << endl << endl;
730 } else if ( verbose & VERB_LABEL )
731 cout << "Calculating a polar image" << endl << endl;
732
733 for ( i=0, zz=zmin; zz<=zmax; zz+=zinc ) {
734 for ( ann=ann_min; ann<=ann_max; ann+=dann ) {
735 for ( ang=0; ang<nangles; ang++, i++ ) {
736 a = dang*ang;
737 cosa = cos(a);
738 sina = sin(a);
739 for ( iz=zz; iz<zz+zinc; iz++ ) {
740 for ( r=ann; r<ann+dann; r++ ) {
741 coor[0] = r*cosa + image->origin()[0];
742 coor[1] = r*sina + image->origin()[1];
743 coor[2] = iz;
744 p->add(i, interpolate(coor));
745 }
746 }
747 }
748 }
749 }
750
751 if ( verbose & VERB_DEBUG )
752 cout << "DEBUG Bimage::polar_transform: Calculating statistics" << endl;
753
754 p->statistics();
755
756 return p;
757}

771Bimage* Bimage::polar_power_spectrum(double resolution, long num_angle)
772{
773 check_resolution(resolution);
774
775 long max_rad = (long) (x*image->sampling()[0]/resolution);
776 if ( max_rad > x/2 ) max_rad = x/2;

788 max_ang = num_angle*r;

1084int Bimage::radial_section(Bimage* prad, long nn,
1085 long zr, double rad_start, double rad_step, double fill)
1086{
1087 long j, j1, xx, yy, zz, zj;
1088 long slice_size = x*y;
1089 long image_size = slice_size*z;
1090 long i = (long) (nn*prad->size().volume() + zr*slice_size);
1091 double r = rad_step*zr + rad_start;
1092 double dr, drp;
1093
1094 for ( yy=0; yy<y; yy++ ) {
1095 for ( xx=0; xx<prad->sizeX(); xx++, i++ ) {
1096 j = nn*image_size + yy*x + xx;
1097 for ( zz=0, zj=z+1, drp=1; zz<z && zj>=z; zz++, j+=slice_size ) {
1098 dr = (*next)[j] - r;
1099 if ( dr > 0 && drp <= 0 ) zj = zz;
1100 drp = dr;
1101 }
1102 if ( zj < z ) {
1103 j1 = nn*image_size + zj*slice_size + yy*x + xx;
1104 j = j1 - slice_size;
1105 dr = ((*next)[j1] - r)/((*next)[j1] - (*next)[j]);
1106 prad->set(i, dr*(*this)[j]);
1107 dr = 1 - dr;
1108 prad->add(i, dr*(*this)[j1]);
1109 } else {
1110 prad->set(i, fill);
1111 }
1112 }
1113 }
