17double rectangle_edge(Vector3<double> d, Vector3<double> hr)
18{
19 d = d.abs() - hr;
20
21 if ( hr[2] > 1 && d[1] < d[2] ) d[1] = d[2];
22 if ( hr[1] > 1 && d[0] < d[1] ) d[0] = d[1];
23
24 return d[0];
25}

27double oval_edge(Vector3<double> d, Vector3<double> hr)
28{
29 double f(0);
30
31 Vector3<double> de(d/hr);
32 double ld(d.length());
33
34 if ( ld < 1 ) return -100;
35
36 double lde(de.length());
37
38 if ( lde > 1e-20 )
39 f = ld*(1.0 - 1.0/lde);
40
41 return f;
42}

44double cylinder_edge(Vector3<double> d, Vector3<double> hr)
45{
46 double f(0), ze(fabs(d[2])-hr[2]);
47
48 d[2] = 0;
49
50 double ld(d.length());
51
52 if ( ld < 1 ) return ze;
53
54 Vector3<double> de(d/hr);
55 double lde(de.length());
56
57 if ( lde > 1e-20 )
58 f = ld*(1.0 - 1.0/lde);
59
60 if ( f < ze ) f = ze;
61
62 return f;
63}

65double shell_edge(Vector3<double> d, double mind, double maxd)
66{
67 double f(0);
68
69 double ld(d.length());
70
71 if ( mind < 0.001 && ld < 0.001 ) return 1; // At origin
72
73 mind = ld - mind ;
74 maxd -= ld;
75
76 if ( mind < maxd ) f = mind;
77 else f = maxd;
78
79 return f;
80}

532int Bimage::hanning_taper(double fill)
533{
534
535 long i, nn, xx, yy, zz, cc;
536 double vol, v;
537 Vector3<double> cx, han;
538
539 cx[0] = M_PI * 2.0L / ( x - 1);
540 cx[1] = M_PI * 2.0L / ( y - 1);
541 cx[2] = M_PI * 2.0L / ( z - 1);
542
543 if ( verbose & VERB_FULL ) {
544 cout << "Hann tapering window:" << endl;
545 cout << "Size: " << size() << endl;
546 cout << "Fill value: " << fill << endl << endl;
547 }
548
549 for ( i=nn=0; nn<n; nn++ ) {
550 for ( zz=0; zz<z; ++zz ) {
551 han[2] = 0.5 * (1. - cos(cx[2]*zz));
552 for ( yy=0; yy<y; ++yy ) {
553 han[1] = 0.5 * (1. - cos(cx[1]*yy));
554 for ( xx=0; xx<x; ++xx ) {
555 han[0] = 0.5 * (1. - cos(cx[0]*xx));
556 vol = han.volume();
557 for ( cc=0; cc<c; cc++, ++i ) {
558 v = ((*this)[i] - fill)*vol + fill;
559 set(i, v);
560 }
561 }
562 }
563 }
564 }
565
566 return 0;

641int Bimage::gaussian_sphere(long nn, Vector3<double> center, double sigma, double amp, bool wrap)
642{
643 if ( sigma < 1 ) sigma = 1;
644 if ( amp == 0 ) amp = 1;
645
646 if ( verbose ) {
647 cout << "Calculating a Gaussian sphere:" << endl;
648 cout << "Center: " << center << endl;
649 cout << "Sigma: " << sigma << endl;
650 cout << "Amplitude: " << amp << endl;
651 if ( wrap ) cout << "Wrapping" << endl;
652 cout << endl;
653 }
654
655 long i, xx, yy, zz, cc;
656 double x2, y2, z2, d2, v;
657 double fac(-0.5/(sigma*sigma));
658 Vector3<double> h(size()/2);
659
660 for ( i=zz=0; zz<z; ++zz ) {
661 z2 = zz - center[2];
662 if ( wrap )
663 if ( fabs(z2) > h[2] ) z2 -= z;
664 z2 *= z2;
665 for ( yy=0; yy<y; ++yy ) {
666 y2 = yy - center[1];
667 if ( wrap )
668 if ( fabs(y2) > h[1] ) y2 -= y;
669 y2 *= y2;
670 for ( xx=0; xx<x; ++xx ) {
671 x2 = xx - center[0];
672 if ( wrap )
673 if ( fabs(x2) > h[0] ) x2 -= x;
674 x2 *= x2;
675 d2 = x2 + y2 + z2;
676 v = amp*exp(fac*d2);
677 for ( cc=0; cc<c; cc++, ++i )
678 add(i, v);
679 }
680 }
681 }
682
683 return 0;
684}
