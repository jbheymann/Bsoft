70/*
71@brief Generates polygons based on a vertex network.
72@param *comp current end vertex used to select next link.
73@param ilink previously selected link index.
74@param nlink number of links so far in polygon.
75@param *poly polygon structure.
76@return int number of vertices in polygon.
77
78 A choice is made between the two non-originating connections by calculating
79 the angle between the vertex normal and the cross product of the originating
80 and non-originating bond. If this angle is smaller than 90 degrees, it is
81 selected to continue following. Each pair of bonds are flagged to not traverse
82 the same polygon more than once.
83
84**/
85int poly_get_connectivity(Bcomponent* comp, int ilink, int nlink, Bpolygon* poly)
86{
87 if ( poly->closed() ) return nlink;
88 if ( !comp->link[ilink] ) return nlink;
89
90 int i, j;
91 double a, mina = TWOPI;
92 Vector3<double> e1, e2, cp;
93 Bcomponent* compsel = NULL;
94
95 poly->comp.push_back(comp);
96 nlink++;
97
98 // Exit when the polygon is too large
99 if ( nlink >= MAXLINK ) return nlink;
100
101 // Find the closest link in one direction
102 for ( i=0, j=-1; i<comp->link.size() && comp->link[i]; i++ ) {
103 if ( i != ilink ) {
104 e1 = comp->link[ilink]->location() - comp->location();
105 e2 = comp->link[i]->location() - comp->location();
106 cp = comp->force().cross(e1);
107 a = e1.angle(e2);
108 if ( e2.angle(cp) > M_PI_2 ) a = TWOPI - a;
109 if ( mina > a ) {
110 mina = a;
111 j = i;
112 }
113 }
114 }
115 if ( j < 0 ) return 0; // No new links found
116 if ( comp->flag[j] ) return 0; // Check if already encountered
117 comp->flag[j] = 1; // Set flag value
118
119 compsel = comp->link[j]; // One more comp in polygon
120 if ( compsel == poly->comp[0] ) { // End of polygon
121 poly->closed(1);
122 } else {
123 for ( i=0; i<compsel->link.size() && compsel->link[i] != comp; i++ ) ;
124 if ( i<compsel->link.size() ) nlink = poly_get_connectivity(compsel, i, nlink, poly);
125 }
126
127 if ( verbose & VERB_DEBUG )
128 cout << tab << compsel->identifier();
129
130 return nlink;
131}
