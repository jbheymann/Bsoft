181void Bplot::resolution_display(vector<double>& fsccut, vector<double>& dprcut)
182{
183 long i;
184 double res(999), FSCsum(0), w, IVR;
185 double *s, *FSC, *DPR;
186 Bstring txt, label(page(0).axis(3).label());
187
188 s = data;
189 FSC = s + nr;
190 DPR = FSC + nr;
191
192 cout << "Resolution curves:" << endl;
193 cout << "Spatial Frequency (1/A)\tRes(A)\t" << label;
194 if ( nc > 2 ) cout << "\tDPR";
195 cout << endl;
196
197 for ( i=0; i<nr; i++ ) {
198 if ( i ) res = 1/s[i];
199 w = M_PI*2*i;
200 if ( label == "FSC" ) w *= 2.0*i;
201 FSCsum += FSC[i]*w;
202 cout << showpoint << fixed << setprecision(5) << setw(7) << s[i] << tab
203 << setprecision(3) << setw(7) << res << tab
204 << setw(7) << FSC[i];
205 if ( nc > 2 ) cout << tab << setw(7) << DPR[i];
206 cout << endl;
207 }
208
209 if ( label == "FSC" ) IVR = pow(4.0*M_PI/(3.0*FSCsum), 1.0/3.0);
210 else IVR = sqrt(M_PI/FSCsum);
211 IVR /= s[1];
212
213 if ( fsccut.size() || dprcut.size() )
214 cout << endl << "Resolution estimates:" << endl;
215
216 for ( i=0; i<fsccut.size(); i++ ) if ( fsccut[i] ) {
217 txt = label + Bstring(fsccut[i], "(%g): ") + Bstring(1/cut(1,fsccut[i],-1), "%g A");
218 page(0).add_text(txt);
219 cout << txt << endl;
220 }
221
222 for ( i=0; i<dprcut.size(); i++ ) if ( dprcut[i] ) {
223 txt = Bstring(dprcut[i], "DPR(%g): ") + Bstring(1/cut(2,dprcut[i],1), "%g A");
224 page(0).add_text(txt);
225 cout << txt << endl;
226 }
227
228 txt = Bstring(IVR, "Information volume radius: %g A");
229 page(0).add_text(txt);
230 cout << txt << endl << endl;
231}
