34int Bimage::kernel_gaussian(double sigma, double max)
35{
36 if ( sigma < 0.001 ) sigma = 1;
37 if ( fabs(max) < 1e-10 ) max = 1;
38
39 if ( verbose & VERB_PROCESS ) {
40 cout << "Generating a gaussian kernel:" << endl;
41 cout << "Size: " << size() << endl;
42 cout << "Sigma: " << sigma << endl;
43 cout << "Maximum: " << max << endl << endl;
44 }
45
46 Vector3<double> h(size()/2);
47 origin(h);
48
49 long i, xx, yy, zz;
50 double x2, y2, z2, s2;
51 double invsigma2(-0.5/(sigma*sigma));
52
53 for ( i=zz=0; zz<z; zz++ ) {
54 z2 = zz - h[2];
55 z2 *= z2;
56 for ( yy=0; yy<y; yy++ ) {
57 y2 = yy - h[1];
58 y2 *= y2;
59 for ( xx=0; xx<x; xx++, i++ ) {
60 x2 = xx - h[0];
61 x2 *= x2;
62 s2 = x2 + y2 + z2;
63 set(i, max*exp(s2*invsigma2));
64 }
65 }
66 }
67
68 return 0;
69}

77int Bimage::kernel_laplacian_of_gaussian(double sigma, double max)
78{
79 if ( sigma < 0.001 ) sigma = 1;
80 if ( max < 1e-10 ) max = 1;
81
82 if ( verbose & VERB_PROCESS ) {
83 cout << "Generating a laplacian-of-gaussian kernel:" << endl;
84 cout << "Size: " << size() << endl;
85 cout << "Sigma: " << sigma << endl;
86 cout << "Maximum: " << max << endl << endl;
87 }
88
89 Vector3<double> h(size()/2);
90 origin(h);
91
92 long i, xx, yy, zz, ndim(0);
93 if ( x > 1 ) ndim++;
94 if ( y > 1 ) ndim++;
95 if ( z > 1 ) ndim++;
96 double x2, y2, z2, s2;
97 double invsigma2 = 1/(sigma*sigma);
98 double invsigma4 = invsigma2*invsigma2;
99 double scale = max/(ndim*invsigma2);
100
101 for ( i=zz=0; zz<z; zz++ ) {
102 z2 = zz - h[2];
103 z2 *= z2;
104 for ( yy=0; yy<y; yy++ ) {
105 y2 = yy - h[1];
106 y2 *= y2;
107 for ( xx=0; xx<x; xx++, i++ ) {
108 x2 = xx - h[0];
109 x2 *= x2;
110 s2 = x2 + y2 + z2;
111 set(i, scale*(s2*invsigma4 - ndim*invsigma2)*exp(-0.5*s2*invsigma2));
112 }
113 }
114 }
115
116 return 0;
117}

120int Bimage::average_line_sum(long nn, long i, long ik, long nk)
121{
122 if ( size()[ik] < 2 ) return 0;
123
124 long ii(i*x), j, j1, j2, jc, cc, inc(1);
125 long nl(size()[ik]), ll(nl*c), hk(nk/2), nv(0);
126 vector<double> s(c,0);
127
128 if ( ik == 1 ) {
129 ii = (i%x) + (i/x) * x * y;
130 inc = x;
131 } else if ( ik == 2 ) {
132 ii = i;
133 inc = x * y;
134 }
135
136 ii += nn * x * y * z;
137
138 double step = inc*hk;
139 vector<double> v(ll);
140
141// cout << ii << tab << ll << tab << hk << tab << inc << tab << s[0] << endl;
142 // Initial sum
143 for ( j=ii, j1=0; j1<hk; j+=inc, j1++, nv++ )
144 for ( cc=0, jc=j*c; cc<c; cc++, jc++ )
145 s[cc] += (*this)[jc];
146// cout << "---" << endl;
147 // Main sums
148 for ( j=ii, j1=0; j1<nl; j+=inc, j1++ ) {
149 if ( j1 < nl-hk ) { // Add the leading values
150 j2 = j + (long) step;
151 for ( cc=0, jc=j2*c; cc<c; cc++, jc++ )
152 s[cc] += (*this)[jc];
153 nv++;
154 }
155 for ( cc=0, jc=j1*c; cc<c; cc++, jc++ )
156 v[jc] = s[cc]/nv;
157 if ( j1 >= hk ) { // Subtract the trailing values
158 j2 = j - (long) step;
159 for ( cc=0, jc=j2*c; cc<c; cc++, jc++ )
160 s[cc] -= (*this)[jc];
161 nv--;
162 }
163 }
164
165 for ( j=ii, j1=0; j1<ll; j+=inc )
166 for ( cc=0, jc=j*c; cc<c; cc++, jc++, j1++ )
167 set(jc, v[j1]);
168
169 return 0;
170}

252Bimage* Bimage::gradient()
253{
254 long i, nn, xx, yy, zz, cc, xy(x*y);
255 Vector3<float> v1, v2;
256
257 if ( verbose & VERB_PROCESS )
258 cout << "Calculating a central difference gradient" << endl << endl;
259
260 Bimage* pg = new Bimage(Float, TVector3, size(), n);
261 pg->sampling(sampling(0));
262 pg->origin(image->origin());
263
264 for ( i=nn=0; nn<n; ++nn ) {
265 for ( zz=0; zz<z; ++zz ) {
266 for ( yy=0; yy<y; ++yy ) {
267 for ( xx=0; xx<x; ++xx, ++i ) {
268 for ( cc=0; cc<3; ++cc )
269 v1[cc] = v2[cc] = (*this)[i];
270 if ( xx ) v1[0] = (*this)[i-1];
271 if ( xx < x-1 ) v2[0] = (*this)[i+1];
272 if ( yy ) v1[1] = (*this)[i-x];
273 if ( yy < y-1 ) v2[1] = (*this)[i+x];
274 if ( zz ) v1[2] = (*this)[i-xy];
275 if ( zz < z-1 ) v2[2] = (*this)[i+xy];
276 pg->set(i, (v2 - v1)/2);
277 }
278 }
279 }
280 }
281
282 pg->statistics();
283
284 return pg;

287Vector3<double> Bimage::gradient_voxel(long i)
288{
289 long xx, yy, zz, iz, iy, ix, jx, jy, jz;
290 long xy(x*y), xyz(xy*z), nn(i/xyz);
291 double len, v(0);
292 Vector3<double> g, w;
293 Vector3<long> coor = coordinates(i%xyz);
294 Vector3<long> coor2;
295
296 zz = (coor[2])? -1: 0;
297 for ( iz=coor[2]+zz; iz<z && zz<2; ++zz, ++iz ) {
298 coor2[2] = zz;
299 jz = iz*xy + nn*xyz;
300 yy = (coor[1])? -1: 0;
301 for ( iy=coor[1]+yy; iy<y && yy<2; ++yy, ++iy ) {
302 coor2[1] = yy;
303 jy = jz + iy*x;
304 xx = (coor[0])? -1: 0;
305 for ( ix=coor[0]+xx; ix<x && xx<2; ++xx, ++ix ) {
306 coor2[0] = xx;
307 len = coor2.length();
308 jx = jy + ix;
309 if ( len ) {
310 v = (*this)[jx]/len;
311 g[0] += v*xx;
312 g[1] += v*yy;
313 g[2] += v*zz;
314 w[0] += fabs(xx);
315 w[1] += fabs(yy);
316 w[2] += fabs(zz);
317 }
318 }
319 }
320 }
321 if ( w[0] ) g[0] /= w[0];
322 if ( w[1] ) g[1] /= w[1];
323 if ( w[2] ) g[2] /= w[2];
324
325 return g;
326}

361int Bimage::gaussian_line_sum(long nn, long i, long ik, Bimage* pk)
362{
363 if ( size()[ik] < 2 ) return 0;
364
365 long ii(i*x), j, j1, j2, jm, jc, ji, cc, inc(1), nl(size()[ik]), ll(nl*c), hk(pk->x/2);
366 double w;
367
368 if ( ik == 1 ) {
369 ii = (i%x) + (i/x) * x * y;
370 inc = x;
371 } else if ( ik == 2 ) {
372 ii = i;
373 inc = x * y;
374 }
375
376 ii += nn * x * y * z;
377 jm = ii + nl*inc;
378
379 vector<double> v(ll);
380
381 // Main sums
382 for ( j1=0; j1<nl; j1++ ) {
383 j2 = ( j1 < hk )? hk - j1: 0;
384 j = ii;
385 if ( j2 < 1 ) j += (j1 - hk)*inc;
386 for ( cc=0, jc=j1*c; cc<c; cc++, jc++ )
387 v[jc] = 0;
388 for ( w=0; j2<pk->x && j<jm; j+=inc, j2++ ) {
389 for ( cc=0, jc=j1*c, ji=j*c; cc<c; cc++, jc++, ji++ )
390 v[jc] += (*pk)[j2] * (*this)[ji];
391 w += (*pk)[j2];
392 }
393 for ( cc=0, jc=j1*c; cc<c; cc++, jc++ )
394 v[jc] /= w;
395 }
396
397 // Transfer back
398 for ( j=ii, j1=0; j1<ll; j+=inc )
399 for ( cc=0, jc=j*c; cc<c; cc++, jc++, j1++ )
400 set(jc, v[j1]);
401
402 return 0;
403}

511int Bimage::convolve_chunk(Bimage* pkernel, float* nudata, long i, long len)
512{
513 long ic, j, jy, jz, jn, m, nn, xx, yy, zz, cc, kx, ky, kz, ix, iy, iz;
514 Vector3<long> hk(pkernel->image->origin());
515 long xmin, ymin, zmin, xmax, ymax, zmax;
516 double v, w;
517
518 coordinates(i, cc, xx, yy, zz, nn);
519
520 if ( cc != 0 )
521 cerr << "Warning: chunk not on channel boundary! (" << i << ")" << endl;
522
523 if ( verbose & VERB_DEBUG )
524 cout << "DEBUG Bimage::convolve_chunk: " << i << tab << cc << tab << xx
525 << tab << yy << tab << zz << tab << nn << endl;
526
527 xmin = kmin(xx, hk[0]);
528 xmax = kmax(xx, pkernel->x, x);
529 ymin = kmin(yy, hk[1]);
530 ymax = kmax(yy, pkernel->y, y);
531 zmin = kmin(zz, hk[2]);
532 zmax = kmax(zz, pkernel->z, z);
533
534 m = i+len;
535 if ( m > datasize ) m = datasize;
536
537 for ( ; i<m; xx++, i+=c ) {
538 if ( xx >= x ) {
539 xx = 0;
540 yy++;
541 if ( yy >= y ) {
542 yy = 0;
543 zz++;
544 if ( zz >= z ) {
545 zz = 0;
546 nn++;
547 }
548 zmin = kmin(zz, hk[2]);
549 zmax = kmax(zz, pkernel->z, z);
550 }
551 ymin = kmin(yy, hk[1]);
552 ymax = kmax(yy, pkernel->y, y);
553 }
554 xmin = kmin(xx, hk[0]);
555 xmax = kmax(xx, pkernel->x, x);
556 for ( cc=0, ic=i; cc<c; cc++, ic++ ) nudata[ic] = 0;
557 w = 0;
558 jn = nn*z;
559 for ( kz=zmin; kz<zmax; kz++ ) {
560 jz = (jn + zz + kz - hk[2])*y;
561 iz = kz*pkernel->y;
562 for ( ky=ymin; ky<ymax; ky++ ) {
563 jy = (jz + yy + ky - hk[1])*x;
564 iy = (iz + ky)*pkernel->x;
565 for ( kx=xmin; kx<xmax; kx++ ) {
566 j = (jy + xx + kx - hk[0])*c;
567 ix = iy + kx;
568 v = (*pkernel)[ix];
569 for ( cc=0, ic=i; cc<c; cc++, ic++, j++ )
570 nudata[ic] += (*this)[j]*v;
571 w += v;
572 }
573 }
574 }
575 if ( w ) for ( cc=0, ic=i; cc<c; cc++, ic++ )
576 nudata[ic] /= w;
577 }
578
579 return 0;
580}

663int Bimage::filter_ortho(int type)
664{
665 change_type(Float);
666
667 long i, xx, yy, zz, nn;
668 double value;
669 long slicesize(x*y*c);
670 float* nudata = new float[datasize];
671 long ksize(1);
672 if ( x > 1 ) ksize += 2;
673 if ( y > 1 ) ksize += 2;
674 if ( z > 1 ) ksize += 2;
675 vector<double> kernel(ksize);
676
677 if ( type == 0 ) {
678 kernel[0] = 0;
679 for ( i=1; i<7; i+=2 )
680 if ( ksize > i ) {
681 kernel[i] = -1;
682 kernel[i+1] = 1;
683 }
684 } else if ( type == 1 ) {
685 kernel[0] = 0;
686 for ( i=1; i<7; i+=2 )
687 if ( ksize > i ) {
688 kernel[0] -= 2;
689 kernel[i] = 1;
690 kernel[i+1] = 1;
691 }
692 }
693
694 if ( verbose & VERB_PROCESS ) {
695 cout << "Convolving with an orthogonal kernel:" << endl;
696 cout << "Kernel type: ";
697 switch ( type ) {
698 case 0: cout << "gradient"; break;
699 case 1: cout << "laplacian"; break;
700 default: break;
701 }
702 cout << endl << endl;
703 }
704
705 for ( i=nn=0; nn<n; nn++ ) {
706 for ( zz=0; zz<z; zz++ ) {
707 for ( yy=0; yy<y; yy++ ) {
708 for ( xx=0; xx<x; xx++, i++ ) {
709 value = kernel[0]*(*this)[i];
710 if ( x > 1 ) {
711 if ( xx > 0 ) value += kernel[1]*(*this)[i-1];
712 else value += kernel[1]*(*this)[i+x-1];
713 if ( xx < x - 1 ) value += kernel[2]*(*this)[i+1];
714 else value += kernel[2]*(*this)[i+1-x];
715 if ( type == 0 ) {
716 nudata[i] = value*value;
717 value = 0;
718 }
719 }
720 if ( y > 1 ) {
721 if ( yy > 0 ) value += kernel[3]*(*this)[i-x];
722 else value += kernel[3]*(*this)[i+slicesize-x];
723 if ( yy < y - 1 ) value += kernel[4]*(*this)[i+x];
724 else value += kernel[4]*(*this)[i+x-slicesize];
725 if ( type == 0 ) {
726 nudata[i] += value*value;
727 value = 0;
728 }
729 }
730 if ( z > 1 ) {
731 if ( zz > 0 ) value += kernel[5]*(*this)[i-slicesize];
732 else value += kernel[5]*(*this)[i+datasize-slicesize];
733 if ( zz < z - 1 ) value += kernel[6]*(*this)[i+slicesize];
734 else value += kernel[6]*(*this)[i+slicesize-datasize];
735 if ( type == 0 ) {
736 nudata[i] += value*value;
737 value = 0;
738 }
739 }
740 if ( type == 1 ) nudata[i] = value;
741 else nudata[i] = sqrt(nudata[i]);
742 }
743 }
744 }
745 }
746
747 data_assign((unsigned char *) nudata);
748
749 statistics();
750
751 return 0;
752}

760int Bimage::filter_dog(double sigma1, double sigma2)
761{
762 if ( sigma1 > sigma2 ) swap(sigma1, sigma2);
763
764 long k(long(6*sigma2+1));
765 Vector3<long> ksize(k, k, k);
766 ksize = ksize.min(size());
767
768 double m1(1/(sqrt(TWOPI)*sigma1));
769 double m2(1/(sqrt(TWOPI)*sigma2));
770
771 Bimage* pk = new Bimage(Float, TSimple, ksize, 1);
772 Bimage* pk2 = new Bimage(Float, TSimple, ksize, 1);
773
774 pk->kernel_gaussian(sigma1, m1);
775 pk2->kernel_gaussian(sigma2, m2);
776 pk->subtract(pk2);
777
778 delete pk2;
779
780 convolve(pk);
781
782 delete pk;
783
784 statistics();
785

836 ymin = kmin(yy, hk[1]);
837 ymax = kmax(yy, pkernel->y, y);

846 jz = (jn + zz + kz - hk[2])*y;

861 case 2: // lorentz
862 acc = 1/(1+lorentz_invsig2*diff2);
863 break;
864 case 3: // tukey
865 if (diff2 <= tukey_invsig2) {
866 f = 1 - tukey_invsig2*diff2;

868 };
869 break;

871 acc *= (*pkernel)[ix];

874 }
875 }
876 }

954int Bimage::filter_rolling_ball(long radius, double scale)
955{
956 if ( radius < 1 ) radius = 1;
957 if ( fabs(scale) < 1e-30 ) scale = 1;
958
959 long i, xx, yy, zz, nn, kx, ky, kz;
960 long ix, iy, iz, jx, jy, jz;
961 double r2(radius*radius);
962 double dx2, dy2, dz2, diff, bcenter;
963
964 Vector3<long> ksize(radius*2+1, radius*2+1, radius*2+1);
965 ksize = ksize.min(size());
966 Vector3<long> hk(ksize/2);
967 Vector3<long> lo, hi;
968
969 long ball_size = (long) ksize.volume();
970 vector<double> ball(ball_size);
971
972 for ( i=zz=0; zz<ksize[2]; zz++ ) {
973 dz2 = (double)zz - hk[2];
974 dz2 *= dz2;
975 for ( yy=0; yy<ksize[1]; yy++ ) {
976 dy2 = (double)yy - hk[1];
977 dy2 *= dy2;
978 for ( xx=0; xx<ksize[0]; xx++, i++ ) {
979 dx2 = (double)xx - hk[0];
980 dx2 *= dx2;
981 diff = r2 - dx2 - dy2 - dz2;
982 if ( diff > 0 ) ball[i] = sqrt(diff)/scale;
983 else ball[i] = 0;
984 }
985 }
986 }
987
988 change_type(Float);
989
990 if ( verbose & VERB_PROCESS ) {
991 cout << "Rolling ball filter:" << endl;
992 cout << "Radius: " << radius << endl;
993 cout << "Scaling: " << scale << endl << endl;
994 }
995
996 vector<float> back(datasize);
997 for ( i=0; i<datasize; i++ ) back[i] = (*this)[i];
998
999 for ( i=nn=0; nn<n; nn++ ) {
1000 for ( zz=0; zz<z; zz++ ) {
1001 lo[2] = kmin(zz, hk[2]);
1002 hi[2] = kmax(zz, ksize[2], z);
1003 for ( yy=0; yy<y; yy++ ) {
1004 lo[1] = kmin(yy, hk[1]);
1005 hi[1] = kmax(yy, ksize[1], y);
1006 for ( xx=0; xx<x; xx++, i++ ) {
1007 lo[0] = kmin(xx, hk[0]);
1008 hi[0] = kmax(xx, ksize[0], x);
1009 if ( scale > 0 ) bcenter = max;
1010 else bcenter = min;
1011 for ( kz=lo[2]; kz<hi[2]; kz++ ) {
1012 iz = (nn*z + zz + kz - hk[2])*y;
1013 jz = kz*ksize[1];
1014 for ( ky=lo[1]; ky<hi[1]; ky++ ) {
1015 iy = (iz + yy + ky - hk[1])*x;
1016 jy = (jz + ky)*ksize[0];
1017 for ( kx=lo[0]; kx<hi[0]; kx++ ) {
1018 ix = iy + xx + kx - hk[0];
1019 jx = jy + kx;
1020 diff = (*this)[ix] - (bcenter + ball[jx]);
1021 if ( ball[jx] > 0 ) {
1022 if ( diff < 0 ) bcenter += diff; // Moving the ball down
1023 } else if ( ball[jx] < 0 ) {
1024 if ( diff > 0 ) bcenter += diff; // Moving the ball up
1025 }
1026 }
1027 }
1028 }
1029 for ( kz=lo[2]; kz<hi[2]; kz++ ) {
1030 iz = (nn*z + zz + kz - hk[2])*y;
1031 jz = kz*ksize[1];
1032 for ( ky=lo[1]; ky<hi[1]; ky++ ) {
1033 iy = (iz + yy + ky - hk[1])*x;
1034 jy = (jz + ky)*ksize[0];
1035 for ( kx=lo[0]; kx<hi[0]; kx++ ) {
1036 ix = iy + xx + kx - hk[0];
1037 jx = jy + kx;
1038 diff = bcenter + ball[jx];
1039 if ( ball[jx] > 0 ) {
1040 if ( back[ix] > diff ) back[ix] = diff; // Assigning the background
1041 } else if ( ball[jx] < 0 ) {
1042 if ( back[ix] < diff ) back[ix] = diff; // Assigning the background
1043 }
1044 }
1045 }
1046 }
1047 }
1048 }
1049 }
1050 }
1051
1052 for ( i=0; i<datasize; i++ ) set(i, (*this)[i] - back[i]);
1053
1054 statistics();
1055
1056 return 0;
1057}
