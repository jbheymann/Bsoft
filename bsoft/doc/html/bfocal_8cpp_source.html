70void ctf_apply_beam_tilt(CTFparam& cp, double tilt, double dir)
71{
72 cp.show_aberration();
73
74 if ( verbose ) {
75 cout << "Beam tilt angle: " << tilt << " rad (" << tilt*180.0/M_PI << " degrees)" << endl;
76 cout << "Beam direction angle: " << dir << " rad (" << dir*180.0/M_PI << " degrees)" << endl;
77 }
78
79 long i;
80 double tamp(sin(tilt)), v;
81 double tx(tamp*cos(dir)), ty(tamp*sin(dir));
82
83 double a = cp.aberration_coefficient(0,0);
84// cout << "lambda=" << cp.lambda() << endl;
85// cout << "amp=" << a << endl;
86 for ( i=1, v=tamp; i<5; ++i, v*=tamp ) a += v/i;
87// cout << "amp=" << a << endl;
88 cp.aberration_coefficient(0,0,a);
89
90 double cs = cp.aberration_coefficient(4,0);
91
92 cp.beam_tilt(tx, ty);
93
94 double d = cp.aberration_coefficient(2,0);
95 d += 2*cs*tamp*tamp;
96 cp.aberration_coefficient(2,0,d);
97
98 double cx = cp.aberration_coefficient(2,-2);
99 double cy = cp.aberration_coefficient(2,2);
100 cp.aberration_coefficient(2, -2, cy + 2*cs*tx*ty);
101 cp.aberration_coefficient(2, 2, cx + cs*(tx*tx-ty*ty));
102
103}
