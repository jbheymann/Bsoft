673 cout << "Size: " << p->size() << endl;
674 cout << endl;
675 }
676
677 vector<Bimage*> imgvec;
678 imgvec.push_back(p);
679 imgvec.push_back(ptemp);
680 imgvec.push_back(pfs_mask);
681
