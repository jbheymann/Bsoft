32int mg_unitcell_vectors(Bmicrograph* mg)
33{
34 long i;
35 vector<double> bx(2,0), by(2,0);
36 Matrix a(2,2);
37 Bstrucfac* sf;
38
39 for ( i = 0, sf = mg->sf; sf; sf = sf->next, i++ ) {
40 a[0][0] += sf->index[0]*sf->index[0];
41 a[0][1] += sf->index[0]*sf->index[1];
42 a[1][1] += sf->index[1]*sf->index[1];
43 bx[0] += sf->index[0]*sf->loc[0];
44 bx[1] += sf->index[1]*sf->loc[0];
45 by[0] += sf->index[0]*sf->loc[1];
46 by[1] += sf->index[1]*sf->loc[1];
47 }
48 a[1][0] = a[0][1];
49
50 if ( i < 2 ) {
51 cerr << "Error: Too few structure factors to calculate unit cell vectors!" << endl;
52 return -1;
53 }
54
55 a.LU_decomposition();
56 a.multiply_in_place(bx);
57 a.multiply_in_place(by);
58
59 mg->hvec[0] = bx[0];
60 mg->hvec[1] = by[0];
61 mg->kvec[0] = bx[1];
62 mg->kvec[1] = by[1];
63
64 return 0;
65}

177Vector3<double> sinc_fit(vector<Vector3<double>> coor, vector<double> value, double& R)
178{
179 R = 0;
180
181 long i;
182 double k(2), d, R1;
183 Vector3<double> ori, bori, off;
184
185 for ( ori[1]=-k; ori[1]<=k; ori[1]+=0.1 ) {
186 for ( ori[0]=-k; ori[0]<=k; ori[0]+=0.1 ) {
187 for ( i=0, R1=0; i<coor.size(); ++i ) {
188 off = coor[i] - ori;
189 d = value[i]*sinc(off[0])*sinc(off[1]);
190 R1 += d;
191 }
192 R1 = sqrt(R1/coor.size());
193// cout << ori << tab << R << endl;
194 if ( R < R1 ) {
195 R = R1;
196 bori = ori;
197 }
198 }
199 }
200
201 if ( verbose & VERB_FULL ) {
202 cout << "Best origin: " << bori << tab << R << endl;
203 }
204
205 return bori;
206}

237vector<Vector3<double>> img_find_reflections(Bimage* p, double ref_res, double threshold, long kedge)
238{
239 if ( threshold < 0.1 ) threshold = 20;
240 if ( kedge < 1 ) kedge = 1;
241
242 long i, na;
243 long ks(kedge/2); // Summation kernel half edge and volume
244 double a;
245 double s(1/ref_res); // Spatial frequency
246 Vector3<double> k(p->real_size()*s); // Frequency space pixel distance
247 k[2] = 0;
248 double rd(1+5/k.length()); // Relative displacement for reference kernel
249 double angsam(ks); // Angular sampling in pixels
250 double da(angsam/k.length()); // Angle increment
251 double pmax, ravg, bkg(0), tR;
252 Vector3<double> tloc, rloc, sloc;
253 vector<Vector3<double>> loc;
254 vector<double> R, Rfit;
255
256 if ( verbose ) {
257 cout << "Finding reflections:" << endl;
258 cout << "Power spectrum size: " << p->size() << endl;
259 cout << "Spatial frequency: " << s << " 1/A" << endl;
260 cout << "Threshold: " << threshold << " e/px" << endl;
261 cout << "Kernel edge size: " << 2*ks+1 << endl;
262 cout << "Frequency space pixels: " << k << endl;
263 cout << "Frequency space pixel size: " << 1/p->real_size()[0] << " 1/Å" << endl;
264 cout << "Angular increment: " << da*180.0/M_PI << " degrees" << endl << endl;
265 }
266
267 for ( a=0, na=0; a<TWOPI; a+=da, ++na ) {
268// tloc = Vector3<double>(p->image->origin()[0] + k[0]*cos(a), p->image->origin()[1] + k[1]*sin(a), 0);
269 tloc = Vector3<double>(k[0]*cos(a), k[1]*sin(a), 0);
270 rloc = tloc * rd;
271 tloc += p->image->origin();
272 rloc += p->image->origin();
273// cout << na << tab << tloc << tab << rloc << endl;
274 i = p->kernel_max(p->index(tloc,0), ks);
275 pmax = (*p)[i];
276 ravg = p->kernel_average(p->index(rloc,0), ks, 0, 1e30);
277 bkg += ravg;
278 if ( pmax > threshold ) {
279 tloc = p->coordinates(i);
280 tloc = img_sinc_fit(p, tloc, ravg, pmax - ravg, tR);
281 if ( loc.size() && tloc.distance(loc.back()) < 2 ) {
282 if ( pmax > R.back() ) {
283 loc.back() = tloc;
284 R.back() = pmax;
285 Rfit.back() = tR;
286 }
287 } else {
288 loc.push_back(tloc);
289 R.push_back(pmax);
290 Rfit.push_back(tR);
291 }
292 }
293 }
294
295 bkg /= na;
296
297 if ( loc.size() < 1 ) {
298 cerr << "Error: No reflections found, check the threshold." << endl;
299 return loc;
300 }
301
302 // Handle wrap-around reflections
303 if ( loc[0].distance(loc.back()) < 2 ) {
304 if ( R[0] < R.back() ) {
305 loc[0] = loc.back();
306 R[0] = R.back();
307 Rfit[0] = Rfit.back();
308 }
309 loc.pop_back();
310 R.pop_back();
311 Rfit.pop_back();
312 }
313
314 if ( verbose )
315 cout << "#\tx\ty\tz\ta\tR\tRfit" << endl;
316 for ( i=0; i<loc.size(); ++i ) {
317 sloc = loc[i] - p->image->origin();
318 if ( verbose )
319 cout << setprecision(2) << i+1 << tab << loc[i] << tab <<
320 atan2(sloc[1], sloc[0])*180.0/M_PI << tab << setprecision(3) << R[i] << tab << Rfit[i] << endl;
321 R[i] = p->kernel_sum(p->index(loc[i],0), 1) - 9*bkg;
322 loc[i] = sloc/p->real_size();
323 }
324
325 if ( verbose ) {
326 cout << "Number of reflections found: " << loc.size() << endl;
327 cout << "Background intensity: " << bkg << " e/px" << endl << endl;
328 }
329
330 return loc;
331}

394Vector3<double> analyze_reflections(vector<Vector3<double>>& loc, double s)
395{
396// long i;
397 double u2(0), v2(0), iso(0);
398 double sf, sf_avg(0), sf_std(0);
399 Matrix amat(2,2);
400 vector<double> f2(2,0);
401
402 if ( verbose )
403 cout << "Analyzing reflections:" << endl;
404
405 for ( auto& l1: loc ) {
406 sf = l1.length();
407 sf_avg += sf;
408 sf_std += sf*sf;
409 u2 = l1[0]*l1[0];
410 v2 = l1[1]*l1[1];
411 f2[0] += u2;
412 f2[1] += v2;
413 amat[0][0] += u2*u2;
414 amat[1][1] += v2*v2;
415 amat[0][1] += u2*v2;
416 amat[1][0] += u2*v2;
417 }
418
419 f2[0] *= s*s;
420 f2[1] *= s*s;
421 amat.singular_value_decomposition(f2);
422 iso = sqrt((f2[0] + f2[1])/2);
423 f2[0] = sqrt(f2[0]);
424 f2[1] = sqrt(f2[1]);
425
426 sf_avg /= loc.size();
427 sf_std = sqrt(sf_std/loc.size() - sf_avg*sf_avg);
428
429 if ( verbose ) {
430 cout << "Spatial frequency average: " << sf_avg << " 1/A" << tab << "(" << sf_std << ")" << endl;
431 cout << "Resolution average: " << 1/sf_avg << " A" << endl;
432 cout << "Scale in u and v: " << f2 << endl;
433 cout << "Isotropic scale: " << iso << tab << "(" << loc.size() << ")" << endl << endl;
434 }
435
436 Vector3<double> scale(f2[0],f2[1],1);
437
438 return scale;
439}
