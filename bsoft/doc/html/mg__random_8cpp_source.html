27int project_random_defocus(Bproject* project, double std)
28{
29 if ( !project ) return 0;
30
31 Bfield* field;
32 Bmicrograph* mg;
33 long n(0);
34 double rnum, rdev(0);
35
36 if ( verbose & VERB_PROCESS ) {
37 cout << "Introducing random errors into defocus average:" << endl;
38 cout << "Target standard deviation: " << std << " A" << endl;
39 }
40
41 random_seed();
42
43 for ( field = project->field; field; field = field->next )
44 for ( mg = field->mg; mg; mg = mg->next ) {
45 if ( !mg->ctf ) mg->ctf = new CTFparam;
46 do {
47 rnum = random_gaussian(0, std);
48 } while ( 2*fabs(rnum) > mg->ctf->defocus_average() );
49 mg->ctf->defocus_average(mg->ctf->defocus_average() + rnum);
50 n++;
51 rdev += rnum*rnum;
52 }
53
54 rdev = sqrt(rdev/n);
55
56 if ( verbose & VERB_PROCESS )
57 cout << "Standard deviation: " << rdev << " A" << endl;
58
59 return 0;
60}

226double particle_random_view(Bparticle* part, double std)
227{
228 Vector3<double> vec = part->view.vector3();
229 Vector3<double> axis = vector3_random(-1.0, 1.0);
230
231 axis.normalize();
232 axis = axis.cross(vec);
233 axis.normalize();
234
235 double angle = random_gaussian(0.0, std);
236 Matrix3 mat = Matrix3(axis, angle);
237
238 axis = mat * vec;
239 part->view = View2<double>(axis[0], axis[1], axis[2], part->view.angle());
240
241 return axis.angle(vec);
242}
