460int project_split_particles(Bproject* project, Bstring partfile, Bstring path, Bstring partext)
461{
462 if ( path.length() && path[-1] != '/' ) path += "/";
463 if ( partext.length() < 1 ) partext = "pif";
464
465 if ( partfile.length() < 3 ) partfile = project->field->mg->fpart;
466 if ( partfile.length() < 3 ) partfile = project->field->mg->part->fpart;
467 if ( partfile.length() < 3 ) {
468 cerr << "Error: No particle image file specified" << endl;
469 bexit(-1);
470 }
471
472 // Directory for individual particle files
473 if ( path.length() ) {
474 mkdir(path.c_str(), O_CREAT );
475 chmod(path.c_str(), 0755);
476 }
477
478 Bfield* field = NULL;
479 Bmicrograph* mg = NULL;
480 Bparticle* part = NULL;
481
482 if ( verbose )
483 cout << "Splitting " << partfile << " into micrograph-associated particle files" << endl << endl;
484
485 Bimage* img_stack = read_img(partfile, 1, -1);
486 Bimage* img_part = NULL;
487
488 size_t n(0), m;
489 Bstring partname;
490
491 for ( field = project->field; field; field = field->next ) {
492 for ( mg = field->mg; mg; mg = mg->next ) {
493 partname = mg->fmg.post_rev('/').pre_rev('.') + "." + partext;
494 if ( path.length() ) partname = path + partname;
495 for ( m=0, part = mg->part; part; part = part->next, m++ )
496 part->ori = img_stack->size()/2;
497 if ( verbose )
498 cout << mg->id << tab << partname << tab << m << endl;
499 if ( m ) {
500// cout << n << tab << m << endl;
501 img_part = img_stack->extract(n, n+m-1);
502 img_part->sampling(mg->pixel_size);
503 img_part->origin(img_part->size()/2);
504 write_img(partname, img_part, 0);
505 delete img_part;
506 mg->fpart = partname;
507 mg->box_size = img_stack->size();
508 n += m;
509 }
510 }
511 }
512
513 delete img_stack;
514
515
516 return 0;
517}
