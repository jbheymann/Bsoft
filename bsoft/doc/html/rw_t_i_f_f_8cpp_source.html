105int tiff_show_tag_values(TIFF* tif)
106{
107 int i, t;
108 double d;
109 uint32_t count;
110 void *data;
111 const TIFFField* tf;
112// TIFFDataType dt;
113
114 int n = TIFFGetTagListCount(tif);
115
116 if ( verbose )
117 cout << "Number of custom tags: " << n << endl;
118
119 for ( i=1; i<n; ++i ) {
120 t = TIFFGetTagListEntry(tif, i);
121 tf = TIFFFieldWithTag(tif, t);
122// dt = TIFFFieldDataType(tf);
123 if ( TIFFFieldPassCount(tf) ) {
124 TIFFGetField(tif, t, &count, &data);
125 if ( verbose )
126 cout << t << tab << count << tab << *(double *)data << endl;
127 } else {
128 TIFFGetField(tif, t, &d);
129 if ( verbose )
130 cout << t << tab << count << tab << d << endl;
131 }
132 }
133
134 return n;
135}

342 double dmin(0), dmax(0);

344 TIFFGetField(fimg, TIFFTAG_MINSAMPLEVALUE, &min);
345 TIFFGetField(fimg, TIFFTAG_MAXSAMPLEVALUE, &max);
346 p->minimum(min);
347 p->maximum(max);
348 } else {
349 TIFFGetField(fimg, TIFFTAG_SMINSAMPLEVALUE, &dmin);
350 TIFFGetField(fimg, TIFFTAG_SMAXSAMPLEVALUE, &dmax);
351 p->minimum(dmin);
352 p->maximum(dmax);

667 case Short:

675 case Integer:

685 TIFFSetField(fimg, TIFFTAG_BITSPERSAMPLE, 64);

693 TIFFSetField(fimg, TIFFTAG_BITSPERSAMPLE, 64);
694 break;

696 TIFFSetField(fimg, TIFFTAG_BITSPERSAMPLE, 8);

700 switch ( p->compound_type() ) {
701 case TSimple:
702 TIFFSetField(fimg, TIFFTAG_PHOTOMETRIC, PHOTOMETRIC_MINISBLACK);

705 case TRGB:

735 page = new char[pagesize];

740 i = y*tilerowsize;

743 }
744 if ( verbose & VERB_DEBUG ) cout << " " << z;

746 }
747 delete[] page;
748 } else {

762 TIFFWriteDirectory(fimg);
763 } // End of multi-image loop
764
765 if ( verbose & VERB_DEBUG )
766 cout << "DEBUG writeTIFF: data written" << endl;
767
768 TIFFClose(fimg);
769
770 if ( verbose & VERB_DEBUG )
771 cout << "DEBUG writeTIFF: Memory freed" << endl;
772
773 return 0;
774}

798{
