310int point_spread(size_t n, vector<double>& a, double step)
311{
312 size_t i, j;
313 double fac = M_PI*1.0L/n, v;
314 vector<double> t(n);
315
316// a[0] = 0;
317
318 for ( i=0; i<n; i++ ) {
319 t[i] = a[0]/2;
320 v = fac*i;
321 for ( j=1; j<n; j++ )
322 t[i] += a[j]*cos(v*j);
323 t[i] *= 2;
324 }
325
326 for ( i=0; i<n; i++ ) a[i] = t[i]*step;
327
328 return 0;
