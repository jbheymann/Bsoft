18int img_stem_lut_apply(Bimage* p, int neg_stain_flag)
19{
20 if ( !p->data_pointer() || p->images() < 2 ) return -1;
21
22 long i;
23 long imgsize = p->sizeX()*p->sizeY();
24 unsigned char* data = p->data_pointer();
25 int j;
26 int prefac(146);
27 int la_gain(10);
28 int sa_gain(10);
29 int bf_gain(0);
30
31 if ( neg_stain_flag ) {
32 la_gain = 2;
33 sa_gain = 0;
34 bf_gain = 1;
35 }
36
37 double ila(0), isa(0), ibf(0), it;
38 float la_lut[LUTSIZE], sa_lut[LUTSIZE], bf_lut[LUTSIZE];
39
40 // Data originating from the microscope
41 for ( i=0; i<LUTSIZE; i++ )
42 la_lut[i] = sa_lut[i] = bf_lut[i] = i;
43
44 for ( j=0; j<7000 && ( ila < LUTSIZE-1 || isa < LUTSIZE-1 ); j++ ) {
45 ila = (1 - exp(-j/2500.0)) * exp(-j/10000.0);
46 isa = (1 - exp(-j/1964.0)) * exp(-j/2000.0);
47 ibf = exp(-j/1100.0);
48 it = ila + isa + ibf;
49// cout << ila << " " << isa << " " << ibf << " " << it << endl;
50 ila = prefac*la_gain*ila/it;
51 isa = prefac*sa_gain*isa/it;
52 ibf = prefac*bf_gain*ibf/it;
53// cout << j << " " << ila << " " << isa << " " << ibf << " " << it << endl;
54 if ( ila < LUTSIZE ) la_lut[(int)ila] = prefac*la_gain*j/2500.0;
55 if ( isa < LUTSIZE ) sa_lut[(int)isa] = prefac*sa_gain*j/1964.0;
56 if ( ibf < LUTSIZE ) bf_lut[(int)ibf] = prefac*bf_gain*(1 - j/1100.0);
57 }
58
59 for ( i=0; i<LUTSIZE; i++ ) {
60// cout << i << " " << la_lut[i] - i << " " << sa_lut[i] - i << " " << bf_lut[i] - i << endl;
61 if ( la_lut[i] >= LUTSIZE ) la_lut[i] = LUTSIZE - 1;
62 if ( sa_lut[i] >= LUTSIZE ) sa_lut[i] = LUTSIZE - 1;
63 if ( bf_lut[i] >= LUTSIZE ) bf_lut[i] = LUTSIZE - 1;
64 if ( bf_lut[i] < 0 ) bf_lut[i] = 0;
65 }
66
67 for ( i=0; i<imgsize; i++ ) data[i] = (int)la_lut[data[i]];
68
69 if ( neg_stain_flag ) {
70 for ( i=0, j=imgsize; i<imgsize; i++, j++ ) {
71 data[j] = (int)bf_lut[data[j]];
72 it = data[j] - data[i]/la_gain;
73 if ( it < 0 ) it = 0;
74 data[i] = (int) it;
75 }
76 } else
77 for ( i=0, j=imgsize; i<imgsize; i++, j++ ) data[j] = (int)sa_lut[data[j]];
78
79 return 0;
80}

188int writeBrookhavenSTEM(Bimage* p)
189{
190 p->change_type(UCharacter);
191
192 char* header = new char[BrookhavenSTEMSIZE];
193 memset(header, ' ', BrookhavenSTEMSIZE);
194
195 p->label().copy(header, BrookhavenSTEMSIZE);
196
197 snprintf(header + 70, BrookhavenSTEMSIZE-70, "SCAN:%6.3f", p->sampling(0)[0]*0.0512);
198
199 p->data_offset(BrookhavenSTEMSIZE);
200
201 long i, j, k;
202 long imgsize = p->sizeX()*p->sizeY()*p->sizeZ();
203 long datasize = p->data_size();
204 unsigned char* data = p->data_pointer();
205 unsigned char* temp = data;
206
207 // Interleaved data
208 if ( p->images() > 1 ) {
209 temp = new unsigned char[datasize];
210 for ( i=0, j=imgsize, k=0; i<imgsize; i++, j++, k+=2 ) {
211 temp[k] = data[i];
212 temp[k+1] = data[j];
213 }
214 }
215
216 ofstream fimg(p->file_name());
217 if ( fimg.fail() ) return -1;
218
219 fimg.write((char *)header, p->data_offset());
220 if ( p->data_pointer() ) fimg.write((char *)temp, datasize);
221
222 fimg.close();
223
224 delete[] header;
225 if ( temp != data ) delete[] temp;
226
227 return 0;
228}
