198 int max = (int) p->maximum();
199 if ( p->data_type() == UShort && max <= UCHAR_MAX ) max = UCHAR_MAX + 1;
200 unsigned char* data = p->data_pointer();
201
202 if ( verbose & VERB_DEBUG )
203 cout << "DEBUG writePNM: datatype=" << p->data_type() << endl;
204
205 for ( n=0; n<p->images(); n++, data += imgsize ) {
206 if ( p->data_type() == Bit ) fimg << "P4" << endl;
207 else if ( p->compound_type() == TSimple ) fimg << "P5" << endl;
208 else if ( p->compound_type() == TRGB ) fimg << "P6" << endl;
209 else cerr << "Error: Compound type " << p->compound_type() << " not supported!" << endl;
210 fimg << p->sizeX() << " " << p->sizeY() << endl;
211 if ( p->data_type() > Bit ) {
212 if ( p->data_type() == UCharacter || p->data_type() == UShort ) fimg << max << endl;
213 else cerr << "Error: Data type " << p->data_type() << " not supported!" << endl;

215 fimg.write((char *)data, imgsize);

218 fimg.close();
