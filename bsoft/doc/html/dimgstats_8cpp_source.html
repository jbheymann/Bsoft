310/************************************************************************
311@brief Calculates the square root of the FOM portion of an image.
312@param *p image.
313@return Bimage* square root of the FOM of the input image.
314
315 The new data is calculated as:
316 new_datum = sqrt(datum)
317 FOM min. must be 0 or greater.
318
319**/
320int img_fom_sqrt(Bimage* p)
321{
322 if ( !p->next ) {
323 cerr << "Error: No FOM block found in " << p->file_name() << endl;
324 return -1;
325 }
326
327 long i, datasize = (long) p->size().volume()*p->images();
328 float* fom = (float *) p->next->data_pointer();
329
330 for ( i=0; i<datasize; i++ ) {
331 if ( fom[i] < 0 ) {
332 cerr << "Cannot take the square root of the FOM of this image because a value is less than zero." << endl;
333 return -1;
334 }
335 else
336 fom[i] = sqrt(fom[i]);
337 }
338
339 return 0;
340}

363Bimage* img_students_t_test(int n, Bimage* p, vector<double>& weight)
364{
365 Bimage* pt = new Bimage(Float, TSimple, p->size(), 1);
366 pt->origin(p->image->origin());
367 pt->sampling(p->sampling(0));
368
369 float* var = (float *) p->next->data_pointer();
370
371 pt->next = new Bimage(Float, TSimple, pt->size(), 1);
372 float* siglev = (float *) pt->next->data_pointer();
373
374 long i, datasize = p->data_size();
375 double ws(0), d;
376
377 for ( i=0; i<n; i++ ) ws += weight[i];
378
379 double dof = ws*(1 - 1.0L/n);
380 double hdof = 0.5*dof;
381 double avg_var = 0;
382
383 for ( i=0; i<datasize; i++ ) avg_var += var[i];
384 avg_var /= datasize;
385
386 if ( verbose ) {
387 cout << "Student's t test:" << endl;
388 cout << "Number: " << n << endl;
389 cout << "Average variance: " << avg_var << endl;
390 cout << "Degrees of freedom: " << dof << endl << endl;
391 }
392
393 avg_var *= avg_var;
394
395 for ( i=0; i<datasize; i++ ) {
396 d = var[i]/n;
397 if ( d < avg_var ) d = avg_var;
398 d = sqrt(d);
399 pt->set(i, (*p)[i]/d);
400 siglev[i] = betai(hdof, 0.5, dof/(dof + (*pt)[i]*(*pt)[i]));
401 }
402
403 return pt;
404}

431Bimage* img_students_t_test_equal(int set1, int set2, Bimage* p1, Bimage* p2, vector<double>& weight)
432{
433 // Set arrays and image parameters
434 Bimage* pt = new Bimage(Float, TSimple, p1->size(), 1);
435 pt->origin(p1->image->origin());
436 pt->sampling(p1->sampling(0));
437 pt->next = new Bimage(Float, TSimple, pt->size(), 1);
438
439 long i, j, datasize = p1->sizeX()*p1->sizeY()*p1->sizeZ()*p1->images();
440
441 float* var1 = (float *) p1->next->data_pointer(); // pointers to variance data
442 float* var2 = (float *) p2->next->data_pointer();
443 float* sig = (float *) pt->next->data_pointer();
444
445 // Compute Student's t-test
446 if ( verbose ) {
447 cout << endl << "Student's t-test, assuming same variance in sets 1 and 2" << endl;
448 cout << "Set 1 contains " << set1 << " images" << endl;
449 cout << "Set 2 contains " << set2 << " images" << endl;
450 }
451
452 double ws1, ws2;
453 for ( i=0, j=0, ws1=0; i<set1; i++, j++ ) ws1 += weight[j];
454 for ( i=0, ws2=0; i<set2; i++, j++ ) ws2 += weight[j];
455
456 double inv_weight_sum = (1.0L / ws1) + (1.0L / ws2);
457
458 double dof1 = ws1*(1.0L - 1.0L/set1); // degrees of freedom
459 double dof2 = ws2*(1.0L - 1.0L/set2);
460 double dof = dof1 + dof2;
461 double hdof = 0.5*dof;
462 double SD;
463 double Vn1, Vn2;
464
465 long zero_count1 = 0, zero_count2 = 0; // Counts numbers of zero values in t-test
466 for ( i=0; i<datasize; i++ ) {
467 Vn1 = var1[i] * dof1; // Compute numerator of variance from the variance
468 Vn2 = var2[i] * dof2;
469 SD = sqrt( ( (Vn1 + Vn2) / dof ) * inv_weight_sum ); // Standard error of the difference of the means
470 if ( SD > 0 ) { // test for zero in denominator
471 pt->set(i, ( (*p1)[i] - (*p2)[i] ) / SD); // t-test value
472 if ( (*pt)[i] == 0 ) zero_count2++;
473 sig[i] = betai(hdof, 0.5L, dof/(dof + (*pt)[i]*(*pt)[i]) ); // significance level
474 } else {
475 pt->set(i, 0);
476 sig[i] = 1;
477 zero_count1++;
478 }
479 }
480
481 if ( verbose & VERB_FULL ) {
482 cout << zero_count1 << " pixels had a zero value for the standard error of the differences of the means, and t was set to zero." << endl;
483 cout << zero_count2 << " pixels had standard deviation > 0 and t = 0." << endl;
484 cout << datasize << " = total pixels in image" << endl;
485 }
486
487 return pt;
488}

513Bimage* img_students_t_test_unequal(int set1, int set2, Bimage* p1, Bimage* p2, vector<double>& weight)
514{
515 // Set arrays and image parameters
516 Bimage* pt = new Bimage(Float, TSimple, p1->sizeX(), p1->sizeY(), p1->sizeZ(), 1);
517 pt->origin(p1->image->origin());
518 pt->sampling(p1->sampling(0));
519 pt->next = new Bimage(Float, TSimple, pt->size(), 1);
520
521 long i, j, datasize = p1->sizeX()*p1->sizeY()*p1->sizeZ()*p1->images();
522
523 float* var1 = (float *) p1->next->data_pointer(); // pointers to variance data
524 float* var2 = (float *) p2->next->data_pointer();
525 float* sig = (float *) pt->next->data_pointer(); // initialize figure-of-merit subimage structure, for significance level
526
527 // Compute Student's t-test
528 if ( verbose ) {
529 cout << endl << "Student's t-test, assuming unequal variance in sets 1 and 2" << endl;
530 cout << "Set 1 contains " << set1 << " images" << endl;
531 cout << "Set 2 contains " << set2 << " images" << endl;
532 }
533
534 double ws1, ws2;
535 for ( i=0, j=0, ws1=0; i<set1; i++, j++ ) ws1 += weight[j];
536 for ( i=0, ws2=0; i<set2; i++, j++ ) ws2 += weight[j];
537
538 double dof1 = ws1*(1.0L - 1.0L/set1); // degrees of freedom
539 double dof2 = ws2*(1.0L - 1.0L/set2);
540 double dof;
541 double Vn1, Vn2;
542
543 long zero_count1 = 0, zero_count2 = 0; // Counts numbers of zero values in t-test
544 for ( i=0; i<datasize; i++ ) {
545 if ( var1[i] > 0 || var2[i] > 0 ) { // test for zero in denominator
546 Vn1 = var1[i]/ws1;
547 Vn2 = var2[i]/ws2;
548 pt->set(i, ( (*p1)[i] - (*p2)[i] ) / sqrt(Vn1 + Vn2));
549 if ( (*pt)[i] == 0 ) zero_count2++;
550 dof = (Vn1 + Vn2)*(Vn1 + Vn2) / (Vn1*Vn1/dof1 + Vn2*Vn2/dof2);
551 sig[i] = betai(0.5L*dof, 0.5L, dof/(dof + (*pt)[i]*(*pt)[i]) ); // significance level
552 } else {
553 pt->set(i, 0);
554 sig[i] = 1;
555 zero_count1++;
556 }
557 }
558
559 if ( verbose & VERB_FULL ) {
560 cout << zero_count1 << " pixels had zero variance and t was set to zero." << endl;
561 cout << zero_count2 << " pixels had variance not equal to 0 and t = 0." << endl;
562 cout << datasize << " = total pixels in image" << endl;
563 }
564
565 return pt;
566}

587Bimage* img_f_test(int set1, int set2, Bimage* p1, Bimage* p2, vector<double>& weight)
588{
589 // Set image parameters and arrays
590 Bimage* pf = new Bimage(Float, TSimple, p1->size(), 1);
591 pf->origin(p1->image->origin());
592 pf->sampling(p1->sampling(0));
593 pf->next = new Bimage(Float, TSimple, pf->size(), 1);
594
595 long i, j, datasize = p1->sizeX()*p1->sizeY()*p1->sizeZ()*p1->images();
596
597 float* var1 = (float *) p1->next->data_pointer(); // pointers to variance data
598 float* var2 = (float *) p2->next->data_pointer();
599 float* sig = (float *) pf->next->data_pointer();
600
601 // Compute F-test
602 if ( verbose ) {
603 cout << endl << "F-test" << endl;
604 cout << "Set 1 contains " << set1 << " images" << endl;
605 cout << "Set 2 contains " << set2 << " images" << endl;
606 }
607
608 double ws1, ws2;
609 for ( i=0, j=0, ws1=0; i<set1; i++, j++ ) ws1 += weight[j];
610 for ( i=0, ws2=0; i<set2; i++, j++ ) ws2 += weight[j];
611
612 double dof1 = ws1*(1.0L - 1.0L/set1);
613 double dof2 = ws2*(1.0L - 1.0L/set2);
614 double df1, df2; // degrees of freedom, sets 1 and 2
615
616 for ( i=0; i<datasize; i++ ) {
617 if ( var1[i] < SMALLFLOAT || var2[i] < SMALLFLOAT ) {
618 pf->set(i, 0);
619 sig[i] = 1;
620 j++;
621 } else {
622 if ( var1[i] > var2[i] ) { // make f the ratio of the larger variance to the smaller one
623 pf->set(i, var1[i] / var2[i]);
624 df1 = dof1;
625 df2 = dof2;
626 } else {
627 pf->set(i, var2[i] / var1[i]);
628 df1 = dof2;
629 df2 = dof1;
630 }
631 sig[i] = 2*betai(0.5L*df2, 0.5L*df1, df2/(df2 + df1*(*pf)[i]) ); // significance level
632 if ( sig[i] > 1 ) sig[i] = 2 - sig[i];
633 }
634 }
635 if ( verbose & VERB_FULL ) {
636 cout << j << " pixels in set 1 or set 2 had zero variance, and the F-values were set to zero." << endl;
637 cout << datasize << " = total pixels in image" << endl;
638 }
639
640 return pf;
641}
