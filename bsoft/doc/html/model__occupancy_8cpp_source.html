125vector<double> model_occupancy_distribution(Bmodel* model, double cutoff, int nfit,
126 long& ncomp, vector<double>& prob, double& R)
127{
128 long i, j, k, h, occ, bins, nsets(3+nfit);
129 double c, s;
130 Bmodel* mp;
131 Bcomponent* comp;
132
133 ncomp = model_maxnum_components(model);
134 bins = ncomp + 1;
135
136 vector<double> dist(bins);
137 vector<double> disterrfit(bins*nsets);
138 vector<double> sum(bins);
139 vector<double> sum2(bins);
140
141 for ( i=k=0; i<bins; i++ ) {
142 dist[i] = sum[i] = sum2[i] = 0;
143 for ( j=0; j<nsets; j++, k++ ) disterrfit[k] = 0;
144 }
145
146 for ( mp = model; mp; mp = mp->next ) {
147 for ( occ=0, comp = mp->comp; comp; comp = comp->next )
148 if ( comp->FOM() >= cutoff ) occ++;
149 disterrfit[occ]++;
150 }
151 for ( s=0, i=0; i<bins; i++ ) s += disterrfit[i];
152 for ( i=0; i<bins; i++ ) disterrfit[i] /= s;
153
154
155 for ( c = cutoff - 0.2; c <= cutoff + 0.2; c += 0.1 ) {
156 for ( i=0; i<bins; i++ ) dist[i] = 0;
157 for ( mp = model; mp; mp = mp->next ) {
158 for ( occ=0, comp = mp->comp; comp; comp = comp->next )
159 if ( comp->FOM() >= c ) occ++;
160 dist[occ]++;
161 }
162 for ( s=0, i=0; i<bins; i++ ) s += dist[i];
163 for ( i=0; i<bins; i++ ) {
164 dist[i] /= s;
165 sum[i] += dist[i];
166 sum2[i] += dist[i]*dist[i];
167 }
168 }
169
170 for ( i=0, j=bins; i<bins; i++, j++ ) {
171 s = sum2[i]/5 - sum[i]*sum[i]/25;
172 if ( s > 0 ) disterrfit[j] = sqrt(s);
173 }
174
175 R = binomial_fit(disterrfit, ncomp, nfit, prob);
176
177 // Order distribution fits in ascending probability order
178 for ( i=0, k=nfit; i<nfit-1; i++, k++ ) {
179 for ( j=i+1, h=j+nfit; j<nfit; j++, h++ ) {
180 if ( prob[k] > prob[h] ) {
181 swap(prob[i], prob[j]);
182 swap(prob[k], prob[h]);
183 }
184 }
185 }
186
187 for ( i=0, j=3; i<nfit; i++, j++ ) {
188 for ( k=0, h=j*bins; k<bins; k++, h++ )
189 disterrfit[h] = Bnpk(ncomp, prob[nfit+i], k, prob[i]);
190 for ( k=0, h=2*bins; k<bins; k++, h++ )
191 disterrfit[h] += disterrfit[j*bins+k];
192 }
193
194 // Print results
195 if ( verbose ) {
196 cout << endl << "Binomial distributions:" << endl;
197 cout << "#\tWeight\tProbability" << endl;
198 for ( i=0; i<nfit; i++)
199 cout << i+1 << tab << prob[i] << tab << prob[nfit+i] << endl;
200 cout << "Residual: " << R << endl;
201 cout << endl << "Occup\t Models\t Std\t Fit";
202 for ( j=0; j<nfit; j++) cout << "\tB" << j+1;
203 cout << endl;
204
205 for ( i=0; i<bins; i++ ) {
206 cout << i << tab << disterrfit[i] << tab << disterrfit[bins+i] << tab << disterrfit[2*bins+i];
207 for ( j=0; j<nfit; j++)
208 cout << "\t " << disterrfit[(3+j)*bins+i];
209 cout << endl;
210 }
211 cout << endl;
212 }
213
214 return disterrfit;

400/*
401@author Daniel Nemecek and Bernard Heymann
402@brief Calculates the coverage of the component in a map.
403@param *p image.
404@param img_num sub-image number.
405@param comp component.
406@param threshold threshold to count covered voxels.
407@return double fraction of examined sphere covered.
408
409 The map must have positive density (higher values are considered density).
410 The coverage for a component is defined as the fraction of voxels within
411 a sphere around the component location that is above the threshold.
412 The radius of the sphere is taken from the component radius.
413**/
414double component_coverage(Bimage* p, int img_num, Bcomponent* comp, double threshold)
415{
416 int n(0), np(0);
417 long i, x, y, z;
418 double a(1), b(1), c(1), fx, fy, fz;
419 double value, fraction(0);
420 Vector3<int> start;
421 Vector3<int> end;
422
423 start[0] = (int) (p->image->origin()[0] + (comp->location()[0] - comp->radius())/p->sampling(0)[0]);
424 start[1] = (int) (p->image->origin()[1] + (comp->location()[1] - comp->radius())/p->sampling(0)[1]);
425 start[2] = (int) (p->image->origin()[2] + (comp->location()[2] - comp->radius())/p->sampling(0)[2]);
426 end[0] = (int) (p->image->origin()[0] + (comp->location()[0] + comp->radius())/p->sampling(0)[0]);
427 end[1] = (int) (p->image->origin()[1] + (comp->location()[1] + comp->radius())/p->sampling(0)[1]);
428 end[2] = (int) (p->image->origin()[2] + (comp->location()[2] + comp->radius())/p->sampling(0)[2]);
429
430 if ( verbose & VERB_DEBUG ) {
431 cout << "DEBUG component_coverage: start=" << start << endl;
432 cout << "DEBUG component_coverage: end=" << end << endl;
433 }
434
435 if ( ( start[0] < 0 ) && ( end[0] < 0 ) ) return 0;
436 if ( ( start[1] < 0 ) && ( end[1] < 0 ) ) return 0;
437 if ( ( start[2] < 0 ) && ( end[2] < 0 ) ) return 0;
438 if ( ( start[0] >= p->sizeX() ) && ( end[0] >= p->sizeX() ) ) return 0;
439 if ( ( start[1] >= p->sizeY() ) && ( end[1] >= p->sizeY() ) ) return 0;
440 if ( ( start[2] >= p->sizeZ() ) && ( end[2] >= p->sizeZ() ) ) return 0;
441 start = start.max(0);
442 end = end.max(0);
443 if ( start[0] >= p->sizeX() ) start[0] = p->sizeX() - 1;
444 if ( start[1] >= p->sizeY() ) start[1] = p->sizeY() - 1;
445 if ( start[2] >= p->sizeZ() ) start[2] = p->sizeZ() - 1;
446 if ( end[0] >= p->sizeX() ) end[0] = p->sizeX() - 1;
447 if ( end[1] >= p->sizeY() ) end[1] = p->sizeY() - 1;
448 if ( end[2] >= p->sizeZ() ) end[2] = p->sizeZ() - 1;
449 if ( start[0] > end[0] ) swap(start[0], end[0]);
450 if ( start[1] > end[1] ) swap(start[1], end[1]);
451 if ( start[2] > end[2] ) swap(start[2], end[2]);
452
453 Vector3<int> size = end - start;
454 Vector3<double> center;
455 center[0] = start[0] + size[0]/2.0;
456 center[1] = start[1] + size[1]/2.0;
457 center[2] = start[2] + size[2]/2.0;
458 if ( size[0] ) a = 2.0/size[0];
459 if ( size[1] ) b = 2.0/size[1];
460 if ( size[2] ) c = 2.0/size[2];
461 a *= a;
462 b *= b;
463 c *= c;
464
465 if ( verbose & VERB_DEBUG ) {
466 cout << "DEBUG component_coverage: start=" << start << endl;
467 cout << "DEBUG component_coverage: end=" << end << endl;
468 cout << "DEBUG component_coverage: center=" << center << endl;
469 }
470
471 for ( n=0, z=start[2]; z<=end[2]; z++ ) {
472 fz = z - center[2];
473 fz = c*fz*fz;
474 for ( y=start[1]; y<=end[1]; y++ ) {
475 fy = y - center[1];
476 fy = b*fy*fy;
477 for ( x=start[0]; x<=end[0]; x++ ) {
478 fx = x - center[0];
479 fx = a*fx*fx;
480 if ( fx + fy + fz <= 1 ) {
481 i = p->index(0, x, y, z, img_num);
482 value = (*p)[i];
483 if ( value > threshold ) np++;
484 n++;
485 }
486 }
487 }
488 }
489
490 if ( n ) fraction = np*1.0L/n;
491
492 return fraction;
493}
