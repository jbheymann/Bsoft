44{

53
54 // Normalize weights
55 for ( j=0, w=0; j<nfit; j++ ) w += simp.parameter(j);
56
57 if ( w > 0 )
58 for ( j=0; j<nfit; j++ ) simp.parameter(j, simp.parameter(j) / w);
59 else
60 for ( j=0; j<nfit; j++ ) simp.parameter(j, 1.0/nfit);
61
62 // Calculate residual
63 for ( i=0; i<simp.points(); i++ ) {
64 fit = 0;
65 for ( j=0; j<nfit; j++ )
66 fit += Bnpk(n, simp.parameter(nfit+j), i, simp.parameter(j));

68 R += df*df;
69 }
