110/*
111@brief Calculates a 3D density map from a set of atomic coordinates.
112@param *molgroup set of molecules with atomic coordinates.
113@param *p image structure for the new map.
114@param *atompar atom type parameters.
115@param resolution resolution (angstrom).
116@param Bfactor global B-factor to use, if 0, use individual atom B-factors
117@param wrap wrapping flag.
118@param gextype type of gaussian used: 0 = single, 1 = atomic potential
119@return int 0.
120
121 A 3D map is calculated from atomic coordinates by placing a gaussian
122 sphere at each set of atomic coordinates. The resolution is set as
123 twice the sigma coefficient of the gaussian function. The amplitude
124 of the gaussian function is set so that the total density calculated
125 equals the atomic mass. The resultant map therefore has the density
126 units of Dalton/voxel.
127 The content and statistics of the new image is not checked.
128
129**/
130int mol_to_image(Bmolgroup* molgroup, Bimage* p, Batomtype* atompar,
131 double resolution, double Bfactor, int wrap, int gextype)
132{
133 long h, i, j, irad, nsig, t;
134 long x, y, z, ix, iy, iz;
135 Vector3<int> lo, hi;
136 Vector3<double> cgrid;
137 double sigma, sigma2, avgsig, dx, dy, dz, dist2;
138 double sigma2_global = resolution*resolution/4 + Bfactor/(8*M_PI*M_PI);
139 double interval = p->sizeX()*p->sampling(0)[0]*0.5/MAXRAD; // Sampling for radial potential curve calculations
140 if ( interval > 0.1 ) interval = 0.1;
141 double sam2 = p->sampling(0)[0]*p->sampling(0)[0];
142 double voxel_size = p->sampling(0).volume();
143 double thisamp, thisdens, thisatom, summass(0), totmass(0);
144 long side = (long) floor((3.0/2.0)*resolution/p->sampling(0)[0]+1);
145 if ( side > MAXSID ) side = MAXSID;
146 if ( gextype ) {
147 side = (long) (MAXRAD*interval/p->sampling(0)[0] - 1);
148 if ( side*p->sampling(0)[0] > 5 ) side = (long) (5/p->sampling(0)[0] + 1);
149 }
150 if ( side > p->sizeX() ) side = p->sizeX();
151 double radius2 = side*side*sam2;
152
153 if ( verbose & VERB_PROCESS ) {
154 cout << "Sigma: " << sqrt(sigma2_global) << " A" << endl;
155 cout << "Cube edge length: " << 2*side+1 << " voxels" << endl;
156 }
157
158 Bmolecule* mol;
159 Bresidue* res;
160 Batom* atom;
161
162 double* apot = NULL;
163 if ( atompar ) apot = get_potential_curves(atompar, interval);
164
165 nsig = 0;
166 avgsig = 0;
167 for ( h=0, mol = molgroup->mol; mol; mol = mol->next ) if ( mol->sel ) {
168 h++;
169 for( res = mol->res; res; res = res->next ) {
170 for ( atom = res->atom; atom; atom = atom->next ) {
171 cgrid[0] = atom->coord[0]/p->sampling(0)[0] + p->image->origin()[0];
172 cgrid[1] = atom->coord[1]/p->sampling(0)[1] + p->image->origin()[1];
173 cgrid[2] = atom->coord[2]/p->sampling(0)[2] + p->image->origin()[2];
174 lo[0] = (long) (cgrid[0] + 0.5 - side);
175 hi[0] = (long) (cgrid[0] + 0.5 + side);
176 lo[1] = (long) (cgrid[1] + 0.5 - side);
177 hi[1] = (long) (cgrid[1] + 0.5 + side);
178 lo[2] = (long) (cgrid[2] + 0.5 - side);
179 hi[2] = (long) (cgrid[2] + 0.5 + side);
180 if ( !wrap ) {
181 if ( lo[0] < 0 ) lo[0] = 0;
182 if ( lo[1] < 0 ) lo[1] = 0;
183 if ( lo[2] < 0 ) lo[2] = 0;
184 if ( hi[0] > (long)p->sizeX() - 1 ) hi[0] = p->sizeX() - 1;
185 if ( hi[1] > (long)p->sizeY() - 1 ) hi[1] = p->sizeY() - 1;
186 if ( hi[2] > (long)p->sizeZ() - 1 ) hi[2] = p->sizeZ() - 1;
187 }
188 if ( Bfactor > 0 ) sigma2 = sigma2_global;
189 else sigma2 = sigma2_global + atom->b/(8*M_PI*M_PI); // B-factors
190// if ( sigma2 < 0.01 ) sigma2 = 0.01;
191 sigma = sqrt(sigma2);
192 nsig++;
193 avgsig += sigma;
194 if ( verbose & VERB_DEBUG )
195 cout << atom->coord << " " << atom->type << " " << atom->mass << endl;
196 thisatom = 0;
197 thisamp = atom->mass*voxel_size/(sqrt(TWOPI*sigma2)*TWOPI*sigma2);
198 sigma2 = -1.0/(2.0*sigma2);
199 ix = (long) (cgrid[0] + 0.5);
200 iy = (long) (cgrid[1] + 0.5);
201 iz = (long) (cgrid[2] + 0.5);
202 i = p->index(0, ix, iy, iz, 0);
203 for ( z=lo[2]; z<=hi[2]; z++ ) {
204 dz = (cgrid[2] - z)*(cgrid[2] - z)*sam2;
205 for ( y=lo[1]; y<=hi[1]; y++ ) {
206 dy = (cgrid[1] - y)*(cgrid[1] - y)*sam2;
207 for ( x=lo[0]; x<=hi[0]; x++ ) {
208 dx = (cgrid[0] - x)*(cgrid[0] - x)*sam2;
209 dist2 = dx + dy + dz;
210 if ( dist2 <= radius2 ) {
211 ix = x;
212 iy = y;
213 iz = z;
214 if ( wrap ) {
215 while ( ix < 0 ) ix += p->sizeX(); /* This wraps the density */
216 while ( iy < 0 ) iy += p->sizeY();
217 while ( iz < 0 ) iz += p->sizeZ();
218 while ( ix >= p->sizeX() ) ix -= p->sizeX();
219 while ( iy >= p->sizeY() ) iy -= p->sizeY();
220 while ( iz >= p->sizeZ() ) iz -= p->sizeZ();
221 }
222 j = p->index(ix, iy, iz);
223 thisdens = 0;
224 if ( gextype ) {
225 t = atom->tnum;
226 if ( j == i ) {
227 thisdens = apot[t*MAXRAD];
228 } else {
229 irad = (long) (sqrt(dist2)/interval + 0.5);
230 if ( irad < MAXRAD ) thisdens = apot[t*MAXRAD+irad];
231 }
232 } else {
233 thisdens = thisamp*exp(dist2*sigma2);
234 }
235 p->add(j, thisdens);
236 thisatom += thisdens;
237 }
238 }
239 }
240 }
241 totmass += thisatom;
242 summass += atom->mass;
243 }
244 }
245 if ( verbose & VERB_TIME )
246 cout << "Molecules done: " << h+1 << "\r" << flush;
247 }
248
249 double scale = POTPREFAC;
250 if ( p->sizeX() > 1 ) scale *= p->sampling(0)[0];
251 if ( p->sizeY() > 1 ) scale *= p->sampling(0)[1];
252 if ( p->sizeZ() > 1 ) scale *= p->sampling(0)[2];
253
254 if ( gextype ) {
255 p->rescale(scale, 0);
256 totmass *= scale;
257 }
258
259 if ( verbose & VERB_PROCESS ) {
260 cout << "Average sigma: " << avgsig/nsig << " A" << endl;
261 cout << "Total mass (sum): " << summass << " Da" << endl;
262 cout << "Total mass (density): " << totmass << " Da" << endl;
263 cout << " Ratio: " << totmass/summass << endl;
264 cout << "Total volume: " << totmass/RHO << " A3" << endl << endl;
265 }
266
267 if ( apot ) delete[] apot;
268
269 return 0;
270}

281{
282 Bmolecule* mol;
283 Bresidue* res;
284 Batom* atom;
285
286 long j, natom(0);
287 long x, y, z;
288 double min = 1e30, max = -1e30, Qavg(0), Qsd(0);
289 double scale = (pcalc->maximum() - pcalc->minimum())/(p->maximum() - p->minimum());
290
291 for ( mol = molgroup->mol; mol; mol = mol->next ) {
292 for( res = mol->res; res; res = res->next ) {
293 for ( atom = res->atom; atom; atom = atom->next ) {
294 x = (long) floor(atom->coord[0]/pcalc->sampling(0)[0] + pcalc->image->origin()[0] + 0.5);
295 y = (long) floor(atom->coord[1]/pcalc->sampling(0)[1] + pcalc->image->origin()[1] + 0.5);
296 z = (long) floor(atom->coord[2]/pcalc->sampling(0)[2] + pcalc->image->origin()[2] + 0.5);
297 while ( x < 0 ) x += pcalc->sizeX();
298 while ( x >= pcalc->sizeX() ) x -= pcalc->sizeX();
299 while ( y < 0 ) y += pcalc->sizeY();
300 while ( y >= pcalc->sizeY() ) y -= pcalc->sizeY();
301 while ( z < 0 ) z += pcalc->sizeZ();
302 while ( z >= pcalc->sizeZ() ) z -= pcalc->sizeZ();
303 j = pcalc->index(x, y, z);
304 if ( (*pcalc)[j] ) atom->q = scale*(*p)[j]/(*pcalc)[j];
305 if ( atom->q > 1 ) atom->q = 1;
306 if ( min > atom->q ) min = atom->q;
307 if ( max < atom->q ) max = atom->q;
308 Qavg += atom->q;
309 Qsd += atom->q*atom->q;
310 atom->b = atom->q;
311 natom++;
312 }
313 }
314 }
315 Qavg /= natom;
316 Qsd = sqrt(Qsd/natom - Qavg*Qavg);
317
318 if ( verbose & VERB_PROCESS ) {
319 cout << "Comparison of molecule " << mol->id << " with map " << p->file_name() << endl;
320 cout << "Origin (original): " << p->image->origin() << endl;
321 cout << "Origin (calculated): " << pcalc->image->origin() << endl;
322 cout << "Scale: " << scale << endl;
323 cout << "Min & max: " << min << " " << max << endl;
324 cout << "Average and standard deviation: " << Qavg << " " << Qsd << endl << endl;
325 } else {
326 cout << "Comparison of molecule " << mol->id << " with map " << p->file_name() << endl;
327 cout << "Average and standard deviation: " << Qavg << " " << Qsd << endl << endl;
328 }
329
330 return 0;
331}

413double one_sf(Bmolgroup* molgroup, Bimage* p, long i, double s, double scale, double* scat)
414{
415 if ( s <= 0 ) return 0;
416
417 long x, y, z;
418 long h, k, l, t, si;
419 double fraction, fraction1, f, phi;
420 Bmolecule* mol;
421 Bresidue* res;
422 Batom* atom;
423 Complex<float> temp;
424
425 p->coordinates(i, x, y, z);
426 if ( x < (p->sizeX() + 1)/2 ) h = x; else h = (long)x - p->sizeX();
427 if ( y < (p->sizeY() + 1)/2 ) k = y; else k = (long)y - p->sizeY();
428 if ( z < (p->sizeZ() + 1)/2 ) l = z; else l = (long)z - p->sizeZ();
429
430 si = (long) s;
431 fraction = s - si;
432 fraction1 = 1 - fraction;
433
434 for ( mol = molgroup->mol; mol; mol = mol->next ) {
435 for( res = mol->res; res; res = res->next ) {
436 for ( atom = res->atom; atom; atom = atom->next )
437 if ( atom->sel ) {
438 t = atom->tnum*MAXSCAT + si;
439 f = atom->q*(fraction1*scat[t] + fraction*scat[t+1]);
440 phi = MIN2PI*(h*atom->vel[0] +
441 k*atom->vel[1] + l*atom->vel[2]);
442 temp = Complex<float>(cos(phi), sin(phi));
443 p->set(i, p->complex(i) + f*temp);
444 }
445 }
446 }
447
448 p->set(i, p->complex(i) * scale);
449
450 if ( verbose & VERB_DEBUG )
451 cout << "DEBUG mol_to_structure_factors: " << x << " " << y << " " << z << " " <<
452 (p->complex(i)).real() << " " << (p->complex(i)).imag() << endl;
453
454 return 0;
455}

457/*
458@brief Calculates a 3D set of structure factors from a set of atomic coordinates.
459@param *molgroup set of molecules with atomic coordinates.
460@param *p image structure for the new structure factors.
461@param *atompar atom type parameters.
462@param resolution resolution (angstrom).
463@param wrap 0=cut atoms outside box, 1=wrap coordinates within unit cell.
464@param Bfactor constant for decay curve.
465@return int 0.
466
467 All structure factors within a given resolution are calculated from
468 all the selected atomic coordinates. The coordinates are fractionalized
469 to fit into the given size box. If the size of the box is not given,
470 it defaults to:
471 x_size = (max(x_coor) - min(x_coor)) / x_sampling
472 The atomic scattering profiles are read from the STAR database as the
473 amplitudes and B-factors of reciprocal space gaussians. For each profile,
474 a lookup table is calculated to speed up further calculations.
475 The content and statistics of the new image is not checked.
476
477**/
478int mol_to_structure_factors(Bmolgroup* molgroup, Bimage* p,
479 Batomtype* atompar, double resolution, int wrap, double Bfactor)
480{
481 long i, j, h, k, l, x, y, z, yi, zi, n, t, nmax;
482 long nscat, nmol(0), natom(0), nsel(0);
483 double s, smax, sx, sy, sz;
484 Bmolecule* mol;
485 Bresidue* res;
486 Batom* atom;
487 Batomtype* at;
488
489 // Set the resolution limit
490 smax = 1;
491 if ( resolution ) smax = 1.0/resolution;
492 resolution = 1.0/smax;
493
494 double recip_interval = 1.1*smax/MAXSCAT;
495 double inv_recip_int = 1/recip_interval;
496 Vector3<double> ori = p->sampling(0)*p->image->origin();
497 Vector3<double> fcoor;
498
499 // Convert to fractional coordinates
500 UnitCell unit_cell(p->real_size());
501 p->unit_cell(unit_cell);
502// cout << unit_cell << endl;
503 Matrix3 frac_mat = unit_cell.skew_matrix();
504
505 double* scat = get_scattering_curves(atompar, Bfactor, recip_interval, nscat);
506
507 for ( nmol = 0, mol = molgroup->mol; mol; mol = mol->next ) if ( mol->sel ) {
508 nmol++;
509 for( res = mol->res; res; res = res->next ) {
510 for ( atom = res->atom; atom; atom = atom->next, natom++ ) {
511 atom->tnum = -1;
512 for ( t=0, at = atompar; at && atom->tnum < 0; at = at->next, t++ )
513 if ( strncmp(atom->el, at->el, 2) == 0 )
514 atom->tnum = t;
515 if ( atom->tnum < 0 ) atom->sel = 0;
516 if ( atom->q <= 0 ) atom->sel = 0;
517 if ( verbose & VERB_DEBUG )
518 cout << "DEBUG mol_to_structure_factors: Atom = " << atom->el << endl;
519 fcoor = atom->coord + ori;
520 atom->vel = frac_mat * fcoor;
521 if ( wrap == 1 ) {
522 while ( atom->vel[0] < 0 ) atom->vel[0] += 1;
523 while ( atom->vel[1] < 0 ) atom->vel[1] += 1;
524 while ( atom->vel[2] < 0 ) atom->vel[2] += 1;
525 while ( atom->vel[0] > 1 ) atom->vel[0] -= 1;
526 while ( atom->vel[1] > 1 ) atom->vel[1] -= 1;
527 while ( atom->vel[2] > 1 ) atom->vel[2] -= 1;
528 } else if ( wrap > 1 ) {
529 if ( atom->vel[0] < 0 ) atom->sel = 0;
530 if ( atom->vel[1] < 0 ) atom->sel = 0;
531 if ( atom->vel[0] > 1 ) atom->sel = 0;
532 if ( atom->vel[1] > 1 ) atom->sel = 0;
533 while ( atom->vel[2] < 0 ) atom->vel[2] += 1;
534 while ( atom->vel[2] > 1 ) atom->vel[2] -= 1;
535 } else {
536 if ( atom->vel[0] < 0 ) atom->sel = 0;
537 if ( atom->vel[1] < 0 ) atom->sel = 0;
538 if ( atom->vel[2] < 0 ) atom->sel = 0;
539 if ( atom->vel[0] > 1 ) atom->sel = 0;
540 if ( atom->vel[1] > 1 ) atom->sel = 0;
541 if ( atom->vel[2] > 1 ) atom->sel = 0;
542 }
543 if ( atom->sel ) nsel++;
544 if ( verbose & VERB_DEBUG )
545 cout << "DEBUG mol_to_structure_factors: "
546 << atom->el << " " << atom->vel << endl;
547 }
548 }
549 }
550
551 if ( verbose & VERB_PROCESS ) {
552 cout << "Molecules and atoms: " << nmol << " " << natom << " (" << nsel << ")" << endl;
553 cout << "Unit cell: " <<
554 unit_cell.a() << " " << unit_cell.b() << " " << unit_cell.c() << " " <<
555 unit_cell.alpha()*180/M_PI << " " << unit_cell.beta()*180/M_PI << " " << unit_cell.gamma()*180/M_PI << endl;
556 cout << "Fractionalization matrix: " << endl << frac_mat << endl;
557 }
558
559 // Calculate the structure factors
560 // Default space group = 1
561 // Estimate the number of structure factors from the spherical volume or circular area
562 n = 0;
563 nmax = (long) ((4.0*M_PI/3.0)*smax*smax*smax/(frac_mat[0][0]*frac_mat[1][1]*frac_mat[2][2]));
564 if ( p->sizeZ() < 2 )
565 nmax = (long) (M_PI*smax*smax/(frac_mat[0][0]*frac_mat[1][1]));
566 // Factor required to calculate the estimated completion from the number of
567 // structure factors already calculated
568// pfac = 100.0/(nmax/2 + M_PI*smax*smax/(2*frac_mat[1][1]*frac_mat[2][2]));
569// if ( p->sizeZ() < 2 )
570// pfac = 100.0/(nmax/2 + smax/frac_mat[1][1]);
571 if ( verbose & VERB_PROCESS )
572 cout << "Reflections estimated: " << nmax << endl;
573
574 // Scaled by the potential prefactor K divided by the volume
575 double scale = POTPREFAC/p->size().volume();
576 if ( p->sizeX() > 1 ) scale /= p->sampling(0)[0];
577 if ( p->sizeY() > 1 ) scale /= p->sampling(0)[1];
578 if ( p->sizeZ() > 1 ) scale /= p->sampling(0)[2];
579
580 if ( verbose & VERB_DEBUG )
581 cout << "DEBUG mol_to_structure_factors: scale = " << scale << endl;
582
583 long imgsize = (long) p->size().volume();
584 long hx = p->sizeX()/2, hy = p->sizeY()/2, hz = p->sizeZ()/2;
585 long hx1 = (p->sizeX()+1)/2, hy1 = (p->sizeY()+1)/2, hz1 = (p->sizeZ()+1)/2;
586 Complex<float> temp;
587
588 p->next = new Bimage(Float, TSimple, p->size(), p->images());
589 float* fom = (float *) p->next->data_pointer();
590
591 // Set up FOM block with s values
592 for ( l=-hz; l<hz1; l++ ) {
593 z = ( l < 0 ) ? l + p->sizeZ() : l;
594 sz = l*frac_mat[2][2];
595 sz *= sz;
596 for ( k=-hy; k<hy1; k++ ) {
597 y = ( k < 0 ) ? k + p->sizeY() : k;
598 sy = k*frac_mat[1][1];
599 sy *= sy;
600 for ( h=-hx; h<=0; h++ ) {
601 x = ( h < 0 ) ? h + p->sizeX() : h;
602 sx = h*frac_mat[0][0];
603 sx *= sx;
604 s = sqrt(sx + sy + sz);
605 if ( s < smax ) {
606 i = p->index(x, y, z);
607 s *= inv_recip_int;
608 fom[i] = s;
609 n++;
610 }
611 }
612 }
613 }
614
615 fom[0] = 1e-10; // Make F0 non-zero
616
617#ifdef HAVE_GCD
618 dispatch_apply(imgsize, dispatch_get_global_queue(0, 0), ^(size_t i){
619 if ( fom[i] ) one_sf(molgroup, p, i, fom[i], scale, scat);
620 });
621#else
622#pragma omp parallel for
623 for ( i=0; i<imgsize; i++ ) if ( fom[i] )
624 one_sf(molgroup, p, i, fom[i], scale, scat);
625#endif
626
627// cout << "first half done" << endl;
628
629 // Fill in the second half from the first
630 for ( l=-hz; l<hz1; l++ ) {
631 z = ( l < 0 ) ? l + p->sizeZ() : l;
632 zi = ( z == 0 ) ? 0 : p->sizeZ() - z;
633 sz = l*frac_mat[2][2];
634 sz *= sz;
635 for ( k=-hy; k<hy1; k++ ) {
636 y = ( k < 0 ) ? k + p->sizeY() : k;
637 yi = ( y == 0 ) ? 0 : p->sizeY() - y;
638 sy = k*frac_mat[1][1];
639 sy *= sy;
640 for ( h=1; h<hx1; h++ ) {
641 sx = h*frac_mat[0][0];
642 sx *= sx;
643 s = sqrt(sx + sy + sz);
644 if ( s < smax ) {
645 i = p->index(p->sizeX() - h, yi, zi);
646 j = p->index(h, y, z);
647 p->set(j, (p->complex(i)).conj());
648 n++;
649 }
650 }
651 }
652 }
653
654 if ( verbose & VERB_PROCESS )
655 cout << "Reflections calculated: " << n << endl << endl;
656
657 delete[] scat;
658
659 return 0;
660}

798 double Bfactor, double recip_interval)
