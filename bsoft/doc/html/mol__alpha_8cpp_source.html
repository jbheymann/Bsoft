68Bmolecule* mol_generate_alpha_helix(int length)
69{
70 int i, j, n;
71 char molname[8] = "A";
72 char restype[8] = "ALA";
73 Bmolecule* mol = NULL;
74 mol = molecule_add(&mol, molname);
75
76 Bsecondary* sec = (Bsecondary *) add_item((char **) &mol->sec, sizeof(Bsecondary));
77 Bresidue* res = NULL;
78 Batom* atom = NULL;
79 Matrix3 mat(1);
80 Vector3<double> shift, axis(0,0,1);
81
82 if ( verbose & VERB_PROCESS )
83 cout << "Generating an aplha helix of length " << length << endl << endl;
84
85 for ( i=n=0; i<length; i++ ) {
86 res = residue_add(&mol->res, restype);
87 res->num = i + 1;
88 mat = Matrix3(axis, HANGLE*i*M_PI/180.0);
89 shift = Vector3<double>(0, 0, HRISE*i);
90 for ( j=0; j<NTEMPATOM; j++ ) {
91 atom = atom_add(&res->atom, alpha_template[j].atomtype);
92 atom->coord[0] = alpha_template[j].coord[0];
93 atom->coord[1] = alpha_template[j].coord[1];
94 atom->coord[2] = alpha_template[j].coord[2];
95 atom->coord = mat * atom->coord;
96 atom->coord += shift;
97 atom->num = ++n;
98 atom->sel = 1;
99 }
100 }
101
102 sec->num = 1;
103 sec->id[0] = 'A';
104 sec->first = mol->res;
105 sec->last = res;
106
107 return mol;
108}

208Vector3<double> alpha_find_orientation(Bresidue* resfirst, Bresidue* reslast)
209{
210 int i, nca;
211 Bresidue* res;
212 Batom* atom;
213 Vector3<double> ca[1000], v1, v2, va, vas;
214
215 for ( nca=0, res = resfirst; res != reslast->next; res = res->next )
216 for ( atom = res->atom; atom; atom = atom->next )
217 if ( strncmp(atom->type, "CA", 2) == 0 )
218 ca[nca++] = atom->coord;
219
220 for ( i=0; i<nca-3; i++ ) {
221 v1 = (ca[i] + ca[i+2]) - (ca[i+1] * 2);
222 v2 = (ca[i+1] + ca[i+3]) - (ca[i+2] * 2);
223 v1.normalize();
224 v2.normalize();
225 va = v1.cross(v2);
226 va.normalize();
227 vas += va;
228 }
229
230 vas.normalize();
231
232 i = nca/2;
233 va = point_on_helix_axis(ca[i-1], ca[i], ca[i+1]);
234
235 if ( verbose & VERB_FULL ) {
236 cout << "Helical center: " << va << endl;
237 cout << "Helical axis: " << vas << endl << endl;
238 }

359long find_closest_mean(long k, Vector3<double>* mc, Vector3<double>* ma, Vector3<double> c, Vector3<double> a, double afac)
360{
361 long i, iclose = -1;
362 double d, dmin = 1e30, angle;
363
364 for ( i=0; i<k; i++ ) {
365 d = c.distance2(mc[i]);
366 angle = afac*a.angle(ma[i]);
367 d = sqrt(d + angle*angle);
368 if ( dmin > d ) {
369 dmin = d;
370 iclose = i;
371 }
372 }

382Bmolgroup* molgroup_consolidate_alpha(Bmolgroup* molgroup)

385 double sampling = 3;
386 Bmolecule* mol, *newmol;
387 Bsecondary* sec;
388
389 for ( nsec=0, mol = molgroup->mol; mol; mol = mol->next )
390 for ( sec = mol->sec; sec; sec = sec->next ) nsec++;

392 if ( nsec < 1 ) {
393 cerr << "Error: No helices found!" << endl;

399 Vector3<double>* a = new Vector3<double>[nsec];

401 for ( i=0, mol = molgroup->mol; mol; mol = mol->next ) {
402 for ( sec = mol->sec; sec; sec = sec->next, i++ ) {
403 c[i] = alpha_find_center(sec->first, sec->last);
404 a[i] = alpha_find_orientation(sec->first, sec->last);
405 }
406 }
407
408 molgroup->box = molgroup->max - molgroup->min;
409
410 // Find the number of density peaks as the number of means and their starting points
411 Bimage* p = new Bimage(Integer, TSimple,
412 (long) (molgroup->box[0]/sampling+1),
413 (long) (molgroup->box[1]/sampling+1),
414 (long) (molgroup->box[2]/sampling+1), 1);
415 p->sampling(sampling, sampling, sampling);
416
417 for ( i=0; i<nsec; i++ ) {
418 x = (long) ((c[i][0] - molgroup->min[0])/p->sampling(0)[0]);
419 y = (long) ((c[i][1] - molgroup->min[1])/p->sampling(0)[1]);
420 z = (long) ((c[i][2] - molgroup->min[2])/p->sampling(0)[2]);
421 j = p->index(0, x, y, z, 0);
422 p->set(j, (*p)[j] + 1);
423 }
424
425 Bimage* pseg = p->regions(1, 0);
426
427 delete p;

429// write_img("t.tif", pseg, 0);

431 long k = (long) pseg->maximum();

434 Vector3<double>* ma = new Vector3<double>[k];
435 Vector3<double>* mcn = new Vector3<double>[k];
436 Vector3<double>* man = new Vector3<double>[k];
437 Vector3<double>* min = new Vector3<double>[k];
438 Vector3<double>* max = new Vector3<double>[k];
439
440 for ( i=0; i<k; i++ ) n[k] = 0;
441
442 for ( i=z=0; z<pseg->sizeZ(); z++ ) {
443 for ( y=0; y<pseg->sizeY(); y++ ) {
444 for ( x=0; x<pseg->sizeX(); x++, i++ ) {
445 if ( (*pseg)[i] > 0 ) {
446 j = (long)(*pseg)[i] - 1;
447 n[j]++;
448 mc[j] += Vector3<double>(x*pseg->sampling(0)[0] + molgroup->min[0],
449 y*pseg->sampling(0)[1] + molgroup->min[1], z*pseg->sampling(0)[2] + molgroup->min[2]);
450 }
451 }
452 }
453 }
454
455 delete pseg;
456
457 if ( verbose & VERB_PROCESS ) {
458 cout << "Consolidating a set of helices:" << endl;
459 cout << "Number of helix groups: " << k << endl;
460 }
461
462 for ( j=0; j<k; j++ )
463 if ( n[j] ) mc[j] /= n[j];
464
465 // Do k-means iterations
466 long it = 0;
467 double afac = 0;
468 double d = 1e30;
469 while ( d > 1e-10 ) {
470 it++;
471 for ( j=0; j<k; j++ ) {
472 mcn[j] = man[j] = 0;
473 min[j] = Vector3<double>(1e30, 1e30, 1e30);
474 max[j] = Vector3<double>(-1e30, -1e30, -1e30);
475 n[j] = 0;
476 }
477 for ( i=0; i<nsec; i++ ) {
478 j = find_closest_mean(k, mc, ma, c[i], a[i], afac);
479 mcn[j] += c[i];
480 man[j] += a[i];
481 min[j] = min[j].min(c[i]);
482 max[j] = max[j].max(c[i]);
483 n[j]++;
484 }
485 d = 0;
486 if ( verbose & VERB_FULL )
487 cout << "Iteration " << it << ":" << endl;
488 for ( j=0; j<k; j++ ) {
489 if ( n[j] ) {
490 mcn[j] /= n[j];
491 man[j] /= n[j];
492 } else {
493 mcn[j] = vector3_random(molgroup->max, molgroup->min);
494 man[j] = vector3_random(-1.0, 1.0);
495 man[j].normalize();
496 }
497 if ( verbose & VERB_FULL )
498 cout << j+1 << tab << n[j] << tab <<
499 mcn[j][0] << tab << mcn[j][1] << tab << mcn[j][2] << tab <<
500 man[j][0] << tab << man[j][1] << tab << man[j][2] << endl;
501 d += mcn[j].distance2(mc[j]);
502 d += man[j].distance2(ma[j]);
503 mc[j] = mcn[j];
504 ma[j] = man[j];
505 }
506 d = sqrt(d/k);
507 afac = 10;
508 if ( verbose )
509 cout << it << tab << d << endl;
510 }

512 molgroup_kill(molgroup);
513
514 // Generate a new set of helices
515 molgroup = molgroup_init();
516 Bmolecule* molalpha = NULL;
517 View2<double> view;
518 Vector3<double> origin;
519 char molname = 'A';
520
521 for ( j=0; j<k; j++ ) if ( n[j] ) {
522 d = max[j].distance(min[j]);
523 molalpha = mol_generate_alpha_helix(14 + (int)(d/1.5));
524 mol_shift_to_center_of_mass(molalpha);
525// view = View2<double>(ma[j][0], ma[j][1], ma[j][2], 0);

529 if ( !molgroup->mol ) molgroup->mol = newmol;

531 mol = newmol;
532 molname++;
533 if ( molname > 'Z' ) molname = 'A';
534 molecule_kill(molalpha);
535 }
536
537 delete[] c;
538 delete[] a;
539 delete[] n;
540 delete[] mc;
541 delete[] ma;
542 delete[] mcn;
543 delete[] man;
544 delete[] min;
545 delete[] max;
546
547 return molgroup;
