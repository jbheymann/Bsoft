442double lambda_R(Bsimplex& simp)
443{
444 long i;
445 double R(0), df, fs(0), fs2(0);
446 vector<double>& f = simp.dependent_values();
447 vector<double>& x = simp.independent_values();
448
449 for ( i=0; i<simp.points(); i++ ) {
450 df = cos(x[i] + simp.parameter(2));
451 if ( df )
452 df = f[i] - simp.parameter(0)*exp(-simp.parameter(1)/df);
453 else
454 df = f[i] - simp.parameter(0);
455 fs += f[i];
456 fs2 += f[i] * f[i];
457 R += df*df;
458 }
459
460 df = i*fs2-fs*fs;
461 if ( df > 0 ) R = sqrt(R/df);
462 else R = sqrt(R/i);
463
464 return R;
465}

483// simp.show();

489 simp.parameter(0, 1.5*Imax);
490 simp.parameter(1, 0.7);
491 simp.parameter(2, 0.01);
492 R = simp.run(10000, 0.0001, lambda_R);
493// simp.show();
494 }
495
496 vector<double> result;
497 for ( i=0; i<3; ++i )
