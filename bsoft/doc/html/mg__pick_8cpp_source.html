18/* flags 1: quadric correct; 2: gaussian filter; 4: mask background */
19Bimage* img_prepare_for_picking(Bimage* p, int flags, long sigma, long bin)
20{
21 Bimage* pb = p->bin_copy(bin);
22
23 if ( flags & 1 ) pb->quadric_correct(pb->quadric_fit());
24
25 long ksize(9);
26 if ( flags & 2 ) {
27 sigma /= bin;
28 if ( sigma < 1 ) sigma = 1;
29 ksize = 2*( (long) (4*sigma + 0.9) ) + 1;
30 pb->filter_gaussian(ksize, sigma);
31 }
32
33 pb->rescale_to_avg_std(0, 1);
34 pb->statistics();
35
36 double threshold(pb->average() - 3*pb->standard_deviation());
37 Bimage* pmask = NULL;
38
39 if ( verbose & VERB_FULL )
40 cout << "threshold = " << threshold << endl;
41
42 if ( flags & 4 ) {
43 pmask = pb->mask_by_threshold(threshold);
44 pmask->mask_erode(5);
45 pmask->change_type(Float);
46 pmask->filter_average(11);
47 pb->multiply(pmask, 1, 0);
48 delete pmask;
49 }
50
51 return pb;
52}

75Bparticle* particles_from_peaks(Bimage* pcc, long bin, double excl_dist,
76 double part_ori, double& fommin, double fommax,
77 long maxnum, double pix_min, double pix_max)
78{
79 if ( part_ori < excl_dist ) part_ori = excl_dist;
80
81 long i, j, ncoor;
82 double fom;
83 Vector3<double> origin(part_ori, part_ori, part_ori);
84 origin = origin.min(pcc->size()-1);
85 Vector3<long> min(origin/bin), max(pcc->size()-min);
86
87 if ( verbose & VERB_PROCESS )
88 cout << "Limits: " << min << tab << max << endl;
89
90 Vector3<long> obin(bin, bin, bin);
91 obin = obin.min(pcc->size());
92
93 if ( excl_dist < 10 ) excl_dist = 1.8*part_ori;
94
95 excl_dist /= bin;
96
97 Vector3<double>* coor = pcc->find_peaks(excl_dist, ncoor, fommin, fommax, pix_min, pix_max);
98
99 Bparticle* partlist = NULL;
100 Bparticle* part = NULL;
101
102 for ( i=j=0; i<ncoor && j<maxnum; i++ ) {
103 fom = pcc->get(0, coor[i]);
104 if ( fom >= fommin && fom <= fommax ) {
105 if ( coor[i] >= min && coor[i] <= max ) {
106 part = particle_add(&part, ++j);
107 if ( !partlist ) partlist = part;
108 part->loc = coor[i] * obin;
109 part->ori = origin;
110 part->fom[0] = fom;
111 }
112 }
113 }
114
115 delete[] coor;
116
117 if ( verbose & VERB_PROCESS )
118 cout << "Number of particles found: " << j << endl << endl;
119
120 return partlist;
121}

182double gauss_find_intersection(long ngauss, vector<double>& gauss)
183{
184 double a1, a2, v, d(1), l((gauss[1]+gauss[2])/(gauss[0]+gauss[3]));
185 double cut((gauss[1] + gauss[4])/2);
186
187 while ( d > 1e-6 ) {
188 v = (cut - gauss[1])/gauss[2];
189 a1 = gauss[0]*exp(-0.5*v*v);
190 v = (cut - gauss[4])/gauss[5];
191 a2 = gauss[3]*exp(-0.5*v*v);
192 d = a2 - a1;
193 cut -= l*d;
194 l *= 0.99;
195 if ( verbose & VERB_DEBUG )
196 cout << a1 << tab << a2 << tab << cut << endl;
197 d = fabs(d);
198 }
199
200 if ( verbose & VERB_DEBUG )
201 cout << "u1=" << gauss[1] << tab << "u2=" << gauss[4] << tab << "cut=" << cut << endl;
202
203 return cut;
204}

387double img_marker_set_difference_at_voxel(Bimage* p, long i, Bmarker* markin, Bmarker* markout)
388{
389 long x, y, z, n;
390 double v, sumin(0), sumout(0);
391 Bmarker* mark = NULL;
392
393 p->coordinates(i, x, y, z);
394 Vector3<float> c(x, y, z);
395
396 for ( n=0, mark = markin; mark; mark = mark->next ) {
397 v = p->get(0, mark->loc + c, 0);
398 if ( v ) {
399 sumin += v;
400 n++;
401 }
402 }
403
404 if ( n ) sumin /= n;
405
406 for ( n=0, mark = markout; mark; mark = mark->next ) {
407 v = p->get(0, mark->loc + c, 0);
408 if ( v ) {
409 sumout += v;
410 n++;
411 }
412 }
413
414 if ( n ) sumout /= n;
415
416 return sumin - sumout;
417}

922 nupart->sel = i;

925 if ( part->loc[2] < 1 ) nupart->loc[2] = 0; // for micrographs
926 if ( verbose & VERB_FULL )

928 }
