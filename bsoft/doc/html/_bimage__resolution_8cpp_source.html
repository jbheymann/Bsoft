92Bplot* Bimage::fsc_dpr(double hi_res, double sampling_ratio, int flag)
93{
94 check_resolution(hi_res);
95 if ( sampling_ratio <= 0 ) sampling_ratio = 1;
96
97 // The frequency scaling is linked to the different dimensions of the
98 // data set (important when the x, y and z dimensions are different)
99 // The radius scaling is set to a value consistent with one pixel width
100 // in reciprocal space for the longest dimension.
101 Vector3<double> freq_scale(1/real_size());
102 double rad_scale = real_size()[0]/sampling_ratio;
103// if ( hi_res < 2*sampling(0)[0] ) hi_res = 2*sampling(0)[0];
104// if ( hi_res > real_size().max()/2.0 ) hi_res = real_size().max()/2.0;
105
106 long maxrad = (long) (2 + rad_scale/hi_res);
107
108 Complex<double> sf1, sf2;
109
110 if ( verbose & VERB_DEBUG )
111 cout << "DEBUG Bimage::fsc_dpr: FFT done" << endl;
112
113 long i, j, xx, yy, zz, ix, iy, iz, iradius, iradius2;
114 double radius, fraction, fraction2, I1, I2, Isum, F1F2re, dphi, Id2;
115 double rx, ry, rz;
116
117 if ( verbose > VERB_RESULT ) {
118 cout << "Resolution limit: " << hi_res << " A (" << maxrad << ")" << endl;
119 cout << "Sampling ratio: " << sampling_ratio << endl << endl;
120 }
121
122 vector<double> F1(maxrad,0);
123 vector<double> F2(maxrad,0);
124 vector<double> Fsum(maxrad,0);
125 vector<double> FSC(maxrad,0);
126 vector<double> DPR(maxrad,0);
127
128 for ( zz=0; zz<z; zz++ ) {
129 rz = zz;
130 if ( rz > (z - 1)/2 ) rz -= z;
131 rz *= freq_scale[2];
132 iz = -zz;
133 if ( iz < 0 ) iz += z;
134 for ( yy=0; yy<y; yy++ ) {
135 ry = yy;
136 if ( ry > (y - 1)/2 ) ry -= y;
137 ry *= freq_scale[1];
138 iy = -yy;
139 if ( iy < 0 ) iy += y;
140 for ( xx=0; xx<x/2; xx++ ) {
141 i = (zz*y + yy)*x + xx;
142 rx = xx;
143 if ( xx > (x - 1)/2 ) rx -= x;
144 rx *= freq_scale[0];
145 ix = -xx;
146 if ( ix < 0 ) ix += x;
147 radius = rad_scale*sqrt(rx*rx + ry*ry + rz*rz);
148 iradius = (long) radius;
149 iradius2 = iradius + 1;
150 if ( iradius2 < maxrad ) {
151 fraction = radius - iradius;
152 fraction2 = 1.0 - fraction;
153 j = (iz*y + iy)*x + ix;
154 sf1 = complex(i).unpack_first(complex(j));
155 sf2 = complex(i).unpack_second(complex(j));
156 I1 = sf1.power();
157 I2 = sf2.power();
158 dphi = angle_set_negPI_to_PI(sf1.phi() - sf2.phi());
159 dphi *= dphi;
160 F1[iradius] += fraction2*I1;
161 F1[iradius2] += fraction*I1;
162 F2[iradius] += fraction2*I2;
163 F2[iradius2] += fraction*I2;
164 F1F2re = sf1.real()*sf2.real() + sf1.imag()*sf2.imag();
165 FSC[iradius] += fraction2*F1F2re;
166 FSC[iradius2] += fraction*F1F2re;
167 Isum = I1 + I2;
168 Fsum[iradius] += fraction2*Isum;
169 Fsum[iradius2] += fraction*Isum;
170 Id2 = Isum*dphi;
171 DPR[iradius] += fraction2*Id2;
172 DPR[iradius2] += fraction*Id2;
173 }
174 }
175 }
176 }
177
178 if ( verbose & VERB_DEBUG )
179 cout << "DEBUG Bimage::fsc_dpr: Creating plot" << endl;
180
181 int ncol(3);
182 if ( flag ) ncol = 2;
183
184 Bstring title("Resolution"), txt;
185 Bplot* plot = new Bplot(1, maxrad, ncol);
186
187 plot->title(title);
188 plot->page(0).title(label());
189 plot->page(0).columns(ncol);
190 for ( i=0; i<ncol; i++ ) plot->page(0).column(i).number(i);
191 plot->page(0).column(0).label("Spatial Frequency (1/A)");
192 plot->page(0).column(0).axis(1);
193
194 if ( z > 1 ) plot->page(0).column(1).label("FSC");
195 else plot->page(0).column(1).label("FRC");
196 plot->page(0).column(1).type(2);
197 plot->page(0).column(1).axis(3);
198 plot->page(0).column(1).color(1,0,0);
199
200 plot->page(0).axis(1).min(0);
201 plot->page(0).axis(1).max(1/hi_res);
202 plot->page(0).axis(1).inc(0.1/hi_res);
203 plot->page(0).axis(1).flags(1);
204 plot->page(0).axis(1).label("Resolution (A)");
205 plot->page(0).axis(3).min(0);
206 plot->page(0).axis(3).max(1);
207 plot->page(0).axis(3).inc(0.1);
208 if ( z > 1 ) plot->page(0).axis(3).label("FSC");
209 else plot->page(0).axis(3).label("FRC");
210
211 if ( ncol > 2 ) {
212 plot->page(0).column(2).label("DPR");
213 plot->page(0).column(2).type(2);
214 plot->page(0).column(2).axis(4);
215 plot->page(0).column(2).color(0,0,1);
216
217 plot->page(0).axis(4).min(0);
218 plot->page(0).axis(4).max(120);
219 plot->page(0).axis(4).inc(10);
220 plot->page(0).axis(4).label("DPR");
221 }
222
223 if ( verbose & VERB_DEBUG )
224 cout << "DEBUG Bimage::fsc_dpr: transferring data, maxrad=" << maxrad << endl;
225
226 (*plot)[0] = 0;
227 (*plot)[maxrad] = 1;
228 for ( i=1; i<maxrad; i++ ) {
229// cout << i << tab << F1[i] << tab << F2[i] << tab << FSC[i] << tab << endl;
230 if ( F1[i]*F2[i] > SMALLFLOAT ) FSC[i] /= sqrt(F1[i]*F2[i]);
231// if ( FSC[i] < SMALLFLOAT ) FSC[i] = 0;
232 if ( FSC[i] < -1 ) FSC[i] = -1;
233 if ( FSC[i] > 1 ) FSC[i] = 1;
234 if ( Fsum[i] > SMALLFLOAT ) DPR[i] = sqrt(DPR[i]/Fsum[i]);
235 else DPR[i] = 0;
236 (*plot)[i] = i/rad_scale;
237 (*plot)[maxrad+i] = FSC[i];
238 if ( ncol > 2 ) (*plot)[2*maxrad+i] = DPR[i]*180.0/M_PI;
239 }
240
241 if ( verbose & VERB_DEBUG )
242 cout << "DEBUG Bimage::fsc_dpr: Plot done" << endl;
243
244 return plot;
245}

422Bimage* Bimage::fsc_shell(Bimage* p, double hi_res, double* cutoff,
423 int thickness, int step, int minrad, int maxrad,

425{
426 long i, j, r;
427 double r1, r2;
428 int fill_type(FILL_AVERAGE); // Fill type for resizing
429 double sampling_ratio(1); // Sampling ratio in reciprocal space
430
431 Bimage* cp1 = NULL;
432 Bimage* cp2 = NULL;
433 Bplot* plot;
434 long ft_size(x), nmap(0), imgsize(image_size());
435 double res_est; // Resolution estimate
436
437 if ( pad > 0 ) {
438 pad = x*(pad+1);
439 ft_size = pad;
440 }
441
442 if ( minrad < 0 ) minrad = 0;
443 if ( maxrad > x/2 ) maxrad = x/2;
444 if ( maxrad < 1 ) maxrad = x/2;

455 cout << "Smoothing: " << smooth << endl;

459 cout << endl;

510 }

512 cout << r << tab << res_est << endl;

514 }
515
516 fft_destroy_plan(plan);
517

546 cp2->hanning_taper(cp2->average());

553 cp1->pad(pad, fill_type, cp1->average());
554 cp2->pad(pad, fill_type, cp2->average());
555 }
556
557 Bimage* pc = cp1->resolution_prepare(cp2, plan);
558 delete cp1;
559 delete cp2;
560
561 Bplot* plot = pc->fsc_dpr(hi_res, sampling_ratio);
562 delete pc;
563
564 double* res_est = new double[4];
565 for ( long j=0; j<4; ++j ) {
566 res_est[j] = 0;
567 if ( cutoff[j] ) {
568 res_est[j] = 1/plot->cut(1, cutoff[j], -1);
569 if ( res_est[j] < hi_res ) res_est[j] = hi_res;
570 if ( res_est[j] > LORESLIM ) res_est[j] = LORESLIM;

572 }
573
574 delete plot;
575
576 return res_est;

794 fft_plan planf, fft_plan planb, long i)

803 Vector3<long> boxsize(size,size,size); // Resolution box
804 Vector3<long> boxhalf(boxsize/2); // Half resolution box
