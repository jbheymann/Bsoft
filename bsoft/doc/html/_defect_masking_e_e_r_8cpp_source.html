36void CreateDefectNeighborInfoMap(const CameraDefects& def, const float* gainImage, CameraDefectNeighborInformation& camDefectNeighborInfo)
37{
38 camDefectNeighborInfo.gainImage = gainImage; // copy the pointer, simply/
39 for (auto it = def.vLineDefects.begin(); it != def.vLineDefects.end(); ++it)
40 {
41 int xStart = it->begin;
42 int xStop = it->end;
43
44 defectNeighborInfo_t defLen = (defectNeighborInfo_t)(xStop - xStart);
45 for (int y = 0; y<4096; ++y)
46 {
47 for (int xx=xStart; xx<=xStop;++xx)
48 {
49 int idx = (y)*4096 + xx;
50 _addDefectNeighborInfo(camDefectNeighborInfo, idx, 1); // 1 is used to mark a defect itself.
51 }
52 int ofs = y*4096;
53 if (xStart>0)
54 {
55 _addDefectNeighborInfo(camDefectNeighborInfo,
56 ofs + xStart - 1, bitMask_Defect1DInterpol | bitMask_DefectRight | defLen);
57 }
58 if (xStop<4095)
59 {
60 _addDefectNeighborInfo(camDefectNeighborInfo,
61 ofs + xStop + 1, bitMask_Defect1DInterpol | bitMask_DefectLeft | defLen);
62 }
63 }
64 }
65 // dont change order! first vlime, then Hline!
66 for (auto it = def.hLineDefects.begin(); it != def.hLineDefects.end(); ++it)
67 {
68 int yStart = it->begin;
69 int yStop = it->end;
70
71 defectNeighborInfo_t defLen = (defectNeighborInfo_t)(yStop - yStart);
72 for (int x = 0; x<4096; ++x)
73 {
74 for (int yy=yStart; yy<=yStop;++yy)
75 {
76 int idx = (yy)*4096 + x;
77 _addDefectNeighborInfo(camDefectNeighborInfo, idx, 1); // 1 is used to mark a defect itself.
78 }
79 if (yStart>0)
80 {
81 int idx = (yStart-1)*4096 + x;
82 defectNeighborInfo_t spec =
83 ((camDefectNeighborInfo.neighborSpec[idx] & bitMask_DefectNeighborTypeMask) && yStart>1)?
84 bitMask_DefectCorner : bitMask_Defect1DInterpol;
85 _addDefectNeighborInfo(camDefectNeighborInfo,
86 idx, spec | bitMask_DefectDown | defLen);
87
88 }
89 if (yStop<4095)
90 {
91 int idx = (yStop+1)*4096 + x;
92 defectNeighborInfo_t spec =
93 ((camDefectNeighborInfo.neighborSpec[idx] & bitMask_DefectNeighborTypeMask) && yStop<4094)?
94 bitMask_DefectCorner : bitMask_Defect1DInterpol;
95 _addDefectNeighborInfo(camDefectNeighborInfo,
96 idx, spec | bitMask_DefectUp | defLen);
97 }
98 }
99 }
100 for (auto it = def.pixelDefects.begin(); it != def.pixelDefects.end(); ++it)
101 {
102 int x = it->x;
103 int y = it->y;
104 // std::cout << "X: "<< x << " - Y: " << y << std::endl;
105
106 const defectNeighborInfo_t defLen = 0; //always for pixels. prep for areas.
107 _addDefectNeighborInfo(camDefectNeighborInfo, y*4096 + x, 1);
108 if (y > 0)
109 _addDefectNeighborInfo(camDefectNeighborInfo, (y-1)*4096 + x, bitMask_Defect2DInterpol | bitMask_DefectDown | defLen);
110
111 if (x > 0)
112 _addDefectNeighborInfo(camDefectNeighborInfo, y*4096 + x-1, bitMask_Defect2DInterpol | bitMask_DefectRight | defLen);
113
114 if (x < 4095)
115 _addDefectNeighborInfo(camDefectNeighborInfo, y*4096 + x+1, bitMask_Defect2DInterpol | bitMask_DefectLeft | defLen);
116
117 if (y < 4095)
118 _addDefectNeighborInfo(camDefectNeighborInfo, (y+1)*4096 + x, bitMask_Defect2DInterpol | bitMask_DefectUp | defLen);
119 }
120 for (auto it = def.areaDefects.begin(); it != def.areaDefects.end(); ++it)
121 {
122 defectNeighborInfo_t lx = (it->endX - it->beginX);
123 defectNeighborInfo_t ly = (it->endY - it->beginY);
124
125 for (int yy=it->beginY; yy<=it->endY;++yy)
126 for (int xx=it->beginX; xx<=it->endX;++xx)
127 _addDefectNeighborInfo(camDefectNeighborInfo, yy*4096 + xx, 1);
128
129 if ((it->beginY) > 0)
130 {
131 int y = it->beginY-1;
132 for (int x=it->beginX; x<=it->endX; ++x)
133 _addDefectNeighborInfo(camDefectNeighborInfo,
134 y*4096 + x, bitMask_Defect2DInterpol | bitMask_DefectDown | ly);
135 }
136 if ((it->endY) < 4095-1)
137 {
138 int y = it->endY+1;
139 for (int x=it->beginX; x<=it->endX; ++x)
140 _addDefectNeighborInfo(camDefectNeighborInfo,
141 y*4096 + x, bitMask_Defect2DInterpol | bitMask_DefectUp | ly);
142 }
143 if ((it->beginX) > 0)
144 {
145 int x = it->beginX-1;
146 for (int y=it->beginY; y<=it->endY; ++y)
147 _addDefectNeighborInfo(camDefectNeighborInfo,

168unsigned DefectElectronAdder::execute(ElectronPos* pListPtr, unsigned nElect, const CameraDefectNeighborInformation& camDefectNeighborInfo)
169{
170 defectCounts.clear();
171 // visit all electrons and add to the map if they are direct neighbors
172 for (unsigned i=0; i<nElect;++i)
173 {
174 uint32_t eOfs = ((pListPtr[i].y >> nSubPixBits) << 12) | (pListPtr[i].x >> nSubPixBits);
175 uint32_t eOfsGain = ((pListPtr[i].y / gainImageSizeFactor) * gainImageSize) | (pListPtr[i].x /gainImageSizeFactor);
176 defectNeighborInfo_t di = camDefectNeighborInfo.neighborSpec[eOfs];
177 if (di==1)
178 {
179 // electron at defect!! make it harmless
180 pListPtr[i].x = 0xffff;
181 pListPtr[i].y = 0xffff;
182 }
183 if (di & bitMask_DefectNeighborTypeMask)
184 {
185 uint16_t subPix = distSubPix(generator);
186 pListPtr[i].x = (pListPtr[i].x & 0xfffc) | (subPix&3);
187 pListPtr[i].y = (pListPtr[i].y & 0xfffc) | (subPix>>2);
188 float pcGain = camDefectNeighborInfo.gainImage[eOfsGain];
189 //std::cout<<"pcGain"<<pcGain<<std::endl;
190 defectNeighborInfo_t defDist = (di & bitMask_Distance)+1;
191 int32_t defOfs = (di & bitMask_DefectUpDown)? 4096:1;
192 defOfs = (di&bitMask_DefectBack)? -defOfs : +defOfs;
193 uint8_t mult = ((di & bitMask_DefectNeighborTypeMask) == bitMask_Defect2DInterpol)? 2 : 1;
194 for (uint8_t k=1; k<=defDist;++k)
195 {
196 defectCounts[eOfs + k*defOfs] += pcGain * ((float)((defDist-k+1)))/(mult*(defDist+1)); // unordered_map creates zero-initialized element on-the-fly if not there yet. so convenient, yet confusing
197 }
198 if ((di & bitMask_DefectNeighborTypeMask) == bitMask_DefectCorner && eOfs >= defOfs && eOfs < 4096*4096+defOfs)
199 {
200 defectNeighborInfo_t diOrt = camDefectNeighborInfo.neighborSpec[eOfs - defOfs];
201 // get the info of the horizontal defect. (by construction in CreateDefectNeighborInfoImage, the corner points now have the vertical one)
202 // note; gain should not be obtained from the nieghbor!
203 defectNeighborInfo_t defDistOrt = (diOrt & bitMask_Distance)+1;
204 int32_t defOfsOrt = (diOrt & bitMask_DefectUpDown)? 4096:1;
205 defOfsOrt = (diOrt & bitMask_DefectBack)? -defOfsOrt : +defOfsOrt;
206 for (uint8_t k=0; k<=defDist;++k)
207 {
208 for (uint8_t m=1; m<=defDistOrt;++m)
209 {
210 defectCounts[eOfs + k*defOfs + m*defOfsOrt] +=
211 pcGain * ((float)((defDist-k+1)))/(defDist+1) * ((float)((defDistOrt-m+1)))/(defDistOrt+1);
212 }
213 }
214 }
215 }
216 }
217
218 // probalistic addition of electrons.
219 unsigned plc = nElect;
220 for (const auto &pair : defectCounts)
221 {
222 uint32_t defectOfs = pair.first;
223 float electProb = pair.second;
224 if (distHit(generator) < electProb)
225 {
226 uint16_t subPix = distSubPix(generator);
227 uint16_t posX = (((defectOfs & 4095) << nSubPixBits) + (subPix>>2));
228 uint16_t posY = (((defectOfs >> 12) << nSubPixBits) + (subPix&3));
229 //if ((defectOfs >> 12)!=2142)
230 // std::cerr<<"HUH "<<posY<<"; "<<(((defectOfs >> 12) << nSubPixBits) + (subPix>>2))<<" ; "<<subPix<<std::endl;
231 pListPtr[plc++] = ElectronPos(posX, posY); // What with subpixel info? random?
232 }
233 }
234 return plc;
235}
