68Bmodel* monte_carlo_metropolis(Bmodel* model, Bmodparam& md, Bimage* map,
69 double beta, double max_angle, double max_shift, long max_iter,
70 double (Efunc)(Bmodel*, Bimage*, Bmodparam&),
71 int (Tfunc)(Bmodel*, double, double, int))
72{
73 if ( !map ) {
74 cerr << "Error: No map specified!" << endl;
75 bexit(-1);
76 }
77
78 int done(0), accept, select_min(0);
79 long i, isel(0), cycle(0), naccept(0);
80 double r, E, pE, bestE, Z(0), Ze, Zi, Emax;
81 double irm = 1.0/get_rand_max();
82 Vector3<double> gloc = models_center_of_coordinates(model);
83 Bmodel* modcopy = models_copy(model);
84 Bmodel* mp = NULL;
85 Bmodel* mpold = NULL;
86
87 vector<double> Elist(max_iter+1,0);
88
89 long nmod(0);
90 for ( mp = modcopy; mp; mp = mp->next ) nmod++;
91 vector<Vector3<double>> iloc(nmod);
92 for ( i=0, mp = model; mp; mp = mp->next, ++i )
93 iloc[i] = mp->center_of_coordinates();
94
95 if ( verbose & VERB_RESULT ) {
96 cout << "Monte carlo metropolis energy minimization:" << endl;
97 cout << "Number of models: " << nmod << endl;
98 cout << "Map file: " << map->file_name() << endl;
99 cout << "Coordinate minima and maxima: " << model->minimum() << " " << model->maximum() << endl;
100 cout << "Bounding box size: " << md.max - md.min << endl;
101 cout << "Bounding box center: " << (md.max + md.min)*0.5 << endl;
102 cout << "Klink: " << md.Klink << endl;
103 cout << "Kangle: " << md.Kangle << endl;
104 cout << "Kelectrostatic: " << md.Kelec << endl;
105 cout << "Kdistance: " << md.Kdistance << endl;
106 cout << "Kmap: " << md.Kmap << endl;
107 cout << "Kseparation: " << md.Ksep << endl;
108 cout << "Kpoint: " << md.Kpoint << " (" << md.pointdecay << ")" << endl;
109 cout << "location: " << md.point << endl;

111 cout << "Cutoff distance: " << md.cutoff << endl;
112 cout << "Separation distance: " << md.sepdist << endl;
113 cout << "Maximum shift: " << max_shift << endl;
114 cout << "Maximum angular increment: " << max_angle*180.0/M_PI << endl;
115 cout << "Steps along link: " << md.linksteps << endl;
116// cout << "Number of links: " << nlink << endl;
117 if ( md.rigid < 1 ) cout << "Treating the whole ensemble as a rigid body" << endl;
118 else if ( md.rigid == 1 ) cout << "Treating each model as a rigid body" << endl;
119 else cout << "Modifying each component's coordinates" << endl;
120 cout << endl;
121 model_calculate_deviations(model, md);
122 }
123
124 pE = (Efunc)(modcopy, map, md);
125
126 bestE = E = pE;
127 Elist[0] = pE;
128
129 if ( verbose & VERB_RESULT ) {
130 display_energy_headers(md);
131 display_energy(0, md, E-pE);
132 }
133 while ( !done && ( cycle < max_iter ) ) {
134 cycle++;
135 mpold = models_copy(modcopy);
136 if ( md.rigid == 1 ) { // Rigid models
137 if ( select_min ) {
138 for ( i=0, Emax=-1e30, mp=modcopy; mp; mp=mp->next, i++ ) {
139 if ( Emax < mp->FOM() ) {
140 Emax = mp->FOM();
141 isel = i;
142 }
143 }
144 } else {
145 isel = (long) ((nmod-0.001)*irm*random());
146 }
147 for ( i=0, mp=modcopy; mp; mp=mp->next, i++ ) {

156 isel = i;
157 }

164 else mp->select(0);
165 }
166 }
167 (Tfunc)(modcopy, max_angle, max_shift, md.rigid);
168 E = (Efunc)(modcopy, map, md);
169 Elist[cycle] = E;

172 bestE = E;

174 model = models_copy(modcopy);
175 if ( verbose & VERB_RESULT )

177 }
178 accept = 1;
179 if ( E > 1e30 ) accept = 0; // High energy indicates the model might be outside the box
180 if ( E > pE ) {
181 r = random()*irm;
182 if ( exp(beta*(pE-E)) < r ) accept = 0;
183 }
184 if ( accept ) {
185 pE = E;
186 naccept++;
187 delete mpold;
188 } else {
189 delete modcopy;
190 modcopy = mpold;
191 }
192// model_test_if_within_box(modcopy, md.min, md.max);
193 }
194
195 delete modcopy;
196
197 E = (Efunc)(model, map, md);
198
199 for ( mp = model; mp; mp = mp->next, ++i )
200 mp->select(1);
201
202 for ( i=0, Z=Ze=0; i<=cycle; i++ ) {
203 Zi = exp(-beta*(Elist[i] - bestE));
204 Z += Zi;
205 if ( Elist[i] - bestE < 0.001 ) Ze += Zi;
206 }
207 if ( Z <= 0 ) Z = 1;
208
209// md.Epot = bestE;
210
211 if ( verbose & VERB_RESULT ) {
212 gloc = models_center_of_coordinates(model) - gloc;
213 cout << "Total shift: " << gloc << endl;
214 for ( i=0, mp = model; mp; mp = mp->next, ++i ) {
215 iloc[i] = mp->center_of_coordinates() - iloc[i];
216 E = model_map_energy(mp, map, md.Kmap);
217 cout << tab << mp->identifier() << tab << iloc[i] << tab << E << endl;
218 }
219 cout << "Fraction of perturbations accepted: " << naccept*1.0/cycle << endl;
220 cout << "Final energy: " << bestE << endl;
221 cout << "Partition function: " << Z/cycle << endl;
222 cout << "Probability of final orientation: " << Ze/Z << endl << endl;
223 model_calculate_deviations(model, md);
224 }
225
226 return model;

399double model_component_overlap(Bmodel* model, Bmodparam& md)
400{
401 if ( md.rigid < 1 || md.Ksep <= 0 ) {
402 md.Esep = 0;
403 return 0;
404 }
405
406 long n(0), i;
407 Vector3<double> min = md.min;
408 Vector3<double> max = md.max;
409 Vector3<double> box = max - min;
410 Vector3<double> sampling(md.sepdist, md.sepdist, md.sepdist);
411 Vector3<long> size = box/sampling + 0.001;
412 size = size.max(1);
413 sampling = box/size + 0.001;
414 long boxsize = (long) size.volume();
415 Vector3<long> coor;
416
417 Bmodel* mp;
418 Bcomponent* comp;
419
420 Bimage* p = new Bimage(Float, TSimple, size, 1);
421 p->sampling(sampling);
422 p->origin(-min/sampling);
423 float* data = (float *) p->data_pointer();
424 p->next = new Bimage(Float, TSimple, p->size(), p->images());
425 float* dataone = (float *) p->next->data_pointer();
426
427 for ( mp = model; mp; mp = mp->next ) {
428 for ( comp = mp->comp; comp; comp = comp->next ) {
429 coor = p->image->image_coordinates(comp->location());
430 if ( p->within_boundaries(coor) ) {
431 i = p->index(coor, 0);
432 dataone[i] = 1;
433 }
434 if ( md.rigid > 1 ) { // Individual components
435 for ( i=0; i<boxsize; i++ ) {
436 data[i] += dataone[i];
437 dataone[i] = 0;
438 }
439 }
440 }
441 if ( md.rigid == 1 ) { // Rigid models
442 for ( i=0; i<boxsize; i++ ) {
443 data[i] += dataone[i];
444 dataone[i] = 0;
445 }
446 }
447 }
448
449 if ( verbose & VERB_DEBUG ) {
450 Bstring filename("overlap.map");
451 write_img(filename, p, 0);
452 }
453
454 md.Esep = 0;
455
456 for ( i=n=0; i<boxsize; i++ ) {
457 if ( data[i] ) {
458 n++;
459 if ( data[i] > 1 ) md.Esep += data[i] - 1;
460 }
461 }
462
463 md.Esep *= md.Ksep * md.sepdist/n;
464
465 delete p;
466
467 return md.Esep;
468}

500 t.origin = model->center_of_coordinates();
501 model_rotate(model, t);
502 }
503 }
504
