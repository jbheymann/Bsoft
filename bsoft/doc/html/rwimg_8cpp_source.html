461int img_convert_fourier_one(Bimage* p, Complex<float>* nufdata, int n,
462 long hermx, long friedel, long oldx, long nux,
463 long xo1, long xo2, long yo1, long yo2,
464 long zo1, long zo2, int z_even, int y_even)
465{
466 Complex<short>* sdata = (Complex<short> *) p->data_pointer();
467 Complex<int>* idata = (Complex<int> *) p->data_pointer();
468 Complex<float>* fdata = (Complex<float> *) p->data_pointer();
469 Complex<double>* ddata = (Complex<double> *) p->data_pointer();
470
471 long x, y, z, nz, i, j, izi, izf, iyi, iyf, jz, jy;
472 long xf, yf, zf, xi, yi, zi;
473 int phisgn = 1;
474 Complex<double>* nuddata = (Complex<double> *) nufdata;
475 Complex<int>* nuidata = (Complex<int> *) nufdata;
476 Complex<short>* nusdata = (Complex<short> *) nufdata;
477
478 nz = n*p->sizeZ();
479 for ( z=0; z<p->sizeZ(); z++ ) {
480 zi = z + zo1 - zo2;
481 if ( zi < 0 ) zi += p->sizeZ();
482 if ( zi >= (int)p->sizeZ() ) zi -= p->sizeZ();
483 if ( z_even )
484 zf = (zi)? p->sizeZ()-zi: zi;
485 else
486 zf = 2*zo1 - zi;
487 if ( zf < 0 ) zf += p->sizeZ();
488 izi = (nz + zi)*p->sizeY();
489 izf = (nz + zf)*p->sizeY();
490 jz = (nz + z)*p->sizeY();
491 for ( y=0; y<p->sizeY(); y++ ) {
492 yi = y + yo1 - yo2;
493 if ( yi < 0 ) yi += p->sizeY();
494 if ( yi >= (int)p->sizeY() ) yi -= p->sizeY();
495 if ( y_even )
496 yf = (yi)? p->sizeY()-yi: yi;
497 else
498 yf = 2*yo1 - yi;
499 if ( yf < 0 ) yf += p->sizeY();
500 iyi = (izi + yi)*oldx;
501 iyf = (izf + yf)*oldx;
502 jy = (jz + y)*nux;
503 for ( x=0; x<nux; x++ ) { //cout << "%d %d %d\n", x, y, z);
504 xi = x + xo1 - xo2;
505 if ( xi < 0 ) xi += p->sizeX();
506 if ( xi >= (int)p->sizeX() ) xi -= p->sizeX();
507 xf = (xi)? p->sizeX()-xi: xi;
508 if ( !friedel || ( xi < (int)hermx ) ) {
509 i = iyi + xi;
510 phisgn = 1;
511 } else {
512 i = iyf + xf;
513 phisgn = -1;
514 }
515 j = jy + x;
516 if ( p->data_type() == Short ) {
517 if ( phisgn > 0 ) nusdata[j] = sdata[i];
518 else nusdata[j] = sdata[i].conj();
519 } else if ( p->data_type() == Integer ) {
520 if ( phisgn > 0 ) nuidata[j] = idata[i];
521 else nuidata[j] = idata[i].conj();
522 } else if ( p->data_type() == Float ) {
523 if ( phisgn > 0 ) nufdata[j] = fdata[i];
524 else nufdata[j] = fdata[i].conj();
525 } else if ( p->data_type() == Double ) {
526 if ( phisgn > 0 ) nuddata[j] = ddata[i];
527 else nuddata[j] = ddata[i].conj();
528 }
529 }
530 }
531 }
532
533 return 0;

555int img_convert_fourier(Bimage* p, FourierType nutransform)
556{
557 if ( !p ) return -1;
558
559 if ( !p->data_pointer() ) return -1;
560
561 if ( verbose & VERB_DEBUG ) {
562 cout << "DEBUG img_convert_fourier: old transform type = " << p->fourier_type() << endl;
563 cout << "DEBUG img_convert_fourier: new transform type = " << nutransform << endl;
564 cout << "DEBUG img_convert_fourier: input size = " << p->size() << endl;
565 }
566
567 if ( p->compound_type() != TComplex ) {
568 p->fourier_type(NoTransform);
569 return 0;
570 }
571
572 if ( p->fourier_type() == NoTransform ) p->fourier_type(Standard);
573
574 if ( nutransform == p->fourier_type() ) return 0;
575
576 if ( verbose & VERB_PROCESS ) {
577 cout << "Converting Fourier transform: ";
578 switch ( p->fourier_type() ) {
579 case Standard: cout << "Standard -> "; break;
580 case Centered: cout << "Centered -> "; break;
581 case Hermitian: cout << "Hermitian -> "; break;
582 case CentHerm: cout << "Centered hermitian -> "; break;
583 default: break;
584 }
585 switch ( nutransform ) {
586 case Standard: cout << "Standard"; break;
587 case Centered: cout << "Centered"; break;
588 case Hermitian: cout << "Hermitian"; break;
589 case CentHerm: cout << "Centered hermitian"; break;
590 default: break;
591 }
592 cout << endl << endl;
593 }
594
595 long hermx = p->sizeX()/2 + 1;
596 long friedel = 0;
597 long oldx = p->sizeX(), nux = p->sizeX();
598 if ( p->fourier_type() == Hermitian || p->fourier_type() == CentHerm ) {
599 oldx = hermx;
600 if ( nutransform == Standard || nutransform == Centered )
601 friedel = 1; // Flag => fill in friedel pairs
602 }
603 if ( nutransform == Hermitian || nutransform == CentHerm )
604 nux = hermx;
605
606 long xo1, yo1, zo1, xo2, yo2, zo2; // Origins
607 switch ( p->fourier_type() ) {
608 case Centered:
609 xo1 = p->sizeX()/2;
610 yo1 = p->sizeY()/2;
611 zo1 = p->sizeZ()/2;
612 break;
613 case CentHerm:
614 xo1 = 0;
615 yo1 = p->sizeY()/2;
616 zo1 = p->sizeZ()/2;
617 break;
618 default:
619 xo1 = yo1 = zo1 = 0;
620 break;
621 }
622
623 switch ( nutransform ) {
624 case Centered:
625 xo2 = p->sizeX()/2;
626 yo2 = p->sizeY()/2;
627 zo2 = p->sizeZ()/2;
628 break;
629 case CentHerm:
630 xo2 = 0;
631 yo2 = p->sizeY()/2;
632 zo2 = p->sizeZ()/2;
633 break;
634 default:
635 xo2 = yo2 = zo2 = 0;
636 break;
637 }
