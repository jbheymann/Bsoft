114int project_get_handedness(Bimage* mapA, Bproject* project, double* mg_ang,
115 int* mg_index, int* mg_select, double rad_min, double rad_max,
116 double res_min, double res_max, double AmB_min, double AB_min,
117 int diff_out, int origins2, Bstring outimg)
118{
119 project->fom_tag[1] = FOM_HAND_A;
120 project->fom_tag[2] = FOM_HAND_B;
121
122 Bfield* field=NULL; // Placeholder for loop over the fields-of-view in project
123 int handedness; // Output to calling function: 1=hand A, 2=hand B, 0=undetermined
124 int ifield; // index for field block
125 Bmicrograph *mg1=NULL, *mg2=NULL; // Structures to hold input data for first and second views of a micrograph tilt pair
126 int n_Global; // Number of particle-image pairs analyzed, for statistical calculations
127 Bimage *mapB=NULL; // Mirror image of input map (mapA)
128
129 Hand_Stats* Global = new Hand_Stats; // Global statistics (i.e. for all micrographs)
130 memset(Global, 0, sizeof(Hand_Stats)); // zero sums for global statistics
131
132 if ( verbose ) {
133 cout << "Minimum Radius (pixels): " << rad_min << endl;
134 cout << "Maximum Radius (pixels): " << rad_max << endl;
135 cout << "High-Pass filter (angs.): " << res_min << endl;
136 cout << "Low-Pass filter (angs.): " << res_max << endl << endl;
137 }
138
139
140 // Copy enantiomer and invert it through origin where equation is
141 // mirror(x,y,z) = enantiomer(-x,-y,-z)
142 // if origin is given as (0,0,0). mapB is hand B map.
143 mapA->calculate_background(); // first be sure background is set correctly
144 Vector3<double> scale(-1,-1,-1), translate;
145 Matrix3 mat(1);
146// mapB = img_transform_copy(mapA, mapA->size(), scale, mapA->image->origin(), translate, mat, FILL_BACKGROUND, 0);
147 mapB = mapA->transform(mapA->size(), scale, mapA->image->origin(), translate, mat, FILL_BACKGROUND, 0);
148
149 if (verbose) {
150 cout << "Hand A map origin: " << mapA->image->origin() << endl;
151 cout << "Hand B map origin: " << mapB->image->origin() << endl;
152 cout << "Map Sampling (angs./pixel): " << mapA->sampling(0) << endl << endl;
153 }
154
155
156 // Determine handedness
157 for ( field = project->field, ifield=1; field; field = field->next, ifield++ ) {
158 // select first view
159 mg1 = field_find_micrograph(field, mg_select[0], mg_index[0], mg_ang[0]);
160 // select second view
161 mg2 = field_find_micrograph(field, mg_select[1], mg_index[1], mg_ang[1]);
162 // find handedness for a field
163 field_get_handedness(mapA, mapB, mg1, mg2, Global, rad_min, rad_max,
164 res_min, res_max, AmB_min, AB_min, diff_out, origins2, outimg, ifield);
165 }
166
167
168 // Calculate and print global statistics to screen
169 hand_compute_stats(Global, diff_out);
170 if (verbose) {
171 cout << endl << endl;
172 hand_print_stats(Global, 1, diff_out);
173 }
174
175
176 // If 95% of particles agree with one handedness, handedness=1 if hand A or 2 if hand B. 0 if < 95%.
177 n_Global = Global->CountA + Global->CountB + Global->Tie;
178 if ( (Global->CountA/n_Global) >= 0.95 ) {
179 handedness = 1;
180 if (verbose) cout << "Output map has hand A (same as input map)." << endl;
181 } else if ( (Global->CountB/n_Global) >= 0.95 ) {
182 handedness = 2;
183 if (verbose) cout << "Output map has hand B (opposite of input map)." << endl;
184 } else {
185 handedness = 0;
186 if (verbose) {
187 cout << "Output map is undetermined because the number of selected" << endl;
188 cout << "particles favoring hand A or B was not 95% or greater." << endl;
189 }
190 }
191
192
193 // Memory clean-up
194 delete mapB;
195 delete Global;
196
197
198 return handedness;
199}

203/*
204@brief Determines handedness for all selected particles (sel > 0) for a
205 field--specifically a selected tilt pair of micrographs from the
206 field.
207@param *mapA 3D map (hand A)
208@param *mapB 3D map (hand B)
209@param *mg1 Micrograph for view 1 of field
210@param *mg2 Micrograph for view 2 of field
211@param *Global Statistics for all fields-of-view
212@param rad_min minimum radius for FOM calculation (pixels)
213@param rad_max maximum radius for FOM calculation (pixels)
214@param res_min minimum resolution for FOM calculation (angs.)
215@param res_max maximum resolution for FOM calculation (angs.)
216@param AmB_min |FOMA - FOMB| must be this value or greater

291 Global->Sum_AbsDiffPhi2A += Pair->Sum_AbsDiffPhi2A;
292 Global->Sum_AbsDiffThe2A += Pair->Sum_AbsDiffThe2A;
293 Global->Sum_AbsDiffPsi2A += Pair->Sum_AbsDiffPsi2A;
294 Global->Sum_AbsDiffPhi2B += Pair->Sum_AbsDiffPhi2B;

302 Global->Sum_AbsDiffPsi2Bsq += Pair->Sum_AbsDiffPsi2Bsq;
303 }
304
305
306 delete Pair;
307
308 return 0;

314/*
315@brief Determines handedness from tilt pair of images, (two views of the
316 same specimen in which the specimen was rotated about a known
317 axis and by a known angle).
318
319@param *mapA 3D map (hand A)
320@param *mapB 3D map (hand B)
321@param *mg1 Micrograph for view 1
322@param *mg2 Micrograph for view 2
323@param *particle1 Particle from view 1
324@param *particle2 Particle from view 2
325@param *Pair Statistics for current micrograph pair
326@param rad_min minimum radius for FOM calculation (pixels)
327@param rad_max maximum radius for FOM calculation (pixels)
328@param res_min minimum resolution for FOM calculation (angs.)
329@param res_max maximum resolution for FOM calculation (angs.)
330@param AmB_min |FOMA - FOMB| must be this value or greater
331@param AB_min min. acceptable value for FOM of correct hand
332@param diff_out output difference of measured & predicted orientations
333@param origins2 flag to determine origins for second view
334@param outimg prefix & suffix for output projection files
335@param fieldnum index of current field (for image output)
336@return int 0.
337
338 -Calculate two predicted orientations for second view from first
339 orientation, the tilt-axis direction, and the rotation angle. One
340 orientation is the predicted orientation for hand A (handedness
341 of input 3D map and first orientation) in the second view. The
342 other is the predicted orientation for hand B (mirror of hand A
343 map).
344 -Projections made of input map at first orientation and predicted
345 second orientation of hand A.
346 -Projection made of mirror map at predicted second orientation of
347 hand B.
348 -Projections of predicted second orientations are compared to image
349 of second view.
350 -As a control, projection of first orientation is compared to image
351 of first view.
352**/
353int get_handedness_from_tilt_pair(Bimage* mapA, Bimage* mapB, Bmicrograph* mg1, Bmicrograph* mg2,
354 Bparticle* particle1, Bparticle* particle2,
355 Hand_Stats* Pair, double rad_min, double rad_max, double res_min, double res_max, double AmB_min, double AB_min,
356 int diff_out, int origins2, Bstring outimg, int fieldnum)
357{
358 double axis_dir(mg1->tilt_axis); // Angle of the tilt axis on the micrograph
359 double rotation_angle(mg2->tilt_angle - mg1->tilt_angle); // Angle of rotation about the tilt axis = tilt_angle (view2) - tilt_angle (view1)
360
361 double max_radius; // radius of largest sphere that will fit inside the 3D map
362 double cc1A; // Correlation coefficient for first view and predicted second views
363 double cc2A, cc2B; // model projections vs. particle image
364 double diff2A_phi(0); // measured angle - predicted angle
365 double diff2A_the(0);
366 double diff2A_psi(0);
367 double diff2B_phi(0);
368 double diff2B_the(0);
369 double diff2B_psi(0);
370 int ThresholdOK(1); // flag for |A-B| threshold
371
372 Vector3<double> translate1, translate2A, translate2B; // Translational-shift arrays for particle images
373 Vector3<double> translate0; // for cross-correlation option
374
375 Euler angles1A; // Euler angles for first view
376 Euler angles2A, angles2B; // Euler angles for second view, hands A & B
377 Euler euler2; // Euler angles for second view (input data)
378 View2<double> view1A(particle1->view); // View vector and angle for first view, hand A
379 View2<double> view1B(view1A);
380 view1B.angle(angle_set_negPI_to_PI(view1B.angle()+M_PI));
381 View2<double> view2; // View vector and angle for second view (input data)
382
383 Matrix3 rot_matrix1A(1); // 3x3 matrix for first and predicted second orientations,
384 Matrix3 rot_matrix1B(1); // for hands A and B
385 Matrix3 rot_matrix2A(1);
386 Matrix3 rot_matrix2B(1);
387 Matrix3 rot_matrixtilt(1); // 3x3 matrix for the tilt axis
388
389 Bimage* img1 = NULL; // a particle image from first micrograph
390 Bimage* img2A = NULL; // a particle image from second micrograph
391 Bimage* img2B = NULL; // a copy of img2A, but it may have a different origin
392 Bimage* prj1A = NULL; // a projection image corresponding to the particle image from first micrograph
393 Bimage* prj2A = NULL; // a projection image from hand A map, corresponding to the particle image from second micrograph
394 Bimage* prj2B = NULL; // a projection image from hand B map, corresponding to the particle image from second micrograph
395
396
397 // General parameters
398 max_radius = mapA->maximum_included_radius(); // reduces unnecessary computations in rotate_project and img_find_shift
399
400
401 // Compute predicted orientations for second view (for both hand A and hand B)
402 // -The addition of Pi for the 1B orientation must be added to the psi 1A value, it cannot be added to the
403 // tilt-axis direction. This is because the tilt axis does not change--and neither does the first view.
404 // Only the psi value for the first view changes.
405 rot_matrix1A = view1A.matrix(); // rotation matrices for views 1A & 1B
406 rot_matrix1B = view1B.matrix();
407
408 rot_matrixtilt = Matrix3(rotation_angle,angle_set_negPI_to_PI(axis_dir)); // rotation matrix for tilting operation
409
410 rot_matrix2A = rot_matrix1A * rot_matrixtilt; // rotation matrices for predicted views 2A & 2B
411 rot_matrix2B = rot_matrix1B * rot_matrixtilt;
412
413 // Get orientation angles
414 angles1A = Euler(view1A);
415 angles2A = Euler(rot_matrix2A); // get predicted angles from matrices
416 angles2B = Euler(rot_matrix2B);
417
418 if ( verbose & VERB_PROCESS ) {
419 cout << "Input orientation (phi, theta, psi) = " <<
420 angles1A.phi()*(180/M_PI) << " " << angles1A.theta()*(180/M_PI) << " " << angles1A.psi()*(180/M_PI) << endl;
421 cout << "Predicted orientation 2A (phi, theta, psi) = " <<
422 angles2A.phi()*(180/M_PI) << " " << angles2A.theta()*(180/M_PI) << " " << angles2A.psi()*(180/M_PI) << endl;
423 cout << "Predicted orientation 2B (phi, theta, psi) = " <<
424 angles2B.phi()*(180/M_PI) << " " << angles2B.theta()*(180/M_PI) << " " << angles2B.psi()*(180/M_PI) << endl;
425 }
426 if ( verbose & VERB_FULL ) {
427 cout << "Matrix for first view, hand A" << endl;
428 cout << rot_matrix1A << endl;
429 cout << "Matrix for tilt" << endl;
430 cout << rot_matrixtilt << endl;
431 cout << "Matrix for second view, hand A" << endl;
432 cout << rot_matrix2A << endl;
433 cout << "Matrix for second view, hand B" << endl;
434 cout << rot_matrix2B << endl;
435 }
436
437
438 // Read images and convert to double point, set pixel size to value from micrograph structure (if different)
439 img1 = read_img(mg1->fpart,1,(particle1->id - 1));
440 img1->change_type(Float);
441 img1->sampling(mg1->pixel_size);
442 img1->origin(particle1->ori);
443
444 img2A = read_img(mg2->fpart,1,(particle2->id - 1));
445 img2A->change_type(Float);
446 img2A->sampling(mg2->pixel_size);
447 img2A->origin(particle2->ori);
448
449 img2B = img2A->copy(); // copy made for hand 2B test because origin may differ

451 Vector3<double> oriA(mapA->image->origin()[0], mapA->image->origin()[1], 0);

454 // Get translational offset (x & y only, z offset = 0) and set origins of data images
455 // -The translational offset is the origin of the particle image minus the center of the map array
456 translate1 = particle1->ori - oriA;
457
458 if (origins2 == 0) { // Get image origins (second view) from input parameter file
459 translate2A = particle2->ori - oriA;
460 translate2B = particle2->ori - oriB;
461 } else if (origins2 == 1) { // Determine origins of second view via cross-correlation with projections
462 prj2A = mapA->rotate_project(rot_matrix2A, translate0, max_radius); // project with no shifts
463 prj2B = mapB->rotate_project(rot_matrix2B, translate0, max_radius);
464
465// img_find_shift(img2A, prj2A, NULL, res_max, res_min, max_radius, 0, 1); // hand A
466 img2A->find_shift(prj2A, NULL, res_max, res_min, max_radius, 0, 1); // hand A
467 translate2A = prj2A->image->origin() - img2A->image->origin();

469
470// img_find_shift(img2B, prj2B, NULL, res_max, res_min, max_radius, 0, 1); // hand B

472 translate2B = prj2B->image->origin() - img2B->image->origin();

474 img2B->origin(translate2B + oriB);
475

477 delete prj2B;
478 } else if (origins2 == 2) { // Get image origins (second view) from Bsubimage structure

487 prj2B = mapB->rotate_project(rot_matrix2B, translate2B, max_radius);

490 // Write projections to disk, if desired
491 if ( outimg.length() ) {
492 hand_write_img(prj1A, "prj1a", outimg, fieldnum, particle1->id);
493 hand_write_img(prj2A, "prj2a", outimg, fieldnum, particle1->id);
494 hand_write_img(prj2B, "prj2b", outimg, fieldnum, particle1->id);
495 }

497
498 // Projections have same origin and pixel size as images
499// prj1A->image->origin(img1->image->origin());
500// prj2A->image->origin(img2A->image->origin());
501// prj2B->image->origin(img2B->image->origin());
502 prj1A->origin(0, img1->image->origin());
503 prj2A->origin(0, img2A->image->origin());
504 prj2B->origin(0, img2B->image->origin());
505
506 prj1A->sampling(img1->sampling(0));
507 prj2A->sampling(img2A->sampling(0));
508 prj2B->sampling(img2B->sampling(0));

512 hand_prepare_image(img1, res_min, res_max);
513 hand_prepare_image(img2A,res_min, res_max);
514 hand_prepare_image(img2B,res_min, res_max);

516 hand_prepare_image(prj2A,res_min, res_max);

522 hand_write_img(prj1A, "prj1a", outimg, fieldnum, particle1->id);
523 hand_write_img(prj2A, "prj2a", outimg, fieldnum, particle1->id);
524 hand_write_img(prj2B, "prj2b", outimg, fieldnum, particle1->id);
525 hand_write_img(img1, "img1", outimg, fieldnum, particle1->id);

530 cout << "General parameters" << endl;

686/*
687@brief Computes statistics from data collected in Hand_Stats structure.
688@param *stats Handedness statistical parameters
689@param diff_out Switch for measured-predicted angle output
690@return int 0.
691
692 Computes average and standard deviation for view1 projection vs.
693 image1, handA-view2 projection vs. image2, and handB-view2 projection
694 vs. image2 correlation coefficients. Also computes these parameters
695 for differences between measured and predicted angles, if user
696 desires.
697**/
698int hand_compute_stats(Hand_Stats* stats, int diff_out)
699{
700 int N; // Number of data points
701 double d;
702
703 N = stats->CountA + stats->CountB + stats->Tie; // total number of selected particles
704 if (N == 0) return -1; // no data = no statistics
705
706 // Averages
707 stats->Avg1A = stats->Sum_1A/N;
708 stats->Avg2A = stats->Sum_2A/N;
709 stats->Avg2B = stats->Sum_2B/N;
710
711 if (diff_out > 0) { // Absolute differences: measured angles - predicted angles
712 stats->AvgAbsDiff_2Aphi = stats->Sum_AbsDiffPhi2A/N;
713 stats->AvgAbsDiff_2Athe = stats->Sum_AbsDiffThe2A/N;
714 stats->AvgAbsDiff_2Apsi = stats->Sum_AbsDiffPsi2A/N;
715 stats->AvgAbsDiff_2Bphi = stats->Sum_AbsDiffPhi2B/N;
716 stats->AvgAbsDiff_2Bthe = stats->Sum_AbsDiffThe2B/N;
717 stats->AvgAbsDiff_2Bpsi = stats->Sum_AbsDiffPsi2B/N;
718 }
719
720 // Standard deviations
721 if (N < 2) { // prevents divide-by-zero errors if only one data point
722 stats->Std1A = 0; stats->Std2A = 0; stats->Std2B = 0;
723 stats->StdAbsDiff_2Aphi = 0; stats->StdAbsDiff_2Athe = 0; stats->StdAbsDiff_2Apsi = 0;
724 stats->StdAbsDiff_2Bphi = 0; stats->StdAbsDiff_2Bthe = 0; stats->StdAbsDiff_2Bpsi = 0;
725 } else {
726 stats->Std1A = sqrt( (stats->Sum_1Asq - (stats->Sum_1A*stats->Sum_1A)/N) / (N - 1) );
727 stats->Std2A = sqrt( (stats->Sum_2Asq - (stats->Sum_2A*stats->Sum_2A)/N) / (N - 1) );
728 stats->Std2B = sqrt( (stats->Sum_2Bsq - (stats->Sum_2B*stats->Sum_2B)/N) / (N - 1) );
729 if (diff_out > 0) {
730 d = stats->Sum_AbsDiffPhi2Asq - (stats->Sum_AbsDiffPhi2A*stats->Sum_AbsDiffPhi2A)/N;
731 if ( d > 0 ) stats->StdAbsDiff_2Aphi = sqrt( d / (N-1) );
732 else stats->StdAbsDiff_2Aphi = 0;
733 d = stats->Sum_AbsDiffPhi2Asq - (stats->Sum_AbsDiffThe2Asq*stats->Sum_AbsDiffThe2A)/N;
734 if ( d > 0 ) stats->StdAbsDiff_2Athe = sqrt( d / (N-1) );
735 else stats->StdAbsDiff_2Athe = 0;
736 d = stats->Sum_AbsDiffPsi2Asq - (stats->Sum_AbsDiffPsi2A*stats->Sum_AbsDiffPsi2A)/N;
737 if ( d > 0 ) stats->StdAbsDiff_2Apsi = sqrt( d / (N-1) );
738 else stats->StdAbsDiff_2Apsi = 0;
739 d = stats->Sum_AbsDiffPhi2Bsq - (stats->Sum_AbsDiffPhi2B*stats->Sum_AbsDiffPhi2B)/N;
740 if ( d > 0 ) stats->StdAbsDiff_2Bphi = sqrt( d / (N-1) );
741 else stats->StdAbsDiff_2Bphi = 0;
742 d = stats->Sum_AbsDiffThe2Bsq - (stats->Sum_AbsDiffThe2B*stats->Sum_AbsDiffThe2B)/N;
743 if ( d > 0 ) stats->StdAbsDiff_2Bthe = sqrt( d / (N-1) );
744 else stats->StdAbsDiff_2Bthe = 0;
745 d = stats->Sum_AbsDiffPsi2Bsq - (stats->Sum_AbsDiffPsi2B*stats->Sum_AbsDiffPsi2B)/N;
746 if ( d > 0 ) stats->StdAbsDiff_2Bpsi = sqrt( d / (N-1) );
747 else stats->StdAbsDiff_2Bpsi = 0;
748 }
749 }
750
751 return 0;
752}
