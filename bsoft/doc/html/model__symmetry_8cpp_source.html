247 cmat = mat * cmat;
248 comp->view(View2<float>(cmat));

333 Bsymmetry sym(symmetry_string);
334
335 int nunits(sym.order());
336 if ( nunits < 2 ) return 0;
337
338 if ( verbose & VERB_PROCESS ) {

342 cout << endl << "Symmetrizing " << sym.label() << endl << endl;
343
344 long ncomp(0), nasu;
345 double tol(1e-4), d, dmin, R(0);
346 Vector3<double> loc;
347 View2<double> view, view_asu;
348
349 Bmodel* mp;
350 Bcomponent* comp;
351 Bcomponent* compasu;
352 Bcomponent* compsel = NULL;
353
354 // Select all components in the ASU
355 for ( nasu=0, mp = model; mp; mp = mp->next ) {
356 for ( comp = mp->comp; comp; comp = comp->next ) {

363 if ( view.distance(view_asu) < tol ) {
364 comp->location(comp->force() * comp->location().length());
365 comp->select(1);
366 nasu++;
367 }
368 }
369 }

432int models_generate_lattice(Bmodel* model, UnitCell uc, Vector3<long> lattice)
433{
434 if ( lattice.volume() < 2 ) return 0;
435
436 int i, x, y, z, nmodel(0);
437 Vector3<double> d;
438 Matrix3 mat = uc.skew_matrix_inverse();
439 Bmodel* mp = model;
440 Bmodel* nu_mp = mp;
441 Bcomponent *comp;
442
443 for ( nmodel=0, mp = model; mp; mp = mp->next ) {
444 nmodel++;
445 nu_mp = mp;
446 }
447
448 if ( verbose ) {
449 cout << "Generating new unit cells for " << nmodel << " models:" << endl;
450 cout << "Lattice: " << lattice << " = " << (long)lattice.volume() << endl;
451 cout << mat << endl;
452 }
453
454 for ( z=0; z<lattice[2]; z++ ) {
455 for ( y=0; y<lattice[1]; y++ ) {
456 for ( x=0; x<lattice[0]; x++ ) {
457 d = Vector3<double>(x,y,z);
458 d = mat * d;
459 if ( verbose & VERB_FULL )
460 cout << "Generating unit cell: " << x << " " << y << " " << z << tab << d << endl;
461 if ( x+y+z > 0 ) for ( i=0, mp = model; i<nmodel; mp = mp->next, i++ ) {
462 nu_mp->next = mp->copy();
463 nu_mp = nu_mp->next;
464 for ( comp = nu_mp->comp; comp; comp = comp->next )
465 comp->location() = comp->location() + d;
466 }
467 }
468 }
469 }
470
471 return 0;
