454double material_scattering_average(map<string,Bmaterial>& material, CTFparam& cp, double area, double thickness)
455{
456 if ( area < 1) {
457 cerr << "Error: The area must be specified!" << endl;
458 bexit(-1);
459 }
460
461 double scat_avg(0);
462 double scut(cp.frequency_cutoff());
463
464 for ( auto m: material ) {
465 Bmaterial& m1 = m.second;
466 scat_avg += m1.elastic_cross_section(cp.volt(), 0, scut);
467 if ( cp.slit_width() < 0.01 )
468 scat_avg += m1.inelastic_cross_section(cp.volt(), 0, scut);
469 }
470
471 return scat_avg/area;
472}

492double micrograph_signal(Bmaterial& material, Bmaterial ice, CTFparam& cp, double area, double thickness)
493{
494 if ( area < 1) {
495 cerr << "Error: The area must be specified!" << endl;
496 bexit(-1);
497 }
498
499 ice.volume(area*thickness - material.volume());
500
501 Bmaterial mat_ice = material_combine(material, ice, 1, 1);
502
503 double cs = material_cross_section(mat_ice, cp);
504
505 double csel = material_elastic_cross_section(mat_ice, cp);
506
507 double signal = (csel/area)*exp(-cs/area);
508

512double particle_signal(map<string,Bmaterial>& material, CTFparam& cp, Vector3<double> vol)
513{
514 if ( material.find("Vitreous ice") == material.end() ) {
515 cerr << "Error: Vitreous ice parameters must be specified!" << endl;
516 return 0;
517 }
518
519 double cs(0), csf(0), cse, cses(0), mu, muf, mus, muss(0), area(vol[0]*vol[1]), I, snr(0);
520
521 for ( auto m: material ) {
522 Bmaterial& m1 = m.second;
523 cs += m1.cross_section(cp.volt());
524 csf += m1.elastic_cross_section(cp.volt(), 0, cp.frequency_cutoff());
525 if ( cp.slit_width() < 0.01 )
526 csf += m1.inelastic_cross_section(cp.volt(), 0, cp.frequency_cutoff());
527 }
528 mu = cs/area;
529 muf = csf/area;
530 I = exp(muf-mu);
531
532 if ( verbose ) {
533 cout << "Calculating the signal for each material:" << endl;
534 cout << "Area: " << area << " A2" << endl;
535 cout << "Thickness: " << vol[2] << " A" << endl;
536 cout << "Intensity recorded: " << I << endl;
537 cout << "Scattering average: " << mu << endl;
538 cout << "Material\tCross(A2))\tSignal\tSNR" << endl;
539 }
540
541 for ( auto m: material ) {
542 Bmaterial& m1 = m.second;
543// cse = m1.cross_section(cp.volt(), 0, cp.frequency_cutoff(), cp.slit_width());
544 cse = m1.elastic_cross_section(cp.volt(), 0, cp.frequency_cutoff());
545// if ( cp.slit_width() < 0.01 )
546// cse += m1.inelastic_cross_section(cp.volt(), 0, cp.frequency_cutoff());
547 cses += cse;
548 mus = (cse/area) * exp(-mu);
549 snr += mus/I;
550 muss += mus;
551 if ( verbose )
552 cout << setw(15) << m1.identifier() << tab << cse << tab << mus << tab << mus/I << endl;
553 }
554
555 if ( verbose )
556 cout << setw(15) << "Total" << tab << cses << tab << muss << tab << snr << endl << endl;
557
558 return mu;
559}

608int micrograph_recorded_intensity(Bmaterial& material, Bmaterial& ice, CTFparam& cp, double area)
609{

612 double radius = pow((3*material.mass())/(4*M_PI*material.density(DA_A3)), 1.0/3.0);
613 if ( area < radius*radius ) area = 4*radius*radius;
614
615 double t_min(100*long(radius/50+1)), t_max(10*t_min);
616
617 cout << "Material mass: " << material.mass() << " Da" << endl;
618 cout << "Area: " << area << " A2" << endl;
619
620 cout << "T(A)\tEMFP\tInt\tS(mg)\tS(part)" << setprecision(5) << endl;
621 for ( t = t_min; t <= t_max; t += t_min ) {
622 emfp = micrograph_emfp(material, ice, cp, area, t);
623 intensity = exp(-t/emfp);

689double particle_snr(Bmaterial& material, double mass, double radius, double thickness, CTFparam& ctf)
690{
691 map<string,Bcomptype>& comp = material.composition();
692
693 if ( mass < 1 ) mass = material.mass();
694
695 // Calculate radius from mass, assuming a globular/spherical protein
696 if ( radius < 1 ) radius = 50;
697 radius = pow((3*mass)/(4*M_PI*material.density(DA_A3)), 1.0/3.0);
698
699 if ( thickness < 1 ) thickness = 2*radius;
700
701 double cs = elastic_cross_section_integrated(comp, ctf);
702 double ics = inelastic_cross_section_langmore(comp, ctf);
703 double emfp = effective_mean_free_path(material, ctf);
704 double snre = 0.5*cs/(M_PI*radius*radius);
705
706 if ( verbose ) {
707 cout << "Particle:" << endl;
708 cout << "Molecular weight: " << mass << " Da" << endl;
709 cout << "Particle radius: " << radius << " A" << endl;
710 cout << "Density: " << material.density(G_CM3) << " g/cm3" << endl;
711 cout << "Density: " << material.density(DA_A3) << " Da/A3" << endl;
712 cout << "Density: " << material.density(NUM_A3) << " #/A3" << endl;
713 cout << "Occupancy: " << mass/(M_PI*radius*radius*thickness) << " Da/A3" << endl;
714 cout << "Elastic cross section: " << cs << " A2" << endl;
715 cout << "Inelastic cross section: " << ics << " A2" << endl;
716 cout << "Particle SNR per electron: " << snre << " /e" << endl;
717 cout << "Effective mean free path: " << emfp << " A" << endl;
718 cout << endl;
719 }
720
721 return snre;
