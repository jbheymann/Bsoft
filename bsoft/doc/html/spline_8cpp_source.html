15Vector3<double> vector3_catmull_rom_interpolate(Vector3<double>* pnt, double t)
16{
17 double s = 1.0 - t;
18 double t2 = t * t;
19 double t3 = t2 * t;
20 double c0 = -t*s*s;
21 double c1 = 2 - 5*t2 + 3*t3;
22 double c2 = t * ( 1 + 4*t - 3*t2 );
23 double c3 = t2*s;
24 Vector3<double> intpnt;
25
26 intpnt[0] = 0.5 * ( c0*pnt[0][0] + c1*pnt[1][0] + c2*pnt[2][0] - c3*pnt[3][0] );

111vector<double> thin_plate_splines(vector< Vector3<double> > points, double lambda)
112{
113 long np(points.size());

115
116 if ( points.size() < 3 )
117 return w;
118
119 long i, j;
120 double a(0);

122
123 for ( i=0; i<np; ++i ) {
124 for ( j=i+1; j<np; ++j ) {
125 Vector3<double> pt_i = points[i];
126 Vector3<double> pt_j = points[j];
127 pt_i[2] = pt_j[2] = 0;
128 double elen = (pt_i - pt_j).length();
129 L[i][j] = L[j][i] = tps_base_func(elen);
