152int img_fix_powerspectrum_cross(Bimage* p, int flag, double fixtol)
153{
154 long i, xx, yy, x(p->sizeX()), y(p->sizeY()), hx(x/2), hy(y/2), npx(0);
155 double a, r(fixtol/2);
156
157 if ( verbose )
158 cout << "Fixing the powerspectrum cross with tolerance " << fixtol << endl;
159
160 if ( flag & 1 ) {
161 if ( verbose ) cout << "Direction x" << endl;
162 for ( i=hy*x, xx=0; xx<x; ++xx, ++i ) {
163 a = (*p)[i-x] + (*p)[i+x];
164 if ( (*p)[i] > r*a ) {
165 p->set(i, a/2);
166 npx++;
167 }
168 }
169 }
170
171 if ( flag & 2 ) {
172 if ( verbose ) cout << "Direction y" << endl;
173 for ( i=hx, yy=0; yy<y; ++yy, i+=x ) {
174 a = (*p)[i-1] + (*p)[i+1];
175 if ( (*p)[i] > r*a ) {
176 p->set(i, a/2);
177 npx++;
178 }
179 }
180 }
181
182 if ( verbose ) cout << "Pixels fixed: " << npx << endl << endl;
183
184 return 0;

212Vector3<double> sinc_fit(vector<Vector3<double>> coor, vector<double> value, double& R)
213{
214 R = 0;
215
216 long i;
217 double k(2), d, R1;
218 Vector3<double> ori, bori, off;
219
220 for ( ori[1]=-k; ori[1]<=k; ori[1]+=0.1 ) {
221 for ( ori[0]=-k; ori[0]<=k; ori[0]+=0.1 ) {
222 for ( i=0, R1=0; i<coor.size(); ++i ) {
223 off = coor[i] - ori;
224 d = value[i]*sinc(off[0])*sinc(off[1]);
225 R1 += d;
226 }

264double ellipse_R(Bsimplex& simp)
265{
266 long i;
267 double R(0), df;
268 vector<double>& f = simp.dependent_values();
269 vector<double>& x = simp.independent_values();
270
271 for ( i=0; i<simp.points(); i++ ) {
272 df = simp.parameter(0)*x[i]*x[i] + simp.parameter(1)*x[i]*f[i] + simp.parameter(2)*f[i]*f[i] - simp.constant(0);
273 R += df*df;
274 }
275
276 R = sqrt(R/i);
277
278 return R;
279}

281double fit_ellipse_iter(vector<Vector3<double>> v)
282{
283 long i;
284 double d2, davg(0), dstd(0);
285 vector<double> vx, vy;
286
287 for ( auto p: v ) {
288 d2 = p.length2();
289 davg += d2;
290 dstd += d2*d2;
291 vx.push_back(p[0]);
292 vy.push_back(p[1]);
293 }
294 davg /= v.size();
295 dstd = sqrt(dstd/v.size() - davg*davg);
296
297// cout << davg << tab << dstd << endl;
298
299 Bsimplex simp(1, 3, 1, vx.size(), vx, vy);
300
301 simp.constant(0, davg);
302 simp.parameter(0, 1);
303 simp.parameter(1, 0);
304 simp.parameter(2, 1);
305 simp.limits(0, 0.9, 1.1);
306 simp.limits(1, -0.1, 0.1);
307 simp.limits(2, 0.9, 1.1);
308
309 double R = simp.run(1000, 1e-6, ellipse_R, 100);
310 R /= dstd;
311
312 vector<double> c(3,0);
313 for ( i=0; i<3; ++i ) c[i] = simp.parameter(i);
314
315 double a = atan((c[2]-c[0]-sqrt((c[0]-c[2])*(c[0]-c[2])+c[1]*c[1]))/c[1]);
316
317 if ( verbose ) {
318 cout << "Constant: " << simp.constant(0) << endl;
319 cout << "Coefficients:" << endl;
320 for ( i=0; i<3; ++i ) cout << c[i] << endl;
321 cout << "Angle: " << a*180.0/M_PI << endl;
322 cout << c[1]/(2*sin(a)*cos(a)) << endl;
323 cout << "R: " << R << endl << endl;
324 }
325
326 return R;
327}
