18View2<double> view_from_spider_euler(SPIDERhead* header)
19{
20 if ( verbose & VERB_DEBUG )
21 cout << "DEBUG view_from_spider_euler: "
22 << header->gamma << "," << header->theta << "," << header->phi << endl;
23
24 View2<double> view;
25
26 view[0] = cos(header->phi)*sin(header->theta);
27 view[1] = sin(header->phi)*sin(header->theta);
28 view[2] = cos(header->theta);
29 view[3] = angle_set_negPI_to_PI(header->gamma + header->phi);

32 cout << "DEBUG view_from_spider_euler: " << view << endl;
