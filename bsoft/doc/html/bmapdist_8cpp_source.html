374int project_set_selection(Bproject* project, long n, vector<long> idx)
375{
376 long i, j;
377 Bimage* p = NULL;
378
379 if ( project->class_avg && project->class_avg->fpart.length() )
380 p = read_img(project->class_avg->fpart, 1, -1);
381 else {
382 cerr << "Error: No class averages!" << endl;
383 return -1;
384 }
385
386 Bstring filename(project->class_avg->fpart);
387 filename = filename.pre_rev('.') + "_cluster." + filename.post_rev('.');
388
389 if ( verbose ) {
390 cout << "Replacing class averages:" << endl;
391 cout << "Old number: " << p->images() << endl;
392 cout << "New number: " << n << endl;
393 cout << "New file name: " << filename << endl;
394 }
395
396 Bfield* field = project->field;
397 Bmicrograph* mg = NULL;
398 Bparticle* part = NULL;
399
400 for ( field = project->field; field; field = field->next )
401 for ( mg = field->mg; mg; mg = mg->next )
402 for ( part = mg->part; part; part = part->next )
403 if ( part->sel > 0 )
404 part->sel = idx[part->sel-1] + 1;
405
406 Bimage* pnu = p->copy_header(n);
407 pnu->data_alloc_and_clear();
408 Bimage* p1;
409 Bparticle* partlist = NULL;
410 Bparticle* nupart = NULL;
411
412 for ( i=j=0, part = project->class_avg; i < p->images() && part; ++i, part = part->next ) {
413 if ( i == idx[i] ) {
414 nupart = particle_copy(&partlist, part);
415 nupart->fpart = filename;
416 nupart->id = ++j;
417 nupart->group = i;
418 nupart->sel = 0;
419 }
420 }
421
422 for ( i=0, part = project->class_avg; i < p->images(); ++i, part = part->next ) {
423 for ( j=0, nupart = partlist; nupart; ++j, nupart = nupart->next )
424 if ( nupart->group == idx[i] ) break;
425 nupart->sel += part->sel;
426 p1 = p->extract(i);
427 pnu->add(j, p1);
428 delete p1;
429 }
430
431 pnu->statistics();
432
433 for ( j=0; j<pnu->images(); ++j )
434 pnu->rescale_to_avg_std(j, 0, 1);
435
436 particle_kill(project->class_avg);
437 project->class_avg = partlist;
438
439 write_img(filename, pnu, 0);
440
441 delete pnu;
442 delete p;
443
444 return 0;
445}
