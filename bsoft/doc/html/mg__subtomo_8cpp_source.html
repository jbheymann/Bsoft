27// It creates two temporary images the size of the input image - usually a tomogram
28// A rotated template
29// The cross correlation map
30Bparticle* img_search_view(Bimage* p, Bimage* ptemp, Bimage* pmask, View2<double> view,
31 double hires, double lores, double shiftlimit, double mindist,
32 double threshold, int maxhits, int refinepeaks,
33 fft_plan planf, fft_plan planb)
34{
35 // View corresponds to a rotation matrix that takes the particle to the standard orientation
36 // need to transpose this matrix to rotate the template from the standard orientation on the particle
37 Matrix3 mat = view.matrix();
38 mat = mat.transpose();
39
40 // Transform the template to the box with size of the reconstruction
41 Bimage* ptemprot = ptemp->rotate(p->size(), mat);
42
43 // Correlate the template and the map
44 Bimage* pcc = ptemprot->cross_correlate(p, hires, lores, pmask, planf, planb);
45
46 delete ptemprot;
47
48 Bparticle* goodpeaks = img_find_refine_peaks(pcc, view, shiftlimit, 0, 0,
49 mindist, threshold, maxhits, refinepeaks);
50
51 delete pcc;
52
53 return goodpeaks;
54}

56// It creates three temporary images the size of the template image
57// A cropped input image
58// An optional real space mask to apply to the cropped input image - deleted before generating more images
59// A rotated template
60// The cross correlation map
61Bparticle* img_refine_view(Bimage* pcrop, Bimage* ptemp, Bimage* pmask, Bimage* pmask2,
62 View2<double> view, double hires, double lores, double shiftlimit,
63 double shiftlimitz, double shiftlimitxy, double mindist, int refinepeaks,
64 fft_plan planf, fft_plan planb)
65{
66// Vector3<double> scale(1,1,1), trans;
67
68 // View corresponds to a rotation matrix that takes the particle to the standard orientation
69 // need to transpose this matrix to rotate the template from the standard orientation on the particle
70 Matrix3 mat = view.matrix();
71 mat = mat.transpose();
72
73 Bimage* pcropmasked = pcrop->copy();
74
75 // Rotate the real space mask to the current view
76 if ( pmask2 ) {
77 Bimage* pmask2rot = pmask2->rotate(pmask2->size(), mat);
78 pcropmasked->multiply(pmask2rot);
79 delete pmask2rot;
80 }
81
82 // Rotate the template to the current view
83 Bimage* ptemprot = ptemp->rotate(ptemp->size(), mat);
84
85 // Correlate the template and the map
86 Bimage* pcc = ptemprot->cross_correlate(pcropmasked, hires, lores, pmask, planf, planb);
87
88 delete ptemprot;
89 delete pcropmasked;
90
91 Bparticle* peak = NULL;
92 long i(0);
93
94 // If no shift allowed, just return the CCC at the origin
95 if ( shiftlimit == 0 && shiftlimitz == 0 && shiftlimitxy == 0) {
96 peak = particle_add(&peak, 1);
97 peak->view2(view);
98 peak->fom[0] = (*pcc)[i];
99 }
100 // Shifts allowed so the hit is looked around origin, with the radius of shiftlimit, or within a cylinder
101 else {
102 peak = img_find_refine_peaks(pcc, view,
103 shiftlimit, shiftlimitz, shiftlimitxy,
104 mindist, -1, -1, refinepeaks);
105 }
106
107 delete pcc;
108
109 return peak;
110}

655double img_find_peak_subtomo(Bimage* p, View2<double> view, double shift,
656 double shift_along, double shift_orthogonal)
657{
658 if ( !p->data_pointer() ) return 0;
659
660 if ( shift < 0 ) shift = 1e30;
661
662 long i(0), x, y, z;
663 long xh = p->sizeX()/2, yh = p->sizeY()/2, zh = p->sizeZ()/2;
664 double x2, y2, z2, r2;
665 double value(0), max;
666 double shift2 = shift*shift;
667 Vector3<double> peak;
668 Vector3<double> origin(0,0,0);
669 Vector3<double> point;
670 Vector3<double> endpoint1;
671 Vector3<double> endpoint2;
672 Vector3<double> closestpoint;
673
674 view = view.backward();
675
676 if ( shift_along > 0 ) {
677 endpoint1 = Vector3<double>(view[0] * shift_along, view[1] * shift_along, view[2] * shift_along);
678 endpoint2 = Vector3<double>(-view[0] * shift_along, -view[1] * shift_along, -view[2] * shift_along);
679 }
680
681 max = -1e37;
682 for ( z=0; z<p->sizeZ(); z++ ) {
683 z2 = origin[2] - z;
684 if ( z2 < -zh ) z2 += p->sizeZ();
685 if ( z2 > zh ) z2 -= p->sizeZ();
686
687 for ( y=0; y<p->sizeY(); y++ ) {
688 y2 = origin[1] - y;
689 if ( y2 < -yh ) y2 += p->sizeY();
690 if ( y2 > yh ) y2 -= p->sizeY();
691
692 for ( x=0; x<p->sizeX(); x++, i++ ) {
693 x2 = origin[0] - x;
694 if ( x2 < -xh ) x2 += p->sizeX();
695 if ( x2 > xh ) x2 -= p->sizeX();
696
697 value = (*p)[i];
698
699 if ( shift_along > 0 ) {
700 point = Vector3<double>((double)x2, (double)y2, (double)z2);
701 // squared distance to the closest point on the line segment
702 r2 = closest_point_line_distance2(point, endpoint1, endpoint2);
703 }
704 else if ( shift_orthogonal > 0 ) {
705 point = Vector3<double>((double)x2, (double)y2, (double)z2);
706 // squared distance to the closest point on the disc
707 r2 = closest_point_disc_distance2(point, Vector3<double>(0,0,0), view, shift_orthogonal);
708 }
709 else {
710 // squared distance to the origin
711 r2 = x2*x2 + y2*y2 + z2*z2;
712 }
713 if ( r2 > shift2 ) value = 0;
714 if ( max < value ) {
715 if ( r2 > shift2 ) value = 0;
716 max = value;
717 peak = Vector3<double>(x,y,z);
718 }
719 }
720 }
721 }
722 p->image->origin(peak);
723 p->image->FOM(max);
724 if ( verbose & VERB_FULL )
725 cout << "Peak: " << peak << tab << max << endl;
726
727 return max;
728}

762 v[1] + t * line[1],
763 v[2] + t * line[2]);
764 }

822double closest_point_disc_distance2( Vector3<double> p, Vector3<double> q, View2<double> view, double radius )
823{
824 Vector3<double> pprime; // projection of p on to the plane
825
826 double p_to_pprime2; // squared distance from p' to the original point p
827
828 double pprime_to_q2; // squared distance from p' to the center of the disc q
829 double pprime_to_q; // distance from p' to the center of the disc q
830
831 double pprime_to_c2; // squared distance from p' to the circumference of the disc c
832 double p_to_c2; // squared distance from p to the circumference of the disc c
833
834 double radius2 = radius * radius;
835 double a, c;
836
837 c = q[0]*view[0] + q[1]*view[1] + q[2]*view[2];
838
839 a = (view[0] * p[0] + view[1] * p[1] + view[2] * p[2]) - c;
840 pprime = Vector3<double>(p[0] - a * view[0], p[1] - a * view[1], p[2] - a * view[2]);
841
842 pprime_to_q2 = (pprime-q).length2();
843 p_to_pprime2 = (p-pprime).length2();
844
845 if (pprime_to_q2 > radius2) {
846
847 pprime_to_q = (pprime-q).length();
848 pprime_to_c2 = (pprime_to_q - radius) * (pprime_to_q - radius);
849
850 p_to_c2 = pprime_to_c2 + p_to_pprime2;
851
852 return p_to_c2;
853 }
854 else {
855 return p_to_pprime2;
856 }

894Sphere locations_fit_sphere(Bparticle* part, int N, double Nstop)
895{
896 Sphere sphere;
897 long Pnpoints(0);
898 Bparticle* p = NULL;
899
900 double PXsum=0, PXsumsq=0, PXsumcube=0;
901 double PYsum=0, PYsumsq=0, PYsumcube=0;
902 double PZsum=0, PZsumsq=0, PZsumcube=0;
903
904 double PXYsum=0, PXZsum=0, PYZsum=0;
905 double PX2Ysum=0, PX2Zsum=0, PY2Xsum=0, PY2Zsum=0, PZ2Xsum=0, PZ2Ysum=0;
906
907 Pnpoints = part->count();
908 double x, y, z;
909
910 for ( p = part; p; p = p-> next ) {
911
912 x = p->loc[0];
913 y = p->loc[1];
914 z = p->loc[2];
915
916 PXsum += x;
917 PXsumsq += x * x;
918 PXsumcube += x * x * x;
919
920 PYsum += y;
921 PYsumsq += y * y;
922 PYsumcube += y * y * y;
923
924 PZsum += z;
925 PZsumsq += z * z;
926 PZsumcube += z * z * z;
927
928 PXYsum += x * y;
929 PXZsum += x * z;
930 PYZsum += y * z;
931
932 PX2Ysum += x * x * y;
933 PX2Zsum += x * x * z;
934 PY2Xsum += y * y * x;
935 PY2Zsum += y * y * z;
936 PZ2Xsum += z * z * x;
937 PZ2Ysum += z * z * y;
938 }
939
940 double Xn = PXsum/Pnpoints; //sum( X[n] )
941 double Xn2 = PXsumsq/Pnpoints; //sum( X[n]^2 )
942 double Xn3 = PXsumcube/Pnpoints; //sum( X[n]^3 )
943 double Yn = PYsum/Pnpoints; //sum( Y[n] )
944 double Yn2 = PYsumsq/Pnpoints; //sum( Y[n]^2 )
945 double Yn3 = PYsumcube/Pnpoints; //sum( Y[n]^3 )
946 double Zn = PZsum/Pnpoints; //sum( Z[n] )
947 double Zn2 = PZsumsq/Pnpoints; //sum( Z[n]^2 )
948 double Zn3 = PZsumcube/Pnpoints; //sum( Z[n]^3 )
949
950 double XY = PXYsum/Pnpoints; //sum( X[n] * Y[n] )
951 double XZ = PXZsum/Pnpoints; //sum( X[n] * Z[n] )
952 double YZ = PYZsum/Pnpoints; //sum( Y[n] * Z[n] )
953 double X2Y = PX2Ysum/Pnpoints; //sum( X[n]^2 * Y[n] )
954 double X2Z = PX2Zsum/Pnpoints; //sum( X[n]^2 * Z[n] )
955 double Y2X = PY2Xsum/Pnpoints; //sum( Y[n]^2 * X[n] )
956 double Y2Z = PY2Zsum/Pnpoints; //sum( Y[n]^2 * Z[n] )
957 double Z2X = PZ2Xsum/Pnpoints; //sum( Z[n]^2 * X[n] )
958 double Z2Y = PZ2Ysum/Pnpoints; //sum( Z[n]^2 * Y[n] )
959
960 //Reduction of multiplications
961 double F0 = Xn2 + Yn2 + Zn2;
962 double F1 = 0.5*F0;
963 double F2 = -8.0*(Xn3 + Y2X + Z2X);
964 double F3 = -8.0*(X2Y + Yn3 + Z2Y);
965 double F4 = -8.0*(X2Z + Y2Z + Zn3);
966
967 //Set initial conditions:
968 double A = Xn;
969 double B = Yn;
970 double C = Zn;
971
972 //First iteration computation:
973 double A2 = A*A;
974 double B2 = B*B;
975 double C2 = C*C;
976 double QS = A2 + B2 + C2;
977 double QB = - 2*(A*Xn + B*Yn + C*Zn);
978
979 //Set initial conditions:
980 double Rsq = F0 + QB + QS;
981
982 //First iteration computation:
983 double Q0 = 0.5*(QS - Rsq);
984 double Q1 = F1 + Q0;
985 double Q2 = 8*( QS - Rsq + QB + F0 );
986 double aA,aB,aC,nA,nB,nC,dA,dB,dC;
987
988 //Iterate N times, ignore stop condition.
989 int n(0);
990 while( n != N ){
991 n++;
992
993 //Compute denominator:
994 aA = Q2 + 16*(A2 - 2*A*Xn + Xn2);
995 aB = Q2 + 16*(B2 - 2*B*Yn + Yn2);
996 aC = Q2 + 16*(C2 - 2*C*Zn + Zn2);
997 aA = (aA == 0) ? 1.0 : aA;
998 aB = (aB == 0) ? 1.0 : aB;
999 aC = (aC == 0) ? 1.0 : aC;
1000
1001
1002 //Compute next iteration
1003 nA = A - ((F2 + 16*( B*XY + C*XZ + Xn*(-A2 - Q0) + A*(Xn2 + Q1 - C*Zn - B*Yn) ) )/aA);
1004 nB = B - ((F3 + 16*( A*XY + C*YZ + Yn*(-B2 - Q0) + B*(Yn2 + Q1 - A*Xn - C*Zn) ) )/aB);
1005 nC = C - ((F4 + 16*( A*XZ + B*YZ + Zn*(-C2 - Q0) + C*(Zn2 + Q1 - A*Xn - B*Yn) ) )/aC);
1006
1007 //Check for stop condition
1008 dA = (nA - A);
1009 dB = (nB - B);
1010 dC = (nC - C);
1011 if( (dA*dA + dB*dB + dC*dC) <= Nstop ){ break; }
1012
1013 //Compute next iteration's values
1014 A = nA;
1015 B = nB;
1016 C = nC;
1017 A2 = A*A;
1018 B2 = B*B;
1019 C2 = C*C;
1020 QS = A2 + B2 + C2;
1021 QB = - 2*(A*Xn + B*Yn + C*Zn);
1022 Rsq = F0 + QB + QS;
1023 Q0 = 0.5*(QS - Rsq);
1024 Q1 = F1 + Q0;
1025 Q2 = 8*( QS - Rsq + QB + F0 );
1026 }
1027
1028 double R = sqrt(Rsq);
1029 sphere.A = A;
1030 sphere.B = B;
1031 sphere.C = C;
1032 sphere.R = R;
1033
1034 return (sphere);
1035}
