271## @brief Selects a layer line.
272#
273# @param x
274# @param y Layer line position in canvas coordinates.

276proc layerLineSelect { x y } {
277 if ![winfo exists .whel] { return }
278 global theimg
279 global ll_selected
280 global fom_cutoff ll_width
281 global px py
282 set c [getImageCanvas $theimg]
283 set wc [getControlWindow $theimg]
284 set ll_selected -1
285 set n [$wc.image.scale get]
286 set mg_item [micrographItem $n]
287 if { [Bmg layerline $mg_item count] < 1 } { return $ll_selected }
288 set scale [$wc.scale.scale get]
289 set px [expr floor( [$c canvasx $x] / $scale ) ]
290 set py [expr floor( [$c canvasy $y] / $scale ) ]
291 set pyi [yFlip $py]
292 set ll_selected [Bmg layerline $mg_item select $px $pyi $ll_width]
293 if { $ll_selected > -1 } {
294 set fom [Bmg layerline $mg_item fom $ll_selected]
295 set order [Bmg layerline $mg_item order $ll_selected]
296 .whel.selected.index config -text "$ll_selected"
297 .whel.selected.fom config -text "$fom"
298 .whel.selected.order_entry delete 0 end
299 .whel.selected.order_entry insert 0 $order
300 layerLineDraw $ll_selected
301 layerLinePlot $ll_selected $order [.whel.radius.entry get]
302 }
303 return $ll_selected
304}

362## @brief Draws a layer line in the original image on the canvas
363#
364# @param id Layer line index.
365
366proc layerLineDraw { id } {
367 if ![winfo exists .whel] { return }
368 global theimg
369 global show_ll_indices
370 global ll_length
371 global ll_selected ll_color ll_select_color
372 if ![winfo exists .whel] { return }
373 set c [getImageCanvas $theimg]
374 set wc [getControlWindow $theimg]
375 if { $id < 0 } { return }
376 if { [llength $id] < 1 } {
377 puts "No ID specified!"
378 return
379 }
380 set n [$wc.image.scale get]
381 set mg_item [micrographItem $n]
382 set d [Bmg layerline $mg_item distance $id]
383 if { [llength $d] < 1 } {
384 puts "No distance for ID = $id!"
385 return
386 }
387 set scale [$wc.scale.scale get]
388 set d [expr $d * $scale ]
389# puts "Drawing $id at $d"
390 set ox [getOriginX $c]
391 set oy [getOriginY $c]
392 set axis [.whel.axis.entry get]
393 set cosax [expr cos($axis)]
394 set sinax [expr sin($axis)]
395 set ll_length [.whel.ll_length.entry get]
396 set x [expr $ox + $d * $cosax]
397 set y [expr $oy - $d * $sinax]
398 set len [expr $scale * $ll_length]
399 set xmin [expr $x - $len*$sinax]
400 set ymin [expr $y - $len*$cosax]
401 set xmax [expr $x + $len*$sinax]
402 set ymax [expr $y + $len*$cosax]
403# puts "$xmin $ymin $xmax $ymax"
404 set color $ll_color
405 if { $id == $ll_selected } { set color $ll_select_color }
406 $c create line $xmin $ymin $xmax $ymax -fill $color -dash . -width 1 -tags ll
407 if { $show_ll_indices } {
408 set x [expr $x + 10]
409 $c create text $x $y -text $id -fill $color -anchor w -tags ll
410 }
411}

534## @brief Plots the selected layer line in the plot canvas
535#
536# @param id Layer line index.
537# @param order Layer line order.
538# @param rad Layer line length.
539
540proc layerLinePlot { id order rad } {
541 global theimg project_item
542 global ll_plot_width ll_plot_height
543 if ![winfo exists .whel] { return }
544 Bmg set $project_item helix_radius $rad
545 if { $id < 0 } { return }
546 set wc [getControlWindow $theimg]
547 set n [$wc.image.scale get]
548 set mg_item [micrographItem $n]
549 if { [Bmg layerline $mg_item count] < 1 } { return }
550 set ll_len [.whel.ll_length.entry get]
551 set data [Bmg layerline $mg_item plot $id $ll_len]
552 set bessel [Bmg layerline $mg_item bessel $order $rad $ll_len]
553 set len [lindex $data 0]
554 if { $len < 1 } { return }
555 set max 0
556 set bmax 0
557 for {set i 1} {$i <= $len} {incr i 1} {
558 set v [lindex $data $i]
559 if { $max < $v } { set max $v }
560 set b [lindex $bessel $i]
561 if { $bmax < $b } { set bmax $b }
562 }
563# set max [expr $max/$bmax]
564# set max [Bimage get $theimg max]
565 set xoff 10
566 set yoff 10
567 set xscale [expr $ll_plot_width*1.0/$len]
568 set yscale [expr $ll_plot_height*1.0/$max]
569 set ybscale [expr $ll_plot_height*1.0/$bmax]
570 set g .whel.plot
571 $g delete llplot
572 set ll ""
573 set bl ""
574 for {set i 1} {$i <= $len} {incr i 1} {
575 set x [expr ($i-1)*$xscale + $xoff]
576 set y [expr ($max - [lindex $data $i]) * $yscale + $yoff]
577# set b [expr (1 - [lindex $bessel $i]) * $ll_plot_height + $yoff]
578 set b [expr ($bmax - [lindex $bessel $i]) * $ybscale + $yoff]
579 append ll " " $x " " $y
580 append bl " " $x " " $b
581 }
582 $g create line $ll -fill blue -smooth yes -width 1 -tags llplot
583 $g create line $bl -fill red -smooth yes -width 1 -tags llplot
584 set orix [expr $ll_plot_width/2 + $xoff]
585 set oriy [expr $ll_plot_height + 2*$yoff]
586 $g create line $orix 0 $orix $oriy \
587 -fill black -smooth yes -width 1 -tags llplot
588}
