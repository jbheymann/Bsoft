33@param n polygon order = number of vertices.

190 cerr << "Error: Polygon incomplete: " << poly->size() << "/" << n << endl;

353 if ( verbose & VERB_PROCESS )
354 cout << endl << "Checking the model:" << endl;

356 for ( ncomp=nlink=npoly=0, mp = model; mp; mp = mp->next ) {
357 for ( comp = mp->comp; comp; comp = comp->next, ncomp++ ) {
358 if ( comp->select() != valence ) {
359 if ( verbose & VERB_PROCESS )
360 cerr << comp->identifier() << ": select = " << comp->select() << endl;

362 }
363// for ( j=0; comp->link[j]; j++ ) ;

367 err++;
368 }
369 }
370 for ( link = mp->link; link; link = link->next, nlink++ ) ;
371 for ( poly = mp->poly; poly; poly = poly->next, npoly++ )
372 if ( !poly->closed() ) {
373 if ( verbose & VERB_PROCESS )
374 cerr << npoly << ": polygon not closed!" << endl;
375 err++;
376 }
377 }
378
379 if ( ncomp - nlink + npoly != 2 ) {
380 if ( verbose & VERB_PROCESS )
381 cerr << "Euler's formula incorrect: " << ncomp << " - " << nlink << " + "

383 err++;

386 if ( verbose & VERB_PROCESS ) {
387 cout << "Number of components: " << ncomp << endl;
388 cout << "Number of links: " << nlink << endl;
