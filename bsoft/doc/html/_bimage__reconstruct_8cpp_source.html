24int Bimage::fspace_2D_interpolate(Complex<float> cv, Vector3<double> m,
25 double part_weight, int interp_type)
26{
27 if ( !next ) next = new Bimage(Float, TSimple, size(), n);
28 if ( !next->next ) next->next = new Bimage(Float, TSimple, size(), n);
29 if ( !next->next->next ) next->next->next = new Bimage(Float, TSimple, size(), n);
30
31 long xx, yy, zz, j, jx, jy, jz;
32 double w(part_weight);
33 Vector3<long> coor;
34 Vector3<double> dist;
35
36 float* power = (float *) next->data_pointer();
37 float* weight = (float *) next->next->data_pointer();
38 float* weight2 = (float *) next->next->next->data_pointer();
39
40 if ( interp_type < 2 ) { // Nearest neighbour
41 coor = Vector3<long>((long) floor(m[0] + 0.5),
42 (long) floor(m[1] + 0.5),
43 (long) floor(m[2] + 0.5));
44 if ( interp_type == 1 ) { // Weighted nearest neighbour
45 dist = m - coor;
46 w *= 1 - dist.length();
47 }
48 if ( w > 0 ) {
49 j = index_wrap(coor);
50 add(j, cv * w);
51 power[j] += w*cv.power();
52 weight[j] += w;
53 weight2[j] += w*w;
54 }
55 } else if ( interp_type == 2 ) { // Trilinear interpolation
56 coor = Vector3<long>((long) floor(m[0]),
57 (long) floor(m[1]),
58 (long) floor(m[2]));
59 dist = m - coor;
60 for ( zz=0; zz<2; zz++ ) {
61 jz = coor[2] + zz;
62 if ( jz < 0 ) jz += z;
63 else if ( jz >= z ) jz -= z;
64 dist[2] = 1.0 - dist[2];
65 for ( yy=0; yy<2; yy++ ) {
66 jy = coor[1] + yy;
67 if ( jy < 0 ) jy += y;
68 else if ( jy >= y ) jy -= y;
69 dist[1] = 1.0 - dist[1];
70 for ( xx=0; xx<2; xx++ ) {
71 jx = coor[0] + xx;
72 if ( jx < 0 ) jx += x;
73 else if ( jx >= x ) jx -= x;
74 dist[0] = 1.0 - dist[0];
75 w = dist.volume() * part_weight;
76 j = index(jx, jy, jz);
77 add(j, cv * w);
78 power[j] += w*cv.power();
79 weight[j] += w;
80 weight2[j] += w*w;
81 }
82 }
83 }
84 }
85
86 return 0;
87}

111int Bimage::fspace_pack_2D(Bimage* p, Matrix3 mat, double hi_res,
112 double lo_res, Vector3<double> scale, double ewald_wavelength,
113 double part_weight, int interp_type)
114{
115 if ( hi_res < 0.1 ) hi_res = 0.1; // Limit on resolution!
116 double mins2 = (lo_res)? 1.0/lo_res: 0;
117 double maxs2 = 1.0/hi_res;
118// max_rad_sq += 1; // Add one pixel width additional interpolation to be removed later
119 mins2 *= mins2;
120 maxs2 *= maxs2;
121
122 if ( verbose & VERB_DEBUG ) {
123 cout << "DEBUG Bimage::fspace_pack_2D: sampling=" << setprecision(4) << sampling(0) << endl;
124 cout << "DEBUG Bimage::fspace_pack_2D: hi_res=" << hi_res << " maxs2=" << maxs2 << endl;
125 }
126
127 long i, xx, yy;
128 long hx = (p->x - 1)/2, hy = (p->y - 1)/2;
129 double s2, d, w;
130 double ew(ewald_wavelength/2);
131 Vector3<double> m, s;
132 Vector3<double> invscale(scale/p->real_size());
133// Vector3<double> mscale(real_size()/scale);
134 Vector3<double> mscale(real_size());
135 Bimage* p2 = p->next;
136
137// cout << setprecision(4) << p->real_size() << tab << invscale << tab << mscale << endl;
138
139 if ( verbose & VERB_FULL ) {
140 cout << "Packing an image into reciprocal space up to " << hi_res << " A resolution" << endl;
141 cout << "Ewald sphere wavelength = " << setprecision(4) << ewald_wavelength << endl;
142 if ( p2 ) cout << "Correcting with a complex CTF" << endl;
143 }
144
145 if ( verbose & VERB_DEBUG )
146 cout << setprecision(4) << mat << endl;
147
148 for ( yy=i=0; yy<p->y; ++yy ) {
149 s[1] = yy;
150 if ( yy > hy ) s[1] -= (double)p->y;
151 s[1] *= invscale[1];
152 for ( xx=0; xx<p->x; ++xx, ++i ) {
153 s[0] = xx;
154 if ( xx > hx ) s[0] -= (double)p->x;
155 s[0] *= invscale[0];
156 s2 = s[0]*s[0] + s[1]*s[1];
157 if ( s2 >= mins2 && s2 <= maxs2 ) {
158 w = part_weight;
159 d = (maxs2 - s2)*mscale[2];
160 if ( d < 1 ) w *= sqrt(d);
161 // Ewald sphere offset: s[2] = ±(lambda/2) * s2
162 if ( ew ) s[2] = ew * s2;
163 m = mat * s;
164 m *= mscale;
165 fspace_2D_interpolate(p->complex(i), m, w, interp_type);
166 if ( ew ) {
167 s[2] = -s[2];
168 m = mat * s;
169 m *= mscale;
170 if ( p2 ) fspace_2D_interpolate(p2->complex(i), m, w, interp_type);
171 else fspace_2D_interpolate(p->complex(i), m, w, interp_type);
172 }
173 }
174 }
175 }
176
177 return 0;
178}

389int Bimage::fspace_pack_3D(Bimage* p, double hi_res, double threshold)
390{
391 if ( !next ) next = new Bimage(Float, TSimple, size(), n);
392 if ( !next->next ) next->next = new Bimage(Float, TSimple, size(), n);
393 if ( !next->next->next ) next->next->next = new Bimage(Float, TSimple, size(), n);
394
395 check_resolution(hi_res);
396
397 long i, xx, yy, zz;
398 double rx, ry, rz, r2, maxrad, maxrad2, pwr, w;
399 Vector3<long> h((size()-1)/2);
400 Vector3<double> freq_scale(1.0/real_size());
401 Complex<float> cv;
402
403 float* power = (float *) next->data_pointer();
404 float* weight = (float *) next->next->data_pointer();
405 float* weight2 = (float *) next->next->next->data_pointer();
406
407 double rad_scale(real_size()[0]), rad_scale2(rad_scale*rad_scale);
408
409 if ( hi_res > 0.1 )
410 maxrad = rad_scale/hi_res;
411 else
412 maxrad = size().max()/2.0;
413 maxrad2 = maxrad*maxrad;
414
415 for ( i=zz=0; zz<z; zz++ ) {
416 rz = zz;
417 if ( zz > h[2] ) rz -= z;
418 rz *= freq_scale[2];
419 rz *= rz;
420 for ( yy=0; yy<y; yy++ ) {
421 ry = yy;
422 if ( yy > h[1] ) ry -= y;
423 ry *= freq_scale[1];
424 ry *= ry;
425 for ( xx=0; xx<x; xx++, i++ ) {
426 rx = xx;
427 if ( xx > h[0] ) rx -= x;
428 rx *= freq_scale[0];
429 rx *= rx;
430 cv = p->complex(i);
431 pwr = cv.power();
432 if ( pwr >= threshold ) {
433 r2 = rad_scale2*(rx + ry + rz);
434 if ( r2 <= maxrad2 ) {
435 w = 1;
436 add(i, cv);
437 power[i] += pwr;
438 weight[i] += w;
439 weight2[i] += w*w;
440 }
441 }
442 }
443 }
444 }
445
446 return 0;
447}

510 }
511 }
