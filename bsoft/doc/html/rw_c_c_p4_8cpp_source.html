81int readCCP4(Bimage* p, int readdata)
82{
83 ifstream* fimg = new ifstream(p->file_name());
84 if ( fimg->fail() ) return -1;
85
86 if ( verbose & VERB_DEBUG )
87 cout << "DEBUG readCCP4: File opened" << endl;
88
89 CCP4head* header = new CCP4head;
90
91 fimg->read((char *)header, CCP4SIZE);
92 if ( fimg->fail() ) return -2;
93
94 // Determine byte order and swap bytes if from little-endian machine
95 unsigned char* b = (unsigned char *) header;
96 long i, sb(0);
97 if ( abs(header->mode) > SWAPTRIG || abs(header->mapc) > SWAPTRIG ) {
98 if ( verbose & VERB_PROCESS )
99 cerr << "Warning: Swapping header byte order for 4-byte types" << endl;
100 sb = 1;
101 long extent = CCP4SIZE - 800; // exclude labels from swapping
102 for ( i=0; i<extent; i+=4 )
103 if ( i < 208 || i > 212 ) swapbytes(b+i, 4);
104 }
105
106 if ( header->nx < 1 || header->ny < 1 || header->nz < 1 || header->mode < 0 ) {
107 cerr << "Error readCCP4: The file " << p->file_name() << " is not a valid CCP4 image!" << endl;
108 cerr << "Size: " << header->nx << tab << header->ny <<
109 tab << header->nz << endl;
110 fimg->close();
111 return -3;
112 }
113
114 // Convert VAX floating point types if necessary
115 // Machine stamp for vax is: 34 65 00 00
116 if ( ( header->machst[0] == 34 ) || ( header->amin > header->amax ) ) {
117 if ( verbose & VERB_PROCESS )
118 cerr << "Warning: Converting VAX floating point" << endl;
119 for ( i=40; i<64; i+=4 ) // Unit cell parameters
120 vax2ieee(b+i, sb);
121 for ( i=76; i<88; i+=4 ) // Min, max, mean
122 vax2ieee(b+i, sb);
123 for ( i=100; i<204; i+=4 ) // Skew matrix and extra
124 vax2ieee(b+i, sb);
125 vax2ieee(b+216, sb); // RMS
126 }
127
128 // Map the parameters
129 if ( strncmp(header->map, "STK ", 4) == 0 ) {
130 p->images(header->nz);
131 p->size(header->nx, header->ny, 1);
132 } else {
133 p->images(1);
134 p->size(header->nx, header->ny, header->nz);
135 }
136 p->channels(1);
137 switch ( header->mode ) {
138 case 0: p->data_type(SCharacter); break;
139 case 1: p->data_type(Short); break;
140 case 2: p->data_type(Float); break;
141 case 3: p->data_type(Short); p->compound_type(TComplex); p->channels(2); break;
142 case 4: p->data_type(Float); p->compound_type(TComplex); p->channels(2); break;
143 default: p->data_type(SCharacter); break;
144 }
145 p->data_offset(CCP4SIZE + header->nsymbt);
146 if ( header->mode > 2 ) {
147 fimg->seekg(0, ios::end);
148 if ( (double) fimg->tellg() > p->data_offset() + 0.8*p->alloc_size() )
149 p->sizeX(2*(p->sizeX() - 1));
150 if ( header->mx%2 == 1 ) p->sizeX(p->sizeX() + 1); // Quick fix for odd x-size maps
151 fimg->clear();
152 fimg->seekg(0, ios::beg);
153 }
154 p->minimum(header->amin);
155 p->maximum(header->amax);
156 p->average(header->amean);
157 p->standard_deviation(header->arms);
158
159 if ( header->mx && header->my && header->mz )
160 p->sampling(header->a/header->mx, header->b/header->my, header->c/header->mz);
161 p->space_group(header->ispg);
162 UnitCell uc(header->a, header->b, header->c, header->alpha, header->beta, header->gamma);
163 p->unit_cell(uc);
164 p->label(header->labels);
165
166 // Allocating the single sub-image and setting its origin
167 p->origin(-header->nxStart, -header->nyStart, -header->nzStart);
168
169 delete header;
170
171 if ( verbose & VERB_DEBUG )
172 cout << "DEBUG rwCCP4: Header setup done" << endl;
173
174 long readsize = p->alloc_size();
175 unsigned char* data = NULL;
176
177 if ( readdata ) {
178 p->data_alloc();
179 if ( p->compound_type() == TSimple ) {
180 fread_large(p->data_pointer(), readsize, p->data_offset(), fimg);
181 if ( sb ) swapbytes(readsize, p->data_pointer(), p->data_type_size());
182 } else {
183 readsize = p->channels()*(p->sizeX()/2+1)*p->sizeY()*p->sizeZ()*p->data_type_size();
184 data = new unsigned char[readsize];
185 fread_large(data, readsize, p->data_offset(), fimg);
186 if ( sb ) swapbytes(readsize, data, p->data_type_size());
187// img_unpack_transform(p, data, CentHerm);
188 p->unpack_transform(data, CentHerm);
189 delete[] data;
190 }
191 }
192
193 fimg->close();
194 delete fimg;
195
196 if ( verbose & VERB_DEBUG )
197 cout << "DEBUG rwCCP4: Data read" << endl;
198
199 return 0;
200}

208int writeCCP4(Bimage* p)
209{
210 if ( verbose & VERB_DEBUG )
211 cout << "DEBUG rwCCP4: Writing CCP4 file" << endl;
212
213 p->color_to_simple();
214
215// if ( p->compound_type() == TComplex ) p->change_type(Float);
216
217 switch ( p->data_type() ) {
218 case Bit: case UCharacter: case SCharacter:
219 if ( p->compound_type() == TComplex ) p->change_type(Short);
220 else p->change_type(SCharacter);
221 break;
222 case UShort: case Short:
223 p->change_type(Short); break;
224 default:
225 p->change_type(Float); break;
226 }
227
228 if ( verbose & VERB_DEBUG )
229 cout << "DEBUG rwCCP4: data type = " << p->data_type() << endl;
230
231 CCP4head* header = new CCP4head;
232 memset(header, 0, sizeof(CCP4head));
233
234 UnitCell uc = p->unit_cell();
235
236 // Map the parameters
237 set_CCP4_machine_stamp(header->machst);
238 header->nx = p->sizeX();
239 header->ny = p->sizeY();
240 if ( p->sizeZ() > 1 || p->images() == 1 ) {
241 memcpy(header->map, "MAP ", 4);
242 header->nz = p->sizeZ();
243 } else {
244 memcpy(header->map, "STK ", 4);
245 header->nz = p->images();
246 }
247 if ( p->compound_type() == TComplex ) header->nx = p->sizeX()/2 + 1; // If a transform, physical storage is nx/2 + 1
248 switch ( p->data_type() ) {
249 case SCharacter: header->mode = 0; break;
250 case Short: header->mode = 1; if ( p->compound_type() == TComplex ) header->mode = 3; break;
251 case Float: header->mode = 2; if ( p->compound_type() == TComplex ) header->mode = 4; break;
252 default: header->mode = 0; break;
253 }
254 if ( p->image->origin()[0] >= 0 )
255 header->nxStart = (int) -(p->image->origin()[0] + 0.5);
256 else
257 header->nxStart = (int) -(p->image->origin()[0] - 0.5);
258 if ( p->image->origin()[1] >= 0 )
259 header->nyStart = (int) -(p->image->origin()[1] + 0.5);
260 else
261 header->nyStart = (int) -(p->image->origin()[1] - 0.5);
262 if ( p->image->origin()[2] >= 0 )
263 header->nzStart = (int) -(p->image->origin()[2] + 0.5);
264 else
265 header->nzStart = (int) -(p->image->origin()[2] - 0.5);
266 header->mx = (int) (uc.a()/p->sampling(0)[0] + 0.5);
267 header->my = (int) (uc.b()/p->sampling(0)[1] + 0.5);
268 header->mz = (int) (uc.c()/p->sampling(0)[2] + 0.5);
269 header->mapc = 1;
270 header->mapr = 2;
271 header->maps = 3;
272 header->amin = p->minimum();
273 header->amax = p->maximum();
274 header->amean = p->average();
275 header->arms = p->standard_deviation();
276// cout << "Unitcell %g %g %g\n", uc.a(), uc.b(), uc.c());
277 header->a = uc.a();
278 header->b = uc.b();
279 header->c = uc.c();
280
281 // This is a band-aid to overcome the limitations of the image format
282 if ( fabs(header->a - p->sampling(0)[0]*header->mx) > 0.001 || fabs(header->b - p->sampling(0)[1]*header->my) > 0.001 ||
283 fabs(header->c - p->sampling(0)[2]*header->mz) > 0.001 ) {
284 header->a = p->sampling(0)[0]*header->mx;
285 header->b = p->sampling(0)[1]*header->my;
286 header->c = p->sampling(0)[2]*header->mz;
287 if ( verbose & VERB_FULL )
288 cerr << "Warning: Resetting the unit cell to: " << header->a << "," << header->b << "," << header->c << endl;
289 }
290
291 long i,j,k;
292 Matrix3 frac_mat = uc.skew_matrix();
293 header->alpha = uc.alpha()*180/M_PI;
294 header->beta = uc.beta()*180/M_PI;
295 header->gamma = uc.gamma()*180/M_PI;
296 for ( i=k=0; i<3; i++ ) for ( j=0; j<3; j++, k++ ) header->skwmat[k] = frac_mat[i][j];
297// for ( i=0; i<3; i++ ) header->skwtrn[i] = frac_mat[i+9];
298 header->ispg = p->space_group();
299
300 if ( verbose & VERB_DEBUG )
301 cout << "DEBUG rwCCP4: Writing labels" << endl;
302
303 header->nlabl = 10;
304 p->label().copy(header->labels, 799);
305 header->labels[799] = 0;
306
307 if ( verbose & VERB_DEBUG ) {
308 cout << "DEBUG rwCCP4: nlabl = " << header->nlabl << endl;
309 for ( i=0; i<header->nlabl; i++ )
310 if ( header->labels[i*80] ) cout << &header->labels[i*80] << endl;
311 }
312
313 int nsym(0);
314 string temp;
315 char* symop = NULL;
316
317#ifndef NOSYMOP
318 if ( p->space_group() > 0 ) symop = read_symop(temp, p->space_group(), nsym);
319#endif
320
321 header->nsymbt = nsym*80;
322
323 if ( verbose & VERB_DEBUG )
324 cout << "DEBUG rwCCP4: nsymbt = " << header->nsymbt << endl;
325
326 p->data_offset(CCP4SIZE + header->nsymbt);
327
328 long datatypesize = p->channels()*p->data_type_size();
329 long datasize = (long)header->nx*header->ny*header->nz*datatypesize;
330
331 if ( verbose & VERB_DEBUG )
332 cout << "DEBUG rwCCP4: offset=" << p->data_offset() << " typesize=" << datatypesize << " datasize=" << datasize << endl;
333
334 unsigned char* data = NULL;
335
336 ofstream fimg(p->file_name());
337 if ( fimg.fail() ) return -1;
338
339 fimg.write((char *)header, CCP4SIZE);
340
341 if ( header->nsymbt )
342 fimg.write(symop, header->nsymbt);
343
344 if ( p->data_pointer() ) {
345 if ( p->compound_type() < TComplex ) {
346 fimg.write((char *)p->data_pointer(), datasize);
347 } else {
348 data = new unsigned char[datasize];
349 p->pack_transform(data, CentHerm);
350 fimg.write((char *)data, datasize);
351 delete[] data;
352 }
353 }
354
355 fimg.close();
356
357 delete header;
358
359 if ( verbose & VERB_DEBUG )
360 cout << "DEBUG rwCCP4: Done writing" << endl;
361
362 return 0;
