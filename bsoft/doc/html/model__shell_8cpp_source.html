195double model_sphericity(Bmodel* model)
196{
197 if ( !model ) return 0;
198 if ( !model->select() ) return 0;
199
200 int ncomp;
201 double d, sph = 0, da = 0, ds = 0;
202
203 Vector3<double> com = model->center_of_coordinates();
204
205 Bcomponent* comp;
206
207 for ( ncomp=0, comp = model->comp; comp; comp = comp->next, ncomp++ ) {
208 d = (comp->location() - com).length();
209 da += d;
210 ds += d*d;
211 }
212
213 if ( ncomp ) {
214 da /= ncomp;
215 ds = ds/ncomp - da*da;
216 if ( ds > 0 ) sph = 1 - sqrt(ds)/da;
217 }
218
219 if ( verbose & VERB_PROCESS )
220 cout << "Sphericity: " << sph << endl << endl;
221
222 return sph;

235double model_ellipsoidicity(Bmodel* model)
236{
237 if ( !model ) return 0;
238 if ( !model->select() ) return 0;
239
240 int i, ncomp;
241 double d, ell = 0;
242 Vector3<double> c, c1, axis[3];
243 Vector3<double> alen = model_principal_axes(model, axis);
244
245 for ( i=0; i<3; i++ ) axis[i].normalize();
246
247 Vector3<double> com = model->center_of_coordinates();
248
249 Bcomponent* comp;
250
251 for ( ncomp=0, comp = model->comp; comp; comp = comp->next, ncomp++ ) {
252 c = comp->location() - com;
253 for ( i=0; i<3; i++ ) c1[i] = c.scalar(axis[i]);
254 c1 /= alen;
255 d = 1 - c1.length2();
256 ell += d*d;
257 }
258
259 ell = sqrt(1 - ell/ncomp);
260
261 if ( verbose & VERB_PROCESS )
262 cout << "Ellipsoidicity: " << ell << endl << endl;
263
264 return ell;
