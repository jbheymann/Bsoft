14/*
15 The intensity is composed of the direct and scattering components.
16 The average for the scattering component (mu) is modified
17 with aperture-based fraction k for the elastic part
18 and the use of an energy filter for the inelastic part.
19*/
20double signal_integrated(double emu, double imu,
21 double ke, double ki, int ef)
22{
23 double sig(1), f(1), mu(ke*emu), pmu(1);
24 if ( !ef ) mu += ki*imu;
25
26 for ( long n=1; n<=10; ++n ) {
27 f *= n; // Factorial
28 pmu *= mu; // Powers of average
29 sig += pmu/f;
30// cout << n << tab << f << tab << pmu << tab << sig << endl;
31 }
32
33 sig *= exp(-emu-imu);
34
35 return sig;
36}
37
38double signal_integrated_new(double emu, double imu, double k, int ef)
39{
40 double sig(0), f(1), pmu(1);
41
42 // Elastic scattering
43 for ( long n=1; n<=10; ++n ) {
44 f *= n; // Factorial
45 pmu *= emu; // Powers of average
46 sig += pmu/f;
47 }
48
49 if ( ef )
50 sig = (1 + k*sig)*exp(-emu-imu);
51 else
52 sig = 1 - (1 - k)*sig*exp(-emu-imu);
53
54 return sig;
55}

110vector<double> calculate_potential_curve(Bcomptype& ct, double dr, double rmax)
111{
112 if ( dr < 1e-10 ) dr = 0.001;
113 if ( rmax < 2 ) rmax = 2;
114
115 long i, j, k, imax(rmax/dr + 0.5);
116 double r2, dr2(dr*dr);
117
118 // Calculate the atomic potential profile
119 if ( verbose & VERB_DEBUG )
120 cout << "DEBUG calculate_potential_curve: z = " << ct.index() << endl;
121
122 vector<double> pot;
123 vector<double> coef(ct.coefficients());
124
125 for ( j=0, k=5; j<5; ++j, ++k ) {
126 coef[j] *= sqrt(M_PI/coef[k]);
127 coef[k] = M_PI*M_PI/coef[k];
128 }
129
130 for ( i=0; i<imax; ++i ) {
131 r2 = i*i*dr2;
132 pot.push_back(coef[10]);
133 for ( j=0, k=5; j<5; ++j, ++k )
134 pot[i] += coef[j]*exp(-coef[k]*r2);
135 }
136
137 return pot;
138}

168double elastic_cross_section_integrated(Bcomptype& ct, CTFparam& ctf)
169{
170 long i(0);
171 double ds(0.001), f, cs(0);
172 double wlr = ctf.lambda()*lorentz(ctf.volt());
173 double scut(ctf.frequency_cutoff());
174 double scale(8*M_PI*ds*ds*wlr*wlr);
175
176// vector<double> scat = calculate_elastic_scattering_curve(ct, ds, scut);
177 vector<double> scat = ct.elastic_scattering_curve(ds, scut);
178
179 for ( auto it = scat.begin(); it != scat.end(); ++it, ++i ) {
180 f = *it;
181 cs += f*f*i;
182// cout << i*ds << tab << cs << endl;
183 }
184
185 cs *= scale;
186
187 return cs;
188}

241double elastic_cross_section_lenz(long Z, double volts)
242{
243 double f = 1e10*PLANCK/(EMASS*LIGHTSPEED);
244 double b2 = beta2(volts);
245
246 double cs = (f*f/(M_PI*b2))*pow(Z, 4.0/3.0);
247
248 return cs;

251double elastic_cross_section_langmore(long Z, double volts)
252{
253 double b2 = beta2(volts), b = sqrt(b2);
254
255 double cs = 1.4e-4*(pow(Z, 1.5)/b2)*(1-0.26*Z/(137*b));
256
257 return cs;
258}

267double inelastic_cross_section_langmore(long Z, double volt)
268{
269 double b2 = beta2(volt);
270
271 double cs = 1.5e-4*
272 (sqrt(Z)/b2)*log(2*b2*
273 (ECHARGE*volt+EMASS*LIGHTSPEED*LIGHTSPEED)/(20*ECHARGE));
274
275 if ( Z == 1 ) cs *= 0.154; // Correction for hydrogen
276
277 return cs;

399double cross_section_half_maximal_frequency(Bcomptype& ct)
400{
401 long i(0);
402 double ds(0.001), f, csi(0);
403 double cs = ct.elastic_scattering_integral(0,1e10);
404// vector<double> scat = calculate_elastic_scattering_curve(ct, ds, 1);
405 vector<double> scat = ct.elastic_scattering_curve(ds, 1);
406
407 if ( verbose & VERB_FULL )
408 cout << "Full scattering integral = " << cs << endl;
409
410 for ( auto it = scat.begin(); it != scat.end() && csi < 0.5*cs; ++it, ++i ) {
411 f = *it * ds;
412 csi += 2*f*f*i;
413// cout << i*ds << tab << *it << tab << f << tab << csi << endl;
414 }
415
416 return ds*i;
417}

424double cross_section_half_maximal_frequency(map<string,Bcomptype>& types)
425{
426 double c, n(0), k(0);
427
428 for ( auto ct: types ) {
429 c = ct.second.component_count();
430 k += c * cross_section_half_maximal_frequency(ct.second);
431// cout << ct.second.identifier() << tab << ct.second.component_count() << tab << k << endl;
432 n += c;
433 }
434
435 if ( n ) k /= n;
436 else k = 1;
437
438 return k;

716vector<double> signal_intensity(Bmaterial& material, double thick_step, double thick_max, CTFparam& ctf)
717{
718 if ( thick_step < 1 ) thick_step = 100;
719 if ( thick_max < 1 ) thick_max = 5000;
720
721 map<string,Bcomptype>& comp = material.composition();
722
723 int ef(ctf.slit_width()>0); // Energy filter
724 double d(material.number_per_angstrom3());
725 double sm = cross_section_half_maximal_frequency(comp);
726 double sc = ctf.frequency_cutoff();
727 double ke = sc*sc/(sc*sc+sm*sm);
728 double ki = sc/(sc+0.054);
729
730 double ecs = elastic_cross_section(comp, ctf.volt());
731
732 double ics = inelastic_cross_section_langmore(comp, ctf);
733
734 double t, emu, imu, i, emfp;
735 vector<double> I;
736
737// cout << "Aperture effect: " << k << endl;
738
739 if ( verbose & VERB_FULL )
740 cout << "t(A)\tI\tEMFP" << endl;
741 for ( t=thick_step; t<=thick_max; t+=thick_step ) {
742 emu = ecs*d*t;
743 imu = ics*d*t;
744 i = signal_integrated(emu, imu, ke, ki, ef);
745 emfp = -t/log(i);
746 I.push_back(i);
747 if ( verbose & VERB_FULL )
748 cout << t << tab << i << tab << emfp << endl;
749 }
750
751 return I;
752}
