99long Bimage::region_assign(Bimage* pmask, long idx,
100 long region_number, double threshold, int sign)
101{
102 if ( verbose & VERB_DEBUG )
103 cout << "DEBUG Bimage::region_assign: Starting with region " <<
104 region_number << " idx=" << idx << endl;
105
106 if ( sign < 0 ) sign = -1;
107 else if ( sign > 0 ) sign = 1;
108 else {
109 sign = 1;
110 if ( threshold < avg ) sign = -1;
111 }
112
113 double abs_threshold(sign*threshold);
114 long i, j, nn, kx, ky, kz;
115 Vector3<long> lo, hi;
116
117 // Set up two lists:
118 // First list is for voxels to loop through for checking
119 // Second list is for new voxels to be checked in a subsequent iteration
120 long number, new_number, count(1);
121 long list_size = 2*(x + y)*z;
122 if ( z > 1 ) list_size += 2*x*y;
123 int* list = new int[list_size];
124 int* new_list = new int[list_size];
125
126 // Initialize the first item in the list
127 pmask->set(idx, region_number);
128 number = 1;
129 list[0] = idx;
130 nn = idx / (x*y*z);
131
132 // Find neighbors in shells around already found neighbors
133 while ( number > 0 ) {
134 if ( verbose & VERB_DEBUG )
135 cout << "DEBUG Bimage::region_assign: region=" << region_number << " list size=" << number << endl;
136 new_number = 0;
137 for ( i=0; i<number; ++i ) {
138 lo = kernel_low(list[i]);
139 hi = kernel_high(list[i]);
140
141 // Loop through the kernel and construct the new list
142 for ( kz=lo[2]; kz<=hi[2]; kz++ ) {
143 for ( ky=lo[1]; ky<=hi[1]; ky++ ) {
144 for ( kx=lo[0]; kx<=hi[0]; kx++ ) {
145 j = index(kx, ky, kz, nn);
146 if ( ((*pmask)[j] == 0) && (sign*(*this)[j] > abs_threshold) ) {
147 pmask->set(j, region_number);
148 count++;
149 new_list[new_number] = j;
150 new_number++;
151 if ( new_number >= list_size ) {
152 cerr << "Error: The list size exceeded its limit ("
153 << list_size << ")" << endl;
154 }
155 }
156 }
157 }
158 }
159 }
160 number = new_number;
161 for ( i=0; i<number; ++i ) list[i] = new_list[i];
162 }
163
164 delete[] list;
165 delete[] new_list;
166
167 if ( verbose & VERB_DEBUG )
168 cout << "DEBUG Bimage::region_assign: Ending with region " << region_number << endl;
169
170 return count;
171}

387{

391

395
396 // Loop through the kernel and check neighbors
397 for ( zz=lo[2]; zz<=hi[2]; ++zz ) {
398 for ( yy=lo[1]; yy<=hi[1]; ++yy ) {
399 for ( xx=lo[0]; xx<=hi[0]; ++xx ) {

401 if ( i != idx ) {

403 if ( nv == 0 )

409 }
