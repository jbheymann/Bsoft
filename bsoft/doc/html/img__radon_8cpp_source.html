247 strcpy (order, "zxy");
248 p->reslice(order);
249
250 filter_X_mask (p, pmask, n_cyc_in, rad_3D, 0);
251
252 // go to rho,phi,theta //
253 strcpy (order, "xzy");
254 p->reslice(order);
255
256 filter_X_mask (p, pmask, n_cyc_in, rad_plane, 1);
257
258 // come back at theta,phi,rho //
259 strcpy (order, "zyx");
260 p->reslice(order);
261 }
262
263 if (support)
264 support_f (p, rad_3D);
265
266 return 0;

454int img_resize_to_next_power2(Bimage* p, int fill_type, double fill)
455{
456 Vector3<int> newsize(1,1,1);
457
458 if ( p->sizeX() > 1 ) newsize[0] = findNextPowerOf(p->sizeX(), 2);
459 if ( p->sizeY() > 1 ) newsize[1] = findNextPowerOf(p->sizeY(), 2);
460 if ( p->sizeZ() > 1 ) newsize[2] = findNextPowerOf(p->sizeZ(), 2);
461
462 if ( newsize[0] == p->sizeX() && newsize[1] == p->sizeY() && newsize[2] == p->sizeZ() )
463 return 0;
464
465// cout << "resizing" << endl;
466
467 Vector3<int> translate(newsize[0]/2 - p->sizeX()/2, newsize[1]/2 - p->sizeY()/2, newsize[2]/2 - p->sizeZ()/2);
468
469 p->resize(newsize, translate, fill_type, fill);
470
471 return 0;
