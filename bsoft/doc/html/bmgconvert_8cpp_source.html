210Bproject* read_project_conv(Bstring* file_list)
211{
212 if ( !file_list ) {
213 cerr << "Error: No micrograph parameter filename!" << endl;
214 return NULL;
215 }
216
217 Bstring ext;
218 Bstring* thisfile;
219
220 if ( verbose & VERB_LABEL ) {
221 cout << "Parameter filenames:";
222 for ( thisfile = file_list; thisfile; thisfile = thisfile->next )
223 cout << " " << *thisfile;
224 cout << endl;
225 }
226
227 int err(0);
228 FileType type;
229 Bproject* project = new Bproject;
230 Bfield* field = NULL;
231 Bmicrograph* mg = NULL;
232 Breconstruction* rec = NULL;
233 project->filename = *file_list;
234
235 for ( thisfile = file_list; thisfile; thisfile = thisfile->next ) {
236 for ( field = project->field; field && field->next; field = field->next ) ;
237 if ( field ) for ( mg = field->mg; mg && mg->next; mg = mg->next ) ;
238 for ( rec = project->rec; rec && rec->next; rec = rec->next) ;
239 ext = thisfile->extension();
240 err = -1;
241 if ( ext.contains("star") ) {
242 type = file_type(*thisfile);
243 if ( type == Micrograph )
244 err = read_project_star(*thisfile, project, 1);
245 else if ( type == MgRelion )
246 err = read_project_relion(*thisfile, project);
247 else
248 cerr << "Error: The STAR file is not a valid form Bsoft can read!" << endl;
249 } else if ( ext.contains("xml") ) {
250 err = read_project_xml(*thisfile, project);
251 } else if ( ext.contains("dat") ) {
252 err = read_project_dat(*thisfile, project);
253 } else if ( ext.contains("crd") ) {
254 err = read_project_crd(*thisfile, project);
255 } else {
256 cerr << "Error: Extension " << ext << " not valid for parameter files!" << endl;
257 }
258 if ( !err ) {
259 if ( !field ) field = project->field;
260 else field = field->next;
261 if ( field ) field_resolve_file_access(field, mg, *thisfile, 8);
262 if ( !rec ) rec = project->rec;
263 else rec = rec->next;
264 if ( rec ) reconstruction_resolve_file_access(rec, *thisfile, 8);
265 } else {
266 cerr << " Parameter file " << *thisfile << " not read." << endl;
267 }
268 }
269
270 if ( err < 0 ) {
271 error_show(file_list->c_str(), __FILE__, __LINE__);
272 project_kill(project);
273 return NULL;
274 }
275
276 project_update_first_zero(project);
277
278// project_check(project);
279
280// if ( verbose & VERB_PROCESS )
281// project_display_counts(project);
282
283 return project;
284}

336int read_project_dat(Bstring& filename, Bproject* project)
337{
338 int err(0), nx(0), ny(0), n(0);
339 char inputLine[1024];
340 int unitsInHeaderI;
341 Bstring unitsInHeaderC; // "angstrom", "nm" or "pixel"
342 double px, volt, amp, def, dev, ast, cs;
343
344 Bstring field_id = filename.base();
345 Bstring mg_id = field_id;
346 Bfield* field = field_add(&project->field, field_id);
347 Bmicrograph* mg = micrograph_add(&field->mg, mg_id);
348 Bparticle* part = NULL;
349 CTFparam* ctf = mg->ctf = new CTFparam;
350
351 if ( verbose ) cout << "Reading a DAT file: " << filename << endl;
352
353 project->comment = "# DAT file: " + filename + "\n";
354
355 /*========================
356 Read DAT-format file
357 ========================*/
358 ifstream fdat(filename.c_str());
359 if ( fdat.fail() ) {
360 cerr << "File " << filename << " does not exist!. Exit program." << endl;
361 bexit(-1);
362 }
363
364 fdat.getline(inputLine, 1023);
365 mg->fpart = inputLine;
366 mg->fpart = mg->fpart.no_lead_space();
367 mg->fpart = mg->fpart.pre(' ');
368 mg->fpart = mg->fpart.remove('\n');
369 if ( mg->fpart.empty() ) {
370 cerr << "ERROR reading line 1 of PFT-format DAT file. Exit program" << endl;
371 exit(-1);
372 }
373
374 // General micrograph parameters
375 sscanf(strchr(inputLine+mg->fpart.length(), ' '), " %d %d %d", &nx, &ny, &n);
376// cout << "\"" << inputLine << "\"" << nx << " " << ny << " " << n << endl;
377
378 fdat.getline(inputLine, 1023);
379 if ( !fdat.good() ) {
380 cout << "ERROR reading line 2 of PFT-format DAT file. Exit program" << endl;
381 exit(-1);
382 }
383 if ( sscanf(inputLine, "%lf,%d,%lf,%lf,%lf,%lf,%lf,%lf\n", &px, &unitsInHeaderI,
384 &volt, &amp, &def, &dev, &ast, &cs) == 8) {
385 switch(unitsInHeaderI) {
386 case 0:
387 unitsInHeaderC = "pixels";
388 break;
389 case 1:
390 unitsInHeaderC = "angstroms";
391 break;
392 case 2:
393 unitsInHeaderC = "nm";
394 default:
395 break;
396 }
397 } else if ( sscanf(inputLine, "%lf,%lf,%lf,%lf,%lf,%lf,%lf\n", &px, &volt,
398 &amp, &def, &dev, &ast, &cs) == 7) {
399 } else {
400 cout << "ERROR interpreting line 2 of PFT-format DAT file. Exit program" << endl;
401 exit(-1);
402 }
403
404 if ( volt < 1000.0 ) volt *= 1000; // Volt units for STAR-format file
405 cs *= 10000000; // Angstrom units for STAR-format file
406 if ( def < 100 ) def *= 1e4; // Angstrom units for STAR-format file
407 if ( dev < 100 ) dev *= 1e4; // Angstrom units for STAR-format file
408 def = (def + dev)/2;
409 dev -= def;
410 if ( dev < 0 ) dev = -dev;
411
412 if ( nx && ny ) mg->box_size = Vector3<int>(nx, ny, 1);
413 mg->pixel_size = Vector3<double>(px, px, 1);
414 ctf->volt(volt);
415 ctf->amp_shift(asin(amp));
416 ctf->defocus_average(def);
417 ctf->astigmatism(dev, ast*M_PI/180.0);
418 ctf->Cs(cs);
419
420 char line[DATLINELENGTH];
421 int numpart(0), pid;
422 float theta, phi, omega, ox, oy, mag, fom1, fom2, fom3;
423
424 while ( fdat.getline(line, DATLINELENGTH) && fdat.good() ) {
425 if ( sscanf(line, "%d,%f,%f,%f,%f,%f,%f,%f,%f,%f", &pid, &theta, &phi, &omega, &ox, &oy, &mag, &fom1, &fom2, &fom3) > 6 ) {
426 part = particle_add(&part, pid);
427 if ( !mg->part ) mg->part = part;
428 part->ori[0] = ox;
429 part->ori[1] = oy;
430 part->mag = mag;
431 part->fom[0] = fom1;
432 part->fom[1] = fom2;
433 part->fom[2] = fom3;
434 part->view = Euler(-omega*M_PI/180.0,
435 theta*M_PI/180.0, phi*M_PI/180.0).view();
436 numpart++;
437 }
438 }
439
440 fdat.close(); // close input DAT file
441
442 if ( n && numpart != n )
443 cerr << "Warning: The number of particles (" << numpart <<
444 ") does not agree with the number in the header (" << n << ")" << endl;
445
446 if ( verbose )
447 cout << "Number of particles read: " << numpart << endl << endl;
448
449 return err;
450}

459int write_project_dat(Bstring& filename, Bproject* project, int flags)
460{
461 int err(0);
462 int mg_select(flags & 2);
463 int i, f, numpart(0);
464 double def_min, def_max, omega;
465 Euler euler;
466 Bfield* field = NULL;
467 Bmicrograph* mg = NULL;
468 Bparticle* part = NULL;
469 CTFparam* ctf;
470 ofstream fdat;
471 Bstring outname = filename;
472
473 if ( verbose ) cout << "Writing a DAT file: " << filename << endl;
474
475 if ( project_count_micrographs(project) < 2 ) mg_select = 1;
476
477 for ( i=1, field = project->field; field; field = field->next ) {
478 for ( mg = field->mg; mg; mg = mg->next, i++ ) if ( mg_select < 1 || mg->select > 0 ) {
479 if ( mg_select < 1 ) outname = filename.pre_rev('.') + Bstring(i, "_%06d.") + filename.post_rev('.');
480 for ( part = mg->part; part; part = part->next ) numpart++;
481 ctf = mg->ctf;
482 if ( !ctf ) ctf = mg->ctf = new CTFparam;
483 def_min = ctf->defocus_average() - ctf->defocus_deviation();
484 def_max = ctf->defocus_average() + ctf->defocus_deviation();
485 fdat.open(outname.c_str());
486 if ( fdat.fail() ) { // make sure input file exists
487 cerr << "Unable to create file " << filename << "!. Exit program." << endl;
488 bexit(-1);
489 }
490 fdat << mg->fpart << " " << mg->box_size[0] << " " << mg->box_size[1] << " " << numpart << endl;
491 fdat << fixed << right << setprecision(4) << setw(9) << mg->pixel_size[0] << ", "
492 << 1 << ", "
493 << setprecision(1) << setw(8) << ctf->volt() << ", "
494 << setprecision(4) << setw(10) << sin(ctf->amp_shift()) << ", "
495 << setprecision(4) << setw(9) << def_max/1e4 << ", "
496 << setprecision(4) << setw(9) << def_min/1e4 << ", "
497 << setprecision(3) << setw(9) << ctf->astigmatism_angle()*180.0/M_PI << ", "
498 << setprecision(2) << setw(7) << ctf->Cs()/1e7 << endl;
499 for ( part = mg->part; part; part = part->next ) {
500 euler = Euler(part->view);
501 omega = -euler.psi()*180.0/M_PI;
502 if ( omega < 0 ) omega += 360;
503 fdat << setw(7) << part->id << ", "
504 << setprecision(3) << setw(7) << euler.theta()*180.0/M_PI << ", "
505 << setprecision(3) << setw(7) << euler.phi()*180.0/M_PI << ", "
506 << setprecision(3) << setw(7) << omega << ", "
507 << setprecision(3) << setw(7) << part->ori[0] << ", "
508 << setprecision(3) << setw(7) << part->ori[1] << ", "
509 << setprecision(3) << setw(5) << part->mag;
510 for ( f=0; f<3; f++ ) fdat << ", " << setprecision(3) << setw(5) << part->fom[f];
511 fdat << endl;
512 }
513 fdat.close();
514 }
515 }
516
517 if ( verbose )
518 cout << "Number of particles written: " << numpart << endl << endl;
519
520 return err;
521}

529int read_project_crd(Bstring& filename, Bproject* project)
530{
531 int err(0);
532 char line[DATLINELENGTH];
533 char* value;
534 double d;
535
536 Bstring field_id = filename.base();
537 Bstring mg_id = field_id;
538 Bfield* field = field_add(&project->field, field_id);
539 Bmicrograph* mg = micrograph_add(&field->mg, mg_id);
540 Bparticle* part = NULL;
541 Bbadarea* bad = NULL;
542 Bstring part_prefix;
543 Bstring part_ext;
544
545 if ( verbose ) cout << "Reading a CRD file: " << filename << endl;
546
547 project->comment = "# CRD file: " + filename + "\n";
548
549 /*========================
550 Read CRD-format file
551 ========================*/
552 ifstream fcrd(filename.c_str());
553 if ( fcrd.fail() ) {
554 cerr << "File " << filename << " does not exist!. Exit program." << endl;
555 bexit(-1);
556 }
557
558 fcrd.getline(line, DATLINELENGTH);
559
560 // General micrograph parameters
561 while ( fcrd.getline(line, DATLINELENGTH) && !strstr(line, "$END") ) {
562 value = strchr(line, '=') + 1;
563 while ( isspace(value[0]) ) value++;
564 if ( strstr(line, "PIC_FILENAME") ) {
565 mg->fmg = value;
566 mg->fmg = mg->fmg.remove('\n');
567 mg->fmg = mg->fmg.remove(',');
568 mg->fmg = mg->fmg.remove('\'');
569 } else if ( strstr(line, "OUTFILE_PREFIX") ) {
570 part_prefix = value;
571 part_prefix = part_prefix.remove('\n');
572 part_prefix = part_prefix.remove(',');
573 part_prefix = part_prefix.remove('\'');
574 } else if ( strstr(line, "OUTFILE_TYPE") ) {
575 part_ext = value;
576 part_ext = part_ext.remove('\n');
577 part_ext = part_ext.remove(',');
578 part_ext = part_ext.remove('\'');
579 } else if ( strstr(line, "ANGSTROMS") ) {
580 sscanf(value, "%lf", &d);
581 if ( d > 0 ) mg->pixel_size[0] = mg->pixel_size[1] = d;
582 } else if ( strstr(line, "RADIUS_PICK") ) {
583 sscanf(value, "%lf", &d);
584 mg->box_size[0] = mg->box_size[1] = 2*d;
585 mg->box_size[2] = 1;
586 } else if ( strstr(line, "RADIUS_BAD") ) {
587 sscanf(value, "%lf", &mg->bad_radius);
588 }
589 }
590
591 if ( part_prefix.length() )
592 mg->fpart = part_prefix + "." + part_ext;
593
594 int numpart(0), numbad(0), pid;
595 float x, y;
596
597 while ( fcrd.getline(line, DATLINELENGTH) ) {
598 if ( sscanf(line, "%d %f %f", &pid, &x, &y) > 2 ) {
599 if ( pid > 0 ) {
600 part = particle_add(&part, pid);
601 if ( !mg->part ) mg->part = part;
602 part->loc[0] = x;
603 part->loc[1] = y;
604 numpart++;
605 } else {
606 bad = (Bbadarea *) add_item((char **) &bad, sizeof(Bbadarea));
607 if ( !mg->bad ) mg->bad = bad;
608 bad->loc[0] = x;
609 bad->loc[1] = y;
610 numbad++;
611 }
612 }
613 }
614
615 fcrd.close(); // close input CRD file
616
617 if ( verbose )
618 cout << "Number of particles written: " << numpart << " (" << numbad << ")" << endl << endl;
619
620 return err;
621}
