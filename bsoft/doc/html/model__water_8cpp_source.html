24Bmodel* model_generate_one_water(string& watername, Vector3<double> Ocoord)
25{
26 double rand_max = 1.0*get_rand_max();
27 double OHbond(0.9572);
28 double HOHangle(109.47*M_PI/180.0);
29 string restype("HOH"), chain("WAT");
30 Vector3<double> H1vec, H2vec;
31 Matrix3 mat(1);
32
33 Bmodel* water = new Bmodel(watername);
34 Bcomptype* cto = water->add_type("O");
35 Bcomptype* cth = water->add_type("H");
36 cto->index(7);
37 cth->index(0);
38 cto->mass(16);
39 cth->mass(1);
40
41 Bcomponent* O = water->add_component(1);
42 O->type(cto);
43 O->description("O");
44 O->add_description("O");
45 O->add_description(restype);
46 O->add_description(chain);
47 O->charge(-0.834);
48 O->location() = Ocoord;
49
50 Bcomponent* H1 = water->add_component(2);
51 H1->type(cth);
52 H1->description("H");
53 H1->add_description("HO1");
54 H1->add_description(restype);
55 H1->add_description(chain);
56 H1->charge(0.417);
57 H1vec[0] = random()/rand_max - 0.5;
58 H1vec[1] = random()/rand_max - 0.5;
59 H1vec[2] = random()/rand_max - 0.5;
60 H1vec.normalize();
61 H1->location() = (H1vec * OHbond) + Ocoord;
62 water->comp->find_and_add_links(O->identifier(), H1->identifier());
63
64 Bcomponent* H2 = water->add_component(3);
65 H2->type(cth);
66 H2->description("H");
67 H2->add_description("HO2");
68 H2->add_description(restype);
69 H2->add_description(chain);
70 H2->charge(0.417);
71 H2vec[0] = random()/rand_max - 0.5;
72 H2vec[1] = random()/rand_max - 0.5;
73 H2vec[2] = random()/rand_max - 0.5;
74 H2vec.normalize();
75
76 H2vec = H1vec.cross(H2vec);
77 mat = Matrix3(H2vec, HOHangle);
78 H2vec = mat * H1vec;
79 H2->location() = (H2vec * OHbond) + Ocoord;
80 water->comp->find_and_add_links(O->identifier(), H2->identifier());
81
82 water->add_angle(O, H1, H2);
83
84 return water;
85}
