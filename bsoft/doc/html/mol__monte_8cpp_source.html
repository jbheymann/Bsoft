71Bmolgroup* monte_carlo_metropolis(Bmolgroup* molgroup, Bmd* md, Bimage* map,
72 double beta, double max_angle, double max_shift,
73 long max_iter, int rigid,
74 double (Efunc)(Bmolgroup*, Bimage*, Bmd*),
75 int (Tfunc)(Bmolgroup*, double, double))
76{
77 if ( !map ) {
78 cerr << "Error: No map specified!" << endl;
79 bexit(-1);
80 }
81
82 int done(0), accept, select_min(0);
83 long i, isel(0), cycle(0), naccept(0);
84 double r, E, pE, bestE, Z(0), Ze, Zi, Emax;
85 double irm = 1.0/get_rand_max();
86 Vector3<double> gloc = molgroup_center_of_mass(molgroup);
87 Bmolgroup* mgc = molgroup_list_copy(molgroup);
88 Bmolgroup* mg = NULL;
89 Bmolgroup* mgold = NULL;
90 Bmolecule* mol;
91
92// double* Elist = new double[max_iter+1];
93 vector<double> Elist(max_iter+1,0);
94
95 long nmg = count_list((char *) molgroup);
96 long nmol = count_list((char *) molgroup->mol);
97 long nbond = count_list((char *) molgroup->bond);
98 Vector3<double>* iloc = new Vector3<double>[nmol];
99
100 if ( verbose & VERB_RESULT ) {
101 cout << "Monte carlo metropolis energy minimization:" << endl;
102 cout << "Molecule file: " << molgroup->filename << endl;
103 cout << "Number of molecule groups: " << nmg << endl;
104 cout << "Number of molecules: " << nmol << endl;
105 if ( map ) cout << "Map file: " << map->file_name() << endl;
106 cout << "Coordinate minima and maxima: " << molgroup->min << " " << molgroup->max << endl;
107 cout << "Bounding box: " << molgroup->box << endl;
108 cout << "Kbond: " << md->Kbond << endl;
109 cout << "Kangle: " << md->Kangle << endl;
110 cout << "Kelectrostatic: " << md->Kelec << endl;
111 cout << "Kvdw: " << md->Kvdw << endl;
112 cout << "Kmap: " << md->Kmap << endl;
113 cout << "Kseparation: " << md->Ksep << endl;
114 cout << "Kpoint: " << md->Kpoint << " (" << md->pointdecay << ")" << endl;
115 cout << "location: " << md->point << endl;
116 cout << "beta constant: " << beta << endl;
117 cout << "Cutoff distance: " << md->cutoff << endl;
118 cout << "Separation distance: " << md->sepdist << endl;
119 cout << "Maximum shift: " << max_shift << endl;
120 cout << "Maximum angular increment: " << max_angle*180.0/M_PI << endl;
121 cout << "Steps along bond: " << md->bondsteps << endl;
122 cout << "Number of bonds: " << nbond << endl;
123 if ( rigid < 1 ) cout << "Treating the whole ensemble as a rigid body" << endl;
124 else if ( rigid == 1 ) cout << "Treating each file with a molecule group as a rigid body" << endl;
125 else if ( rigid == 2 ) cout << "Treating each molecule as a rigid body" << endl;
126 else cout << "Modifying each atom's coordinates" << endl;
127 cout << endl;
128 md_calculate_deviations(molgroup, md->wrap);
129 }
130
131 if ( rigid < 1 ) {
132 for ( mg = mgc; mg; mg = mg->next ) mg->select = "ALL";
133 } else if ( rigid == 1 ) {
134 for ( mg = mgc; mg; mg = mg->next ) {
135 mg->sel = 0;
136 mg->select = "FILE";
137 }
138 } else if ( rigid == 2 ) {
139 for ( i=0, mol = mgc->mol; mol; mol = mol->next, ++i ) {
140 mol->sel = 0;
141 iloc[i] = mol_center_of_mass(mol);
142 }
143 mgc->select = "MOL";
144 } else mgc->select = "ATOM";
145
146 pE = (Efunc)(mgc, map, md);
147
148 bestE = E = pE;
149 Elist[0] = pE;
150
151 if ( verbose & VERB_RESULT ) {
152 display_energy_headers(md);
153 display_energy(0, md, E-pE);
154 }
155 while ( !done && ( cycle < max_iter ) ) {
156 cycle++;
157 mgold = molgroup_list_copy(mgc);
158 if ( rigid == 1 ) {
159 if ( select_min ) {
160 for ( i=0, Emax=-1e30, mg=mgc; mg; mg=mg->next, i++ ) {
161 if ( Emax < mg->fom ) {
162 Emax = mg->fom;
163 isel = i;
164 }
165 }
166 } else {
167 isel = (long) ((nmg-0.001)*irm*random());
168 }
169 for ( i=0, mg=mgc; mg; mg=mg->next, i++ ) {
170 if ( i == isel ) mg->sel = 1;
171 else mg->sel = 0;
172 }
173 } else if ( rigid == 2 ) {
174 if ( select_min ) {
175 for ( i=0, Emax=-1e30, mol=mgc->mol; mol; mol=mol->next, i++ ) {
176 if ( Emax < mol->fom ) {
177 Emax = mol->fom;
178 isel = i;
179 }
180 }
181 } else {
182 isel = (long) ((nmol-0.001)*irm*random());
183 }
184 for ( i=0, mol=mgc->mol; mol; mol=mol->next, i++ ) {
185 if ( i == isel ) mol->sel = 1;
186 else mol->sel = 0;
187 }
188 }
189 (Tfunc)(mgc, max_angle, max_shift);
190 E = (Efunc)(mgc, map, md);
191 Elist[cycle] = E;
192// display_energy(cycle, md, E-pE);
193 if ( bestE > E ) {
194 bestE = E;
195 molgroup_list_kill(molgroup);
196 molgroup = molgroup_list_copy(mgc);
197 if ( verbose & VERB_RESULT )
198 display_energy(cycle, md, E-pE);
199 }
200 accept = 1;
201 if ( E > 1e30 ) accept = 0; // High energy indicates the molecule might be outside the box
202 if ( E > pE ) {
203 r = random()*irm;
204 if ( exp(beta*(pE-E)) < r ) accept = 0;
205 }
206 if ( accept ) {
207 pE = E;
208 naccept++;
209 molgroup_list_kill(mgold);
210 } else {
211 molgroup_list_kill(mgc);
212 mgc = mgold;
213 }
214// molgroup_test_if_within_box(mgc, mgc->min, mgc->max);
215 }
216
217 molgroup_list_kill(mgc);
218
219 for ( i=0, Z=Ze=0; i<=cycle; i++ ) {
220 Zi = exp(-beta*Elist[i]);
221 Z += Zi;
222 if ( Elist[i] - bestE < 0.001 ) Ze += Zi;
223 }
224 if ( Z <= 0 ) Z = 1;
225
226// delete[] Elist;
227
228 md->Epot = bestE;
229
230 gloc = molgroup_center_of_mass(molgroup) - gloc;
231
232 if ( verbose & VERB_RESULT ) {
233 cout << "Total shift: " << gloc << endl;
234 for ( i=0, mol = molgroup->mol; mol; mol = mol->next, ++i ) {
235 iloc[i] = mol_center_of_mass(mol) - iloc[i];
236 E = mol_map_energy(mol, map, md->Kmap);
237 cout << tab << mol->id << tab << iloc[i] << tab << E << endl;
238 }
239 cout << "Fraction of perturbations accepted: " << naccept*1.0/cycle << endl;
240 cout << "Final energy: " << bestE << endl;
241 cout << "Partition function: " << Z << endl;
242 cout << "Probability of final orientation: " << Ze/Z << endl << endl;
243 md_calculate_deviations(molgroup, md->wrap);
244 }
245
246 delete[] iloc;
247
248 return molgroup;
249}

671double monte_rigid_body_fit_energy(Bmolgroup* molgroup, Bimage* map, Bmd* md)
672{

674

677 md->Epot += md->Emap;
678 }
679
680 if ( md->Kpoint ) {
681 md->Epoint = md->Kpoint*md->point.distance(molgroup_center_of_mass(molgroup));
682 md->Epot += md->Epoint;
683 }
684

688 }
689

704{

769double molgroup_atom_overlap(Bmolgroup* molgroup, Bmd* md)
770{
771 if ( md->Ksep <= 0 ) return 0;
772
773 long n(0), i, x, y, z;
774 Vector3<double> box = molgroup->box;
775 Vector3<double> sampling(md->sepdist, md->sepdist, md->sepdist);
776 Vector3<int> size;
777 size[0] = (int) (box[0]/sampling[0] + 0.001);
778 size[1] = (int) (box[1]/sampling[1] + 0.001);
779 size[2] = (int) (box[2]/sampling[2] + 0.001);
780 size = size.max(1);
781 sampling[0] = box[0]/size[0] + 0.001;
782 sampling[1] = box[1]/size[1] + 0.001;
783 sampling[2] = box[2]/size[2] + 0.001;
784 long boxsize = (long) size.volume();
785
786 Bmolgroup* mg;
787 Bmolecule* mol;
788 Bresidue* res;
789 Batom* atom;
790
791 Bimage* p = new Bimage(Float, TSimple, size, 1);
792 float* data = (float *) p->data_pointer();
793 p->next = new Bimage(Float, TSimple, p->size(), p->images());
794 float* dataone = (float *) p->next->data_pointer();
795
796 for ( mg = molgroup; mg; mg = mg->next ) {
797 for ( mol = mg->mol; mol; mol = mol->next ) {
798 for ( res = mol->res; res; res = res->next ) {
799 for ( atom = res->atom; atom; atom = atom->next ) {
800 x = (long) ((atom->coord[0] - mg->min[0])/sampling[0]);
801 y = (long) ((atom->coord[1] - mg->min[1])/sampling[1]);
802 z = (long) ((atom->coord[2] - mg->min[2])/sampling[2]);
803 if ( x>=0 && x<size[0] && y>=0 && y<size[1] && z>=0 && z<size[2] ) {
804 i = p->index(x, y, z);
805 dataone[i] = 1;
806 }
807 }
808 }
809 if ( !mg->select.contains("FILE") ) {
810 for ( i=0; i<boxsize; i++ ) {
811 data[i] += dataone[i];
812 dataone[i] = 0;
813 }
814 }
815 }
816 if ( mg->select.contains("FILE") ) {
817 for ( i=0; i<boxsize; i++ ) {
818 data[i] += dataone[i];
819 dataone[i] = 0;
820 }
821 }
822 }
823
824 if ( verbose & VERB_DEBUG ) {
825 Bstring filename("overlap.map");
826 write_img(filename, p, 0);
827 }
828
829 md->Esep = 0;
830
831 for ( i=n=0; i<boxsize; i++ ) {
832 if ( data[i] ) {
833 n++;
834 if ( data[i] > 1 ) md->Esep += data[i] - 1;
835 }
836 }
837
838 md->Esep *= md->Ksep * md->sepdist/n;
839
840 delete p;
841
842 return md->Esep;
843}
