145long mg_replace_parameters(Bmicrograph* mg, Bmicrograph* mg_source, Bstring& which, Vector3<double> origin)
146{
147 Bparticle *part, *part_source;
148 Vector3<double> shift = mg->origin - origin;
149
150 if ( which.contains("ctf") ) {
151 if ( mg_source->fps.length() > 0 ) mg->fps = mg_source->fps;
152 if ( mg_source->magnification ) mg->magnification = mg_source->magnification;
153 if ( mg_source->sampling ) mg->sampling = mg_source->sampling;
154 if ( mg_source->ctf ) {
155 if ( !mg->ctf ) mg->ctf = new CTFparam;
156 mg->ctf->update(mg_source->ctf);
157 }
158 }
159
160 if ( which.contains("part") ) {
161 mg->box_size = mg_source->box_size;
162 mg->bad_radius = mg_source->bad_radius;
163 particle_kill(mg->part);
164 mg->part = particle_copy(mg_source->part);
165 for ( part = mg->part; part; part = part->next )
166 part->loc += shift;
167 }
168
169 if ( which.contains("ori") ) {
170 for ( part = mg->part, part_source = mg_source->part; part && part_source; part = part->next, part_source = part_source->next )
171 part->ori = part_source->ori;
172 }
173
174 return 1;
175}

177long field_replace_parameters(Bfield* field, Bfield* field_source, Bstring& which)
178{
179 Bmicrograph *mg, *mg_source;
180 Vector3<double> origin = field->mg->origin; // Assumes the first micrograph is the reference
181
182 long nmg = field_count_micrographs(field);
183 long nmg_source = field_count_micrographs(field_source);
184
185 if ( nmg_source == 1 ) { // Replaces parameters from one micrograph into many
186 mg_source = field_source->mg;
187 for ( nmg=0, mg = field->mg; mg; mg = mg->next )
188 nmg += mg_replace_parameters(mg, mg_source, which, origin);
189 } else if ( nmg_source == nmg ) { // Replaces parameters from into corresponding micrographs
190 for ( nmg=0, mg = field->mg, mg_source = field_source->mg; mg && mg_source; mg = mg->next, mg_source = mg_source->next )
191 nmg += mg_replace_parameters(mg, mg_source, which, origin);
192 } else {
193 cerr << "Error in updating parameters: The numbers of micrographs are unsuitable!" << endl;
194 cerr << " Master field " << field->id << " has " << nmg << " micrographs" << endl;
195 cerr << " Source field " << field_source->id << " has " << nmg_source << " micrographs" << endl;
196 }
197
198 return nmg;
199}
