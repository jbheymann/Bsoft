16bool set_tile_size(Vector3<long>& tile_size, Vector3<long> img_size)
17{
18 if ( tile_size == img_size ) return 0;
19
20 // Minimum and maximum tile sizes allowed
21 long min_size(64);
22 long max_size(4096);
23
24 tile_size = tile_size.max(min_size);
25 tile_size = tile_size.min(img_size);
26 tile_size = tile_size.min(max_size);
27
28 if ( tile_size[0] != tile_size[1] ) {
29 if ( tile_size[0] > tile_size[1] ) tile_size[0] = tile_size[1];
30 else tile_size[1] = tile_size[0];
31 if ( img_size[2] > 1 ) tile_size[2] = tile_size[0];
32 }
33
34 return 1;
35}

161Bimage* Bimage::powerspectrum_tiled(long img_num, Vector3<long> tile_size, int flags)
162{
163 // If the image is square it is assumed to be a CCD image or extracted to be square
164 // and no padding around the edges is necessary
165 int pad(0);
166 long max_size(4096);
167
168 // If the image is not square, it is assumed to be a micrograph with severe edge
169 // features that will affect the power spectrum
170 if ( x != y && x > max_size && y > max_size) {
171 pad = x/10;
172 if ( pad < y/10 ) pad = y/10;
173 }
174
175 Vector3<long> start, ext_size, step_size;
176
177 bool do_tile = set_tile_size(tile_size, size());
178
179 ext_size[0] = tile_size[0]*((x - 2*pad)/tile_size[0]);
180 ext_size[1] = tile_size[1]*((y - 2*pad)/tile_size[1]);
181 ext_size[2] = tile_size[2]*((z - 2*pad)/tile_size[2]);
182 ext_size = ext_size.max(1);
183
184 start[0] = (x - ext_size[0])/2;
185 start[1] = (y - ext_size[1])/2;
186 start[2] = (z - ext_size[2])/2;
187
188 if ( verbose & VERB_DEBUG ) {
189 cout << "DEBUG Bimage::powerspectrum_tiled: img_num=" << img_num << endl;
190 cout << "DEBUG Bimage::powerspectrum_tiled: ext_size=" << ext_size << endl;
191 cout << "DEBUG Bimage::powerspectrum_tiled: start=" << start << endl;
192 cout << "DEBUG Bimage::powerspectrum_tiled: tile_size=" << tile_size << endl;
193 }
194
195 if ( verbose & VERB_PROCESS )
196 cout << "Tile size: " << tile_size << endl << endl;
197
198 // If a single image assume a micrograph that needs to be tiled
199 Bimage* pex = NULL;
200 if ( do_tile )
201 pex = extract_tiles(img_num, start, ext_size, tile_size, step_size, 0);
202 else
203 pex = copy();
204
205 pex->power_spectrum(flags);
206
207 return pex;

241Bimage* Bimage::powerspectrum_tilt_axis(long img_num, Vector3<long> tile_size,
242 double tilt_axis, double tilt_offset, int flags)
243{
244 long i, xx, yy, ntiles(0);
245 double tan_a(tan(tilt_axis));
246 double inv_tan_a(1.0/tan_a);
247 double ratang(atan2(y, x));
248 Vector3<long> origin(size()/2);
249 vector<Vector3<long>> tile;
250
251 bool do_tile = set_tile_size(tile_size, size());
252
253 if ( fabs(tilt_axis) < ratang || fabs(tilt_axis) > M_PI - ratang ) {
254 origin[1] += (long)tilt_offset;
255 ntiles = x/tile_size[0];
256 tile.resize(ntiles);
257 for ( i=0, xx = origin[0] - (ntiles * tile_size[0])/2; i<ntiles; i++, xx += tile_size[0] ) {
258 tile[i][0] = xx;
259 tile[i][1] = (long) (origin[1] + (xx - origin[0])*tan_a);
260 }
261 } else {
262 origin[0] += (long)tilt_offset;
263 ntiles = y/tile_size[1];
264 tile.resize(ntiles);
265 for ( i=0, yy = origin[1] - (ntiles * tile_size[1])/2; i<ntiles; i++, yy += tile_size[1] ) {
266 tile[i][0] = (long) (origin[0] + (yy - origin[1])*inv_tan_a);
267 tile[i][1] = yy;
268 }
269 }
270
271 if ( verbose & VERB_DEBUG ) {
272 cout << "DEBUG Bimage::powerspectrum_tilted: img_num=" << img_num << endl;
273 cout << "DEBUG Bimage::powerspectrum_tilted: origin=" << origin << endl;
274 cout << "DEBUG Bimage::powerspectrum_tilted: tile_size=" << tile_size << endl;
275 cout << "DEBUG Bimage::powerspectrum_tilted: ntiles=" << ntiles << endl;
276// for ( i=0; i<ntiles; i++ )
277// cout << "DEBUG Bimage::powerspectrum_tilted: " << tile[i][0] << " " << tile[i][1] << endl;
278 }
279
280 // If a single image assume a micrograph that needs to be tiled
281 Bimage* pex = NULL;
282 if ( do_tile )
283 pex = extract_tiles(img_num, tile, tile_size);
284 else
285 pex = extract(img_num);
286
287 Vector3<long> translate;
288 Vector3<long> nusize(1,1,1);
289 if ( pex->sizeX() != pex->sizeY() ) {

307Bimage* Bimage::defocus_scale(long nn, double df, double df2, double iCL2, int fill_type)
308{
309 Bimage* ps = new Bimage(Float, TSimple, size(), 1);
310
311 long i, j, xx, yy;
312 double interval(1.0/(image[nn].sampling()[0]*x));
313 double s1, s2, f1(iCL2*df), f2(2*iCL2*df2), len, fill(image[nn].background());
314 vector<double> vs(x,0);
315
316 for ( size_t i=0; i<vs.size(); ++i ) {
317 s2 = interval*i;
318 s2 *= s2;
319 s1 = f1 - sqrt(f1*f1 + s2*s2 - f2*s2);
320 vs[i] = sqrt(s1)/interval;
321 }
322
323 Vector3<double> dis, coor, ori(size()/2);
324
325 for ( i=0, yy=0; yy<y; ++yy ) {
326 dis[1] = double(yy) - ori[1];
327 for ( xx=0; xx<x; ++xx, ++i ) {
328 dis[0] = double(xx) - ori[0];
329 len = dis.length();
330 coor = ori;
331 if ( len ) {
332 j = long(len);
333 f2 = len - j;
334 f1 = 1 - f2;
335 coor += dis * (f1*vs[j] + f2*vs[j+1])/len;
336 }
337 ps->set(i, interpolate(coor, nn, fill));
338 }
339 }
340
341 return ps;
342}

495double isotropy_R(Bsimplex& simp)
496{
497 long i;
498 double R(0), df;
499 vector<double>& f = simp.dependent_values();
500 vector<double>& x = simp.independent_values();
501
502 for ( i=0; i<simp.points(); i++ ) {
503 df = f[i] - (simp.parameter(0) + simp.parameter(1)*cos(2*(simp.parameter(2)+x[i])));
504 R += df*df;
505 }
506
507 R = sqrt(R/i);
508
509 return R;
510}

523vector<double> Bimage::powerspectrum_isotropy(long n, double& lores, double& hires)
524{
525 if ( lores <= 0 || lores > real_size()[0] ) lores = real_size()[0];
526 if ( lores < hires ) swap(lores, hires);
527 if ( hires < 2*sampling(0)[0] ) hires = 2*sampling(0)[0];
528
529 long i, j, xx, yy;
530 long kmin(real_size()[0]/lores);
531 long kmax(real_size()[0]/hires);
532 long na(TWOPI*kmax), h(x/2);
533
534 if ( verbose & VERB_PROCESS ) {
535 cout << "Power spectrum anisotropy:" << endl;
536 cout << "Origin: " << image[n].origin() << endl;
537 cout << "Resolution: " << hires << " - " << lores << " A" << endl;
538 cout << "Pixel radii: " << kmin << " - " << kmax << endl;
539 cout << "Angles: " << na << endl;
540 }
541
542 double dx, dy, r, a(0), f, v, y_avg(0), y_var(0), xy_scale(x/y);
543 vector<double> vx(na,0), vy(na,0);
544
545 for ( i=n*size().volume(), yy=0; yy<y; ++yy ) {
546 dy = (yy - image[n].origin()[1])*xy_scale;
547 if ( dy > h ) dy -= x;
548 for ( xx=0; xx<x; ++xx, ++i ) {
549 dx = xx - image[n].origin()[0];
550 if ( dx > h ) dx -= x;
551 r = sqrt(dx*dx + dy*dy);
552 if ( r >= kmin && r <= kmax ) {
553 v = (*this)[i];
554 a = (na/TWOPI)*atan2(dy, dx);
555 if ( a < 0 ) a += na;
556 j = long(a);
557 if ( j > na-1 ) cerr << "j too large!" << endl;
558 f = a - j;
559 vx[j] += 1-f;
560 vy[j] += (1-f)*v;
561 j++;
562 vx[j] += f;
563 vy[j] += f*v;
564 }
565 }
566 }
567
568 for ( i=0; i<na; ++i) {
569// cout << i << tab << x[i] << tab << y[i] << endl;
570 vy[i] /= vx[i];
571 vx[i] = TWOPI*i*1.0/na;
572 y_avg += vy[i];
573 y_var += vy[i]*vy[i];
574 }
575 y_avg /= na;
576 y_var /= na;
577 y_var -= y_avg*y_avg;
578
579// cout << "avg=" << y_avg << " var=" << y_var << endl;
580
581 Bsimplex simp(1, 3, 0, na, vx, vy);
582
583 simp.parameter(0, y_avg);
584 simp.parameter(1, 2*sqrt(y_var));
585 simp.parameter(2, 0);
586 simp.limits(0, 0.7*y_avg, 1.5*y_avg);
587 simp.limits(1, y_var, 5*sqrt(y_var));
588 simp.limits(2, -M_PI, M_PI);
589
590 double R = simp.run(10000, 1e-5, isotropy_R);
591
592 double aniso = simp.parameter(1)/simp.parameter(0);
593 vector<double> fit(3,0);
594 fit[0] = simp.parameter(0);
595 fit[1] = simp.parameter(1);
596 fit[2] = simp.parameter(2);
597 if ( fit[2] < -M_PI_2 ) fit[2] += M_PI;
598 if ( fit[2] > M_PI_2 ) fit[2] -= M_PI;
599
600// cout << "limit_lo=" << simp.limit_low(1) << endl;
601
602 if ( verbose & VERB_PROCESS ) {
603 cout << "Fitted function: " << fit[0] << " + " << fit[1] <<
604 " * cos(2*(a + " << fit[2]*180/M_PI << "))" << endl;
605 cout << "Anisotropy: " << aniso << endl;
606 cout << "R: " << R << endl << endl;
607 }
608
609 if ( verbose & VERB_FULL ) {
610 cout << "Angle\tPower\tFit" << endl;
611 for ( i=0; i<na; ++i)
612 cout << vx[i]*180.0/M_PI << tab << vy[i] << tab << fit[0] + fit[1]*cos(2*(vx[i]+fit[2])) << endl;
613 cout << endl;
614 }
615
616 return fit;
617}

619double Bimage::average_line(long xx, long yy, long zz, long nn, long len, int dir)
620{
621 long i, j;
622 double val(0);
623
624 if ( dir & 1 ) {
625 if ( xx < 0 ) {
626 len += xx;
627 xx = 0;
628 } else if ( xx + len >= x ) {
629 len += x - xx - 1;
630 }
631 for ( i=index(xx, yy, zz, nn), j=0; j<len; ++i, ++j ) val += (*this)[i];
632 } else if ( dir & 2 ) {
633 if ( yy < 0 ) {
634 len += yy;
635 yy = 0;
636 } else if ( yy + len >= y ) {
637 len += y - yy - 1;
638 }
639 for ( i=index(xx, yy, zz, nn), j=0; j<len; i+=x, ++j ) val += (*this)[i];
640 }
641
642 if ( len ) val /= len;
643
644 return val;

656long Bimage::fix_power_spectrum(int dir, double ratio)
657{
658 long nfix(0);
659 long i, xx, yy, zz, nn;
660 double v;
661
662 if ( dir & 1 ) { // Horizontal
663 yy = y/2;
664 for ( nn=0; nn<n; ++nn ) {
665 for ( zz=0; zz<z; ++zz ) {
666 for ( xx=0; xx<x; ++xx ) {
667 i = index(xx, yy, zz, nn);
668 v = (average_line(xx-1, yy-1, zz, nn, 3, dir) +
669 average_line(xx-1, yy+1, zz, nn, 3, dir))/2;
670 if ( (*this)[i] - v > ratio ) {
671 set(i, v);
672 nfix++;
673 }
674 }
675 }
676 }
677 }
678
679 if ( dir & 2 ) { // Vertical
680 xx = x/2;
681 for ( nn=0; nn<n; ++nn ) {
682 for ( zz=0; zz<z; ++zz ) {
683 for ( yy=0; yy<y; ++yy ) {
684 i = index(xx, yy, zz, nn);
685 v = (average_line(xx-1, yy-1, zz, nn, 3, dir) +
686 average_line(xx+1, yy-1, zz, nn, 3, dir))/2;
687 if ( (*this)[i] - v > ratio ) {
688 set(i, v);
689 nfix++;
690 }
691 }
692 }
693 }
694 }
695
696 return nfix;
697}
