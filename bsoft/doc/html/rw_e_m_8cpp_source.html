41int readEM(Bimage* p, int readdata)
42{
43 ifstream* fimg = new ifstream(p->file_name());
44 if ( fimg->fail() ) return -1;
45
46 EMhead* header = new EMhead;
47
48 fimg->read((char *)header, EMSIZE);
49 if ( fimg->fail() ) return -2;
50
51 // Determine byte order and swap bytes if necessary
52 unsigned char* b = (unsigned char *) header;
53 int i, sb = 0;
54 if ( abs(header->nz) > SWAPTRIG || header->nz < 1 ) {
55 if ( verbose & VERB_PROCESS )
56 cerr << "Warning: Swapping header byte order for 4-byte types" << endl;
57 sb = 1;
58 int extent = EMSIZE; // exclude labels from swapping
59 for ( i=4; i<16; i+=4 ) swapbytes(b+i, 4); // Image size fields
60 for ( i=96; i<extent; i+=4 ) swapbytes(b+i, 4); // Extras
61 }
62
63 // Map the parameters
64 p->size(header->nx, header->ny, header->nz);
65 p->images(1);
66 p->channels(1);
67
68 switch( header->stamp[3] ){
69 case 1 : p->data_type(UCharacter); break; // Images
70 case 2 : p->data_type(Short); break;
71 case 4 : p->data_type(Integer); break;
72 case 5 : p->data_type(Float); break;
73 case 8 : p->data_type(Float); // Transform
74 p->compound_type(TComplex); p->channels(2);
75 break;
76 default : p->data_type(UCharacter);
77 }
78
79 p->data_offset(EMSIZE);
80 p->label(header->label);
81
82
83 long readsize = p->alloc_size();
84 unsigned char* data = NULL;
85
86 if ( readdata ) {
87 p->data_alloc();
88 if ( p->compound_type() == TSimple ) {
89 fread_large(p->data_pointer(), readsize, p->data_offset(), fimg);
90 if ( sb ) swapbytes(readsize, p->data_pointer(), p->data_type_size());

95 if ( sb ) swapbytes(readsize, data, p->data_type_size());
96 p->unpack_transform(data, Hermitian);
97 delete[] data;
98 }
99 }
100
101 fimg->close();

103

115int writeEM(Bimage* p)
116{
117 p->color_to_simple();
118
119 if ( p->compound_type() == TComplex ) p->change_type(Float);
120
121 switch ( p->data_type() ) {
122 case Bit: case UCharacter: case SCharacter:
123 p->change_type(UCharacter); break;
124 case UShort: case Short:

129 p->change_type(Float); break;
130 }
131
132 switch ( p->data_type() ) {
133 case SCharacter: p->change_type(UCharacter); break;
134 case UShort: p->change_type(Short); break;
135 default: break;
136 }
137
138 EMhead* header = new EMhead;
139 memset(header, 0, sizeof(EMhead));
140
141 // Map the parameters
142 header->nx = p->sizeX();
143 header->ny = p->sizeY();
144 header->nz = p->sizeZ();
145 switch ( systype(0) ) {
146 case BigIEEE: header->stamp[0] = 3; break; // SGI machine stamp
147 case LittleIEEE: header->stamp[0] = 6; break; // Alpha, Intel
148 case LittleVAX: header->stamp[0] = 1; break; // VAX
149 default: break;
150 }
151 switch ( p->data_type() ) {
152 case UCharacter : header->stamp[3] = 1; break; // Images
153 case Short : header->stamp[3] = 2; break;
154 case Integer : header->stamp[3] = 4; break;
155 case Float : header->stamp[3] = 5;
156 if ( p->compound_type() == TComplex ) header->stamp[3] = 8;
157 break; // Transform
158 default : header->stamp[3] = 1;
159 }
160
161 p->label().copy(header->label, 80);
162
163 p->data_offset(EMSIZE);
164
165 // If a transform, then the physical storage in x is only half+1

168
169 int datatypesize = p->channels()*p->data_type_size();
170 long datasize = (long) xstore*header->ny*header->nz*datatypesize;

179 if ( p->data_pointer() ) {

181 fimg.write((char *)p->data_pointer(), datasize);
182 } else {
183 data = new unsigned char[datasize];
