31Bimage* Bimage::project(char axis, int flags)
32{
33 long i, j, xx, yy, zz, cc, nn;
34 double v, scale;
35 Vector3<double> u(image->sampling());
36
37 Bimage* proj = NULL;

46 } else {

48 scale = 1.0L/z;
49 proj->sampling(u[0], u[1], 1);

52 long ds(n*x*y*c);
53
54 if ( flags==3 ) proj->fill(min);
55 else if ( flags==2 ) proj->fill(max);
56
57 if ( verbose & VERB_LABEL )
58 cout << "Projecting along " << axis << " (" << flags << ")" << endl << endl;
59
60 // First calculate projection
61 for ( i=nn=0; nn<n; nn++ ) {
62 if ( axis == 'x' ) {
63 proj->image[nn].origin(image[nn].origin()[2], image[nn].origin()[1], 0.0);
64 } else if ( axis == 'y' ) {
65 proj->image[nn].origin(image[nn].origin()[0], image[nn].origin()[2], 0.0);
66 } else {
67 proj->image[nn].origin(image[nn].origin()[0], image[nn].origin()[1], 0.0);
68 }
69 for ( zz=0; zz<z; zz++ )
70 for ( yy=0; yy<y; yy++ )
71 for ( xx=0; xx<x; xx++ )
72 for ( cc=0; cc<c; cc++, i++ ) {
73 if ( axis == 'x' )
74 j = ((nn*y + yy)*z + zz)*c + cc;
75 else if ( axis == 'y' )
76 j = ((nn*z + zz)*x + xx)*c + cc;
77 else

81 if ( (*proj)[j] < v ) proj->set(j, v);
82 } else if ( flags==2 ) {
83 if ( (*proj)[j] > v ) proj->set(j, v);
84 } else {
85 proj->add(j, v);
86 }
87 }
88 }
89

115Bimage* Bimage::rotate_project(Matrix3 mat, Vector3<double> translate,
116 double radial_cutoff, bool norm_flag)
117{
118 long i, j, xx, yy, zz;
119 long xo, yo, xp, yp, yw;
120 double oldz_sq, oldyz_sq;
121 double xf, yf, maxlen(0);
122 Vector3<double> old, newvec;
123
124 if ( radial_cutoff < 1 ) radial_cutoff = x;
125 double rad_sq = radial_cutoff*radial_cutoff;
126
127 Vector3<double> oldorigin(image->origin());
128
129 Vector3<double> neworigin = oldorigin + translate;
130 neworigin[2] = 0;
131
132 Matrix3 matinv = mat.transpose();
133
134 if ( verbose & VERB_FULL ) {
135 cout << "Rotating around the map center and projecting:" << endl;
136 cout << "Rotation origin: " << oldorigin << endl;
137 cout << matinv << endl;
138 if ( translate[0] != 0 || translate[1] != 0 || translate[2] != 0 )
139 cout << "Translation: " << translate << endl;
140 cout << "Radial cutoff: " << radial_cutoff << " pixels" << endl;
141 cout << endl;
142 }
143
144 Bimage* proj = new Bimage(Float, TSimple, x, y, 1, 1);
145 proj->sampling(image->sampling()[0], image->sampling()[1], 1);
146 proj->origin(neworigin[0], neworigin[1], 0);
147
148 double bkg(background(long(0)));
149 double value;
150 long slice_size(x*y);
151
152 for ( zz=0; zz<z; zz++ ) {
153 old[2] = zz - oldorigin[2];
154 oldz_sq = old[2]*old[2];
155 if ( oldz_sq <= rad_sq ) for ( yy=0; yy<y; yy++ ) {
156 old[1] = yy - oldorigin[1];
157 oldyz_sq = oldz_sq + old[1]*old[1];
158 if ( oldyz_sq <= rad_sq ) for ( xx=0; xx<x; xx++ ) {
159 old[0] = xx - oldorigin[0];
160 if ( old[0]*old[0] + oldyz_sq <= rad_sq ) {
161 newvec = (matinv * old) + neworigin;
162 yo = (long) newvec[1];
163 if ( ( yo >= 0 ) && ( yo < y - 1 ) ) {
164 xo = (long) newvec[0];
165 if ( ( xo >= 0 ) && ( xo < x - 1 ) ) {
166 if ( maxlen < fabs(newvec[2]) ) maxlen = fabs(newvec[2]);
167 yf = newvec[1] - yo;
168 xf = newvec[0] - xo;
169 i = (zz*y + yy)*x + xx;
170 value = (*this)[i] - bkg;
171 for ( yp=yo; yp<yo+2; yp++ ) {
172 yw = yp*x;
173 yf = 1 - yf;
174 for ( xp=xo; xp<xo+2; xp++ ) {
175 j = yw + xp;
176 xf = 1 - xf;
177 proj->add(j, value*xf*yf);
178 }
179 }
180 }
181 }
182 }
183 }
184 }
185 }
186
187 if ( norm_flag )
188 for ( i=0; i<slice_size; i++ ) proj->set(i, (*proj)[i] / maxlen);
189
190 proj->statistics();
191
192 return proj;
193}

278Bimage* Bimage::central_section(Matrix3 mat, double resolution, FSI_Kernel* kernel, double wavelength)
279{
280 if ( image->sampling()[0] < 1e-10 ) sampling(1,1,1);
281 if ( resolution < 1e-10 ) resolution = image->sampling()[0];
282
283 long i, xx, yy;
284 long hx((x - 1)/2), hy((y - 1)/2);
285 Vector3<double> inv_scale(1.0/real_size());
286 Vector3<double> m, iv;
287 double maxs2(1.0/(resolution*resolution));
288 double sx2, sy2, s2;
289 double d2(wavelength/2.0);
290
291 if ( verbose & VERB_FULL ) {
292 cout << "Calculating a central section:" << endl;
293 cout << "Resolution limit: " << resolution << " A "<< endl;
294 if ( wavelength )
295 cout << "Ewald sphere for wavelength: " << wavelength << " A" << endl;
296 cout << mat << endl << endl;
297 }
298
299 Bimage* proj = new Bimage(Float, TComplex, x, y, 1, 1);
300 proj->fourier_type(Standard);
301 proj->image[0] = image[0];
302 proj->sampling(image->sampling()[0], image->sampling()[1], 1);
303 proj->origin(image->origin()[0], image->origin()[1], 0);
304
305 for ( i=yy=0; yy<y; yy++ ) {
306 iv[1] = yy;
307 if ( iv[1] > hy ) iv[1] -= y;
308 iv[1] *= inv_scale[1];
309 sy2 = iv[1]*iv[1];
310 for ( xx=0; xx<x; xx++, i++ ) {
311 iv[0] = xx;
312 if ( iv[0] > hx ) iv[0] -= x;
313 iv[0] *= inv_scale[0];
314 sx2 = iv[0]*iv[0];
315 s2 = sx2 + sy2;
316 if ( s2 <= maxs2 ) {
317 if ( wavelength ) iv[2] = d2*s2;
318 m = mat * iv;
319 m *= real_size();
320 proj->set(i, fspace_interpolate(0, m, kernel));
321 }
322 }
323 }
324
325 proj->statistics();
326
327 return proj;
328}

443/*
444@author Bernard Heymann and Peter Leong
445@brief Back-projects a 2D image into a 3D volume.
446@param *p a 2D image.
447@param resolution high resolution limit (angstrom).
448@param axis tilt axis for single tilt series (radians).
449@param planf 2D forward Fourier transform plan.
450@param planb 2D backward Fourier transform plan.
451@return int 0.
452
453 Requirements:
454 The image must be 2D with the orientation parameters in
455 the sub-image structure.
456 The orientation parameters are defined as those giving the projection,
457 and is reversed for backprojection.
458 The 2D image is converted to floating point and rescaled to an average
459 of zero and a standard deviation of one. The 3D map is traversed and
460 the density added from the 2D image projected according to the orientation
461 parameters. The origin of the new map is in the center of the volume.
462
463**/
464int Bimage::back_project(Bimage* p, double resolution, double axis,
465 fft_plan planf, fft_plan planb)
466{
467 if ( !p->data_pointer() ) return 0;
468
469 p->change_type(Float);
470 if ( axis ) p->fspace_weigh_ramp(resolution, axis, planf, planb);
471 else p->fspace_weigh_ramp(resolution, planf, planb);
472
473 p->statistics();
474 p->rescale_to_avg_std(0, 1);
475 p->calculate_background();
476
477 long i, xx, yy, zz;
478 double fill(p->background(long(0)));
479 Vector3<double> oldorigin(p->image->origin()), neworigin(image->origin());
480 Vector3<double> oldcoor, newcoor;
481 Vector3<double> scale(p->image->magnification(), p->image->magnification(), 1);
482
483 // The rotation matrix is transposed because the orientation parameters
484 // give the orientation as rotated from the reference view
485 Matrix3 mat = p->image->view().matrix();
486 mat = mat.transpose();
487 mat = scale*mat;
488// cout << mat << endl;
489
490 if ( verbose & VERB_LABEL )
491 cout << "Backprojecting" << endl << endl;
492
493 for ( i=zz=0; zz<z; zz++ ) {
494 newcoor[2] = zz - neworigin[2];
495 for ( yy=0; yy<y; yy++ ) {
496 newcoor[1] = yy - neworigin[1];
497 for ( xx=0; xx<x; xx++, i++ ) {
498 newcoor[0] = xx - neworigin[0];
499 oldcoor = mat * newcoor + oldorigin;
500 oldcoor[2] = 0;
501 add(i, p->interpolate(oldcoor, 0, fill));
502 }
503 }
504 }
505
506 return 0;
507}

522int Bimage::ewald_sphere(double volt, double t)
523{
524 double wl = electron_wavelength(volt);
525 double dz = image->sampling()[2];
526
527 if ( t < 1 || t > dz*z ) t = dz*z;
528
529 fft(FFTW_FORWARD, 1);
530
531 phase_shift_to_origin();
532
533 bool use;
534 long i, xx, yy, zz;
535 double u, v, w, s2, fac, phi;
536 Vector3<long> h(size()/2);
537 Vector3<double> fscale(1.0/real_size());
538
539 if ( verbose ) {
540 cout << "Imposing Ewald spheres:" << endl;
541 cout << "Acceleration voltage: " << volt << " V" << endl;
542 cout << "Wavelength: " << wl << " A" << endl;
543 cout << "Thickness: " << t << " A" << endl << endl;
544 }
545
546 Bimage* prop = new Bimage(Float, TComplex, x, y, z, 1);
547
548 for ( i=zz=0; zz<z; ++zz ) {
549 w = (zz < h[2])? zz: zz-z;
550 w *= dz;
551 if ( fabs(w) <= t/2 ) use = 1;
552 else use = 0;
553 fac = M_PI*wl*w;
554 for ( yy=0; yy<y; ++yy ) {
555 v = (yy < h[1])? yy: yy-y;
556 v *= fscale[1];
557 for ( xx=0; xx<x; ++xx, ++i ) {
558 u = (xx < h[0])? xx: xx-x;
559 u *= fscale[0];
560 s2 = u*u + v*v;
561 phi = fac*s2;
562 if ( use ) prop->set(i, Complex<float>(cosl(phi), sinl(phi)));
563 }
564 }
565 }
566
567 prop->fftz(FFTW_FORWARD, 2);
568
569 complex_product(prop);
570
571 delete prop;
572
573 return 0;
574}
