322 change = 0;
323 ucount++;
324 }
325 }

328 particle->fom[0] = particleo->fom[0];

342 cout << "Total number of particles: " << total << endl;
343 cout << "Number of unchanged particles: " << ucount << endl;
344 cout << "FOM-difference threshold: " << fom_diff << endl << endl;

347 }
