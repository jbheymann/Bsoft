39/*
40@brief Functions for 2D crystallographic indexing.
41 Diffraction spot interface syntax:
42 Bmg spot [item] [action] [params]
43 Actions:
44 count counts the number of spots
45 ids returns the spot indices (3 per spot)
46 reindex [h1] [k1] [l1] [h2] [k2] [l2] re-indexes a spot
47 location [h] [k] [l] returns the location of a spot
48 fom [h] [k] [l] returns the FOM of a spot
49 select [x] [y] [z] returns the spot indices at the given location
50 move [h] [k] [l] [dx] [dy] [dz] updates a spot location by the given vector
51 create [h] [k] [l] [x] [y] [z] creates a new spot at the given location
52 delete [all|mg|rec|h] [k] [l] deletes spots
53 unitcell calculates unit cell vectors
54 generate [m] generates spots
55 mask [r] masks spots
56 Parameters:
57 h k l Miller indices
58 x x location
59 y y location
60 z z location
61 dx change in x
62 dy change in y
63 dz change in z
64 m limit in reciprocal space for generating spots
65 r mask radius
66*/

68Tcl_Obj* do_spot(Bproject* project, int objc, Tcl_Obj *CONST objv[])
69{
70 Tcl_Obj* returnObj = NULL;
71
72 Bstring item = Tcl_GetStringFromObj(objv[2], NULL);
73 if ( verbose & VERB_DEBUG )
74 cout << "DEBUG do_spot: Item: " << item << " (" << item.length() << ")" << endl;
75
76 Bstring id = item.post(':');
77 id = id.remove(' ');
78 item = item.pre(':');
79
80 if ( !item.contains("Field") && !item.contains("Micrograph") ) {
81 cerr << "Error in do_spot: Item " << item << " must be a micrograph!" << endl;
82 returnObj = Tcl_NewObj();
83 return returnObj;
84 }
85
86 Bfield* field = NULL;
87 Bmicrograph* mg = NULL;
88 Bimage* p = NULL;
89 Bstring filename, filename2, base, delete_what;
90
91 if ( item.contains("Field") ) {
92 for ( field = project->field; field && field->id != id; field = field->next ) ;
93 if ( field ) mg = field->mg;
94 } else if ( item.contains("Micrograph") ) {
95 for ( field = project->field; field; field = field->next ) {
96 for ( mg = field->mg; mg && mg->id != id; mg = mg->next ) ;
97 if ( mg ) break;
98 }
99 }
100
101 if ( mg ) {
102 filename = mg->fft;
103 if ( filename.length() < 1 ) filename = mg->fps;
104 }
105
106 if ( filename.length() ) {
107 base = filename.base();
108 for ( p = imglist; p; p = p->next ) {
109 filename2 = p->file_name();
110 if ( filename2.base() == base ) break;
111 }
112 }
113
114 Bstring action = Tcl_GetStringFromObj(objv[3], NULL);
115 if ( verbose & VERB_DEBUG )
116 cout << "DEBUG do_spot: Action: " << action << " (" << action.length() << ")" << endl;
117
118 if ( action == "count" ) {
119 returnObj = spot_count(mg);
120 } else if ( action == "ids" ) {
121 returnObj = spot_ids(mg, objc, objv);
122 } else if ( action == "reindex" ) {
123 returnObj = spot_reindex(mg, objc, objv);
124 } else if ( action == "location" ) {
125 returnObj = spot_location(mg, objc, objv);
126 } else if ( action == "fom" ) {
127 returnObj = spot_fom(mg, objc, objv);
128 } else if ( action == "select" ) {
129 returnObj = spot_select(mg, objc, objv);
130 } else if ( action == "move" ) {
131 returnObj = spot_move(mg, objc, objv);
132 } else if ( action == "create" ) {
133 returnObj = spot_create(mg, objc, objv);
134 } else if ( action == "delete" ) {
135 returnObj = spot_delete(mg, objc, objv);
136 } else if ( action == "unitcell" ) {
137 returnObj = spot_unitcell_vectors(mg, objc, objv);
138 } else if ( action == "generate" ) {
139 returnObj = spot_generate(mg, p, objc, objv);
140 } else if ( action == "mask" ) {
141 if ( p ) returnObj = spot_mask(mg, p, objc, objv);
142 } else {
143 cerr << "Error: Action " << action << " not recognized!" << endl;
144 }
145
146 if ( !returnObj ) {
147 returnObj = Tcl_NewObj();
148 Tcl_SetIntObj(returnObj, 0);
149 }
150
151 return returnObj;
152}

297 dmin = 2*rad;
298 for ( sf = sflist; sf; sf = sf->next ) {

303 }
304 }
