330## @brief Draws a box or bad area
331#
332# Draws a box and circle with the current box size and number
333# for a selected box, or draws a red circle with the current bad size
334# for a selected bad area.
335#
336# @param id Box or bad area identifier.
337
338proc boxDraw { id } {
339 global project_item theimg
340 global box_selected box_color_mode
341 global show_box_labels box_color box_select_color bad_color bad_select_color
342 global fom_min fom_max
343 global sel_min sel_max
344 if ![winfo exists .wbox] { return }
345 if { $id == 0 } { return }
346 set c [getImageCanvas $theimg]
347 set wc [getControlWindow $theimg]
348 set n [$wc.image.scale get]
349 set z [$wc.slice.scale get]
350 set mg_item [micrographItem $n]
351 set loc [Bmg box $mg_item location $id]
352# puts "Location for $mg_item $id: $loc"
353 if { [lindex $loc 0] < 0 } {
354 puts "ID $id not found!"
355 return
356 }
357 set scale [$wc.scale.scale get]
358 set scale_1 [expr $scale - 1]
359 set yi [yFlip [lindex $loc 1]]
360 set xr [expr [lindex $loc 0] * $scale ]
361 set yr [expr $yi * $scale ]
362 set zr [expr abs([lindex $loc 2] - $z) ]
363 if { $id > 0 } {
364 set box_radius_x [expr [.wbox.box_size.x get] * 0.5]
365 set box_radius_y [expr [.wbox.box_size.y get] * 0.5]
366 set box_radius_z [expr [.wbox.box_size.z get] * 0.5]
367 if { $zr > $box_radius_z } { return }
368 set zr2 1
369 if { $box_radius_z > 0 } {
370 set zr [expr $zr*1.0/$box_radius_z]
371 set zr2 [expr sqrt(1 - $zr*$zr)]
372 }
373# puts "$xr $yr"
374 set box_radius_x [expr $box_radius_x * $scale ]
375 set box_radius_y [expr $box_radius_y * $scale ]
376 set oval_radius_x [expr $box_radius_x*$zr2]
377 set oval_radius_y [expr $box_radius_y*$zr2]
378 set xmin [expr $xr - $box_radius_x ]
379 set xmax [expr $xr + $box_radius_x + $scale_1 ]
380 set ymin [expr $yr - $box_radius_y ]
381 set ymax [expr $yr + $box_radius_y + $scale_1 ]
382 set cxmin [expr $xr - $oval_radius_x ]
383 set cxmax [expr $xr + $oval_radius_x + $scale_1 ]
384 set cymin [expr $yr - $oval_radius_y ]
385 set cymax [expr $yr + $oval_radius_y + $scale_1 ]
386 set idtag [format "box%d" $id]
387 if { [string compare $box_color_mode "FOM"] == 0 } {
388 set fom [Bmg box $mg_item fom $id]
389 set color [spectrum $fom $fom_min $fom_max]
390 } elseif { [string compare $box_color_mode "Selection"] == 0 } {
391 set sel [Bmg box $mg_item select $id]
392 set color [spectrum $sel $sel_min $sel_max]
393 } elseif { [lsearch $box_selected $id] >= 0 } {
394 set color $box_select_color;
395 } else {
396 set color $box_color;
397 }
398 $c create rect $xmin $ymin $xmax $ymax -outline $color -width 1 -tags [list box $idtag]
399 $c create oval $cxmin $cymin $cxmax $cymax -outline $color -width 1 -tags [list box $idtag]
400 if { $show_box_labels } {
401 $c create text $xr $yr -text $id -fill $color -tags [list box $idtag]
402 }
403 } elseif { $id < 0 } {
404 set bad_radius [.wbox.bad_radius.e get]
405 if { $zr > $bad_radius } { return }
406 set bad_radius [expr $scale * sqrt($bad_radius*$bad_radius - $zr*$zr)]
407 set xmin [expr $xr - $bad_radius ]
408 set xmax [expr $xr + $bad_radius + $scale_1 ]
409 set ymin [expr $yr - $bad_radius ]
410 set ymax [expr $yr + $bad_radius + $scale_1 ]
411 if { [lsearch $box_selected $id] >= 0 } {
412 set color $bad_select_color;
413 } else {
414 set color $bad_color;
415 }
416 $c create oval $xmin $ymin $xmax $ymax -outline $color -width 1 -tags bad
417 }
418}
