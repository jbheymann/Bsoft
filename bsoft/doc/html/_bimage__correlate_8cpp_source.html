90double Bimage::correlate(Bimage* p, double rmin, double rmax, Bimage* pmask, int flag)
91{
92 if ( n > 1 || p->images() > 1 ) {
93 cerr << "Error: Only single images are supported for correlation!" << endl;
94 return -1.0;
95 }
96
97 if ( !check_if_same_size(p) ) {
98 error_show("Bimage::correlate", __FILE__, __LINE__);
99 return -1;
100 }
101
102 if ( std <= 0 || p->std <= 0 ) return -1;
103
104 if ( rmin > x/2 ) rmin = 0;
105 if ( rmax <= rmin ) rmax = size().max();
106
107 if ( verbose & VERB_PROCESS ) {
108 cout << "Calculating a correlation map:" << endl;
109 cout << "Origin: " << image->origin() << endl;
110 cout << "Radii: " << rmin << " - " << rmax << endl;
111 if ( pmask )
112 cout << "Mask: " << pmask->file_name() << endl;
113 }
114
115 int use_this;
116 long i, xx, yy, zz, ns(0);
117 double v1, v2, dx, dy, dz, d2, rmin2(rmin*rmin), rmax2(rmax*rmax);
118 double avg1(0), avg2(0), std1(0), std2(0), c12(0), denom, cc(0);
119
120 for ( i=zz=0; zz<z; zz++ ) {
121 dz = (double)zz - image->origin()[2];
122 dz *= dz;
123 for ( yy=0; yy<y; yy++ ) {
124 dy = (double)yy - image->origin()[1];
125 dy *= dy;
126 for ( xx=0; xx<x; xx++, i++ ) {
127 dx = (double)xx - image->origin()[0];
128 dx *= dx;
129 d2 = dx + dy + dz;
130 use_this = 1;
131 if ( pmask && (*pmask)[i] < 1 ) use_this = 0;
132 if ( d2 < rmin2 || d2 > rmax2 ) use_this = 0;
133 if ( use_this ) {
134 v1 = (*this)[i];
135 v2 = (*p)[i];
136 avg1 += v1;
137 avg2 += v2;
138 std1 += v1*v1;
139 std2 += v2*v2;
140 c12 += v1*v2;
141 ns++;
142 } else if ( flag ) {
143 p->set(i, 0);
144 }
145 }
146 }
147 }
148
149 if ( ns ) {
150 avg1 /= ns;
151 avg2 /= ns;
152 std1 = std1/ns - avg1*avg1;
153 std2 = std2/ns - avg2*avg2;
154 if ( std1 > 0 ) std1 = sqrt(std1);
155 else std1 = 0;
156 if ( std2 > 0 ) std2 = sqrt(std2);
157 else std2 = 0;
158 c12 /= ns;
159 }
160
161 denom = std1*std2;
162 if ( denom <= 0 ) {
163// cerr << "Error in Bimage::correlate: One or both standard deviations are zero!" << endl;
164 return 0;
165 }
166
167 denom = 1/denom;
168
169 cc = (c12 - avg1*avg2) * denom;
170
171 if (flag ) {
172 for ( i=0; i<datasize; i++ ) if ( (*p)[i] )
173 p->set(i, ((*this)[i] - avg1)*((*p)[i] - avg2)*denom);
174 p->statistics();
175 }
176
177 if ( verbose & VERB_PROCESS ) {
178 cout << "Averages: " << avg1 << " " << avg2 << endl;
179 cout << "Standard deviations: " << std1 << " " << std2 << endl;
180 cout << "Correlation: " << cc << " (" << ns << ")" << endl << endl;
181 }
182
183 return cc;

265int Bimage::auto_correlate(double hires, double lores)
266{
267 if ( hires > lores ) swap(hires, lores);
268
269 long nn, i, xx, yy, zz;
270 long ix, iy, iz;
271 double s2, sx2, sy2, sz2, smin2(0), smax2(0), sum, scale;
272 Vector3<double> u(image->sampling());
273 Vector3<double> realsize(real_size());
274
275 if ( lores > 0 ) smin2 = 1/(lores*lores);
276
277 if ( hires > 0 ) {
278 smax2 = 1/(hires*hires);
279 } else {
280 if ( x > 1 ) smax2 += 1.0/(4.0*u[0]*u[0]);
281 if ( y > 1 ) smax2 += 1.0/(4.0*u[1]*u[1]);
282 if ( z > 1 ) smax2 += 1.0/(4.0*u[2]*u[2]);
283 hires = 1/sqrt(smax2);
284 }
285
286 if ( verbose & VERB_FULL ) {
287 cout << "Auto-correlation:" << endl;
288 if ( lores > 0 )
289 cout << "Resolution range: " << hires << " - " << lores << " A" << endl;
290 else
291 cout << "No resolution limits" << endl;
292 cout << endl;
293 }
294
295 fft();
296
297 Complex<double> cv;
298// set(0, cv);
299
300 for ( i=nn=0; nn<n; nn++ ) {
301 sum = 0;
302 for ( zz=0; zz<z; zz++ ) {
303 iz = zz;
304 if ( zz > (z - 1)/2 ) iz -= z;
305 sz2 = iz/realsize[2];
306 sz2 *= sz2;
307 for ( yy=0; yy<y; yy++ ) {
308 iy = yy;
309 if ( yy > (y - 1)/2 ) iy -= y;
310 sy2 = iy/realsize[1];
311 sy2 *= sy2;
312 for ( xx=0; xx<x; xx++, i++ ) {
313 ix = xx;
314 if ( xx > (x - 1)/2 ) ix -= x;
315 sx2 = ix/realsize[0];
316 sx2 *= sx2;
317 s2 = sx2 + sy2 + sz2;
318 if ( s2 >= smin2 && s2 <= smax2 ) {
319 cv = (complex(i)).power();
320 sum += cv.real();
321 } else {
322 cv = 0;
323 }
324 set(i, cv);
325 }
326 }
327 }
328 if ( sum ) {
329 scale = sqrt(x*y*z)/sum;
330// cout << "scale=" << scale << endl;
331 multiply(nn, scale);
332 }
333 image[nn].origin(0.0,0.0,0.0);
334 }
335
336// cout << "backtransforming" << endl;
337
338 fft_back();
339
340 return 0;
341}
