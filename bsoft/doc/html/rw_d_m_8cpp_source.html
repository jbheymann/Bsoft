41int readDM(Bimage* p, int readdata, int img_select)
42{
43 ifstream* fimg = new ifstream(p->file_name());
44 if ( fimg->fail() ) return -1;
45
46 // Read a small block at the beginning to see if it is the fixed or tagged format
47 char buf[1024];
48 fimg->read(buf, 4);
49 if ( fimg->fail() ) return -2;
50
51// version = *((int *) buf);
52 version = buf[0];
53
54 if ( version < 1 || version > 4 ) version = buf[3];
55
56 if ( show == 1 )
57 cout << "Magic number = " << version << endl;
58
59 fimg->seekg(0, ios::beg);
60
61 switch ( version ) {
62 case 0: readFixedDMHeader(fimg, p, readdata); break;
63 case 3:
64 case 4: readTagGroupWithVersion(fimg, p, readdata, img_select); break;
65 default:
66 cout << "Digital Micrograph format version " << version << " not supported!" << endl;
67 }
68
69 fimg->close();
70 delete fimg;
71
72 return 0;
73}

81int writeDM(Bimage* p)
82{
83 DMhead* header = new DMhead;
84 memset(header, 0, sizeof(DMhead));
85
86 long datatypesize = p->channels()*p->data_type_size();
87 long datasize = p->data_size();
88
89 // Map the parameters
90 header->endian = 65535;
91 header->xSize = p->sizeX();
92 header->ySize = p->sizeY();
93 header->zSize = p->sizeZ();
94 header->depth = datatypesize;
95 switch ( p->data_type() ) {
96 case UCharacter : header->type = BINARY_DATA; break;
97 case UShort : header->type = UNSIGNED_INT16_DATA; break;
98 case Short : header->type = SIGNED_INT16_DATA; break;
99 case Integer : header->type = SIGNED_INT32_DATA; break;
100 case Float : header->type = ( p->compound_type() == TComplex )? COMPLEX8_DATA: REAL4_DATA; break;
101 case Double : header->type = ( p->compound_type() == TComplex )? COMPLEX16_DATA: REAL8_DATA; break;
102 default : header->type = NULL_DATA;
103 }
104
105 p->data_offset(24);
106
107 ofstream fimg(p->file_name().c_str());
108 if ( fimg.fail() ) return -1;
109
110 fimg.write((char *)header, p->data_offset());
111 fimg.write((char *)p->data_pointer(), datasize);
112
113 fimg.close();
114
115 delete header;
116
117 return 0;
118}

189int readFixedDMHeader(ifstream* fimg, Bimage* p, int readdata)
190{
191 DMhead* header = new DMhead;
192 DMMachead* macheader = (DMMachead *) header;
193
194// if ( fread( header, sizeof(DMhead), 1, fimg ) < 1 ) return -2;
195 fimg->read((char *)header, sizeof(DMhead));
196 if ( fimg->fail() ) return -2;
197
198 if ( verbose & VERB_DEBUG || show )
199 cout << "DEBUG readFixedDMHeader: macheader=" << macheader <<
200 " width=" << macheader->width << " height=" << macheader->height << endl;
201
202 p->data_offset(8);
203 if ( ( macheader->width < 1 ) || ( macheader->height < 1 ) ) {
204 macheader = (DMMachead *) ((char*)header + 6);
205 p->data_offset(p->data_offset() + 6);
206 }
207
208 if ( verbose & VERB_DEBUG )
209 cout << "DEBUG readFixedDMHeader: macheader=" << macheader <<
210 " width=" << macheader->width << " height=" << macheader->height << endl;
211
212 // Determine header type
213 int i, fixed(1), sb(0);
214/* if ( macheader->width == 0 && macheader->height == -1 ) {
215 fixed = 1;
216 sb = 0;
217 } else if ( macheader->width == -1 && macheader->height == 0 ) {
218 fixed = 1;
219 sb = 1;
220 } else if ( systype(0) >= LittleIEEE ) sb = 1;
221*/
222 // Swap bytes if necessary
223 unsigned char* b = (unsigned char *) header;
224 if ( sb ) {
225 if ( fixed ) {
226 if ( verbose & VERB_PROCESS )
227 cerr << "Warning: Swapping header byte order for 4-byte types" << endl;
228 for ( i=0; i<24; i+=4 ) swapbytes(b+i, 4);
229 } else {
230 b = (unsigned char *) macheader;
231 if ( verbose & VERB_PROCESS )
232 cerr << "Warning: Swapping header byte order for 2-byte types" << endl;
233 for ( i=0; i<8; i+=2 ) swapbytes(b+i, 2);
234 }
235 }
236
237 // Map the parameters
238 p->images(1);
239 p->channels(1);
240 p->compound_type(TSimple);
241 if ( fixed ) {
242 p->size(header->xSize, header->ySize, header->zSize);
243 p->data_offset(24);
244 datatype_from_dm3_type(header->type, p);
245 } else {
246 p->size(macheader->width, macheader->height, 1);
247 switch( macheader->type ) {
248 case 6 : p->data_type(UCharacter); break;
249 case 14 : p->data_type(UCharacter); break;
250 case 9 : p->data_type(SCharacter); break;
251 case 10 : p->data_type(UShort); break;
252 case 1 : p->data_type(Short); break;
253 case 7 : p->data_type(Integer); break;
254 case 2 : p->data_type(Float); break;
255 case 3 : p->data_type(Float); p->compound_type(TComplex); p->channels(2); break;
256 case 13 : p->data_type(Double); p->compound_type(TComplex); p->channels(2); break;
257 default : p->data_type(UCharacter);
258 }
259 }
260
261
262 delete header;
263
264 if ( readdata ) {
265 fimg->read((char *)p->data_pointer(), p->alloc_size());
266 if ( fimg->fail() ) return -3;
267 if ( sb ) swapbytes(p->alloc_size(), p->data_pointer(), p->data_type_size());
268 }
269
270 fimg->close();
271
272 return sb;
273}

318 case 7: dvalue = *buf.d; break;
319 case 8: ivalue = *buf.uc; break;
320 case 9: ivalue = *buf.uc; break;
321 case 10: ivalue = *buf.uc; break;
322 case 11: ivalue = *buf.sl; break;
323 case 12: ivalue = *buf.ul; break;

328 if ( !dvalue ) dvalue = ivalue;

330 if ( show == 1 ) cout << tab << dvalue;

334 if ( dm3_type == 6 || dm3_type == 7 )
