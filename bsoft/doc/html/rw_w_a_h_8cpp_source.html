28int readWAH(Bstring& filename, Bmolgroup *molgroup)
29{
30 if ( verbose & VERB_DEBUG )

52 strncpy(restype, aline + 3, 4);
53 resnum = get_integer(aline + 7, 4);
54 strncpy(atomtype, aline + 11, 4);
55 restype[4] = 0; // Residue names only 4 characters
56 atomtype[4] = 0; // Atom names only 4 characters
57 if ( n < 1 || resnum != prev_resnum ) {
58 if ( verbose & VERB_DEBUG )
59 cout << "DEBUG readWAH: resnum=" << resnum << " restype=" << restype << endl;
60 if ( mol->res ) res = residue_add(&res, restype);
61 else res = residue_add(&mol->res, restype);
62 res->num = resnum;
63 prev_resnum = resnum;
64 }
65 if ( verbose & VERB_DEBUG )
66 cout << "DEBUG readWAH: atomnum=" << atomnum << " atomtype=" << atomtype << endl;
67 if ( !res->atom ) atom = atom_add(&res->atom, atomtype);
68 else atom = atom_add(&atom, atomtype);
69 atom->num = ++atomnum;
70 atom->coord[0] = get_float(aline + 15, 10);
71 atom->coord[1] = get_float(aline + 25, 10);
72 atom->coord[2] = get_float(aline + 35, 10);
73 atom->q = get_float(aline + 45, 10);
74 atom->b = get_float(aline + 55, 10);
75 atom->sel = 1; /* All atoms read selected */
76 n++;
77 }
78
79 if ( n ) nmol = 1;
80 else nmol = -1;
81
82 fwah.close();
83

130 atom->coord[0], atom->coord[1], atom->coord[2], atom->q, atom->b);
131 fwah << string;
132 nw++;
133 }
134 }
135 }
136 if ( nw > 0 ) nmol++;
137 }
138
139 fwah.close();
140
