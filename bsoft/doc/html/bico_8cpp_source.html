66int main(int argc, char **argv)
67{
68 // Initialize variables
69 Bstring type; // Type of analysis: worst or opposite
70 int analyze(0); // Flag to calculate statistics
71 double threshold(-1); // Threshold for finding odd vertices
72 double hires(0), lores(0); // Limiting resolution for cross-correlation
73 double maxres(0); // Maximum reconstruction resolution
74 Bstring ref_file; // Reference template file name
75 Bstring mask_file; // Real space mask
76 Bstring fsmask_file; // Frequency space missing mask

78 Bstring outfile; // Output parameter file name
79 Bstring reconsfile; // Output reconstruction file
80
81 int optind;
82 Boption* option = get_option_list(use, argc, argv, optind);
83 Boption* curropt;
84 for ( curropt = option; curropt; curropt = curropt->next ) {
85 if ( curropt->tag == "find" )
86 type = curropt->value;
87 if ( curropt->tag == "analyze" ) {
88 analyze = 1;
89 if ( ( threshold = curropt->value.real() ) < -0.99 )
90 cerr << "-analyze: A threshold must be specified!" << endl;
91 }
92 if ( curropt->tag == "resolution" )
93 if ( curropt->values(hires, lores) < 1 )
94 cerr << "-resolution: A high resolution limit must be specified!" << endl;
95 if ( curropt->tag == "maxresolution" )
96 if ( ( maxres = curropt->value.real() ) < 0.1 )
97 cerr << "-maxresolution: A resolution limit must be specified!" << endl;
98 if ( curropt->tag == "reference" )
99 ref_file = curropt->filename();
100 if ( curropt->tag == "mask" )
101 mask_file = curropt->filename();
102 if ( curropt->tag == "Mask" )
103 fsmask_file = curropt->filename();
104 if ( curropt->tag == "special" )
105 specvert_file = curropt->filename();
106 if ( curropt->tag == "output" )
107 outfile = curropt->filename();
108 if ( curropt->tag == "reconstruction" )
109 reconsfile = curropt->filename();
110 }
111 option_kill(option);
112
113 double ti = timer_start();
114
115 // Read all the parameter files
116 Bstring* file_list = NULL;
117 while ( optind < argc ) string_add(&file_list, argv[optind++]);
118 if ( !file_list ) {
119 cerr << "Error: No parameter files specified!" << endl;
120 bexit(-1);
121 }
122
123 Bproject* project = read_project(file_list);
124 string_kill(file_list);
125
126 if ( project == NULL ) {
127 cerr << "Error: No input file read!" << endl;
128 bexit(-1);
129 }
130
131 Vector3<long> size;
132 Bimage* pref = NULL;
133 if ( ref_file.length() ) {
134 pref = read_img(ref_file, 1, 0);
135 } else if ( !analyze ) {
136 pref = project_reconstruct_3D(project, -1, size, maxres);
137 write_img("reference.mrc", pref, 0);
138 }
139
140 if ( !analyze && !pref ) {
141 cerr << "Error: No template map!" << endl;
142 bexit(-1);
143 }
144
145 Bimage* pmask = NULL;
146 if ( mask_file.length() ) {
147 pmask = read_img(mask_file, 1, 0);
148 pref->multiply(pmask);
149 }
150
151 Bimage* pfsmask = NULL;
152 if ( fsmask_file.length() ) {
153 pfsmask = read_img(fsmask_file, 1, 0);
154 if ( pfsmask->image->origin().length() < 1 )
155 pfsmask->center_wrap(); // Make sure the origin is in the center
156 }
157
158 if ( maxres > hires ) maxres = hires;
159
160 int done(0);
161 Bimage* pavg = NULL;
162
163 if ( type.length() ) {
164 if ( type[0] == 'w' ) {
165 done = project_ico_worst_vertex(project, pref, pmask, pfsmask, hires, lores);
166 } else if ( type[0] == 'o' ) {
167 done = project_ico_opposite_vertex(project, pmask, pfsmask, hires, lores);
168 } else if ( type == "c5" ) {
169 done = project_ico_best_c5(project, pref, pmask, pfsmask, hires, lores);
170 }
171 if ( done ) pavg = img_vertex_reconstruct(project, size, maxres);
172 }
173
174 if ( specvert_file.length() ) {
175 Bproject* project_specvert = read_project(specvert_file);
176 project_assign_special_vertex_orientations(project, project_specvert);
177 project_kill(project_specvert);
178 }
179
180 if ( analyze ) project_analyze_vertices3(project, threshold);
181
182
183 if ( project && outfile.length() )
184 write_project(outfile, project, 0, 0);
185
186 long i(0);
187 Bstring filename(reconsfile);
188 if ( pavg && reconsfile.length() ) {
189 for ( Bimage* p = pavg; p; p = p->next ) {
190 if ( pavg->next ) filename = reconsfile.pre_rev('.') + Bstring(++i, "_%02d.") + reconsfile.post_rev('.');
191 if ( verbose )
192 cout << "Writing " << filename << endl;
193 write_img(filename, p, 0);
194 }
195 }
196
197 project_kill(project);
198 delete pref;
199 delete pavg;
200 delete pmask;
201 delete pfsmask;
202
203
204 timer_report(ti);
205
206 bexit(0);

265long particle_worst_vertex(Bparticle* vert, Bimage* pref, Bimage* pmask, Bimage* pfsmask, double hires, double lores)
266{
267 long i, psel(0);
268 double cc_worst(1);
269 Bparticle* worst = NULL;
270
271 vector<Bparticle*> varr(12);
272
273 for ( i=0, worst = vert; worst && i<12; worst = worst->next, ++i )
274 varr[i] = worst;
275
276#ifdef HAVE_GCD
277 dispatch_apply(11, dispatch_get_global_queue(0, 0), ^(size_t i){
278 Bparticle* v = varr[i];
279 Bimage* p = img_read_orient_mask(v->fpart, pmask, v->view2());
280 v->fom[0] = img_cc(p, pref, pfsmask, hires, lores);
281 delete p;
282 });
283#else
284#pragma omp parallel for
285 for ( i=0; i<12; ++i ) {
286 Bparticle* v = varr[i];
287 Bimage* p = img_read_orient_mask(v->fpart, pmask, v->view2());
288 v->fom[0] = img_cc(p, pref, pfsmask, hires, lores);
289 delete p;
290 }
291#endif
292
293 for ( i=0; vert && i<12; vert = vert->next, ++i ) {
294 if ( verbose & VERB_FULL )
295 cout << vert->sel << tab << vert->fom[0] << endl;
296 if ( cc_worst > vert->fom[0] ) {
297 psel = vert->sel;
298 cc_worst = vert->fom[0];
299 worst = vert;
300 }
301 vert->sel = 1;
302 }
303
304 if ( worst )
305 worst->sel = 2;
306
307 return psel;
308}

310long particle_cc_opposite_vertices(Bparticle* vert, Bimage* pmask,
311 Bimage* pfsmask, double hires, double lores)
312{
313 long i, j, psel(0);
314 double at(M_PI - 0.02), cc, cc_worst(1);
315 Bparticle* vert1 = vert;
316 Bparticle* vert2;
317 Bparticle* worst = NULL;
318 Bparticle* worst2 = NULL;
319 Bimage *p1, *p2;
320
321 for ( i=0, vert2 = vert1; vert2 && i<12; vert2 = vert2->next, ++i )
322 vert2->fom[0] = -2;
323
324 for ( i=0; vert && i<11; vert = vert->next, ++i ) if ( vert->fom[0] < -1 ) {
325 p1 = img_read_orient_mask(vert->fpart, pmask, vert->view2());
326 for ( j=i+1, vert2 = vert->next; vert2 && j<12; vert2 = vert2->next, ++j )
327 if ( vert->view.backward().angle(vert2->view.backward()) > at ) break;
328 if ( j<12 && vert2 ) {
329 p2 = img_read_orient_mask(vert2->fpart, pmask, vert2->view2());
330 cc = img_cc(p1, p2, pfsmask, hires, lores);
331 delete p2;
332 vert->fom[0] = vert2->fom[0] = cc;
333 if ( verbose & VERB_FULL )
334 cout << vert->sel << " - " << vert2->sel << tab << cc << endl;
335 if ( cc_worst > cc ) {
336 psel = vert->sel;
337 cc_worst = cc;
338 worst = vert;
339 worst2 = vert2;
340 }
341 }
342 delete p1;
343 }
344
345 for ( i=0, vert2 = vert1; vert2 && i<12; vert2 = vert2->next, ++i )
346 vert2->sel = 1;
347
348 if ( worst )
349 worst->sel = worst2->sel = 2;
350
351 return psel;
352}

355 Bimage* pmask, Bimage* pfsmask, double hires, double lores)

357 long i, psel(0);

361 Bimage* p;
362

367 delete p;

370 vert1->sel = 1;
371 } else {
372 vert2->sel = 1;
373 psel = i;

377 vert1 = vert2;
378 cc1 = cc;
379 psel = i;
380 }
381 }
382 }

387long particle_best_c5(Bparticle* part, Bimage* pref, Bimage* pmask,
388 Bimage* pfsmask, double hires, double lores)
389{
390 long i, psel(0);
391
392 Bsymmetry sym("C5");
393 vector<View2<double>> views = sym.get_all_views(part->view2());
394
395#ifdef HAVE_GCD
396 __block vector<double> cc(5);
397 dispatch_apply(4, dispatch_get_global_queue(0, 0), ^(size_t i){
398 Bimage* p = img_read_orient_mask(part->fpart, pmask, views[i]);
399 cc[i] = img_cc(p, pref, pfsmask, hires, lores);
400 delete p;
401 });
402#else
403 vector<double> cc(5);
404#pragma omp parallel for
405 for ( i=0; i<5; ++i ) {
406 Bimage* p = img_read_orient_mask(part->fpart, pmask, views[i]);
407 cc[i] = img_cc(p, pref, pfsmask, hires, lores);
408 delete p;
409 }
410#endif
411
412 part->fom[0] = -1;
413 for ( i=0; i<5; ++i ) {
414 if ( verbose & VERB_FULL )
415 cout << i << tab << cc[i] << endl;
416 if ( part->fom[0] < cc[i] ) {
417 psel = i+1;
418 part->fom[0] = cc[i];
419 part->view2(views[i]);
420 }
421 }
422
423 return psel;
424}

701long project_analyze_vertices(Bproject* project, double threshold)
702{
703 long i, i2, imin, npart(0), nodd, noddtot(0);
704 long h, nh(40), bins(nh+1);
705 double avg, max, min, min2, fommax(2), scale(nh/fommax);
706 vector<double> fom(12,0);
707 vector<long> fomhis(bins,0);
708
709 Breconstruction* rec;
710 Bparticle* part, *vert;
711
712 if ( verbose )
713 cout << "Analyzing vertices:" << endl;
714
715 for ( rec = project->rec; rec; rec = rec->next ) if ( rec->part ) {
716 if ( verbose )
717 cout << "Reconstruction: " << rec->id << endl;
718 for ( part = vert = rec->part; part; part = vert, npart++ ) {
719 avg = 0;
720 min = min2 = 1;
721 max = -1;
722 for ( vert = part, i = i2 = 0; vert && vert->group == part->group;
723 vert = vert->next, ++i ) {
724// cout << vert->group << tab << vert->id << tab << vert->fom[0] << endl;
725 fom[i] = vert->fom[0];
726 if ( min > vert->fom[0] ) {
727 min = vert->fom[0];
728 imin = i;
729 }
730 if ( max < vert->fom[0] ) max = vert->fom[0];
731 if ( vert->sel == 1 ) {
732 avg += vert->fom[0];
733 } else {
734 i2 = i;
735 min2 = vert->fom[0];
736 }
