88vector<string> read_symop(string& symopfile, int spacegroup)
89{
90 vector<string> symop;
91
92 // No space group has more than 192 operators
93 if ( spacegroup < 1 || spacegroup > 100000 ) return symop;
94
95 if ( symopfile.length() < 1 ) {
96 symopfile = "symop.star";
97 symopfile = parameter_file_path(symopfile);
98 }
99
100 if ( verbose & VERB_DEBUG )
101 cout << "DEBUG read_symop: Reading symmetry operator file " << symopfile << endl;
102
103 string ext = extension(symopfile);
104
105 if ( spacegroup > 1 ) {
106 if ( ext.find("star") != string::npos || ext.find("cif") != string::npos ) {
107 symop = read_symop_star(symopfile, spacegroup);
108 } else {
109 symop = read_symop_lib(symopfile, spacegroup);
110 }
111 if ( !symop.size() )
112 cerr << "Error: No symmetry operator file read!" << endl;
113 }
114
115 if ( verbose & VERB_DEBUG )
116 cout << "DEBUG read_symop: Symmetry operator file read: " << symopfile << endl;
117
118 return symop;

280int write_symop_star(string& filename, int spacegroup, vector<string>& symop, int line_len)
281{
282 int i;
283 Bstar star;
284
285 star.comment("# Symmetry operators\n\n");
286
287 BstarBlock& block = star.add_block(to_string(spacegroup));
288
289 block[SYMMETRY_NUMBER] = to_string(spacegroup);
290
291 BstarLoop& loop = block.add_loop();
292 loop.tags()[SYMMETRY_EQUIVID] = 0;
293 loop.tags()[SYMMETRY_EQUIVXYZ] = 1;
294
295 for ( i=0; i<symop.size(); ++i ) {
296 vector<string>& vs = loop.add_row(2);
297 vs[0] = to_string(i+1);
298 vs[1] = symop[i];
299 }
300
301 return star.write(filename);

304int write_pointgroup_star(string filename, Bsymmetry& sym, View2<double> ref_view)
305{
306 Bstar star;
307
308 star.comment("# Symmetry from bsym\n\n");
309
310 int i;
311 Matrix3 mat = ref_view.matrix();
312 Vector3<double> v;
313
314 BstarBlock& block = star.add_block(sym.label());
315
316 block[SYMMETRY_POINT_GROUP] = sym.label();
317 block[SYMMETRY_PG_NUMBER] = to_string(sym.point());
318
319 BstarLoop& loop = block.add_loop();
320 loop.tags()[SYMMETRY_AXIS_ORDER] = 0;
321 loop.tags()[SYMMETRY_AXIS_X] = 1;
322 loop.tags()[SYMMETRY_AXIS_Y] = 2;
323 loop.tags()[SYMMETRY_AXIS_Z] = 3;
324
325 for ( i=0; i<sym.operations(); i++ ) {
326 v = mat * sym[i].axis();
327 vector<string>& vs = loop.add_row(4);
328 vs[0] = to_string(sym[i].order());
329 vs[1] = to_string(v[0]);
330 vs[2] = to_string(v[1]);
331 vs[3] = to_string(v[2]);
332 }
333
334 return star.write(filename);
