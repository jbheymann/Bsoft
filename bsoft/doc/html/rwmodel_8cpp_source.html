543{

553 }
554

557
558 for ( k=i; k+1<comp->link.size() && comp->link[k+1]; k++ ) {
559 comp->link[k] = comp->link[k+1];
560 comp->flag[k] = comp->flag[k+1];
561 }
562 comp->link[k] = NULL;
563 comp->flag[k] = 0;
564
565 for ( k=j; k+1<comp2->link.size() && comp2->link[k+1]; k++ ) {
566 comp2->link[k] = comp2->link[k+1];
567 comp2->flag[k] = comp2->flag[k+1];
568 }
569 comp2->link[k] = NULL;
570 comp2->flag[k] = 0;
571
572 for ( link = plink = *link_list; link; ) {
573 if ( ( link->comp[0] == comp && link->comp[1] == comp2 ) ||
574 ( link->comp[1] == comp && link->comp[0] == comp2 ) ) {
575 if ( link == plink ) {
576 *link_list = plink = link->next;
577 delete link;
578 link = plink;
579 } else {
580 plink->next = link->next;
581 delete link;
582 link = plink->next;
583 }
584 } else {
585 plink = link;
586 link = link->next;
587 }
588 }
