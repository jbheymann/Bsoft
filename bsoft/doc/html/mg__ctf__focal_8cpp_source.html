674double focus_cs_amp_section_fit_R(Bsimplex& simp)
675{
676 long i, j;
677 double v, v2(0), f2(0), w(1), CC(0);
678 double fac = -simp.constant(0)/4; // B-factor
679 vector<double>& f = simp.dependent_values();
680 vector<double>& x = simp.independent_values();
681
682 for ( i=j=0; i<simp.points(); ++i, ++j ) {
683 v = simp.parameter(0); // Amp
684 v += simp.parameter(1) * x[j]; // Defocus
685 v += simp.parameter(2)*x[j]*x[j]; // Cs
686 w = exp(fac*x[j]); // Envelope
687 v += x[++j]; // Relative focus
688 v = sinl(v);
689 v = 2*v*v - 1;
690 v *= w;
691 CC += f[i]*v;
692 v2 += v*v;
693 f2 += f[i]*f[i];
694 }
695
696 CC /= sqrt(v2*f2);
697
698 return 1-CC;
699}
