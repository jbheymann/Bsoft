781int Bimage::gradient_correction()
782{
783 change_type(Float);
784
785 long i, j, nn, xx, yy, zz;
786 double v;
787 vector<double> b(4);
788 Matrix a(4,4), a2(4,4);
789
790 if ( verbose & VERB_PROCESS ) {
791 cout << "Correcting the image linear gradient:" << endl;
792 cout << "Image\tb0\tb1\tb2\tb3" << endl;
793 }
794
795 // Precalculate the a matrix which depends only on the size
796 // Keep it like this in case we want to use a mask later
797 for ( zz=0; zz<z; zz++ ) {
798 for ( yy=0; yy<y; yy++ ) {
799 for ( xx=0; xx<x; xx++ ) {
800 a[0][0] += 1;
801 a[0][1] += xx;
802 a[0][2] += yy;
803 a[0][3] += zz;
804 a[1][1] += xx*xx;
805 a[1][2] += xx*yy;
806 a[1][3] += xx*zz;
807 a[2][2] += yy*yy;
808 a[2][3] += yy*zz;
809 a[3][3] += zz*zz;
810 }
811 }
812 }
813 for ( i=1; i<4; i++ ) {
814 for ( j=0; j<i; j++ )
815 a[i][j] = a[j][i];
816 if ( a[i][i] < 1 ) a[i][i] = 1; // The diagonals cannot be zero
817 }
818
819 for ( nn=0; nn<n; nn++ ) {
820 for ( i=0; i<4; i++ ) b[i] = 0;
821 a2 = a;
822 for ( i = nn*x*y*z, zz=0; zz<z; zz++ ) {
823 for ( yy=0; yy<y; yy++ ) {
824 for ( xx=0; xx<x; xx++, i++ ) {
825 v = (*this)[i];
826 b[0] += v;
827 b[1] += v*xx;
828 b[2] += v*yy;
829 b[3] += v*zz;
830 }
831 }
832 }
833 a2.LU_decomposition(b);
834 if ( verbose & VERB_PROCESS )
835 cout << n << tab << b[0] << tab << b[1] << tab << b[2] << tab << b[3] << endl;
836 for ( i = nn*x*y*z, zz=0; zz<z; zz++ ) {
837 for ( yy=0; yy<y; yy++ ) {
838 for ( xx=0; xx<x; xx++, i++ ) {
839 v = (*this)[i] - (b[0] + b[1]*xx + b[2]*yy + b[3]*zz);
840 set(i, v);
841 }
842 }
843 }
844 }
845
846 return 0;

891vector<double> Bimage::quadric_fit()
892{
893 long i, j, nn, nterm(7);
894 long xx, yy, zz;
895 double dx, dy, dz;
896 Matrix a(nterm,nterm);
897 vector<double> b(nterm);
898 vector<double> vec(nterm);
899
900 for ( nn=0; nn<n; nn++ ) {
901 for ( i=0; i<nterm; i++ ) { // Clear the matrix and vector
902 for ( j=0; j<nterm; j++ ) a[i][j] = 0;
903 b[i] = 0;
904 }
905 vec[0] = 1;
906 for ( zz=0; zz<z; zz++ ) {
907 dz = -image[nn].origin()[2] + zz;
908 vec[3] = dz;
909 vec[6] = dz*dz;
910 for ( yy=0; yy<y; yy++ ) {
911 dy = -image[nn].origin()[1] + yy;
912 vec[2] = dy;
913 vec[5] = dy*dy;
914 for ( xx=0; xx<x; xx++ ) {
915 dx = -image[nn].origin()[0] + xx;
916 vec[1] = dx;
917 vec[4] = dx*dx;
918 i = index(0,xx,yy,zz,nn);
919 for ( j=0; j<nterm; j++ ) b[j] += vec[j]*(*this)[i];
920 for ( i=0; i<nterm; i++ )
921 for ( j=0; j<=i; j++ ) a[i][j] += vec[i]*vec[j];
922 }
923 }
924 }
925 for ( i=0; i<nterm-1; i++ )
926 for ( j=i+1; j<nterm; j++ ) a[i][j] = a[j][i];
927 for ( i=0; i<nterm; i++ )
928 if ( fabs(a[i][i]) < 1e-37 ) a[i][i] = 1;
929 a.LU_decomposition(b);
930 if ( verbose & VERB_PROCESS )
931 cout << "Quadric coefficients: " <<
932 b[0] << tab << b[1] << tab << b[2] << tab << b[3] << tab << b[4] << tab << b[5] << tab << b[6] << endl << endl;
933 }
934
935 return b;
