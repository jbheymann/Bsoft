140 if ( model ) mp = model->add(id);

144 mp->image_number(to_integer(ib.at(MODEL_MAP_NUMBER)));

173 id = ir[k];
174 if ( comp ) comp = comp->add(id);
175 else mp->comp = comp = new Bcomponent(id);
176 htab[id] = comp;
177 id = ir[l];
178 comp->type(mp->type->find(id));
179 if ( ( j = il.find(COMPONENT_X) ) >= 0 ) loc[0] = to_real(ir[j]);
180 if ( ( j = il.find(COMPONENT_Y) ) >= 0 ) loc[1] = to_real(ir[j]);

195 if ( ( j = il.find(COMPONENT_FOM) ) >= 0 ) comp->FOM(to_real(ir[j]));
196 if ( ( j = il.find(COMPONENT_SELECT) ) >= 0 ) comp->select(to_integer(ir[j]));
197 if ( ( j = il.find(COMPONENT_DESCRIPTION) ) >= 0 ) {

199 comp->description() = split(s);

201 }
202 } else if ( ( k = il.find(COMPLINK_1) ) >= 0 ) {
203// cout << "COMPLINK_1 found: " << COMPLINK_1 << endl;
204 if ( htab.size() < 1 ) {
205 cerr << "No components found to link up!" << endl;
206 bexit(-1);
207 }
208 l = il.find(COMPLINK_2);
209 Blink* link = NULL;
210 Bcomponent* comp1 = NULL;

213// comp1 = mp->comp->find(ir[k]);
214// comp2 = mp->comp->find(ir[l]);
215 comp1 = htab[ir[k]];
216 comp2 = htab[ir[l]];
217 if ( link ) link = link->add(comp1, comp2);

219 if ( ( j = il.find(COMPLINK_ANGLE) ) >= 0 ) link->angle(to_real(ir[j]));
220 if ( ( j = il.find(COMPLINK_RADIUS) ) >= 0 ) link->radius(to_real(ir[j]));
221 if ( ( j = il.find(COMPLINK_LENGTH) ) >= 0 ) link->length(to_real(ir[j]));
222 if ( ( j = il.find(COMPLINK_RED) ) >= 0 ) rgba[0] = to_real(ir[j]);
223 if ( ( j = il.find(COMPLINK_GREEN) ) >= 0 ) rgba[1] = to_real(ir[j]);

226 link->color(rgba);

229 }
230// cout << "Setting up link lengths" << endl;
231 for ( link=mp->link; link; link=link->next )
232 link->length(link->comp[0]->location().distance(link->comp[1]->location()));
233 }
234 }
235 }

237 model->comment(star.comment());
238
239 models_setup_links(model);
240
241 return model;

251int write_model_star(string& filename, Bmodel* model, int splt)
252{
253 string id;
254 Bstar star;
255
256 Bmodel* mp = NULL;
257 Bcomptype* comptype = NULL;
258 Bcomponent* comp = NULL;
259 Blink* link = NULL;
260
261 star.comment(model->comment());
262 star.line_length(200);
263
264// item_change_tag(star, COMPTYPE_MAP_NUMBER, COMPTYPE_NUMBER);
265
266 int n, err(0);
267 for ( n=0, mp = model; mp; mp = mp->next, n++ ) {
268// cout << "writing model " << mp->identifier() << endl;
269 BstarBlock& block = star.add_block(mp->identifier());
270 block[MODEL_ID] = mp->identifier();
271 block[MODEL_TYPE] = mp->model_type();
272 block[MODEL_SYM] = mp->symmetry();
273 block[MODEL_HAND] = to_string(mp->handedness());
274 block[MODEL_MAP_FILENAME] = mp->mapfile();
275 block[MODEL_MAP_NUMBER] = to_string(mp->image_number());
276 block[MODEL_FOM] = to_string(mp->FOM());
277 block[MODEL_SELECT] = to_string(mp->select());
278// cout << "writing component types" << endl;
279 if ( mp->type ) {
280 BstarLoop& loop = block.add_loop();
281 loop.tags() = comptype_tags();
282 for ( comptype = mp->type; comptype; comptype = comptype->next ) {
283 vector<string>& vs = loop.add_row();
284 vs[0] = comptype->identifier();
285 vs[1] = comptype->file_name();
286 vs[2] = to_string(comptype->image_number());
287 vs[3] = to_string(comptype->mass());
288 vs[4] = to_string(comptype->charge());
289 vs[5] = to_string(comptype->FOM());
290 vs[6] = to_string(comptype->select());
291 }
292 }
293// cout << "writing components" << endl;
294 if ( mp->comp ) {
295 BstarLoop& loop = block.add_loop();
296 loop.tags() = component_tags();
297 for ( comp = mp->comp; comp; comp = comp->next ) {
298 vector<string>& vs = loop.add_row();
299 vs[0] = comp->identifier();
300 if ( comp->type() ) id = comp->type()->identifier();
301 else id = "?";
302 vs[1] = id;
303 vs[2] = to_string(comp->location()[0]);
304 vs[3] = to_string(comp->location()[1]);
305 vs[4] = to_string(comp->location()[2]);
306 vs[5] = to_string(comp->view()[0]);
307 vs[6] = to_string(comp->view()[1]);
308 vs[7] = to_string(comp->view()[2]);
309 vs[8] = to_string(comp->view()[3]*180.0/M_PI);
310 vs[9] = to_string(comp->radius());
311 vs[10] = to_string(comp->color()[0]);
312 vs[11] = to_string(comp->color()[1]);
313 vs[12] = to_string(comp->color()[2]);
314 vs[13] = to_string(comp->color()[3]);
315 vs[14] = to_string(comp->density());
316 vs[15] = to_string(comp->FOM());
317 vs[16] = to_string(comp->select());
318 vs[17] = "\"" + concatenate(comp->description()) + "\"";
319 }
320 }
321// cout << "writing links" << endl;
322 if ( mp->link ) {
323 BstarLoop& loop = block.add_loop();
324 loop.tags() = link_tags();
325 for ( link = mp->link; link; link = link->next ) {
326 vector<string>& vs = loop.add_row();
327 vs[0] = link->comp[0]->identifier();
328 vs[1] = link->comp[1]->identifier();
329 vs[2] = to_string(link->angle()*180.0/M_PI);
330 vs[3] = to_string(link->radius());
331 vs[4] = to_string(link->length());
332 vs[5] = to_string(link->color()[0]);
333 vs[6] = to_string(link->color()[1]);
334 vs[7] = to_string(link->color()[2]);
335 vs[8] = to_string(link->color()[3]);
336 vs[9] = to_string(link->FOM());
337 vs[10] = to_string(link->select());
338 }
339 }
340 }
341
342 if ( verbose & VERB_DEBUG )
343 cout << "DEBUG write_model_star: " << filename << endl;
344
345 err = star.write(filename, splt);
346
347 if ( err < 0 ) return err;
348
349 return n;
350}
