240Bresidue_type* read_residue_prop_star(Bstring& filename)
241{
242 Bstar star;
243
244 if ( star.read(filename.str()) < 0 )
245 error_show(filename.c_str(), __FILE__, __LINE__);
246
247 if ( star.blocks().size() < 0 ) {
248 cerr << "No data blocks found in the STAR file!" << endl;
249 return NULL;
250 }
251
252 if ( verbose & VERB_DEBUG )
253 cout << "DEBUG read_residue_prop_star: " << filename << endl;
254
255 int i, j, n(0);
256 Bresidue_type* rt = NULL;
257 Bresidue_type* rt_first = NULL;
258
259 for ( auto ib: star.blocks() ) {
260 for ( auto il: ib.loops() ) {
261 if ( ( i = il.find(RESPROP_CODE1) ) >= 0 ) {
262 for ( auto ir: il.data() ) {
263 rt = residue_type_add(&rt, ir[i][0]);
264 if ( !rt_first ) rt_first = rt;
265 if ( ( j = il.find(RESPROP_MASS) ) >= 0 )
266 rt->mass = to_real(ir[j]);
267 if ( ( j = il.find(RESPROP_VOLUME) ) >= 0 )
268 rt->vol = to_real(ir[j]);
269 if ( ( j = il.find(RESPROP_EXTENSION) ) >= 0 )
270 rt->ext = to_real(ir[j]);
271 if ( ( j = il.find(RESPROP_EXTSTDEV) ) >= 0 )
272 rt->extsd = to_real(ir[j]);
273 if ( ( j = il.find(RESPROP_HYDROPHOBICITY) ) >= 0 )
274 rt->hphob = to_real(ir[j]);
275 if ( ( j = il.find(RESPROP_CHARGE) ) >= 0 )
276 rt->charge = to_real(ir[j]);
277 if ( ( j = il.find(RESPROP_H) ) >= 0 )
278 rt->comp[0] = to_real(ir[j]);
279 if ( ( j = il.find(RESPROP_C) ) >= 0 )
280 rt->comp[1] = to_real(ir[j]);
281 if ( ( j = il.find(RESPROP_N) ) >= 0 )
282 rt->comp[2] = to_real(ir[j]);
283 if ( ( j = il.find(RESPROP_O) ) >= 0 )
284 rt->comp[3] = to_real(ir[j]);
285 if ( ( j = il.find(RESPROP_S) ) >= 0 )
286 rt->comp[4] = to_real(ir[j]);
287 n++;
288 }
289 }
290 }
291 }
292
293 if ( verbose & VERB_DEBUG )
294 cout << "DEBUG read_residue_prop_star: n=" << n << endl;
295
296 return rt_first;
297}

342Bresidue_type* read_residue_properties(Bstring& filename)
343{
344 Bresidue_type* rt_curr = NULL;
345 Bresidue_type* rt_first = NULL;
346
347 // Open property file read only
348 FILE *fprop;
349 if ( ( fprop = fopen(filename.c_str(), "r") ) == NULL ) return NULL;
350
351 if ( verbose )
352 cout << "Reading file: " << filename << endl;
353
354 // The property list column headers are recognized by one of a set of
355 // expected strings (such "Mass" or "MW" or "VOL")
356 int i, n;
357 int headflag(0), nhead(0);
358 int imass = -1, ivol = -1, iext = -1, iextsd = -1, icharge = -1, ihphob = -1;
359 float val[32];
360 int nrechead = 8;
361 const char* rechead[] = {" MW"," MAS"," VOL"," EXT"," EXTSD"," HP"," HYD"," CH"};
362
363 // Parse the file
364 char aline[128], *aptr;
365 Bstring string, *itemlist, *item;
366 while ( fgets(aline, 100, fprop) != NULL ) {
367 string = aline;
368 if ( string.length() < 3 ) // Blank lines = end of section
369 headflag = 0;
370 if ( headflag > 1 ) { // Read a line of properties
371 rt_curr = residue_type_add(&rt_curr, aline[0]);
372 if ( !rt_first ) rt_first = rt_curr;
373 aptr = aline + 1;
374 for ( i=0; i<nhead; i++ ) {
375 sscanf(aptr, "%f%n", &val[i], &n);
376 aptr += n;
377 }
378 if ( imass > -1 ) rt_curr->mass = val[imass];
379 if ( ivol > -1 ) rt_curr->vol = val[ivol];
380 if ( iext > -1 ) rt_curr->ext = val[iext];
381 if ( iextsd > -1 ) rt_curr->extsd = val[iextsd];
382 if ( icharge > -1 ) rt_curr->charge = val[icharge];
383 if ( ihphob > -1 ) rt_curr->hphob = val[ihphob];
384 }
385 if ( headflag == 0 ) // Check for property headers
386 for ( i=0; i<nrechead; i++ )
387 if ( string.contains(rechead[i]) ) headflag = 1;
388 if ( headflag == 1 ) {
389 itemlist = string.split();
390 for ( item = itemlist, nhead = 0; item; item = item->next, nhead++ ) {
391 if ( item->contains("MW") || item->contains("MAS") ) imass = nhead;
392 if ( item->contains("VOL") ) ivol = nhead;
393 if ( item->contains("EXT") ) iext = nhead;
394 if ( item->contains("EXTSD") ) iextsd = nhead;
395 if ( item->contains("HP") || item->contains("HYD") ) ihphob = nhead;
396 if ( item->contains("CH") || item->contains("Q") ) icharge = nhead;
397 }
398 string_kill(itemlist);
399 headflag = 100;
400 }
401 }
402
403 fclose(fprop);
404
405 return rt_first;
406}
