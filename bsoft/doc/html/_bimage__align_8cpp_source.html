281vector<Vector3<double>> interpolate_shifts(Vector3<double>* sh, long nimg, long window, long step)
282{
283 long nsh = (nimg+step-1)/step;
284
285 if ( nsh != nimg ) {
286 cerr << "Error in interpolate_shifts: The number of shifts must equal the number of images!" << endl;
287 bexit(-1);
288 }
289
290 vector<Vector3<double>> shfull(nsh);
291
292 if ( verbose & VERB_PROCESS )
293 cout << "Interpolating shifts over window of " << window << " and step of " << step << endl;
294
295 long i, j, h((step-1)/2), w, ws, we;
296 double f;
297 double c = (window%2 || step==1)? 0: 0.5; // Center of window
298
299 for ( i = 0; i < nsh; ++i, c += step ) {
300 ws = i*step - h;
301 we = ws + step;
302 if ( ws < 0 ) ws = 0;
303 if ( we > nimg ) we = nimg;
304 if ( i == nsh-1 ) we = nimg;

307// cout << c << tab << w << tab << ws << tab << f << endl;
308 if ( i ) {
309 j = i - 1;
310 shfull[i] = sh[i] - (sh[i] - sh[j])*f;

312 shfull[0] = sh[0] - (sh[1] - sh[0])*f;
313 }
314 if ( verbose & VERB_FULL )
315 cout << i << tab << sh[i][0] << tab << sh[i][1] << tab

867double mtf_gaussian_R(Bsimplex& simp)
868{
869 long i;
870 double R(0), df, s2;
871 double invsig2(-0.5/(simp.parameter(2)*simp.parameter(2)));
872 vector<double>& f = simp.dependent_values();
873 vector<double>& x = simp.independent_values();
874
875 for ( i=1; i<simp.points(); i++ ) {
876 s2 = x[i]*x[i];
877 df = f[i] - (simp.parameter(0) + simp.parameter(1)*exp(invsig2*s2));
878 R += df*df;
879 }
880
881 R = sqrt(R/i);
882 R /= simp.dependent_variance();
883
884 return R;
885}

887double coincidence_loss_R(Bsimplex& simp)
888{
889 long i;
890 double R(0), sf, df;
891 vector<double>& f = simp.dependent_values();
892 vector<double>& x = simp.independent_values();
893
894 for ( i=1; i<simp.points(); ++i ) {
895 sf = sin(simp.parameter(2)*x[i])/(simp.parameter(2)*x[i]);
896 df = f[i] - simp.parameter(0)*(1 - simp.parameter(1)*sf*sf);
897 R += df*df;
898 }
899
900 R /= i;
901 R /= simp.dependent_variance();
902
903 return R;

1240/*
1241@brief Real space cross-correlation of two power spectrum annuli in two directions.
1242@param n number of values in the annulus.
1243@param *x first annulus.
1244@param *y second annulus.
1245@return double* array of correlation coefficients.
1246
1247 The first halves of the annuli is cross-correlated forward and the second
1248 halves backward, representing the correlations of the original annulus and
1249 its inverse.
1250**/
1251double* cc_two_way(long nn, double* x, double* y)
1252{
1253 long i, j, k, n2(nn/2);
1254 double sx(0), sx2(0), sy(0), sy2(0), sxy, d;
1255
1256 double* cc = new double[nn];
1257 for ( i=0; i<nn; i++ ) cc[i] = 0;
1258
1259 for ( i=0; i<n2; i++ ) {
1260 sx += x[i];
1261 sx2 += x[i]*x[i];
1262 sy += y[i];
1263 sy2 += y[i]*y[i];
1264 }
1265
1266 d = (n2*sx2 - sx*sx)*(n2*sy2 - sy*sy);
1267 if ( d > 0 ) d = sqrt(d);
1268 else {
1269 cerr << "Error in cc_two_way: denominator is negative!" << endl;
1270 return cc;
1271 }
1272
1273 for ( i=0; i<n2; i++ ) {
1274 sxy = 0;
1275 for ( j=0, k=i; j<n2; j++, k++ ) {
1276 if ( k >= nn ) k = 0;
1277 sxy += x[j]*y[k];
1278 }
1279 cc[i] = (n2*sxy - sx*sy)/d;
1280 }
1281
1282 for ( i=n2; i<nn; i++ ) {
1283 sxy = 0;
1284 for ( j=n2, k=i; j<nn; j++, k-- ) {
1285 if ( k < 0 ) k = nn - 1;
1286 sxy += x[j]*y[k];
1287 }
1288 cc[i] = (n2*sxy - sx*sy)/d;
1289 }
1290
1291 return cc;
1292}

1316double Bimage::correlate_annuli(Bimage* polref,
1317 int ann_min, int ann_max, double ang_min, double ang_max,
1318 fft_plan planf, fft_plan planb, double& cc_max)
1319{
1320 if ( ann_min > ann_max ) swap(ann_min, ann_max);
1321 if ( ann_min < 0 ) ann_min = 0;
1322 if ( ann_max >= y ) ann_max = y - 1;
1323
1324 while ( ang_min < -TWOPI ) ang_min += TWOPI;
1325 while ( ang_max > TWOPI ) ang_max -= TWOPI;
1326
1327 if ( fabs(ang_min - ang_max) < 1e-10 ) {
1328 ang_min = 0;
1329 ang_max = TWOPI;
1330 }
1331
1332 Bimage* pcc = polref->pack_two_in_complex(this);
1333
1334 if ( verbose & VERB_PROCESS )
1335 cout << "Doing row cross-correlations from annulus " << ann_min << " to annulus " << ann_max << endl;
1336
1337 // Do FFT's in place
1338 long i, j, ij, icc;
1339 long offset, linesize(pcc->channels()*pcc->sizeX());
1340 double scale, sum1, sum2;
1341 Complex<float> d1, d2;
1342
1343 float* cc = new float[2*pcc->sizeX()];
1344 for ( i=0; i<2*pcc->sizeX(); i++ ) cc[i] = 0;
1345
1346 // Specify FFTW arrays
1347 Complex<float>* data;
1348 Complex<float>* temp1 = new Complex<float>[pcc->sizeX()];
1349 Complex<float>* temp2 = new Complex<float>[pcc->sizeX()];
1350
1351 // Correlate polar images line by line
1352 for ( i=ann_min, offset=i*linesize; i<=ann_max; i++, offset+=linesize ) {
1353 data = (Complex<float> *) (pcc->data_pointer(offset));
1354 fftw(planf, data);
1355 sum1 = sum2 = 0;
1356 for ( j=0; j<pcc->sizeX(); j++ ) temp1[j] = 0;
1357 for ( j=0; j<pcc->sizeX()/2; j++ ) {
1358 ij = -j;
1359 if ( ij < 0 ) ij += pcc->sizeX();
1360 d1 = data[j].unpack_first(data[ij]);
1361 d2 = data[j].unpack_second(data[ij]);
1362 temp1[ij] = d1 * d2.conj();
1363 temp1[i] = temp1[ij].conj();
1364 sum1 += d1.power();
1365 sum2 += d2.power();
1366 }
1367 if ( sum1 && sum2 ) {
1368 scale = 0.5/sqrt(sum1*sum2);
1369 for ( j=0; j<pcc->sizeX(); j++ ) temp1[j] *= scale;
1370 }
1371 fftw(planb, temp1);
1372 for ( j=0; j<pcc->sizeX(); j++ ) cc[j] += temp1[j].real();
1373 }
1374
1375 delete[] temp1;
1376 delete[] temp2;
1377
1378 icc = 0;
1379 cc_max = cc[0];
1380
