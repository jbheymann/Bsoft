204Bimage* Bimage::extract_shell(long nn, double minrad, double maxrad)
205{
206 long i, j, xx, yy, zz, cc;
207 double rx2, ry2, rz2, r2;
208
209 if ( minrad > maxrad ) swap(minrad, maxrad);
210 if ( minrad < 0 ) minrad = 0;
211 if ( maxrad > x ) maxrad = x - 1;
212
213 double minrad2 = minrad*minrad;
214 double maxrad2 = maxrad*maxrad;
215
216 Bimage* pex = copy_header(1);
217 pex->image[0] = image[nn];
218 pex->data_alloc();
219 pex->fill(avg);
220
221 if ( verbose & VERB_PROCESS ) {
222 cout << "Extracting subimage: " << nn << endl;
223 cout << "Shell: " << minrad << " " << maxrad << endl;
224 cout << endl;
225 }
226
227 for ( zz=0; zz<z; zz++ ) {
228 if ( z > 1 ) {
229 rz2 = zz - image[nn].origin()[2];
230 rz2 *= rz2;
231 } else rz2 = 0;
232 for ( yy=0; yy<y; yy++ ) {
233 ry2 = yy - image[nn].origin()[1];
234 ry2 *= ry2;
235 for ( xx=0; xx<x; xx++ ) {
236 rx2 = xx - image[nn].origin()[0];
237 rx2 *= rx2;
238 r2 = rx2 + ry2 + rz2;
239 if ( r2 >= minrad2 && r2 <= maxrad2 ) {
240 i = pex->index(xx, yy, zz);
241 j = index(xx, yy, zz, nn);
242 for ( cc=0; cc<c; cc++, i++, j++ ) pex->set(i, (*this)[j]);
243 }
244 }
245 }
246 }
247
248 pex->statistics();
249
250 return pex;
251}

496long find_optimal_tiling(long target, long tile_size, double fraction=0.2)
497{
498 if ( fraction > 0.5 ) fraction = 0.5;
499
500 long overlap(fraction*tile_size), i, imin(overlap/2), imax(overlap*2);
501 long step(tile_size - overlap);
502 long num(target/step);
503 long d, dmin(target);
504
505 for ( i = imin; i < imax; i++ ) {
506 step = tile_size - i;
507 num = target/step;
508 d = num*step + i - target;
509 if ( d >= 0 && dmin > d ) {
510 overlap = i;
511 dmin = d;
512 if ( verbose & VERB_FULL )
513 cout << overlap << tab << num << tab << step << tab << d << endl;
514 }
515 }
516
517 return overlap;
518}

535{
536 tile_size = tile_size.min(size());
537
538 Vector3<long> num, overlap, step, icoor;
539 Vector3<long> maxori(size() - tile_size);
540
541 for ( long j=0; j<3; j++ )
542 if ( size()[j] > 1 ) overlap[j] = find_optimal_tiling(size()[j], tile_size[j], fraction);
543
544 step = tile_size - overlap;
545
546 num = size()/step;
547 num = num.max(1);
548
549 if ( verbose & VERB_FULL ) {
550 cout << size() << endl;
551 cout << num * step + overlap << endl;
552 }
553
554 long i, ntiles(num.volume());
555
556 vector<Vector3<long>> coords(ntiles);
557
558 for ( i=icoor[2]=0; icoor[2]<num[2]; icoor[2]++ )
559 for ( icoor[1]=0; icoor[1]<num[1]; icoor[1]++ )
560 for ( icoor[0]=0; icoor[0]<num[0]; icoor[0]++, i++ ) {
561 coords[i] = icoor * step;
562 coords[i] = coords[i].min(maxori);
563 }
564
565
566 if ( verbose & VERB_FULL ) {
567 cout << "Extracting " << ntiles << " tiles:" << endl;
568 cout << "Tile size: " << tile_size << endl;
569 cout << "Number of tiles: " << num << endl;

710Bimage* Bimage::extract_tetrahedron(Vector3<double>* tet, int fill_type, double fill)
711{
712 if ( fill_type == FILL_AVERAGE ) fill = avg;
713
714 long i, j, k, nn, xx, yy, zz, cc;
715
716 Vector3<double> center;
717 Vector3<double> normal[4];
718 double origin[4];
719
720 for ( i=0; i<4; i++ ) center = center + tet[i];
721 center /= 4.0;
722
723 normal[0] = tet[0].normal(tet[1], tet[2]);
724 normal[1] = tet[1].normal(tet[2], tet[3]);
725 normal[2] = tet[2].normal(tet[3], tet[0]);
726 normal[3] = tet[3].normal(tet[0], tet[1]);
727
728 // The distance of a point from a plane is given by the inner product sum of
729 // the normal to the plane and the vector from that point to any point on the plane
730 for ( i=0; i<4; i++ ) {
731 origin[i] = normal[i].scalar(tet[i]);
732 if ( normal[i].scalar(center) < origin[i] ) {
733 normal[i] = -normal[i];
734 origin[i] = -origin[i];
735 }
736 }
737
738 Bimage* ptet = copy();
739
740 if ( verbose & VERB_PROCESS ) {
741 cout << "Extracting a tetrahedron:" << endl;
742 cout << "Vertices: " << tet[0] << endl;
743 cout << " " << tet[1] << endl;
744 cout << " " << tet[2] << endl;
745 cout << " " << tet[3] << endl;
746 cout << "Center: " << center << endl;
747 cout << "Normals: " << normal[0] << " (" << origin[0] << ")" << endl;
748 cout << " " << normal[1] << " (" << origin[1] << ")" << endl;
749 cout << " " << normal[2] << " (" << origin[2] << ")" << endl;
750 cout << " " << normal[3] << " (" << origin[3] << ")" << endl;
751 cout << endl;
752 } else if ( verbose & VERB_LABEL )
753 cout << "Extracting a tetrahedron" << endl << endl;
754
755 for ( i=nn=0; nn<n; nn++ ) {
756 if ( fill_type == FILL_BACKGROUND ) fill = background(nn);
757 for ( zz=0; zz<z; zz++ ) {
758 for ( yy=0; yy<y; yy++ ) {
759 for ( xx=0; xx<x; xx++, i++ ) {
760 for ( j=0; j<4; j++ ) {
761 if ( normal[j][0]*xx + normal[j][1]*yy + normal[j][2]*zz < origin[j] )
762 for ( cc=0, k=i*c; cc<c; cc++, k++ )
763 ptet->set(k, fill);
764 }
765 }
766 }
767 }
768 }
769
770 return 0;
771}

784Bimage* Bimage::orthogonal_slices(long nn, Vector3<long> voxel,
785 Vector3<long> ext_size)
786{
787 if ( voxel[0] < 0 || voxel[1] < 0 || voxel[2] < 0 ) return NULL;
788 if ( voxel[0] >= x || voxel[1] >= y || voxel[2] >= z ) return NULL;
789
790 if ( ext_size.volume() < 1 ) ext_size = size();
791
792 int xf, yf;
793 long i, j, iy, xx, yy, zz, cc;
794 long xo, yo, zo;
795 long h, nh(1);
796 if ( z > 1 ) nh = 3;
797 long nx = (long) (ext_size[0]);
798 long ny = (long) (ext_size[1]);
799
800 Vector3<long> start(voxel - (ext_size * 0.5));
801 Vector3<long> end(start + ext_size);
802 Vector3<double> ori;
803
804 if ( verbose & VERB_PROCESS ) {
805 cout << "Extracting orthogonal views:" << endl;
806 cout << "Start: " << start << endl;
807 cout << "End: " << end << endl;
808 cout << "Fill value: " << image[nn].background() << endl;
809 cout << endl;
810 }
811
812 Bimage* porth = new Bimage(datatype, compoundtype, nx, ny, 1, nh);
813 porth->sampling(sampling(0));
814 porth->fill(image[nn].background());
815
816 for ( h=0; h<nh; h++ ) {
817 porth->background(h, background(nn));
818 if ( h%2 == 0 ) ori[0] = image[nn].origin()[0] - start[0];
819 else ori[0] = image[nn].origin()[2] - start[2];
820 if ( h < 2 ) ori[1] = image[nn].origin()[1] - start[1];
821 else ori[1] = image[nn].origin()[2] - start[2];
822 porth->origin(h, ori);
823 if ( h == 0 ) porth->image[h].view(0,0,1,0);
824 else if ( h == 1 ) porth->image[h].view(1,0,0,0);
825 else porth->image[h].view(0,1,0,0);
826 xo = voxel[0];
827 yo = voxel[1];
828 zo = voxel[2];
829 for ( yy=0; yy<ny; yy++ ) {
830 yf = 1;
831 if ( h < 2 ) {
832 yo = (long) (yy + start[1]);
833 if ( yo < 0 ) yf = 0;
834 else if ( yo >= y ) yf = 0;
835 } else {
836 zz = yy;
837 zo = (long) (zz + start[2]);
838 if ( zo < 0 ) yf = 0;
839 else if ( zo >= z ) yf = 0;
840 }
841 iy = (h*ny + yy)*nx;
842 for ( xx=0; xx<nx; xx++ ) {
843 xf = 1;
844 if ( h%2 == 0 ) {
