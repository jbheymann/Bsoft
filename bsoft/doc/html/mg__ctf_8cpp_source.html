422double aberration(long n, long m, double s, double p)
423{
424 if ( n <= 0 ) return 1;
425 if ( n > 4 ) return 0;
426
427 double s2(s*s);
428 if ( n == 2 ) s = s2;
429 else if ( n == 3 ) s *= s2;
430 else if ( n == 4 ) s = s2*s2;
431
432 if ( m == 0 ) return s;
433 if ( m < 0 ) return s * sin(-m*p);
434 else return s * cos(m*p);
435}
