18## @brief Writes a command to a file.
19#
20# @param filename file name.
21# @param mode access mode.
22proc saveCommand { filename mode } {
23 global command
24
25 if { [catch {open $filename $mode} f] } { return }
26
27 puts $f "$command\n"
28
29 close $f
30}

32## @brief Selects a file for reading or writing using a standard dialog
33#
34# @param operation "save" or "append".
35
36proc fileDialog { operation } {
37 set filename [tk_getSaveFile]
38
39 if { [string length $filename] < 1 } { return }
40
41 if { [string compare $operation "save"] == 0 } {
42 saveCommand $filename "w"
43 } elseif { [string compare $operation "append"] == 0 } {
44 saveCommand $filename "a"
45 }

48## @brief Constructs the command line from a list containing the program name, the options, and the file names
49#
50# @param c the entry for the command
51# @param f the frame for files
52# @param t the text with options
53
54proc constructCommand { c f t } {
55 set cmd [lindex [$c get] 0]
56
57 foreach w [winfo children $t] {
58 if [winfo exists $w.label] {
59 upvar [$w.label cget -variable] on
60 set opt [$w.label cget -text]
61 if { $on == 1 } {
62 append cmd " " $opt " " [$w.e1 get]
63 }
64 }
65 }
66
67 foreach w [winfo children $f] {
68 upvar [$w.label cget -variable] on
69 if { $on == 1 } {
70 append cmd " " [$w.name get]
71 }
72 }
73
74 $c delete 0 end
75 $c insert 0 $cmd
76}

78## @brief Parses the usage line
79#
80# @param c the entry for the command
81# @param f the frame for files
82# @param t the text with options
83# @param line the line
84
85proc parseUsageLine { c f t line } {
86 global font
87
88 destroy .usage
89 label .usage -font $font -justify left -text $line -background yellow
90
91 set words [split $line]
92 for { set i 3 } { $i < [llength $words] } { incr i } {
93 set w $f.$i
94 frame $w
95 set fn [lindex $words $i]
96 checkbutton $w.label -text $fn -variable $fn -relief flat \
97 -background cyan -command "constructCommand $c $f $t"
98 $w.label deselect
99 entry $w.name -width 50
100 button $w.button -text "Browse ..." -command "getFilename $w.name"
101 pack $w.label $w.name $w.button -side left -pady 1 -padx 4
102 pack $w -side top -fill x -expand no
103 bind $w.name <KeyRelease> "constructCommand $c $f $t"
104 }
105
106 if { $i > 3 } { pack $f -side bottom -pady 2 -fill x -expand no }
107 pack .usage -side bottom -pady 2 -anchor w
108}

110## @brief Parses one line starting with an option
111#
112# @param c the entry for the command
113# @param f the frame for files
114# @param t the text with options
115# @param line the line
116
117proc parseOptionLine { c f t line } {
118 global font smallfont
119 if [regexp "^--" $line] { return }
120
121 set words [split $line]
122 set b [lindex $words 0]
123 set v [lindex $words 1]
124
125 set d ""
126 foreach word [lrange $words 2 end] {
127 if { $word != "" } {
128 lappend d $word
129 }
130 }
131
132 set w $t.$b
133 frame $w
134 entry $w.e1 -width [string length $v]
135 $w.e1 insert 0 $v
136
137 checkbutton $w.label -text "$b" -variable $b -relief flat \
138 -background cyan -command "constructCommand $c $f $t"
139 $w.label deselect
140
141 if { $v > " " } {
142 pack $w.label $w.e1 -side left -pady 1 -padx 4
143 } else {
144 pack $w.label -side left -pady 1 -padx 4
145 }
146
147 $t window create end -window $w -padx 4 -pady 1
148 $t insert end [join $d " "]
149 $t insert end "\n"
150
151 bind $w.e1 <KeyRelease> "constructCommand $c $f $t"
152}

154## @brief Parses a command line to fill in options
155#
156# @param t text window name
157# @param cmd command line
158
159proc parseCommand { t cmd } {
160 set vf 0
161 foreach word $cmd {
162 if ![string first "-" $word] {
163 set vf 0
164# set word [string trimleft $word "-"]
165 foreach w [winfo children $t] {
166 if [regexp -- $word $w] {
167 $w.label select
168 if { [$w.e1 get] > " " } {
169 set vf 1
170 }
171 break
172 }
173 }
174 } elseif { $vf } {
175 $w.e1 delete 0 end
176 $w.e1 insert 0 $word
177 set vf 0
178 }
179 }
180}

182## @brief Selects the program and get the options from the program
183#
184# @param w window name
185# @param cmd command line
186
187proc selectProgram { w cmd } {
188 global font
189 global smallfont
190
191 set wn $w
192 if [string equal "." $w] { set wn "" }
193
194 set c $wn.cmd
195 set f $wn.file
196 set t $wn.txt
197 set g $wn.grid
198
199 destroy $f
200 destroy $t
201 destroy $g
202 destroy $wn.hscroll
203 destroy $wn.vscroll
204
205 set prog [lindex $cmd 0]
206# puts "Program: $prog"
207 if { [ string first "brun" $cmd ] > -1 } { return }
208 if ![ file executable $prog ] { return }
209
210 catch { exec $prog } result
211
212 $c.e delete 0 end
213 $c.e insert 0 $cmd
214
215 frame $f
216
217 frame $g
218 pack $g -side bottom -expand yes -fill both -padx 1 -pady 1
219
220 grid rowconfig $g 0 -weight 1 -minsize 0
221 grid columnconfig $g 0 -weight 1 -minsize 0
222
223 text $t -wrap none -relief sunken -bd 2 \
224 -xscrollcommand "$wn.hscroll set" \
225 -yscrollcommand "$wn.vscroll set" \
226 -setgrid 1 -highlightthickness 0 -pady 2 -padx 3
227
228 scrollbar $wn.hscroll -orient horizontal -command "$t xview"
229 scrollbar $wn.vscroll -orient vertical -command "$t yview"
230
231 grid $t -padx 1 -in $g -pady 1 \
232 -row 0 -column 0 -rowspan 1 -columnspan 1 -sticky news
233 grid $wn.vscroll -in $g -padx 1 -pady 1 \
234 -row 0 -column 1 -rowspan 1 -columnspan 1 -sticky news
235 grid $wn.hscroll -in $g -padx 1 -pady 1 \
236 -row 1 -column 0 -rowspan 1 -columnspan 1 -sticky news
237
238 $t insert end "Description:\n"
239 foreach line [split $result "\n"] {
240 if ![string first "-" $line] {
241 parseOptionLine $c.e $f $t $line
242 } elseif ![string first "Usage" $line] {
243 parseUsageLine $c.e $f $t $line
244 } else {
245 $t insert end "$line\n"
246 }
247 }
248
249 parseCommand $t $cmd
250}

263 if [string equal "." $w] { set wn "" }

271 -underline 0 -accelerator "Ctrl-s"

273 -underline 0 -accelerator "Ctrl-a"
274 $wn.menu.file add command -label "Quit" -command { exit } -underline 0 \
275 -accelerator "Ctrl-q"
276
277 bind all <Control-q> { exit }
278 bind all <Command-q> { exit }
279
280# puts "cl: $cl"
281
282 set command "Select a program and fill in the options"
283 set program ""
284 if [string length $cl] {
285 set command "$bsoft_bin/$cl"
286 set p "$bsoft_bin/"
287 append p [lindex $cl 0]
288 if [ file executable $p ] { set program $p }
289 }
290# puts "Command: $command"
291# puts "Program: $program"
292
293 set c $wn.cmd
294 frame $c
295 entry $c.e -width 100 -xscrollcommand "$c.s set" -textvariable command
296 scrollbar $c.s -relief sunken -orient horiz -command "$c.e xview"
297 pack $c.s $c.e -side bottom -fill x -expand 1 -padx 1m -pady 1
298
299 frame $wn.buttons
300 menubutton $wn.buttons.program -text "Program" -underline 0 -direction below \
301 -menu $wn.buttons.program.m -relief raised -width 20 \
302 -textvariable program
303 menu $wn.buttons.program.m -tearoff 0
304 set file_list [lsort [glob [file join $bsoft_bin/*]]]
305 set i 0
306 foreach p [split $file_list] {
307 set prog [file tail $p]
308 if { [file executable $p] && [file isfile $p] } {
309 $wn.buttons.program.m add command -label $prog \
310 -command "selectProgram $w $p" -columnbreak [expr ($i % 20) == 0]
311 incr i
312 }
313 }
314
315 if [string length $command] { selectProgram $w $command }
316
317# set logfile "b.log"
318
319 button $wn.buttons.run -text Execute -command { executeCommand $command "b.log" }
320 button $wn.buttons.dismiss -text Dismiss -command "destroy $w"
321
322 pack $wn.buttons.program $wn.buttons.run $wn.buttons.dismiss -side left -expand yes
323 pack $wn.buttons $c -side top -fill x -pady 2m -expand no
324
325 bind . <Return> { selectProgram . [.cmd.e get] }
326}
