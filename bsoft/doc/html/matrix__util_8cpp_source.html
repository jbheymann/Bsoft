27int matrix_normalize(Matrix& m)
28{
29 long i, r, c;
30 double err(1);
31 double* rw = new double[m.rows()];
32 double* cw = new double[m.columns()];
33
34 if ( verbose & VERB_FULL )
35 cout << "Cycle\tError" << endl;
36 for ( i=0; i<100 && err > 1e-20; i++ ) {
37 for ( r=0; r<m.rows(); r++ ) rw[r] = 0; // Row scaling
38 for ( r=0; r<m.rows(); r++ )
39 for ( c=0; c<m.columns(); c++ )
40 rw[r] += m[r][c];
41 for ( r=0; r<m.rows(); r++ ) rw[r] = 1/rw[r];
42 for ( r=0; r<m.rows(); r++ )
43 for ( c=0; c<m.columns(); c++ )
44 m[r][c] *= rw[r];
45
46 for ( c=0; c<m.columns(); c++ ) cw[c] = 0; // Column scaling
47 for ( r=0; r<m.rows(); r++ )
48 for ( c=0; c<m.columns(); c++ )
49 cw[c] += m[r][c];
50 for ( c=0; c<m.columns(); c++ ) cw[c] = 1/cw[c];
51 for ( r=0; r<m.rows(); r++ )
52 for ( c=0; c<m.columns(); c++ )
53 m[r][c] *= cw[c];
54
55 for ( err=0, r=0; r<m.rows(); r++ ) err += rw[r];
56 for ( c=0; c<m.columns(); c++ ) err += cw[c];
57 err = fabs(err/(m.rows() + m.columns()) - 1);
58 if ( verbose & VERB_FULL )
59 cout << i+1 << tab << err << endl;
60 }
61
62 delete[] rw;
63 delete[] cw;
64
65 return 0;
66}

182int matrix_permute(int i, int j, Matrix matrix, int* order,
183 double* best_R, int* best_order)
184{

186 int* this_order;

193 for ( m=0; m<n; m++ ) best_order[m] = order[m];
194 }

196 this_order = new int[n];
197 for ( m=0; m<n; m++ ) this_order[m] = order[m];
198 for( k=i; k<=j; k++) {
199 matrix_permute(i+1, j, matrix, order, best_R, best_order);
200 if( k != j ) {
201 swap(this_order[i-1], this_order[k]);
202 for ( m=0; m<n; m++ ) order[m] = this_order[m];
203 }
204 }
205 delete[] this_order;
206 }
207
208 return 0;

232double matrix_calc_dist_fit(Matrix matrix, int* order)
233{
234 long i, j, k, n = matrix.columns();
235 double min(1), max(0), new_min(1), new_max(0);
236 Matrix new_mat(n,n);
237
238 // Set up the lower triangle with the original distances
239 for ( i=1; i<n; i++ ) {
240 for ( j=0; j<i; j++ ) {
241 new_mat[i][j] = matrix[order[i]][order[j]];
242 if ( min > new_mat[i][j] ) min = new_mat[i][j];
243 if ( max < new_mat[i][j] ) max = new_mat[i][j];
244 }
245 }
246
247 // Set up the second upper diagonal
248 for ( j=1; j<n; j++ )
249 new_mat[j-1][j] = new_mat[j][j-1];
250
251 // Calculate the other upper diagonals
252 for ( i=0; i<n-1; i++ ) {
253 for ( j=i+2; j<n; j++ ) {
254 for ( k=i+1; k<=j; k++ )
255 new_mat[i][j] += new_mat[k][k-1];
256 if ( new_min > new_mat[i][j] ) new_min = new_mat[i][j];
257 if ( new_max < new_mat[i][j] ) new_max = new_mat[i][j];
258 }
259 }
260
261 if ( verbose & VERB_DEBUG ) {
262 cout << "DEBUG matrix_calc_dist_fit: Order:" << endl;
263 for ( i=0; i<n; i++ ) cout << " " << order[i];
264 cout << endl << "DEBUG matrix_calc_dist_fit: New matrix:" << endl;
265 cout << new_mat << endl;
266 }
267
268 // Calculate a figure-of-merit weighted by diagonal
269 double diff, weight, wsum(0), R(0);
270 double scale = 1/(max - min);
271 double new_scale = 1/(new_max - new_min);
272 double offset = min/(max - min) - new_min/(new_max - new_min);
273 for ( i=2; i<n; i++ ) {
274 for ( j=0; j<i-1; j++ ) {
275 diff = scale*new_mat[i][j] - new_scale*new_mat[j][i] - offset;
276 weight = 1.0/(i - j);
277 R += weight*diff*diff;
278 wsum += weight;
279 }
280 }
281 R = sqrt(R/wsum);
282
283 if ( verbose & VERB_PROCESS ) {
284 cout << "R = " << R << endl;
285 for ( j=0; j<n; j++ ) cout << " " << order[j]+1;
286 cout << endl;
287 }
288
289 return R;
290}
