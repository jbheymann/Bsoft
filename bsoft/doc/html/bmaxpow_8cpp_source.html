92 reset = 1;

196double img_reconstruct_one(Bimage* p, Vector3<long> size, Bsymmetry& sym,
197 View2<double> view, double hi_res, double lo_res, double scale)
198{
199 long i, n;
200 double fom(0);
201 Vector3<double> vscale(scale, scale, scale);
202
203 Bimage* prec = new Bimage(Float, TComplex, size, 1);
204
205 prec->fspace_pack_2D(p, view, sym, hi_res, lo_res, vscale, 0, 1, 0);
206
207 prec->fspace_reconstruction_weigh();
208
209 float* power = (float *) prec->next->data_pointer();
210 float* weight = (float *) prec->next->next->data_pointer();
211
212 for ( i=n=0; i<prec->image_size(); i++ ) {
213 if ( weight[i] > 2.5 ) {
214 fom += prec->complex(i).power()/power[i];
215 n++;
216 }
217 }
218
219 if ( n ) fom /= n;
220 else cout << "No overlapping voxels!" << endl;
221
222 delete prec;
223
224 return fom;
225}

227double img_reconstruct_one_v3(Bimage* p, Vector3<long> size, Bsymmetry& sym,
228 View2<double> view, double hi_res, double lo_res, double scale)
229{
230 long i, n;
231 double pwr(0), avg(0), w(0), fom(0);
232 Vector3<double> vscale(scale, scale, scale);
233
234 Bimage* prec = new Bimage(Float, TComplex, size, 1);
235
236 prec->fspace_pack_2D(p, view, sym, hi_res, lo_res, vscale, 0, 1, 0);
237
238 float* power = (float *) prec->next->data_pointer();
239 float* weight = (float *) prec->next->next->data_pointer();
240
241 for ( i=n=0; i<prec->image_size(); i++ ) {
242 if ( weight[i] > 1.5 ) {
243 avg = prec->complex(i).power();
244 pwr = power[i];
245 w = pwr - avg/weight[i];
246 if ( w > SMALLFLOAT ) {
247 fom += (avg - pwr)/w;
248 n++;
249 }
250 }
251 }
252 if ( n ) fom /= n;
253 else cout << "No overlapping voxels!" << endl;
254
255 delete prec;
256
257 return fom;

354double img_find_maximum_power(Bimage* p, Vector3<long> size, Bsymmetry& sym,
355 double theta_step, double phi_step, double alpha_step,
356 double hi_res, double lo_res, double scale, int pad_factor)
357{
358 if ( !p->data_pointer() ) return 0;
359
360 if ( hi_res < 2*p->sampling(0)[0]/scale ) hi_res = 2*p->sampling(0)[0]/scale;
361
362 if ( verbose & VERB_FULL )
363 cout << "Finding the orientation based on maximum power up to " << hi_res << " A resolution" << endl;
364
365 long iv, nviews;
366 double fomavg, bestfom;
367
368 vector<View2<double>> view = sym.asymmetric_unit_views(theta_step, phi_step, 1);
369 view = view_list_expand_angles(view, -M_PI, M_PI - alpha_step/2, alpha_step);
370 View2<double> bestview;
371 nviews = view.size();
372
373 if ( verbose & VERB_FULL )
374 cout << "x\ty\tz\ta\tFOM" << endl;
375
376#ifdef HAVE_GCD
377 __block vector<double> fom_arr(nviews,0);
378 dispatch_apply(nviews, dispatch_get_global_queue(0, 0), ^(size_t iv){
379 fom_arr[iv] = img_reconstruct_one(p, size, sym, view[iv], hi_res, lo_res, scale);
380 });
381#else
382 vector<double> fom_arr(nviews,0);
383#pragma omp parallel for
384 for ( iv = 0; iv < nviews; iv++ )
385 fom_arr[iv] = img_reconstruct_one(p, size, sym, view[iv], hi_res, lo_res, scale);
386#endif
387
388 for ( iv = 0, bestfom = -1e30, fomavg = 0; iv < nviews; iv++ ) {
389 fomavg += fom_arr[iv];
390 if ( bestfom < fom_arr[iv] ) {
391 bestfom = fom_arr[iv];
392 bestview = view[iv];
393 }
394 }
395
396 fomavg /= nviews;
397
398 if ( verbose & VERB_FULL )
399 cout << bestview << tab << fomavg << endl;
400
401 p->view(bestview);
402 p->image->FOM(fomavg);
403
404 return bestfom;
405}

407double part_find_maximum_power(Bparticle* part,
408 Vector3<long> size, Bimage* pmask, long ft_size, Bsymmetry& sym,
409 double theta_step, double phi_step, double alpha_step,
410 double hi_res, double lo_res, double scale, int pad_factor,
411 int ctf_action, double wiener, int flags)
412{
413 FOMType fom_tag[NFOM] = {FOM_CC, FOM_CC_AVG};
414
415 if ( flags & CHECK_PPX )
416 if ( ppx_check(part, fom_tag) )
417 return part->fom[0];
418
419 double part_scale(1);
420 Bimage* p = NULL;
421 Bmicrograph* mg = part->mg;
422
423 if ( ( p = read_img(mg->fpart, 1, part->id - 1) ) == NULL ) {
424 error_show("part_find_maximum_power", __FILE__, __LINE__);
425 return -1;
426 }
427
428 CTFparam em_ctf;
429 if ( mg->ctf ) em_ctf.update(mg->ctf);
430 if ( part->def > 0 ) em_ctf.defocus_average(part->def);
431
432 if ( pmask ) p->multiply(pmask);
433
434 if ( part->ori[0] <= 0 ) {
435 if ( p->image->origin()[0] <= 0 ) part->ori = p->size()/2;
436 else part->ori = p->image->origin();
437 }
438 p->image->origin(part->ori);
439
440 p->view(part->view);
441
442 if ( size[2] < 2 ) {
443 if ( p->image->view()[2] >= 0 ) p->image->view(0,0,1,0);
444 else p->image->view(0,0,-1,0);
445 }
446
447 if ( part->mag > 0 ) part_scale = scale/part->mag;
448 else part_scale = scale;
449
450 p->rescale_to_avg_std(0, 1);
451 p->calculate_background();
452
453 p->pad(ft_size, FILL_BACKGROUND, p->background(long(0)));

455 p->fft();
456
457 p->phase_shift_to_origin();
458
459 p->sampling(mg->pixel_size);
460
461 if ( mg->ctf && ctf_action )
462 img_ctf_apply(p, em_ctf, ctf_action, wiener, 0, hi_res, 0);
463
464 part->fom[0] = img_find_maximum_power(p, size, sym, theta_step,
465 phi_step, alpha_step, hi_res, lo_res, part_scale, pad_factor);
466
467 part->view = p->image->view();
468 part->fom[1] = p->image->FOM();
469
470 delete p;
471// cout << "Particle done" << endl;
472
473 if ( flags & WRITE_PPX ) {
474 Bstring ppx_name = ppx_filename(part->mg->id, part->id);
475 write_particle(ppx_name, part, 0, 0, fom_tag);
476 ppx_name = 0;
477 }
478
479 return part->fom[0];
