50Bplot* read_rps(Bstring& filename)
51{
52 if ( verbose & VERB_PROCESS )
53 cout << "# Reading radial power spectrum: " << filename << endl;
54
55 ifstream f(filename.c_str());
56 if ( f.fail() ) return NULL;
57
58 int i, j, ncol(2), nrow(0);
59 string str;
60 double s, p;
61 vector<double> vs, vp;
62
63 getline(f, str);
64
65 while ( !f.eof() ) {
66 f >> s >> p;
67 if ( p ) {
68 vs.push_back(s);
69 vp.push_back(p);
70 nrow++;
71 }
72 }
73
74 f.close();
75
76 Bstring title = "RPS Plot";
77 Bplot* plot = new Bplot(1, nrow, ncol);
78 plot->title(title);
79 plot->page(0).title(title);
80 plot->page(0).columns(ncol);
81 for ( i=0; i<ncol; i++ ) plot->page(0).column(i).number(i);
82 plot->page(0).column(0).label("Spatial Frequency (A)");
83 plot->page(0).column(1).label("Power");
84// plot->page(0).axis(1).min(0);
85// plot->page(0).axis(1).max(1/hi_res);
86// plot->page(0).axis(1).inc(0.1/hi_res);
87 plot->page(0).axis(1).flags(1);
88 plot->page(0).axis(1).label("Spatial Frequency (1/Å)");
89// plot->page(0).axis(3).min(0);
90// plot->page(0).axis(3).max(1);
91// plot->page(0).axis(3).inc(0.1);
92 plot->page(0).column(1).type(2);
93 plot->page(0).column(0).axis(1);
94 plot->page(0).column(1).axis(3);
95
96 for ( i=0, j=nrow; i<nrow; ++i, ++j ) {
97 (*plot)[i] = vs[i];
98 (*plot)[j] = vp[i];
99 if ( verbose & VERB_DEBUG )
100 cout << vs[i] << tab << vp[i] << endl;
101 }
102
103 return plot;
104}

110 double weigh_dose(0); // Dose per frame to weigh by dose
111 int normalize(0); // Flag to normalize amplitudes

123 Bstring maskfile; // Mask to exclude regions in reciprocal space
124 Bstring realmask; // Real space mask
125
126 int optind;
127 Boption* option = get_option_list(use, argc, argv, optind);
128 Boption* curropt;
129 for ( curropt = option; curropt; curropt = curropt->next ) {
130 if ( curropt->tag == "invert" ) setinvert = 1;
131 if ( curropt->tag == "dose" )
132 if ( ( weigh_dose = curropt->value.real() ) < 0.001 )
133 cerr << "-dose: The dose per frame must be specified!" << endl;
134 if ( curropt->tag == "normalize" )
135 if ( ( normalize = curropt->value.integer() ) < 1 )
136 cerr << "-normalize: A normalization option must be specified!" << endl;
137 if ( curropt->tag == "root" ) root = 1;
138 if ( curropt->tag == "square" ) square = 1;
139 if ( curropt->tag == "datatype" )
140 nudatatype = curropt->datatype();
141 if ( curropt->tag == "sampling" )
142 sam = curropt->scale();
143 if ( curropt->tag == "origin" ) {
144 if ( curropt->value[0] == 'c' ) {
145 set_origin = 2;
146 } else {
147 origin = curropt->origin();
148 set_origin = 1;
149 }
150 }
151 if ( curropt->tag == "resolution" )
152 if ( ( resolution = curropt->value.real() ) < 0.01 )
153 cerr << "-resolution: A resolution must be specified!" << endl;
154 if ( curropt->tag == "rescale" ) {
155 if ( curropt->values(nuavg, nustd) < 2 )
156 cerr << "-rescale: Both average and standard deviation must be specified!" << endl;
157 else if ( nustd <= 0 )
158 cerr << "-rescale: A positive standard deviation must be specified!" << endl;
159 }
160 if ( curropt->tag == "RPS" )
161 rpsfile = curropt->filename();
162 if ( curropt->tag == "FSC" )
163 fscfile = curropt->filename();
164 if ( curropt->tag == "Reference" )
165 reffile = curropt->filename();
166 if ( curropt->tag == "Mask" )
167 maskfile = curropt->filename();
168 if ( curropt->tag == "mask" )
169 realmask = curropt->filename();
170 }
171 option_kill(option);
172
173 double ti = timer_start();
174
175#ifdef HAVE_GCD
176 fftwf_init_threads();
177 fftwf_plan_with_nthreads(system_processors());
178#endif
179 if ( verbose )
180 cout << "Number of threads: " << system_processors() << endl;

187 nudatatype = p->data_type(); // Preserve old type

233

235 p->fspace_weigh(pref, pmask, resolution);
236 delete pref;

238 if ( fscfile.length() )
239 p->fspace_weigh_FSC_curve(plot, resolution);
240// else if ( rpsfile.length() )
241 else
242 p->fspace_weigh_RPS_curve(plot, resolution);
243 } else if ( weigh_dose ) {
244 p->fspace_weigh_dose(weigh_dose);
245 } else if ( normalize > 0 ) {
246 p->fspace_normalize_radial(pmask, resolution, (normalize==1));
247 } else if ( root ) {
248 p->fspace_sqrt_amp();
249 } else if ( square ) {
250 p->fspace_square_amp();
251 } else {
252 p->fspace_weigh_C_curve(resolution);
253 }
254
255 if ( plot ) delete plot;
256 if ( pmask ) delete pmask;
257
258 if ( nustd > 0 ) p->rescale_to_avg_std(nuavg, nustd);
259
260 if ( optind < argc ) {
261 p->change_type(nudatatype);
262 write_img(argv[optind], p, 0);
263 }
264
265 delete p;
266
267#ifdef HAVE_GCD
268 fftwf_cleanup_threads();
269#endif
270
271
272 timer_report(ti);
273
274 bexit(0);
