228 double elastic_scattering_amplitude(double s) {
229 long j, k;
230 double s2(s*s), amp(coef[10]);
231 for ( j=0, k=5; j<5; ++j, ++k )
232 amp += coef[j]*exp(-coef[k]*s2);
233 return amp;
234 }
235 vector<double> elastic_scattering_curve(double ds, double scut) {
236 if ( ds < 1e-10 ) ds = 0.01;
237 long i, imax(scut/ds + 1);
238 vector<double> scat;
239 for ( i=0; i<imax; ++i )
240 scat.push_back(elastic_scattering_amplitude(ds*i));
241 return scat;
242 }
243 // Burge & Smith (1962) formula
244 double elastic_scattering_integral(double slo, double shi) {
245 long i, j, k, l;
246 double a, b, cs(0);
247 for ( i=0, k=5; i<5; ++i, ++k ) {
248 for ( j=0, l=5; j<5; ++j, ++l ) {
249 b = coef[k] + coef[l];
250 a = coef[i]*coef[j]/b;
251 cs += a*(exp(-b*slo*slo) - exp(-b*shi*shi));
252 }
253 }
254 return cs;
255 }

259 double elastic_cross_section(double volt, double slo, double shi) {
260 double wlr = electron_wavelength_relativistic(volt);
261 return 4*M_PI*wlr*wlr*elastic_scattering_integral(slo, shi);
262 }
263 double inelastic_cross_section_langmore(double volt) {
264 return inelastic_cross_section_langmore(volt, 0, 1e10);
265 }
266 double inelastic_cross_section_langmore(double volt, double slo, double shi) {
267 double wlr = electron_wavelength_relativistic(volt);
268 double g = lorentz(volt);
269 double cs = 0.25*wlr*wlr*sqrt(ind)*log(2*(1+1/g)*volt/20);
270 if ( ind == 1 ) cs *= 0.154; // Correction for hydrogen
271 cs *= shi/(shi + 0.054) - slo/(slo + 0.054);
272 return cs;
273 }
