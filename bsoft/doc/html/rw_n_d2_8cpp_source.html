118{
119 char c(0);

153 break;
154 case 6:

161 break;

225 fimg.seekg(it.pos, fimg.beg);
226 chunk_list.push_back(read_chunk_header(fimg));
227 ChunkHeader& header = chunk_list.back();
228 start = it.pos + 18 + header.off;
229 fimg.seekg(start, fimg.beg);
230 param = read_param(fimg);
231 fimg.read((char *)&vi, sizeof(int));
232 if ( verbose & VERB_DEBUG_ND2 ) {
233 cout << start << endl;
234 cout << param << endl;
235 cout << fimg.tellg() << endl;
236 cout << "Attributes = " << vi << endl;
237 }
238 fimg.read((char *)&len, sizeof(long));
239 for ( i=0; i<vi; ++i )
240 attr.push_back(read_attribute(fimg));
241 }
242 if ( it.tag.find("ImageDataSeq") != string::npos ) {
243 vi = stoi(it.tag.substr(it.tag.find("|")+1));
244 fimg.seekg(it.pos, fimg.beg);
245 chunk_list.push_back(read_chunk_header(fimg));

271 cout << it.first << "\t" << it.second << endl;
272
273 for ( auto it: attr ) {
274 if ( it.tag == "uiWidth" ) p->sizeX(it.val);
275 if ( it.tag == "uiWidthBytes" ) line_size = it.val;
276 if ( it.tag == "uiHeight" ) p->sizeY(it.val);
277 if ( it.tag == "uiComp" ) p->channels(it.val);
278 if ( it.tag == "uiBpcInMemory" ) {
279 switch (int(it.val)) {
280 case 8: p->data_type(UCharacter); break;
281 case 16: p->data_type(UShort); break;
282 case 32: p->data_type(UInteger); break;
283 default: p->data_type(UCharacter);
284 }
285 }
286 if ( it.tag == "uiSequenceCount" ) p->images(it.val);
287 if ( it.tag == "dCalibration" ) px = 1.0e4*it.val;
288 }
289
290 p->sampling(px, px, 1);
291 p->sizeZ(1);
292 if ( p->channels() == 1 ) p->compound_type(TSimple);
293 else p->compound_type(TMulti);
294 p->data_offset(img_pos[0]);
295
296 if ( readdata ) {
297 p->data_alloc();
298 unsigned char* data = p->data_pointer();
299 // Should this be line_size?
300 long readsize(p->sizeX()*p->channels()*p->data_type_size());
301 for ( auto it: img_pos ) {
302 for ( i=0, j=it.second; i<p->sizeY(); ++i, j+=line_size ) {
303 fimg.seekg(j, fimg.beg);
304 fimg.read((char *)data, readsize);
305 data += readsize;
306 }
307 }
308 }
309
310 fimg.close();
311
312 return 0;
