435int Bimage::shrink_wrap(Vector3<long> nusize, Vector3<long> translate)
436{
437 if ( !data_pointer() ) return -1;

439 if ( translate[0] == 0 && translate[1] == 0 && translate[2] == 0 &&

444 long nux, nuy, nuz;
445
446 if ( verbose & VERB_PROCESS ) {
447 cout << "Shrinking and wrapping:" << endl;
448 cout << "New size: " << nusize << endl;
449 cout << "New origin: " << translate << endl << endl;

451 cout << "Shrinking and wrapping" << endl << endl;

455 long nudatasize = nusize[0]*nusize[1]*nusize[2]*n*c;

461 nuz = zz + translate[2];
462 if ( nuz < 0 ) nuz += nusize[2];
463 if ( nuz >= nusize[2] ) nuz -= nusize[2];
464 for ( yy=0; yy<y; yy++ ) {

470 if ( nux < 0 ) nux += nusize[0];
471 if ( nux >= nusize[0] ) nux -= nusize[0];
472 j = (((nn*nusize[2] + nuz)*nusize[1] + nuy)*nusize[0] + nux)*c;
473 for ( cc=0; cc<c; cc++, i++, j++ )
