143Batomtype* atom_type_add(Batomtype** atype, const char* aname)
144{
145 Batomtype* this_atype = *atype;
146 Batomtype* new_atype = new Batomtype;
147 memset(new_atype, 0, sizeof(Batomtype));
148
149 if ( aname ) strncpy(new_atype->name, aname, 8);
150 if ( aname ) {
151 while ( isspace(aname[0]) ) aname++;
152 strncpy(new_atype->name, aname, 6);
153 if ( isalpha(aname[0]) ) {
154 new_atype->el[0] = toupper(aname[0]);
155 if ( isalpha(aname[1]) ) {
156 if ( islower(aname[1]) ) new_atype->el[1] = aname[1];
157 }
158 }
159 }
160
161 if ( !this_atype )
162 *atype = new_atype;
163 else {
164 while ( this_atype->next ) this_atype = this_atype->next;
165 this_atype->next = new_atype;
166 }
167
168 return new_atype;
169}

192Batomtype* read_atom_prop_star(Bstring& filename)
193{
194 Bstar star;
195
196 if ( star.read(filename.c_str()) < 0 ) {
197 error_show(filename.c_str(), __FILE__, __LINE__);
198 return NULL;
199 }
200
201 BstarBlock block = star.block(0);
202 BstarLoop loop = block.loop(0);
203
204 // Check if it contains atom parameters
205 if ( loop.find(ATOM_TYPE_SYMBOL) < 0 ) {
206 cerr << "Error: Star to atom properties conversion unsuccessful!" << endl;
207 return NULL;
208 }
209
210 Batomtype* at = NULL;
211 Batomtype* at_first = NULL;
212 Bstring symbol;
213 long j, k;
214
215 if ( verbose & VERB_DEBUG )
216 cout << "DEBUG read_atom_prop_star: Reading " << filename << endl;
217
218 // Convert the atomic data
219 for ( auto il = block.loops().begin(); il != block.loops().end(); ++il ) {
220 if ( ( j = il->find(ATOM_TYPE_SYMBOL) ) >= 0 ) {
221 for ( auto ir = il->data().begin(); ir != il->data().end(); ++ir ) {
222 symbol = (*ir)[j];
223 if ( at ) at = atom_type_add(&at, symbol.c_str());
224 else at = atom_type_add(&at_first, symbol.c_str());
225 if ( ( k = il->find(ATOM_TYPE_NUMBER) ) >= 0 ) at->z = to_integer((*ir)[k]);
226 if ( ( k = il->find(ATOM_TYPE_MASS) ) >= 0 ) at->mass = to_real((*ir)[k]);
227 if ( ( k = il->find(ATOM_TYPE_OXIDATION) ) >= 0 ) at->oxid = to_real((*ir)[k]);
228 if ( ( k = il->find(ATOM_TYPE_RADIUS_BOND) ) >= 0 ) at->bond = to_real((*ir)[k]);
229 if ( ( k = il->find(ATOM_TYPE_RADIUS_VDW) ) >= 0 ) at->vdw = to_real((*ir)[k]);
230 if ( ( k = il->find(ATOM_TYPE_SCAT_A1) ) >= 0 ) at->sfa[0] = to_real((*ir)[k]);
231 if ( ( k = il->find(ATOM_TYPE_SCAT_A2) ) >= 0 ) at->sfa[1] = to_real((*ir)[k]);
232 if ( ( k = il->find(ATOM_TYPE_SCAT_A3) ) >= 0 ) at->sfa[2] = to_real((*ir)[k]);
233 if ( ( k = il->find(ATOM_TYPE_SCAT_A4) ) >= 0 ) at->sfa[3] = to_real((*ir)[k]);
234 if ( ( k = il->find(ATOM_TYPE_SCAT_A5) ) >= 0 ) at->sfa[4] = to_real((*ir)[k]);
235 if ( ( k = il->find(ATOM_TYPE_SCAT_B1) ) >= 0 ) at->sfb[0] = to_real((*ir)[k]);

272int write_atom_prop_star(Bstring& filename, Batomtype* at_first)
273{
274 if ( !at_first ) {
275 cerr << "Error: No atom parameters to write!" << endl;
276 return -1;
277 }
278
279 Bstar star;
280 BstarBlock& block = star.add_block("Atomic_parameters");
281 BstarLoop& loop = block.add_loop();
282
283 loop.tags() = atom_type_tags();
284
285 Batomtype* at;
286
287 star.line_length(120);
288
289 // Count the number of atom types
290 int n, err(0);
291 for ( n=0, at = at_first; at; at = at->next, n++ ) {
292 vector<string>& vs = loop.add_row(6);
293 vs[0] = at->name;
294 vs[1] = to_string(at->z);
295 vs[2] = to_string(at->mass);
296 vs[3] = to_string(at->oxid);
297 vs[4] = to_string(at->bond);
298 vs[5] = to_string(at->vdw);
299 vs[6] = to_string(at->sfa[0]);
300 vs[7] = to_string(at->sfa[1]);
301 vs[8] = to_string(at->sfa[2]);
302 vs[9] = to_string(at->sfa[3]);
303 vs[10] = to_string(at->sfa[4]);
304 vs[11] = to_string(at->sfb[0]);
305 vs[12] = to_string(at->sfb[1]);
306 vs[13] = to_string(at->sfb[2]);
307 vs[14] = to_string(at->sfb[3]);
308 vs[15] = to_string(at->sfb[4]);
309 vs[16] = to_string(at->sfc);
310 }
311
312 err = star.write(filename.c_str());
313
314 if ( err < 0 ) return err;
