190double md_bond_forces(Bmolgroup* molgroup, double Kbond, int wrap)
191{
192 if ( Kbond <= 0 ) return 0;
193
194 double dist, dev, fac, energy(0);
195 Vector3<double> d, Force;
196 Bbond* bond;
197 Bbond* bondlist = molgroup->bond;
198
199 for ( bond = bondlist; bond; bond = bond->next ) {
200 if ( wrap )
201 d = vector3_difference_PBC(bond->atom1->coord, bond->atom2->coord, molgroup->box);
202 else
203 d = bond->atom1->coord - bond->atom2->coord;
204 dist = d.length();
205 dev = dist - bond->l;
206 energy += Kbond*dev*dev;
207 fac = -2*Kbond*dev/dist;
208 Force = d * fac;
209 bond->atom1->F += Force;
210 bond->atom2->F -= Force;
211 }
212
213 return energy;
214}

284double md_nonbonded_forces(Bmolgroup* molgroup, Bmd* md)
285{
286 if ( md->Kvdw < 0 ) md->Kvdw = 0;
287 if ( md->Kelec < 0 ) md->Kelec = 0;
288 if ( md->Kvdw <= 0 && md->Kelec <= 0 ) return 0;

291 long i, ii, x, y, z, xx, yy, zz, ix, iy, iz;
292 Vector3<double> box = molgroup->box;
293 Vector3<double> sampling(md->cutoff, md->cutoff, md->cutoff);
294 Vector3<int> size((int) (box[0]/sampling[0] + 0.001),
295 (int) (box[1]/sampling[1] + 0.001), (int) (box[2]/sampling[2] + 0.001));
296 size = size.max(1);
297 for ( i=0; i<3; i++ ) sampling[i] = box[i]/size[i] + 0.001;

300

302
303 md->Evdw = md->Eelec = 0;
304
305 for ( z=0; z<size[2]; z++ ) {
306 for ( y=0; y<size[1]; y++ ) {
307 for ( x=0; x<size[0]; x++ ) {
308 i = (z*size[1] + y)*size[0] + x;
309 for ( latom = alist[i]; latom; latom = latom->next ) {
310 for ( zz=z-1; zz<=z+1; zz++ ) {
311 iz = zz;
312 if ( iz < 0 ) iz += size[2];
313 if ( iz >= size[2]) iz -= size[2];
314 for ( yy=y-1; yy<=y+1; yy++ ) {
315 iy = yy;
316 if ( iy < 0 ) iy += size[1];
317 if ( iy >= size[1] ) iy -= size[1];
318 for ( xx=x-1; xx<=x+1; xx++ ) {
319 ix = xx;
320 if ( ix < 0 ) ix += size[0];
321 if ( ix >= size[0] ) ix -= size[0];
322 ii = (iz*size[1] + iy)*size[0] + ix;
323 for ( latom2 = alist[ii]; latom2; latom2 = latom2->next ) {
324 dononbond = 1;
325 if ( latom2->atom == latom->atom ) dononbond = 0; // Same atom
326 else if ( latom2->atom->next == latom->atom ) dononbond = 0; // Bonded atoms
327 else if ( latom2->atom == latom->atom->next ) dononbond = 0; // Bonded atoms

329 if ( latom2->atom->next->next == latom->atom ) dononbond = 0;

369int atom_nonbonded_forces(Batom* atom, Batom* atom2, Bmd* md, Vector3<double> box)
370{
371 if ( md->Kvdw < 0 ) md->Kvdw = 0;
372 if ( md->Kelec < 0 ) md->Kelec = 0;
373 if ( md->Kvdw <= 0 && md->Kelec <= 0 ) return 0;
374
375 Vector3<double> d, Force;
376 double rd2, rd6, rd12, dist2, invdist2, fac;
377 Bbondtype* bt = NULL;
378
379 if ( md->wrap )
380 d = vector3_difference_PBC(atom2->coord, atom->coord, box);
381 else
382 d = atom2->coord - atom->coord;
383 dist2 = d.length2();
384 if ( dist2 < md->cutoff*md->cutoff ) {
385 invdist2 = 1/dist2;
386 if ( md->Kvdw ) {
387 bt = md_find_bond_type(atom, atom2, md->bond);
388 if ( bt && bt->vdwdist > 0 ) {
389 rd2 = bt->vdwdist*bt->vdwdist*invdist2;
390 rd6 = rd2*rd2*rd2;
391 rd12 = rd6*rd6;
392 md->Evdw += md->Kvdw*(md->VdWcoeff1*rd12 - md->VdWcoeff2*rd6);
393 fac = -md->Kvdw*(rd12 - rd6)*invdist2;
394 Force = d * fac;
395 atom->F += Force;
396 }
397 }
398 if ( md->Kelec ) {
399 fac = md->Kelec*atom->chrg*atom2->chrg*sqrt(invdist2);
400 md->Eelec += fac;
401 fac *= invdist2;
402 Force = d * fac;
403 atom->F += Force;
404 }
405 }
406
407 return 0;

429double md_point_force(Bmolgroup* molgroup, Vector3<double> point, double Kpoint, double decay)
430{
431 if ( Kpoint <= 0 ) return 0;
432
433 double dist, en1, energy = 0;
434 Vector3<double> dir;
435 Bmolecule* mol;
436 Bresidue* res;
437 Batom* atom;
438
439 for ( mol = molgroup->mol; mol; mol = mol->next ) {
440 for( res = mol->res; res; res = res->next ) {
441 for ( atom = res->atom; atom; atom = atom->next ) {
442 dir = atom->coord - point;
443 dist = dir.length();
444 en1 = Kpoint*exp(-decay*dist);
445 energy += en1;
446 atom->F += dir * (decay*en1/dist);
447 }
448 }
449 }
450
451 return energy;
