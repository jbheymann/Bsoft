39{

42 double voxel_density(density*sampling(0).volume());
43 int local_psd(0);

51 mn = mx = resolution*0.5;
52 psd->show_minimum(resolution*0.5);
53 psd->show_maximum(resolution*0.5);
54 scale = 1;
55 psd->data_alloc();
56 for ( i=0; i<slice_size; ++i )
57 psd->set(i, psd->show_minimum());
58 } else {
59 mn = psd->minimum();
60 mx = psd->maximum();
61 scale = (psd->show_maximum() - psd->show_minimum())/(mx - mn);
62 for ( i=0; i<slice_size; ++i )
63 psd->set(i, scale*(*psd)[i] + psd->show_minimum());
64 }

71 show_maximum((nz - 1)*image->sampling()[2] - resolution);
72 }
73
74 double loz(show_minimum()/image->sampling()[2]);
75 scale = (show_maximum()-show_minimum())/(image->sampling()[2]*(max-min));
76 if ( show_minimum() > show_maximum() ) factor *= -1;
77
78 if ( verbose & VERB_PROCESS ) {
79 cout << "Converting a 2D image into a 3D surface density" << endl;
80 cout << "New z dimension: " << nz << endl;
81 cout << "Dimension scale: " << sampling(0) << " A/pixel" << endl;
82 cout << "Density scale: " << scale << " A/(unit density)" << endl;
83 cout << "Z min & max: " << show_minimum() << " A " << show_maximum() << " A" << endl;
84 cout << "Z thickness: " << show_maximum() - show_minimum() << " A" << endl;

88 cout << "Density within surface: " << density << " Da/A3" << endl << endl;
89 } else if ( verbose & VERB_LABEL )
90 cout << "Converting a 2D image into a 3D surface density" << endl << endl;
91
92 for ( i=yy=0; yy<y; ++yy ) {
93 for ( xx=0; xx<x; ++xx, ++i ) {
94 zdis = loz + scale*((*this)[i]-min);
95 for ( j=i, zz=0; zz<psurf->z; ++zz, j+=slice_size )
96 psurf->set(j, voxel_density/(1+exp(factor*(zz-zdis)/(*psd)[i])));

98 }
99
100 if ( local_psd ) delete psd;

199

201 nuvec[1] = yy - nuorigin[1];
202 for ( xx=0; xx<x; ++xx, ++i ) {
203 nuvec[0] = xx - nuorigin[0];
204 for ( zz=z-1; zz; zz-- ) {
205 nuvec[2] = zz - nuorigin[2];
206 old = mat * nuvec + oldorigin;
207 d = interpolate(old);
208 if ( d > threshold ) {
209 dz = u[2]*zz;
210 if ( (*pheight)[i] <= 0 ) pheight->set(i, dz);
211 if ( bottom > dz ) bottom = dz;
212 }
213 }
214 }
215 }
216
217 for ( i=0; i<x*y; ++i ) if ( (*pheight)[i] ) pheight->add(i, -bottom);
218
219 pheight->statistics();
220
221 return pheight;
222}
