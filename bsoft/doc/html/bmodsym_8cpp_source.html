120 if ( curropt->tag == "parameters" )
121 paramfile = curropt->filename();

123 asu_sym = curropt->symmetry_string();

125 symmetry_apply = curropt->symmetry_string();
126 if ( curropt->tag == "consolidate" )

128 if ( curropt->tag == "lattice" ) {

138 if ( curropt->tag == "componentradius" )

144 if ( curropt->tag == "origin" )
145 origin = curropt->origin();
146 if ( curropt->tag == "annuli" )
147 if ( curropt->values(ann_min, ann_max) < 1 )

149 if ( curropt->tag == "width" )
150 if ( ( ann_width = curropt->value.integer() ) < 1 )
151 cerr << "-width: An annular width must be specified!" << endl;
152 if ( curropt->tag == "reference" )
153 ref_view = curropt->view();
154 if ( curropt->tag == "output" )

199

203 models_apply_point_group(model, symmetry_apply, origin, ref_view);
204
205 if ( lattice.volume() > 1 )
206 models_generate_lattice(model, uc, lattice);
207
208 if ( maxorder )
209 model_component_symmetry(model, 360, ann_min, ann_max, ann_width, 0, 0, 0, minorder, maxorder);
210
211 if ( reset ) models_select_all(model);
212
213 if ( merge ) model_merge(model);
214
215 models_selection_stats(model);
216
217 // Write an output parameter format file if a name is given
218 if ( model && ( outfile.length() || split == 9 ) )
219 write_model(outfile.str(), model, split);

221 delete model;
222
223 timer_report(ti);
224
225 bexit(0);
