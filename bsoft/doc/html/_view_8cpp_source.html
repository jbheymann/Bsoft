200 double alpha_start = theview.angle();

202

205 alpha_start -= alpha;
206 alpha_end += alpha;
207 }

213 theview.normalize();

223 axis2.normalize();
224

227 Quaternion qv(0, theview[0], theview[1], theview[2]);
228

231
232 if ( verbose & VERB_DEBUG ) {

235 << ", phi_step: " << phi_step*180/M_PI
236 << ", alpha_step: " << alpha_step*180/M_PI << endl;
237 cout << "DEBUG views_within_limits: Axis1: " << axis1 << endl;
238 cout << "DEBUG views_within_limits: Axis2: " << axis2 << endl;
239 }
240
241 if ( theta_step < 1e-6 ) theta_step = 1e6;
242 if ( phi_step < 1e-6 ) phi_step = 1e6;
243 if ( alpha_step < 1e-6 ) alpha_step = 1e6;
244 for ( theta = theta_start; theta <= view_angle_limit; theta += theta_step ) {
245 q1 = Quaternion(axis1, theta);
246 for ( phi = phi_start; phi <= view_angle_limit; phi += phi_step ) {
247 q = Quaternion(axis2, phi);
248 q *= q1;
249 q = q.rotate(qv);
250 for ( alpha = alpha_start; alpha <= alpha_end; alpha += alpha_step ) {
251 v.push_back(View2<float>(q[1], q[2], q[3], alpha));
252 if ( verbose & VERB_DEBUG )
253 cout << "DEBUG views_within_limits: " << v.back() << " - " << theview.angle(v.back())*180/M_PI << endl;
254 }
255 }
256 }
257
258 return v;
259}
