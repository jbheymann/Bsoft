181int Bimage::noise_spectral(double alpha)
182{
183 long i, j, nn, xx, yy, zz, cc, imgsize(x*y*z);
184 double x2, y2, z2, s2, w, sum;
185 double fac(-alpha/4); // Factor of 2 because we work with s2, and another 2 for amplitude scaling
186 double range(2.0/get_rand_max());
187
188 if ( verbose & VERB_PROCESS ) {
189 cout << "Generating a spectral noise image:" << endl;
190 cout << "Alpha: " << alpha << endl << endl;
191 }
192
193 for ( i=nn=0; nn<n; nn++ ) {
194 sum = 0;
195 for ( zz=0; zz<z; zz++ ) {
196 z2 = zz*1.0L/z;
197 if ( z2 >= 0.5 ) z2 -= 1;
198 z2 *= z2;
199 for ( yy=0; yy<y; yy++ ) {
200 y2 = yy*1.0L/y;
201 if ( y2 >= 0.5 ) y2 -= 1;
202 y2 *= y2;
203 for ( xx=0; xx<x; xx++, i++ ) {
204 x2 = xx*1.0L/x;
205 if ( x2 >= 0.5 ) x2 -= 1;
206 x2 *= x2;
207 s2 = x2 + y2 + z2;
208 if ( s2 ) {
209 w = pow(s2, fac);
210 for ( cc=0, j=i*c; cc<c; cc++, j++ )
211 set(j, w*(random()*range-1));
212 sum += w*w;
213 }
214 }
215 }
216 }
217 sum = sqrt(sum/(3*imgsize));
218 for ( j=nn*imgsize; j<(nn+1)*imgsize; j++ ) set(j, complex(j)/sum);
219 }
220
221 fft_back();
222
223 statistics();
224
225 return 0;
226}
