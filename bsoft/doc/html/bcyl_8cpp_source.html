59 int optind;
60 Boption* option = get_option_list(use, argc, argv, optind);
61 Boption* curropt;
62 for ( curropt = option; curropt; curropt = curropt->next ) {
63 if ( curropt->tag == "symmetrize" ) symmetrize = 1;
64 if ( curropt->tag == "unwrap" ) {

66 cerr << "-unwrap: An angle must be specified!" << endl;

70 if ( curropt->tag == "autocorrelate" ) autocorrelate = 1;

74 cerr << "-rescale: Both average and standard deviation must be specified!" << endl;
75 else
76 setrescale = 1;

83 order = curropt->value;

86 }

149int img_plot_max_vs_angle(Bimage* p)
150{
151 long i, x, y(0), z, z1, z2, zm, n, dz(10);
152 double a, v, cc_max;
153
154 for ( n=0; n<p->images(); n++ ) {
155 z1 = 0; z2 = dz;
156 cout << "Image " << n+1 << ":" << endl << "Angle\tz\tCC" << endl;
157 for ( x=0; x<p->sizeX(); x++ ) {
158 a = x*TWOPI/p->sizeX();
159 for ( z=z1, zm=0, cc_max = p->minimum(); z<z2; z++ ) {
160 i = p->index(0, x, y, z, n);
161 v = (*p)[i];
162 if ( cc_max < v ) {
163 cc_max = v;
164 zm = z;
165 }
166 }
167 z1 = (zm > dz)? zm - dz: 0;
168 z2 = zm + dz;
169 cout << a*180.0/M_PI << tab << zm*p->sampling(0)[2] << tab << cc_max << endl;
170 }
171 cout << endl;
172 }
173
174 return 0;
