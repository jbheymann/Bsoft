18vector<pair<string,string>> tag_value_parse(string s)
19{
20// cout << s << endl;
21 vector<pair<string,string>> tv;
22
23 size_t i = s.find("<") + 1;
24 s = s.substr(i, s.find(">") - i);
25
26 vector<string> slist = split(s);
27 string tag, value;
28
29 for ( auto sp: slist ) {
30 i = sp.find("=");
31// cout << "sp = " << sp << endl;
32 if ( i != string::npos ) {
33 tag = sp.substr(0, i);
34 if ( sp[++i] == '\"' ) i++;
35 value = sp.substr(i, sp.rfind("\"") - i);
36 } else {
37 tag = sp;
38 value = "";
39 }
40 tv.push_back(make_pair(tag, value));
41// cout << "tv: " << tag << tab << value << endl;
42 }
43
44 return tv;
45}
