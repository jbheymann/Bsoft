253int atom_clean_type(Batom* atom, const char* type)
254{
255 if ( !type ) return 0;
256
257 while ( isspace(type[0]) ) type++;
258 if ( type[0] == '"' ) type++;
259
260 strncpy(atom->type, type, 8);
261
262 if ( atom->el[0] ) return 0;
263
264 if ( isdigit(type[0]) ) {
265 if ( type[1] == 'H' ) atom->el[0] = 'H';
266 else atom->el[0] = 'C';
267 } else if ( isalpha(type[0]) ) {
268 atom->el[0] = toupper(type[0]);
269 if ( isalpha(type[1]) ) {
270 if ( islower(type[1]) ) atom->el[1] = type[1];
271 // Special cases
272 if ( atom->el[0] == 'M' )
273 atom->el[1] = tolower(type[1]);
274 }
275 } else {
276 cout << "Warning: No element assigned for atom " << atom->type << endl;
277 }
278
