154

173Matrix seq_dot_plot_mov_avg(Matrix dot_plot, int window)
174{
175 int isize = dot_plot.columns(), jsize = dot_plot.rows();
176
177 if ( window < 1 ) window = 1;
178 if ( window > 1000 ) window = 1000;
179 int i, j, w, start, end;
180 int shortest = isize;
181 if ( shortest > jsize ) shortest = jsize;
182 int low_half = window/2;
183 int up_half = window - low_half - 1;
184
185 Matrix mov_avg(isize, jsize);
186
187 for ( j=0; j<jsize; j++ ) { // Diagonals in upper triangle
188 for ( i=0; i<jsize-j && i<shortest; i++ ) {
189 start = i - low_half;
190 if ( start < 0 ) start = 0;
191 end = i + up_half;
192 if ( end >= shortest ) end = shortest - 1;
193 if ( end >= jsize-j ) end = jsize - j - 1;
194 for ( w=start; w<=end; w++ )
195 mov_avg[i+j][i] += dot_plot[w+j][w];
196 }
197 }
198
199 for ( i=1; i<isize; i++ ) { // Diagonals in lower triangle
200 for ( j=0; j<isize-i && j<shortest; j++ ) {
201 start = j - low_half;
202 if ( start < 0 ) start = 0;
203 end = j + up_half;
204 if ( end >= shortest ) end = shortest - 1;
205 if ( end >= isize-i ) end = isize - i - 1;
206 for ( w=start; w<=end; w++ )
207 mov_avg[j][i+j] += dot_plot[w][i+w];
208 }
209 }
210
211 return mov_avg;
212}
