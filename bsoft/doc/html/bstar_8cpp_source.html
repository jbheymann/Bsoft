50int main(int argc, char **argv)
51{
52 // Initialize variables
53 long header(0); // Show header lines using this length
54 int split(0); // Flag to split into blocks
55 int showflag(1); // Default show tags
56 string tag_to_list; // STAR item which should be listed
57 string tag_to_be_deleted; // STAR item which should be deleted
58 string block_to_be_deleted; // STAR item in block which should be deleted
59 string tag_integer; // Integer STAR item which should be scaled and shifted
60 int iscale(1);

67 // Initialize the STAR database
68 Bstar star;
69 star.line_length(120); // Set the output line length
70
71 int optind;
72 Boption* option = get_option_list(use, argc, argv, optind);
73 Boption* curropt;
74 Bstring* strarr;
75 for ( curropt = option; curropt; curropt = curropt->next ) {
76 if ( curropt->tag == "header" )
77 if ( ( header = curropt->value.integer() ) < 1 )
78 cerr << "-header: A line length must be specified!" << endl;
79 strarr = curropt->value.split(",");
80 if ( curropt->tag == "split" ) {
81 if ( curropt->value.contains("dat") ) split = 9;
82 else if ( ( split = curropt->value.integer() ) < 1 )
83 cerr << "-split: An integer must be specified!" << endl;
84 else
85 if ( split > 4 ) split = 4;
86 }
87 if ( curropt->tag == "linelength" ) {
88 if ( curropt->value.integer() < 1 )
89 cerr << "-linelength: A line length must be specified!" << endl;
90 else
91 star.line_length(curropt->value.integer());
92 }
93 if ( curropt->tag == "list" ) {
94 showflag = 0;
95 tag_to_list = curropt->value.c_str();
96 if ( tag_to_list.size() < 1 )
97 cerr << "-list: A STAR tag must be specified!" << endl;
98 }
99 if ( curropt->tag == "delete" ) {
100 showflag = 0;
101 tag_to_be_deleted = curropt->value.c_str();
102 if ( tag_to_be_deleted.length() < 1 )
103 cerr << "-delete: A STAR tag must be specified!" << endl;
104 }
105 if ( curropt->tag == "blockdelete" ) {
106 showflag = 0;
107 block_to_be_deleted = curropt->value.c_str();
108 if ( block_to_be_deleted.length() < 1 )
109 cerr << "-blockdelete: A STAR tag must be specified!" << endl;
110 }
111 if ( curropt->tag == "integerscale" ) {
112 showflag = 0;
113 tag_integer = strarr->c_str();
114 if ( tag_integer.length() < 1 )
115 cerr << "-integerscale: A STAR tag must be specified!" << endl;
116 else {
117 if ( strarr->next ) iscale = strarr->next->integer();
118 if ( strarr->next->next ) ishift = strarr->next->next->integer();
119 }
120 }
121 if ( curropt->tag == "floatscale" ) {
122 showflag = 0;
123 tag_float = strarr->c_str();
124 if ( tag_float.length() < 1 )
125 cerr << "-floatscale: A STAR tag must be specified!" << endl;
126 else {
127 if ( strarr->next ) scale = strarr->next->real();
128 if ( strarr->next->next ) shift = strarr->next->next->real();
129 }
130 }
131 if ( curropt->tag == "output" )
132 outstar = curropt->filename().c_str();
133 string_kill(strarr);
134 }
135 option_kill(option);
136
137 double ti = timer_start();
138
139 // Read all the STAR format files
140 while ( optind < argc )
141 star.read(argv[optind++]);
142
143 if ( header )
144 star.list_comments(header);
145
146 if ( tag_to_list.length() > 0 )
147 item_list(star, tag_to_list);
148
149 if ( tag_to_be_deleted.length() > 0 )
150 item_delete_all(star, tag_to_be_deleted);
151
152 if ( block_to_be_deleted.length() > 0 )
153 star.erase(block_to_be_deleted);
154
155 if ( tag_integer.length() > 0 )
156 item_scale_shift(star, tag_integer, iscale, ishift, 0);
157
158 if ( tag_float.length() > 0 )
159 item_scale_shift(star, tag_float, scale, shift, 1);
160
161 if ( showflag && verbose ) show_tags(star);
162
163 // write an output STAR format file if a name is given
164 if ( outstar.length() ) {
165 star.comment(star.comment() + command_line_time().c_str());
166 star.write(outstar, split);
167 }
168
169 timer_report(ti);
170
171 bexit(0);
172}

179int show_tags(Bstar& star)
180{
181 int ntags(0);
182 string tag;
183
184 for ( auto ib: star.blocks() ) {
185 cout << endl << "data_" << ib.tag() << endl;
186 cout << endl << "loop_" << endl;
187 cout << "_star_tag" << endl;
188 cout << "_star_loop" << endl;
189 cout << "_star_type" << endl;
190 for ( auto it: ib.items() ) {
191 if ( verbose & VERB_DEBUG )
192 cout << "DEBUG show_tags: tag = " << it.first << tab << it.second << endl;
193 tag = "\"_" + it.first + "\"";
194 cout << left << setw(40) << tag << " no ";
195 if ( check_for_number(it.second) )
196 cout << "Number" << endl;
197 else
198 cout << "String" << endl;
199 ntags++;
200 }
201 for ( auto il: ib.loops() ) {
202 map<string, int>& t = il.tags();
203 for ( auto it: t ) {
204 if ( verbose & VERB_DEBUG )
205 cout << "DEBUG show_tags: loop tag = " << it.first << tab << it.second << endl;
206 tag = "\"_" + it.first + "\"";
207 cout << left << setw(40) << tag << " yes ";
208 if ( check_for_number(il[0][it.second]) )
209 cout << "Number" << endl;
210 else
211 cout << "String" << endl;
212 ntags++;
213 }
214 }
215 cout << endl;
216 }
217 cout << endl;
218
219 return ntags;
