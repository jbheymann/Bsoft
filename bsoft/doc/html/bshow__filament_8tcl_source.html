275## @brief Draws a circle for a node on the canvas.
276#
277# @param fid Filament identifier.
278# @param nid Node identifier.
279
280proc nodeDraw { fid nid } {
281 global theimg
282 global node_selected show_node_labels
283 global filament_color filament_first_color filament_select_color
284 set c [getImageCanvas $theimg]
285 set wc [getControlWindow $theimg]
286 set n [$wc.image.scale get]
287 set z [$wc.slice.scale get]
288 set mg_item [micrographItem $n]
289 set loc [Bmg node $mg_item location $fid $nid]
290# puts "drawing $fid $nid: $loc"
291 set scale [$wc.scale.scale get]
292 set x [expr [lindex $loc 0] * $scale ]
293 set y [expr [yFlip [lindex $loc 1]] * $scale ]
294 set zr [expr abs([lindex $loc 2] - $z) ]
295 set radius [expr [.wfil.node_radius.entry get] * $scale]
296 set xmin [expr $x - $radius ]
297 set xmax [expr $x + $radius + $scale - 1 ]
298 set ymin [expr $y - $radius ]
299 set ymax [expr $y + $radius + $scale - 1 ]
300 set color $filament_color
301 if { $nid == 1 } {
302 set color $filament_first_color
303 $c create text $xmin $y -text $fid -fill blue -anchor e -tags node
304 }
305 if { $fid == [lindex $node_selected 0] && $nid == [lindex $node_selected 1] } {
306 set color $filament_select_color
307 }
308 $c create oval $xmin $ymin $xmax $ymax -outline $color -width 1 -tags node
309 if { $show_node_labels } {
310 $c create text $xmax $y -text $nid -fill $color -anchor w -tags node
311 }
312}

314## @brief Draws a line between two nodes on the canvas.
315#
316# @param fid Filament identifier.
317# @param nid1 Node 1 identifier.
318# @param nid2 Node 2 identifier.
319
320proc nodeLinkDraw { fid nid1 nid2 } {
321 global theimg
322 global filament_color
323 set c [getImageCanvas $theimg]
324 set wc [getControlWindow $theimg]
325 set n [$wc.image.scale get]
326 set z [$wc.slice.scale get]
327 set mg_item [micrographItem $n]
328 set loc1 [Bmg node $mg_item location $fid $nid1]
329 set loc2 [Bmg node $mg_item location $fid $nid2]
330 set scale [$wc.scale.scale get]
331 set x1 [expr [lindex $loc1 0] * $scale ]
332 set y1 [expr [yFlip [lindex $loc1 1]] * $scale ]
333 set x2 [expr [lindex $loc2 0] * $scale ]
334 set y2 [expr [yFlip [lindex $loc2 1]] * $scale ]
335 set fz -1.0
336 set dz [expr [lindex $loc2 2] - [lindex $loc1 2]]
337 if { $dz == 0 } {
338 set fz [expr [lindex $loc2 2] - $z + 0.5]
339 } else {
340 set fz [expr ([lindex $loc2 2] - $z)/($dz*1.0)]
341 }
342 if { $fz >= 0.0 && $fz <= 1.0 } {
343 $c create line $x1 $y1 $x2 $y2 -fill $filament_color -width 1 -tags fil
344 }
345}

626 for {set i 1} {$i <= $len} {incr i 1} {
627 set v [lindex $prof $i]
628 if { $min > $v } { set min $v }
629 if { $max < $v } { set max $v }
630 }
631 if { $max <= $min } { set max [expr $min + 1] }
632# puts "profile: $min $max"
633 set xoff 5
634 set yoff [expr $fil_plot_height * ( (3*$max - $min) * 0.25 / ($max - $min)) + 5]
635 set xscale [expr $fil_plot_width*1.0/$len]
636 set yscale [expr $fil_plot_height*0.5/($max - $min)]
637
638# puts "$yoff $yscale"
639
640 set fp ""
641 set fps ""
642 for {set i 1} {$i <= $len} {incr i 1} {
643 set j [expr $i + $len]
644 set x [expr ($i-1)*$xscale + $xoff]
645 set y [expr $yoff - [lindex $prof $i] * $yscale]
646 set ys [expr $yoff - [lindex $prof $j] * $yscale]
647 append fp " " $x " " $y
648 append fps " " $x " " $ys
649 }
650 $g create line $fp -fill blue -smooth yes -width 1 -tags filplot
651 $g create line $fps -fill red -smooth yes -width 1 -tags filplot
652 set orix [expr $fil_plot_width/2 + $xoff]
653 set oriy $yoff
654 $g create line $orix 0 $orix [expr 2*$yoff + $fil_plot_height] \
655 -fill black -smooth yes -width 1 -tags filplot
656 $g create line 0 $oriy [expr 2*$xoff + $fil_plot_width] $oriy \
657 -fill black -smooth yes -width 1 -tags filplot
658}
