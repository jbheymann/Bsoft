220 for ( y=0; y<p->sizeY(); y++ ) {
221 vec[1] = y - center[1];
222 for ( x=0; x<p->sizeX(); x++, i++ ) {
223 vec[0] = x - center[0];
224 d = vec.length();
225 if ( d < min ) p->set(i,1);
226 else if ( d > max ) p->set(i,0);
