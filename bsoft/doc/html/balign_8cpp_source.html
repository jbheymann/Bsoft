156Transform img_align(Bimage* p1, Bimage* p2, Vector3<long> tile_size,
157 double res_lo, double res_hi, double max_shift, int filter_flag, int refine_flag)
158{
159 long i, iter, maxiter(5);
160 Vector3<double> shift;
161 Transform t, best_t;
162 best_t.fom = 1e37;
163
164 Vector3<long> start1, common_size, step_size;
165
166 // Find the largest area covering both images
167 common_size = p1->size().min(p2->size());
168 common_size = common_size.max(1);
169
170 // Ensure the tile fits into the images
171 tile_size = tile_size.max(1);
172 tile_size = tile_size.min(common_size);
173 if ( tile_size[0] >= common_size[0] && common_size[0] > 1 ) {
174 tile_size[0] = (int) (0.4*common_size[0]);
175// start1[0] = (common_size[0] - tile_size[0])/2;
176 }
177 if ( tile_size[1] >= common_size[1] && common_size[1] > 1 ) {
178 tile_size[1] = (int) (0.4*common_size[1]);
179// start1[1] = (common_size[1] - tile_size[1])/2;
180 }
181 if ( tile_size[2] >= common_size[2] && common_size[2] > 1 ) {
182 tile_size[2] = (int) (0.4*common_size[2]);
183// start1[2] = (common_size[2] - tile_size[2])/2;
184 }
185
186 if ( max_shift < 1 ) max_shift = tile_size[0]/4;
187
188 Vector3<double> half_tile;
189 half_tile[0] = (tile_size[0] - 1)/2;
190 half_tile[1] = (tile_size[1] - 1)/2;
191 half_tile[2] = (tile_size[2] - 1)/2;
192
193 Bimage* pex1 = NULL;
194 Bimage* pex2 = NULL;
195 Bmarker* set1 = NULL;
196 Bmarker* set2 = NULL;
197 Bmarker* m1 = NULL;
198 Bmarker* m2 = NULL;
199 Bstring fn1(p1->file_name()), fn2(p2->file_name());
200 Bstring filename = fn1.pre_rev('.') + "_" + fn2.pre_rev('.') + ".ps";
201
202 if ( verbose & VERB_LABEL ) {
203 cout << "Aligning images ";
204 if ( refine_flag ) cout << "with ";
205 else cout << "without ";
206 cout << "shift refinement." << endl << endl;
207 }
208
209 if ( common_size[0] > 2*tile_size[0] )
210 start1[0] = (common_size[0]
211 - (int) (tile_size[0]*(common_size[0]/tile_size[0])))/2;
212 if ( common_size[1] > 2*tile_size[1] )
213 start1[1] = (common_size[1]
214 - (int) (tile_size[1]*(common_size[1]/tile_size[1])))/2;
215 if ( common_size[2] > 2*tile_size[2] )
216 start1[2] = (common_size[2]
217 - (int) (tile_size[2]*(common_size[2]/tile_size[2])))/2;
218
219 if ( verbose & VERB_PROCESS ) {
220 cout << "Common size: " << common_size << endl;
221 cout << "Tile size: " << tile_size << endl;
222 cout << "Maximum shift: " << max_shift << endl;
223 cout << "Reference start: " << start1 << endl << endl;
224 }
225
226 if ( res_hi < 0.1 || res_hi > tile_size[0]*p1->sampling(0)[0] ) res_hi = 4.0*p1->sampling(0)[0];
227
228 if ( filter_flag ) {
229 p1->filter_extremes();
230 p2->filter_extremes();
231 }
232
233 pex1 = p1->extract_tiles(0, start1, common_size, tile_size, step_size, 0);
234
235 vector<Vector3<long>> tile_start(pex1->images());
236
237 for ( i=0; i<pex1->images(); i++ ) {
238 m1 = (Bmarker *) add_item((char **) &m1, sizeof(Bmarker));
239 m2 = (Bmarker *) add_item((char **) &m2, sizeof(Bmarker));
240 if ( !set1 ) set1 = m1;
241 if ( !set2 ) set2 = m2;
242 m1->sel = m2->sel = 1;
243 m1->fom = m2->fom = 1;
244 m1->loc = pex1->image[i].origin() + half_tile;
245 tile_start[i] = pex1->image[i].origin();
246 }
247
248 for ( iter=0; iter<maxiter && best_t.fom>0.1; iter++ ) {
249 pex2 = p2->extract_tiles(0, tile_start, tile_size);
250 // Determines the shift of the second image onto the first
251 pex1->find_shift(pex2, NULL, res_hi, res_lo, max_shift, 0, refine_flag);
252 for ( i=0, m2=set2; i<pex1->images(); i++, m2=m2->next ) {
253 shift = pex1->image[i].origin() - pex2->image[i].origin();
254 if ( shift[0] > half_tile[0] ) shift[0] -= tile_size[0];
255 if ( shift[1] > half_tile[1] ) shift[1] -= tile_size[1];
256 if ( shift[2] > half_tile[2] ) shift[2] -= tile_size[2];
257 // To get the location in the second image, the shift must be subtracted
258 m2->loc = tile_start[i] + half_tile - shift;
259 }
260 t = markers_find_rottrans(set1, set2, 0.01);
261// if ( verbose & VERB_PROCESS )
262 cout << "Fit " << iter+1 << ":\t" << p1->file_name() << tab << p2->file_name() << tab
263 << t.trans << tab
264 << t.scale << tab
265 << t.angle*180.0/M_PI << tab << t.fom << endl;
266 if ( best_t.fom > t.fom ) best_t = t;
267 else iter = maxiter;
268 for ( i=0, m1=set1, m2=set2; i<pex1->images(); i++, m1=m1->next, m2=m2->next ) {
269 m2->loc[0] = best_t.scale[0]*(m1->loc[0]*cos(best_t.angle) -
270 m1->loc[1]*sin(best_t.angle)) + t.trans[0] + 0.5;
271 m2->loc[1] = best_t.scale[1]*(m1->loc[0]*sin(best_t.angle) +
272 m1->loc[1]*cos(best_t.angle)) + t.trans[1] + 0.5;
273 m2->loc[2] = best_t.scale[2]*m1->loc[2] + t.trans[2];
274 tile_start[i] = m2->loc - half_tile;
275 }
276 delete pex2;
277 }
278
279 if ( verbose & VERB_RESULT )
280 cout << "Best fit:\t" <<
281 p1->file_name() << tab << p2->file_name() << tab <<
282 best_t.trans << tab <<
283 best_t.scale << tab <<
284 best_t.angle*180.0/M_PI << tab << best_t.fom << endl;
285
286 delete pex1;
287
288 ps_marker_errors(filename, set1, set2, t, p1->size(), 20);
289
290 kill_list((char *) set1, sizeof(Bmarker));
291 kill_list((char *) set2, sizeof(Bmarker));
292
293 return best_t;
294}
