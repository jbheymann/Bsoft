432/*
433@brief Calculates the harmonic potential between components.
434@param *link link.
435@param d0 reference distance between components.
436@param Kd distance force constant.
437@param wrap flag to wrap coordinates.
438@param box simulation box boundaries.
439@return double link energy.
440
441 The harmonic energy function:
442 E = Kd*(d - d0)^2
443 The force is zero when d = reference link length (d0).
444**/
445double component_harmonic_potential(Bcomponent* comp1, Bcomponent* comp2, double d0, double Kd, bool wrap, Vector3<double> box)
446{
447 if ( d0 <= 0 || Kd <= 0 ) return 0;
448
449 double d, dev, fac, E(0);
450 Vector3<double> v, F;
451
452 if ( wrap ) v = vector3_difference_PBC(comp1->location(), comp2->location(), box);
453 else v = comp1->location() - comp2->location();
454 d = v.length();
455 dev = d - d0;
456 E = Kd*dev*dev;
457 if ( d ) {
458 fac = -2*Kd*dev/d;
459 F = v * fac;
460 comp1->force(comp1->force() + F);
461 comp2->force(comp2->force() - F);
462 }
463
464 if ( verbose & VERB_DEBUG )
465 cout << "DEBUG component_harmonic_potential: c1=" << comp1->identifier() <<
466 " c2=" << comp2->identifier() << " Kd=" << Kd << " d0=" << d0 << " E=" << E << endl;
467
468 return E;
469}

471/*
472@brief Calculates the soft sphere potential between components.
473@param comp1 first component.
474@param comp2 second component.
475@param d0 reference distance between components.
476@param Kd distance force constant.
477@param wrap flag to wrap coordinates.
478@param box simulation box boundaries.
479@return double distance energy.
480
481 The soft sphere potential is given by:
482 E = Kd*(d0/d)^12
483 The potential is set to zero for d > 3*d0;
484**/
485double component_soft_potential(Bcomponent* comp1, Bcomponent* comp2, double d0, double Kd, bool wrap, Vector3<double> box)
486{

552/*
553@brief Calculates the Morse potential between components.
554@param comp1 first component.
555@param comp2 second component.
556@param d0 reference distance between components.
557@param Kd distance force constant.
558@param wrap flag to wrap coordinates.
559@param box simulation box boundaries.
560@return double distance energy.
561
562 The Morse potential is given by:
563 E = Kd*((1-exp((a/d0)*(d0-d)))^2 - 1)
564 The width of the energy well is given by a, set here to 6.
565 The force is zero when d = d0.
566 The potential is set to zero for d > 3*d0;
567**/
568double component_morse_potential(Bcomponent* comp1, Bcomponent* comp2, double d0, double Kd, bool wrap, Vector3<double> box)
569{
570 if ( d0 <= 0 || Kd <= 0 ) return 0;
571
572 double d, fac, E(0);
573 double ef, ef1, a = 6/d0;
574 Vector3<double> v, F;
575
576 if ( wrap ) v = vector3_difference_PBC(comp1->location(), comp2->location(), box);
577 else v = comp1->location() - comp2->location();
578// v = comp2->location() - comp1->location(); // Points from 1 to 2
579 d = v.length();
580 if ( d < 3*d0 ) {
581 ef = exp(a*(d0 - d));
582 ef1 = 1 - ef;
583 E = Kd*(ef1*ef1 - 1);
584 fac = 2*a*Kd*ef1*ef/d;
585 F = v * fac; // Force on 1 in direction of 2
586 comp1->force(comp1->force() + F);
587 comp2->force(comp2->force() - F);
588 }
589
590 return E;
591}

593/*
594@brief Calculates the potential between two links to a component.
595@param comp component.
596@param comp1 first component linked to.
597@param comp2 second component linked to.
598@param a0 reference angle between linked components.
599@param Ka angular force constant.
600@return double angular energy.
601
602 The angular potential is given by:
603 E = Ka*(a - a0)^2
604 The force is zero when a = a0.
605**/
606double component_angular_potential(Bcomponent* comp, Bcomponent* comp1, Bcomponent* comp2, double a0, double Ka)
607{
608 if ( Ka <= 0 ) return 0;
609
610 double v1len, v2len, da, fac, E(0);
611 Vector3<double> v1, v2, n;
612
613 v1 = comp1->location() - comp->location();
614 v2 = comp2->location() - comp->location();
615 n = v1.cross(v2);
616 n.normalize();
617 v1len = v1.normalize();
618 v2len = v2.normalize();
619 da = acos(v1.scalar(v2)) - a0;
620 fac = -2*Ka*da;
621 E = Ka*da*da*v1len*v2len;
622 comp1->force(comp1->force() - n.cross(v1) * fac * v1len);
623 comp2->force(comp2->force() + n.cross(v2) * fac * v2len);
624
625 return E;
626}
