58int emx_get_transformation(xmlNodePtr parent, Matrix3& mat, Vector3<double>& origin)
59{
60 xmlNodePtr node = xml_find_node(parent, "transformationMatrix");
61
62 if ( !node ) return -1;
63
64 mat[0][0] = xml_get_real(node, "t11");
65 mat[0][1] = xml_get_real(node, "t12");
66 mat[0][2] = xml_get_real(node, "t13");
67 mat[1][0] = xml_get_real(node, "t21");
68 mat[1][1] = xml_get_real(node, "t22");
69 mat[1][2] = xml_get_real(node, "t23");
70 mat[2][0] = xml_get_real(node, "t31");
71 mat[2][1] = xml_get_real(node, "t32");
72 mat[2][2] = xml_get_real(node, "t33");
73
74 origin[0] = xml_get_real(node, "t14");
75 origin[1] = xml_get_real(node, "t24");
76 origin[2] = xml_get_real(node, "t34");
77
78 Matrix3 mattr = mat.transpose();
79
80#if CONV_SWITCH
81// 20150218 - convention T
82 origin = mattr * origin;
83 origin = -origin;
84#else
85// 20150323 - convention T-1
86 mat = mattr;
87#endif
88
89 return 0;
90}

159int read_project_emx(Bstring& filename, Bproject* project, Bstring& xsdfile)
160{
161 if ( filename.length() < 1 ) {
162 error_show("No file names found!", __FILE__, __LINE__);
163 return -1;
164 }
165
166 int err(0);
167 xmlDocPtr doc;
168 xmlNodePtr emx_node;
169
170 detect_and_fix_carriage_return(filename.c_str());
171
172 doc = xmlParseFile(filename.c_str());
173 if (doc == NULL) return -1;
174
175 if ( verbose & VERB_PROCESS )
176 cout << "# Reading EMX file: " << filename << endl;
177
178 if ( xsdfile.length() ) xml_validate(doc, xsdfile);
179
180 emx_node = xmlDocGetRootElement(doc);
181 if ( !emx_node ) {
182 error_show(filename.c_str(), __FILE__, __LINE__);
183 xmlFreeDoc(doc);
184 return -1;
185 }
186
187 if ( xmlStrcmp(emx_node->name, (const xmlChar *) "EMX") ) {
188 cerr << "Error: The document " << filename << " is not an EMX file!" << endl;
189 xmlFreeDoc(doc);
190 return -1;
191 }
192
193 xmlChar* content = xmlGetProp(emx_node, BAD_CAST "version");
194 if ( verbose ) {
195 if ( content ) cout << "EMX version " << content << endl;
196 else cout << "EMX version ?" << endl;
197 }
198
199// xmlDocDump(stderr, doc);
200// xmlElemDump(stderr, doc, node);
201
202 err = emx_to_project(doc, project, filename);
203
204 xmlFreeDoc(doc);
205
206 xmlCleanupParser();
207
208 project_calculate_angles(project);
209
210 return err;
211}

327/*
328@brief Converts micrograph data from an EMX document to a project structure.
329@param xmlDocPtr doc XML document pointer.
330@param *project project structure.
331@param &filename file name to resolve paths.
332@return int error code (<0 means failure).
333
334 The function sets up the project hierarchy from a XML file.
335 All angles given in the document are assumed to be in degrees,
336 and are converted here only to radians.
337
338**/
339int emx_to_project(xmlDocPtr doc, Bproject* project, Bstring& filename)
340{
341 int err(0);
342
343 xmlNodePtr emx_node = xmlDocGetRootElement(doc);
344 xmlNodePtr mg_node = NULL;
345 xmlNodePtr part_node = NULL;
346 xmlNodePtr part_mg_node = NULL;
347 xmlNodePtr node = NULL;
348
349 Bfield* field = project->field;
350 Bmicrograph* mg = NULL;
351// Breconstruction* rec = NULL;
352 Bparticle* part = NULL;
353
354 Bstring id, mg_file, part_file, prev_mg_file;
355 long i, pid, img_num;
356 double defU, defV;
357 Vector3<double> origin, center;
358 Matrix3 mat;
359
360 for ( i=0, node = emx_node->xmlChildrenNode; node; node = node->next ) {
361 mg_file = 0;
362 part_file = 0;
363 if ( !xmlStrcmp(node->name, BAD_CAST "micrograph") ) {
364 mg_node = node;
365 mg_file = (char *) xmlGetProp(mg_node, BAD_CAST "fileName");
366 if ( verbose & VERB_DEBUG )
367 cout << "DEBUG emx_to_project: mg=" << mg_file << endl;
368 id = Bstring(++i, "%d");
369 if ( mg_file.length() ) id = mg_file.pre_rev('.');
370 if ( id.length() < 1 ) id = Bstring(i, "%d");;
371// if ( !field ) field = field_add(&project->field, id);
372 if ( prev_mg_file != mg_file ) field = field_add(&project->field, id);
373 if ( verbose & VERB_DEBUG )
374 cout << "DEBUG emx_to_project: id=" << id << endl;
375 mg = micrograph_add(&field->mg, id);
376 mg->block = i;
377 mg->fmg = mg_file;
378 mg->img_num = xml_get_integer_attribute(mg_node, "index") - 1;
379 if ( mg->img_num < 0 ) {
380 if ( verbose & VERB_FULL )
381 cerr << "Warning: " << mg->id << " index number is less than 1" << endl;
382 mg->img_num = 0;
383 }
384 if ( verbose & VERB_DEBUG )
385 cout << "DEBUG emx_to_project: mg: mg=" << mg_file << " img_num=" << mg->img_num << endl;
386
387 center = micrograph_get_nominal_origin(mg);
388
389 emx_get_transformation(mg_node, mg->matrix, origin);
390 mg->origin = center + origin;
391
392 mg->select = xml_get_integer(mg_node, "activeFlag");
393 mg->fom = xml_get_real(mg_node, "fom");
394
395 if ( xml_find_node(mg_node, "pixelSpacing") )
396 mg->pixel_size = emx_get_real_vector2(mg_node, "pixelSpacing");
397
398 if ( xml_check_for_node(mg_node, "defocusU") || xml_check_for_node(mg_node, "acceleratingVoltage") ) {
399 mg->ctf = new CTFparam;
400 mg->ctf->volt(1e3 * xml_get_real(mg_node, "acceleratingVoltage"));
401 mg->ctf->amp_shift(asin(xml_get_real(mg_node, "amplitudeContrast")));
402 mg->ctf->Cs(1e7 * xml_get_real(mg_node, "cs"));
403 defU = 10*xml_get_real(mg_node, "defocusU");
404 defV = 10*xml_get_real(mg_node, "defocusV");
405 if ( defU < 100 ) defU *= 1e3;
406 if ( defV < 100 ) defV *= 1e3;
407 mg->ctf->defocus_average((defU + defV)/2);
408// mg->ctf->defocus_deviation(fabs(defU - defV)/2);
409// mg->ctf->astigmatism_angle(xml_get_real(node, "defocusUAngle") * M_PI/180.0);
410 mg->ctf->astigmatism(fabs(defU - defV)/2, xml_get_real(node, "defocusUAngle") * M_PI/180.0);
411 }
412 prev_mg_file = mg_file;
413 } else if ( !xmlStrcmp(node->name, BAD_CAST "particle") ) {
414 part_node = node;
415 part_file = (char *) xmlGetProp(part_node, BAD_CAST "fileName");
416 pid = xml_get_integer_attribute(part_node, "index");
417 if ( verbose & VERB_DEBUG )
418 cout << "DEBUG emx_to_project: part=" << part_file << " pid=" << pid << endl;
419
420 part_mg_node = xml_find_node(part_node, "micrograph");
421 if ( part_mg_node) {
422 mg = NULL;
423 mg_file = (char *) xmlGetProp(part_mg_node, BAD_CAST "fileName");
424// cout << "***" << mg_file << "***" << endl;
425 if ( mg_file.length() ) {
426 img_num = xml_get_integer_attribute(part_mg_node, "index") - 1;
427 if ( img_num < 0 ) img_num = 0;
428// cout << "***" << mg_file << "***" << img_num << "***" << endl;
429// cout << "***" << field->mg->fmg << "***" << field->mg->img_num << "***" << endl;
430 for ( field = project->field; field; field = field->next ) {
431 for ( mg = field->mg; mg; mg = mg->next ) {
432 if ( mg->fmg == mg_file && mg->img_num == img_num ) break;
433 }
434 if ( mg ) break;
435 }
436 if ( verbose & VERB_DEBUG )
437 cout << "DEBUG emx_to_project: part: mg=" << mg->fmg << " img_num=" << mg->img_num << endl;
438 } else {
439 cerr << "Warning: No micrograph file name specified!" << endl;
440 }
441 }
442
443 if ( !mg ) {
444 if ( verbose & VERB_FULL )
445 cerr << "Warning: No micrograph for particle " << pid << " found!" << endl;
446 id = Bstring(++i, "%d");
447 if ( !field ) {
448 if ( verbose )
449 cout << "Generating a new field-of-view record with id: " << id << endl;
450 field = field_add(&project->field, id);
451 }
452 if ( verbose )
453 cout << "Generating a new micrograph record with id: " << id << endl;
454 mg = micrograph_add(&field->mg, id);
455 }
456
457// mg->fpart = (char *) xmlGetProp(part_node, BAD_CAST "fileName");
458
459 if ( xml_find_node(part_node, "boxSize") )
460 mg->box_size = emx_get_integer_vector2(part_node, "boxSize");
461
462 if ( verbose & VERB_DEBUG )
463 cout << "Particle " << pid << endl;
464 part = particle_add(&mg->part, pid);
465
466 part->fpart = part_file;
467
468 if ( mg->box_size.volume() < 1 )
469 mg->box_size = particle_get_box_size(part);
470
471 if ( xml_find_node(part_node, "pixelSpacing") )
472 part->pixel_size = emx_get_real_vector2(part_node, "pixelSpacing");
473
474 if ( part->pixel_size[0] <= 0 )
475 part->pixel_size = Vector3<double>(1,1,1);
476
477 part->loc = emx_get_real_vector2(part_node, "centerCoord");
478
479 emx_get_transformation(part_node, mat, origin);
480 part->ori = origin + mg->box_size/2;
481 part->view = View2<double>(mat);
482
483 defU = 10*xml_get_real(part_node, "defocusU");
484 defV = 10*xml_get_real(part_node, "defocusV");
485 if ( defU < 100 ) defU *= 1e3;
486 if ( defV < 100 ) defV *= 1e3;
487 part->def = (defU + defV)/2;
488 part->dev = fabs(defU - defV)/2;
489 part->ast = xml_get_real(part_node, "defocusUAngle") * M_PI/180.0;
490
491 part->sel = xml_get_integer(part_node, "activeFlag");
492 part->fom[0] = xml_get_real(part_node, "fom");
493 }
494 }
495
496 return err;
497}

500/*
501@brief Converts micrograph data from an project structure to a XML document.
502@param Bproject* project parameter structure.
503@param xmlDocPtr doc XML document.
504@param mg_select flag to only write selected micrographs.
505@param rec_select flag to only convert selected reconstructions.
506@return int error code (<0 means failure).
507
508 The function packes new micrograph data back into an existing XML
509 document from which the old project parameters were obtained.
510 All angles used internally are in radians, and are converted here
511 to degrees for output. The angles within the micrograph structure
512 are left in radians.
513 The stylesheet written into the document is "Bsoft_micrograph.xsl".
514**/
515int project_to_emx(Bproject* project, xmlDocPtr doc, int mg_select, int rec_select)
516{
517 int err(0);
518
519 Bfield* field = NULL;
520 Bmicrograph* mg = NULL;
521 Bparticle* part = NULL;
522
523 xmlNodePtr root_node = xmlNewDocPI(doc, BAD_CAST "xml-stylesheet",
524 BAD_CAST "href=\"emx.xsl\" type=\"text/xsl\"");
525 xmlDocSetRootElement(doc, root_node);
526
527 xmlNodePtr emx_node = xmlNewNode(NULL, BAD_CAST "EMX");
528
529 xmlAddSibling(root_node, emx_node);
530
531 xmlNewProp(emx_node, BAD_CAST "version", BAD_CAST "1.0");
532
533 Bstring cmnt(" Written by Bsoft ");
534 cmnt += BVERSION;
535 xmlNodePtr cmnt_node = xmlNewComment(BAD_CAST cmnt.c_str());
536 xmlAddChild(emx_node, cmnt_node);
537
538 xmlNodePtr mg_node = NULL;
539 xmlNodePtr part_node = NULL;
540 xmlNodePtr node = NULL;
541
542 double angle;
543 Vector3<double> center;
544 Bstring filename;
545
546 for ( field=project->field; field; field=field->next ) {
547 for ( mg=field->mg; mg; mg=mg->next ) {
548 if ( mg->fmg.length() ) filename = mg->fmg;
549 else filename = "dummy.mrc";
550 mg_node = xmlNewChild(emx_node, NULL, BAD_CAST "micrograph", NULL);
551 xmlNewProp(mg_node, BAD_CAST "fileName", BAD_CAST filename.c_str());
552 xml_set_integer_attribute(mg_node, "index", mg->img_num+1, "%d");
553
554 center = micrograph_get_nominal_origin(mg);
555
556 emx_set_real_vector2(mg_node, "pixelSpacing", mg->pixel_size[0], mg->pixel_size[1], "A/px");
557
558 emx_set_transformation(mg_node, mg->matrix, mg->origin - center);
559
560 if ( mg->ctf ) {
561 emx_set_real_unit(mg_node, "acceleratingVoltage", mg->ctf->volt()/1e3, "kV");
562 emx_set_real_unit(mg_node, "amplitudeContrast", sin(mg->ctf->amp_shift()), NULL);
563 emx_set_real_unit(mg_node, "cs", mg->ctf->Cs()/1e7, "mm");
564 emx_set_real_unit(mg_node, "defocusU", (mg->ctf->defocus_average() + mg->ctf->defocus_deviation())/10, "nm");
565 emx_set_real_unit(mg_node, "defocusV", (mg->ctf->defocus_average() - mg->ctf->defocus_deviation())/10, "nm");
566 angle = mg->ctf->astigmatism_angle()*180.0/M_PI;
567 if ( angle < 0 ) angle += 180;
568 emx_set_real_unit(mg_node, "defocusUAngle", angle, "deg");
569 }
570
571 xml_set_integer(mg_node, "activeFlag", mg->select, "%d");
572 xml_set_real(mg_node, "fom", mg->fom, "%g");
573 }
574 }
575
576 for ( field=project->field; field; field=field->next ) {
577 for ( mg=field->mg; mg; mg=mg->next ) {
578 if ( mg->fmg.length() ) filename = mg->fmg;
579 else filename = "dummy.mrc";
580 center = mg->box_size/2;
581 for ( part=mg->part; part; part=part->next ) {
582 part_node = xmlNewChild(emx_node, NULL, BAD_CAST "particle", NULL);
583 if ( part->fpart.length() )
584 xmlNewProp(part_node, BAD_CAST "fileName", BAD_CAST part->fpart.c_str());
585 else
586 xmlNewProp(part_node, BAD_CAST "fileName", BAD_CAST mg->fpart.c_str());
587 xml_set_integer_attribute(part_node, "index", part->id, "%d");
588
589 node = xmlNewChild(part_node, NULL, BAD_CAST "micrograph", NULL);
590 xmlNewProp(node, BAD_CAST "fileName", BAD_CAST filename.c_str());
591 xml_set_integer_attribute(node, "index", mg->img_num+1, "%d");
592
593 emx_set_real_vector2(part_node, "pixelSpacing", part->pixel_size[0], part->pixel_size[1], "A/px");
594
595 emx_set_real_vector2(part_node, "boxSize", mg->box_size[0], mg->box_size[1], "px");
596 emx_set_real_vector2(part_node, "centerCoord", part->loc[0], part->loc[1], "px");
597
598 emx_set_transformation(part_node, part->view, part->ori - center);
599
600 if ( part->def ) {
601 emx_set_real_unit(part_node, "defocusU", (part->def + part->dev)/10, "nm");
602 emx_set_real_unit(part_node, "defocusV", (part->def - part->dev)/10, "nm");
603 angle = part->ast*180.0/M_PI;
604 if ( angle < 0 ) angle += 180;
605 emx_set_real_unit(part_node, "defocusUAngle", angle, "deg");
606 }
607
608 xml_set_integer(part_node, "activeFlag", part->sel, "%d");
609 xml_set_real(part_node, "fom", part->fom[0], "%g");
610 }
611 }
612 }
613
614 return err;
615}
