336/*
337@brief Adds a bond type to a linked list.
338@param **bond pointer to any bond type in the list.
339@param *symbol1 atom1 of bond.
340@param *symbol2 atom2 of bond.
341@param covlength reference covalent bond length.
342@param vdwdist reference Van der Waals distance.
343@return Bbondtype* new bond type.
344
345 The function allocates memory for a new bond type structure.
346 If the content of the pointer is null, the new structure is
347 the first in the list. Otherwise, the end of the list is found
348 and the new structure added to it.
349
350**/
351Bbondtype* bond_type_add(Bbondtype** bond, char* symbol1, char* symbol2, float covlength, float vdwdist)
352{
353 Bbondtype* this_bond = *bond;
354 Bbondtype* new_bond = new Bbondtype;
355 memset(new_bond, 0, sizeof(Bbondtype));
356
357 strncpy(new_bond->type1, symbol1, 8);
358 strncpy(new_bond->type2, symbol2, 8);
359 new_bond->covlength = covlength;
360 new_bond->vdwdist = vdwdist;
361
362 if ( verbose & VERB_DEBUG )
363 cout << "DEBUG bond_type_add: covlength=" << covlength << " vdwdist=" << vdwdist << endl;
364
365 if ( !this_bond )
366 *bond = new_bond;
367 else {
368 while ( this_bond->next ) this_bond = this_bond->next;
369 this_bond->next = new_bond;
370 }
371
372 return new_bond;
373}
