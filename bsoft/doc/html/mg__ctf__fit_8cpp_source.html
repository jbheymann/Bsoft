66Bimage* img_ctf_radial_average(Bimage* p, long n, CTFparam& em_ctf)

69 cerr << "in img_ctf_radial_average" << endl;

71 long nrad = p->sizeX()/2;

73 prad->sampling(p->sampling(0)[0], 1, 1);

75 long i, x, y, z;
76 long iradius;
77 double dx, dy, dz, dr, radius, angle, fraction, cosang, sinang;
78 double yscale(p->sizeX()*1.0L/p->sizeY());
79 vector<double> num(nrad);
80 double* rdata = (double *) prad->data_pointer();
81
82 double smin2(1), smax2(1);
83 if ( fabs(em_ctf.defocus_average()) > 1 ) {
84 smin2 = 1-em_ctf.defocus_deviation()/em_ctf.defocus_average();
85 smax2 = 1+em_ctf.defocus_deviation()/em_ctf.defocus_average();
86 }
87
88 if ( verbose & VERB_FULL ) {
89 cout << "Radial average for CTF fitting:" << endl;
90 cout << "Size: " << nrad << endl;
91 cout << "Power spectrum origin: " << p->image[n].origin() << endl;
92 cout << "Defocus, deviation, angle: " <<
93 em_ctf.defocus_average() << " " << em_ctf.defocus_deviation() << " " << em_ctf.astigmatism_angle() << endl;
94 }
95
96 if ( p->sizeZ() < 2 ) {
97 if ( verbose & VERB_DEBUG )
98 cout << "DEBUG img_ctf_radial_average: doing image " << n+1 << endl;
99 for ( i=0; i<nrad; i++ ) num[i] = 0;
100 for ( y=0; y<p->sizeY(); y++ ) {
101 dy = ((double)y - p->image[n].origin()[1])*yscale;
102 for ( x=0; x<p->sizeX(); x++ ) {
103 dx = (double)x - p->image[n].origin()[0];
104 radius = sqrt(dx*dx + dy*dy);
105 iradius = (long) radius;
106 if ( iradius < nrad ) {
107 angle = atan2(dy, dx) - em_ctf.astigmatism_angle();
108 cosang = cos(angle);
109 sinang = sin(angle);
110 dr = smax2*cosang*cosang + smin2*sinang*sinang;
111 if ( dr >= 0) {
112 radius *= sqrt(dr);
113// if ( !isfinite(radius) )
114// cerr << "Warning: radius for " << iradius << " not finite!" << endl;
115 iradius = (long) radius;
116 if ( iradius < nrad - 1 ) {
117 fraction = radius - iradius;
118 num[iradius] += 1.0 - fraction;
119 i = p->index(0,x,y,0,n);
120 rdata[iradius] += (1.0 - fraction)*(*p)[i];
121 iradius++;
122 if ( iradius < nrad ) {
123 num[iradius] += fraction;
124 rdata[iradius] += fraction*(*p)[i];
125 }
126 }
127 }
128 }
129 }
130 }
131 for ( iradius=0; iradius<nrad; iradius++ )
132 if ( num[iradius] ) rdata[iradius] /= num[iradius];
133 else rdata[iradius] = rdata[iradius-1];
134 } else {
135 if ( verbose & VERB_DEBUG )
136 cout << "DEBUG img_ctf_radial_average: calculating 3D RPS" << endl;
137 for ( i=0; i<nrad; i++ ) num[i] = 0;
138 for ( z=0; z<p->sizeZ(); z++ ) {
139 dz = z - p->image[n].origin()[2];
140 for ( y=0; y<p->sizeY(); y++ ) {
141 dy = y - p->image[n].origin()[1];
142 for ( x=0; x<p->sizeX(); x++ ) {
143 dx = x - p->image[n].origin()[0];
144 radius = sqrt(dx*dx + dy*dy + dz*dz);
145 iradius = (long) radius;
146 if ( iradius < nrad - 1 ) {
147 fraction = radius - iradius;
148 num[iradius] += 1.0 - fraction;
149 i = p->index(0,x,y,z,n);
150 rdata[n*nrad+iradius] += (1.0 - fraction)*(*p)[i];
151 iradius++;
152 if ( iradius < nrad ) {
153 num[iradius] += fraction;
154 rdata[iradius] += fraction*(*p)[i];
155 }
156 }
157 }
158 }
159 }
160 for ( iradius=0; iradius<nrad; iradius++ )
161 if ( num[iradius] ) rdata[iradius] /= num[iradius];
162 }
163
164 prad->statistics();
165
166 return prad;

169double img_ctf_fit_residual(Bimage* p, long n, CTFparam& em_ctf, double lores, double hires)
170{
171 long rmin = (long) (p->real_size()[0]/lores);
172 long rmax = (long) (p->real_size()[0]/hires);
173
174 long i;
175 double s, s2, b, e, c, d, R(0), v(0);
176
177 Bimage* prad = img_ctf_radial_average(p, n, em_ctf);
178
179 for ( i=rmin; i<=rmax; ++i ) {
180 s = i/p->real_size()[0];
181 s2 = s*s;
182 b = em_ctf.calc_baseline(s);
183 e = em_ctf.calc_envelope(s);
184 c = em_ctf.calculate(s2, 0);
185 c *= c;
186 d = b + e;
187 v += d*d;
188 d = b + e*c - (*prad)[i];
189 R += d*d;
190 if ( verbose & VERB_FULL )
191 cout << i << tab << (*prad)[i] << tab << b + e*c << tab << d << endl;
192 }
193
194 delete prad;
195
196 R = sqrt(R/v);
197
198 return R;
199}

1018/*
1019 Expects a radial power spectrum
1020 Measures the astigmatism between the resolution limits
1021 The variable real_size is the box x-size in angstrom
1022 Zeroes are calculated, given in spatial frequency units
1023 For every two successive zeroes between the resolution limits:
1024 the minimum within a kernel is found at each zero: [ib1]:b1 and [ib2]:b2
1025 the maximum halfway between the zeroes within a kernel is found: [im]:m
1026 Measure: sum(im*(m - (b1+b2)/2))
1027 the intent is that this measure should be maximum for the correct astigmatism parameters
1028*/
1029double img_ctf_astigmatism_measure(Bimage* p, long n, double real_size, CTFparam& em_ctf, double lores, double hires)
1030{
1031 Bimage* prad = img_ctf_radial_average(p, n, em_ctf);
1032
1033 if ( !prad ) {
1034 cerr << "Error in img_ctf_astigmatism_measure: No radial power spectrum calculated!" << endl;
1035 return -1;
1036 }
1037
1038 double pixel_size = prad->sampling(0)[0];
1039 if ( pixel_size <= 0 ) pixel_size = 1;
1040
1041 long i, ib1, ib2, im, j, nn(0);
1042 long k = prad->sizeX()/100;
1043 if ( k < 2 ) k = 2;
1044 long ilo = (long) (real_size/lores + 0.5);
1045 long ihi = (long) (real_size/hires + 0.5);
1046 if ( ihi >= prad->sizeX() ) ihi = prad->sizeX() - 1;
1047 double b1, b2, m, OM(0);
1048 double max_s = 0.45/pixel_size;
1049
1050 Bimage* prmb = img_ctf_radial_subtract_baseline(prad, real_size, em_ctf);
1051
1052 vector<double> zero = em_ctf.zeroes(max_s);
1053 long nz(zero.size());
1054
1055 if ( nz > prad->sizeX() ) nz = prad->sizeX();
1056
1057 for ( i=1; i<nz; i++ ) {
1058 ib1 = (long) (zero[i-1]*real_size + 0.5);
1059 ib2 = (long) (zero[i]*real_size + 0.5);
1060 im = (ib1 + ib2)/2;
1061 if ( im >= ilo && im <= ihi ) {
1062 for ( b1=(*prmb)[ib1], j=ib1-k; j<=ib1+k; j++ )
1063 if ( b1 > (*prmb)[j] ) b1 = (*prmb)[j];
1064 for ( b2=(*prmb)[ib2], j=ib2-k; j<=ib2+k; j++ )
1065 if ( b2 > (*prmb)[j] ) b2 = (*prmb)[j];
1066 for ( m=(*prmb)[im], j=im-k; j<=im+k; j++ )
1067 if ( m < (*prmb)[j] ) m = (*prmb)[j];
1068 OM += im*(m - (b1 + b2)/2);
1069 nn += im;
1070 }
1071 }
1072
1073 OM /= nn;
1074
1075 delete prad;
1076 delete prmb;
1077
1078 if ( verbose & VERB_FULL )
1079 cout << "Fitting measure: " << OM << endl;
1080
1081 return OM;
1082}

1178int img_add_aberration_terms(Bimage* psum, Bimage* psec, Bimage* p, CTFparam& cp, bool ew_flag)
1179{
1180 long i, j, n, x, y, z;
1181 double sx, sy, sz, s, a, dphi, pwr;
1182 Complex<double> cv, ca;
1183 Vector3<double> freq_scale(1.0/psum->real_size());
1184 Vector3<double> h((psum->sizeX() - 1)/2, (psum->sizeY() - 1)/2, (psum->sizeZ() - 1)/2);
1185
1186 if ( verbose & VERB_DEBUG ) {
1187 cout << "Calculating aberration terms" << endl;
1188 cout << endl;
1189 }
1190
1191 for ( i=j=n=0; n<psum->images(); n++ ) {
1192 for ( z=0; z<psum->sizeZ(); z++ ) {
1193 sz = z;
1194 if ( z > h[2] ) sz -= psum->sizeZ();
1195 sz *= freq_scale[2];
1196 for ( y=0; y<psum->sizeY(); y++ ) {
1197 sy = y;
1198 if ( y > h[1] ) sy -= psum->sizeY();
1199 sy *= freq_scale[1];
1200 for ( x=0; x<psum->sizeX(); x++, i++ ) if ( psec->complex(i).not_zero() ) {
1201 sx = x;
1202 if ( x > h[0] ) sx -= psum->sizeX();
1203 sx *= freq_scale[0];
1204 a = atan2(sy,sx);
1205 if ( !ew_flag ) { // Original
1206 s = sx*sx + sy*sy + sz*sz;
1207 dphi = cp.calculate_aberration_even(s, a);
1208 ca = Complex<double>(cosl(dphi),sinl(dphi));
1209 } else { // Ewald sphere
1210 s = sqrt(sx*sx + sy*sy + sz*sz);
1211 ca = cp.calculate_complex(s, a);
1212 }
1213 cv = psec->complex(i);
1214 pwr = cv.power();
1215 cv *= p->complex(i).conj();
1216 j = 6*i;
1217 psum->add(j++, cv.real() * ca.imag()); // (S*P')re * sin g
1218 psum->add(j++, cv.imag() * ca.imag()); // (S*P')im * sin g
1219 psum->add(j++, cv.real() * ca.real()); // (S*P')re * cos g
1220 psum->add(j++, pwr * ca.real() * ca.real()); // S2 * cos2 g
1221 psum->add(j++, pwr * ca.real() * ca.imag()); // S2 * cos g * sin g
1222 psum->add(j++, pwr * ca.imag() * ca.imag()); // S2 * sin2 g
1223 }
1224 }
1225 }
1226 }
1227
1228 return 0;
1229}
