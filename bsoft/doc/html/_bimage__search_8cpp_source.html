194Bimage* Bimage::search_volume(Bimage* ptemp, vector<View2<double>>& views, double alpha,
195 double alpha_step, double hires, double lores,
196 Bimage* pmask, double threshold)
197{
198 int mode(0); // Mode: 0=global, 1=refine, 2=symmetry
199 long i, nviews(0);
200 double alpha_min(0);
201 double alpha_max(TWOPI - alpha_step/2);
202 double a;
203 double fommin(1), fommax(0), fomavg(0), fomstd(0);
204
205 if ( alpha >= 0 ) { // Case of refinement
206 mode = 1;
207 alpha_min = alpha - alpha_step;
208 alpha_max = alpha + alpha_step;
209 } else if ( alpha_step >= 1.999*M_PI ) mode = 2; // Symmetry
210
211 change_type(Float);
212 ptemp->change_type(Float);
213 rescale_to_avg_std(0, 1);
214 ptemp->rescale_to_avg_std(0, 1);
215 calculate_background();
216 ptemp->calculate_background();
217
218 Vector3<double> ori(image->origin());
219
220 if ( ori.sum() == 0 ) {
221 ori = size()/2;
222 origin(ori);
223 }
224
225 if ( mode < 2 ) views = view_list_expand_angles(views, alpha_min, alpha_max, alpha_step);
226
227 nviews = views.size();
228
229 if ( verbose ) {
230 cout << "Finding a template in a map:" << endl;
231 cout << "Number of views: " << nviews << endl;
232 cout << "Alpha step size: " << alpha_step*180.0/M_PI << endl;
233 cout << "Origin: " << ori << endl;
234 }
235
236 Bimage* pfit = new Bimage(Float, TView, size(), 1);
237 pfit->next = new Bimage(Float, TSimple, size(), 1);
238 pfit->sampling(sampling(0));
239
240 if ( verbose & VERB_RESULT )
241 cout << "View\tCCmax" << endl;
242
243 for ( i=0; i<nviews; i++ ) {
244 fommax = search_volume_view(ptemp, views[i], hires, lores, pmask, threshold, pfit);
245 if ( verbose )
246 cout << views[i] << tab << fommax << endl;
247 }
248
249 for ( i=0, fommax=-1, fommin=1, fomavg=fomstd=0; i<datasize; i++ ) {
250 a = (*pfit->next)[i];
251 if ( fommin > a ) fommin = a;
252 if ( fommax < a ) fommax = a;
253 fomavg += a;
254 fomstd += a*a;
255 }
256
257 if ( verbose ) {
258 fomavg /= datasize;
259 fomstd = fomstd/datasize - fomavg*fomavg;
260 if ( fomstd > 0 ) fomstd = sqrt(fomstd);
261 else fomstd = 0;
262 cout << "FOM minimum and maximum: " << fommin << " " << fommax << endl;
263 cout << "FOM average: " << fomavg << endl;
264 cout << "FOM standard deviation: " << fomstd << endl << endl;
265 }
266
267 return pfit;
268}
